    47→    47→function formatFileSize(bytes: number): string {
    48→    48→  if (bytes < 1024) return `${bytes} B`;
    49→    49→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
    50→    50→  return `${(bytes / (1024 * 1024)).toFixed(1)} MB`;
    51→    51→}
    52→    52→
    53→    53→function formatLastSeen(lastSeen: string | null): string {
    54→    54→  if (!lastSeen) return "Offline";
    55→    55→  const diffMs = Date.now() - new Date(lastSeen).getTime();
    56→    56→  const diffSec = Math.floor(diffMs / 1000);
    57→    57→  if (diffSec < 60) return "Seen just now";
    58→    58→  const diffMin = Math.floor(diffSec / 60);
    59→    59→  if (diffMin < 60) return `Seen ${diffMin}m ago`;
    60→    60→  const diffHr = Math.floor(diffMin / 60);
    61→    61→  if (diffHr < 24) return `Seen ${diffHr}h ago`;
    62→    62→  const diffDay = Math.floor(diffHr / 24);
    63→    63→  if (diffDay < 7) return `Seen ${diffDay}d ago`;
    64→    64→  return `Seen ${new Date(lastSeen).toLocaleDateString()}`;
    65→    65→}
