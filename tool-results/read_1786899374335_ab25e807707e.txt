    38→function formatFileSize(bytes: number): string {
    39→  if (bytes < 1024) return `${bytes} B`;
    40→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
    41→  return `${(bytes / (1024 * 1024)).toFixed(1)} MB`;
    42→}
    43→
    44→function formatLastSeen(lastSeen: string | null): string {
    45→  if (!lastSeen) return "Offline";
    46→  const diffMs = Date.now() - new Date(lastSeen).getTime();
    47→  const diffSec = Math.floor(diffMs / 1000);
    48→  if (diffSec < 60) return "Seen just now";
    49→  const diffMin = Math.floor(diffSec / 60);
    50→  if (diffMin < 60) return `Seen ${diffMin}m ago`;
    51→  const diffHr = Math.floor(diffMin / 60);
    52→  if (diffHr < 24) return `Seen ${diffHr}h ago`;
    53→  const diffDay = Math.floor(diffHr / 24);
    54→  if (diffDay < 7) return `Seen ${diffDay}d ago`;
    55→  return `Seen ${new Date(lastSeen).toLocaleDateString()}`;
    56→}

   231→export function ChatLayout({ currentUserId, otherUserId, conversationId, partner, initialMessages }: ChatLayoutProps) {
   232→  const { messages, isLoading, isPartnerTyping, partnerStatus, sendMessage, onTyping, markAsRead, vanishMessage, deleteMessage, sendReaction, loadMore, hasMore, loadingMore } =
   233→    useChat({ currentUserId, otherUserId, conversationId, initialMessages });
   234→
   235→  const [draft, setDraft] = useState("");
   236→  const [sendBtnKey, setSendBtnKey] = useState(0);
   237→  const [isSending, setIsSending] = useState(false);
   238→  const [vanishMode, setVanishMode] = useState(false);
   239→  const [replyTo, setReplyTo] = useState<ChatMessage | null>(null);
   240→  const [ephemeralSeconds, setEphemeralSeconds] = useState<number | null>(null);
   241→  const [infoPanelOpen, setInfoPanelOpen] = useState(false);
   242→  const inputRef = useRef<HTMLTextAreaElement>(null);
   243→  const viewport = useVisualViewport();
   244→
   245→  /* Background theme + wallpaper */
   246→  const { themeId, theme, selectTheme, wallpaper, setWallpaper } = useChatBackground();
   247→
   248→  /* Voice */
   249→  const voice = useVoiceRecorder();
   250→  const [showVoiceWaveform, setShowVoiceWaveform] = useState(false);
   251→  const [isSendingVoice, setIsSendingVoice] = useState(false);
   252→
   253→  /* Attachment */
   254→  const [attachmentFile, setAttachmentFile] = useState<File | null>(null);
   255→  const [attachmentPreview, setAttachmentPreview] = useState<string | null>(null);
   256→  const [isUploading, setIsUploading] = useState(false);
   257→  const fileInputRef = useRef<HTMLInputElement>(null);
   258→
   259→  const containerStyle = viewport.height > 0 ? ({ height: `${viewport.height}px` } as React.CSSProperties) : ({ height: "100dvh" } as React.CSSProperties);
   260→  const kbOffset = viewport.isKeyboardVisible && viewport.offsetTop > 0 ? ({ marginTop: `-${viewport.offsetTop}px` } as React.CSSProperties) : undefined;
   261→
   262→  /* Compute background style — wallpaper overrides theme */
   263→  const bgStyle = wallpaper
   264→    ? { backgroundImage: `url(${wallpaper})`, backgroundSize: "cover", backgroundPosition: "center" }
   265→    : theme.style;
   266→
   267→  useEffect(() => { inputRef.current?.focus(); }, []);
   268→
   269→  /* Show voice waveform when recording */
   270→  useEffect(() => { setShowVoiceWaveform(voice.isRecording); }, [voice.isRecording]);
   271→
   272→  /* Global keyboard shortcuts */
   273→  useEffect(() => {
   274→    function handleGlobalKey(e: KeyboardEvent) {
   275→      if (e.key === "Escape") {
   276→        if (replyTo) { setReplyTo(null); e.preventDefault(); }
   277→        else if (infoPanelOpen) { setInfoPanelOpen(false); e.preventDefault(); }
   278→      }
   279→    }
   280→    window.addEventListener("keydown", handleGlobalKey);
   281→    return () => window.removeEventListener("keydown", handleGlobalKey);
   282→  }, [replyTo, infoPanelOpen]);
   283→
   284→  /* Derived values — MUST be declared before any useCallback that
   285→     references them in its dependency array to avoid TDZ errors
   286→     in the production webpack bundle. */
   287→  const hasAttachment = attachmentFile !== null;
   288→  const hasText = draft.trim().length > 0;
   289→
   290→  /* ---- Attachment upload helpers (declared BEFORE handleSubmit
   291→     because handleSubmit's dependency array includes uploadAttachment;
   292→     accessing a const before its declaration causes a TDZ crash in the
   293→     production webpack bundle deployed to Cloudflare Workers). ---- */
   294→  const clearAttachment = useCallback(() => {
   295→    if (attachmentPreview) URL.revokeObjectURL(attachmentPreview);
   296→    setAttachmentFile(null);
   297→    setAttachmentPreview(null);
   298→  }, [attachmentPreview]);
   299→
   300→  const uploadAttachment = useCallback(async () => {
   301→    if (!attachmentFile || isUploading) return;
   302→    setIsUploading(true);
   303→    try {
   304→      const fd = new FormData();
   305→      fd.append("file", attachmentFile);
   306→      fd.append("conversation_id", conversationId);
   307→      if (replyTo?.id) fd.append("reply_to_id", replyTo.id);
   308→      if (vanishMode) fd.append("vanish_mode", "true");
   309→      if (ephemeralSeconds) fd.append("ephemeral_seconds", String(ephemeralSeconds));
   310→
   311→      const res = await fetch("/api/messages/upload", { method: "POST", body: fd });
   312→      if (!res.ok) {
   313→        const data = await res.json().catch(() => ({ error: "Upload failed" }));
   314→        throw new Error(data.error || `Upload failed: ${res.status}`);
   315→      }
   316→      /* Message is inserted server-side and pushed via realtime;
   317→         the useChat hook will pick it up automatically. */
   318→      clearAttachment();
   319→      setReplyTo(null);
   320→    } catch (err) {
   321→      console.error("[ChatLayout] upload error:", err);
   322→      toast({
   323→        title: "Upload failed",
   324→        description: err instanceof Error ? err.message : "Could not send the file.",
   325→        variant: "destructive",
   326→      });
   327→    } finally {
   328→      setIsUploading(false);
   329→    }
   330→  }, [attachmentFile, isUploading, conversationId, replyTo, vanishMode, ephemeralSeconds, clearAttachment]);
   331→
   332→  const handleSubmit = useCallback(async (e: FormEvent) => {
   333→    e.preventDefault();
   334→    /* If an attachment is selected, upload it instead of sending text */
   335→    if (hasAttachment && !hasText) {
   336→      await uploadAttachment();
   337→      return;
   338→    }
   339→    /* If there's both text and an attachment, send text first, then upload */
   340→    const trimmed = draft.trim();
   341→    if (!trimmed || isSending) return;
   342→    setIsSending(true);
   343→    const opts: SendMessageOptions = { vanish_mode: vanishMode, ephemeral_seconds: ephemeralSeconds, reply_to_id: replyTo?.id ?? null };
   344→    setDraft(""); setReplyTo(null);
   345→    if (inputRef.current) inputRef.current.style.height = "auto";
   346→    try {
   347→      await sendMessage(trimmed, opts);
   348→      setSendBtnKey((k) => k + 1);
   349→      /* Upload attachment after text is sent */
   350→      if (hasAttachment) await uploadAttachment();
   351→    }
   352→    finally { setIsSending(false); requestAnimationFrame(() => inputRef.current?.focus()); }
   353→  }, [draft, isSending, sendMessage, vanishMode, ephemeralSeconds, replyTo, hasAttachment, uploadAttachment]);
   354→
   355→  const handleVoiceSend = useCallback(async () => {
   356→    const result = await voice.stopRecording();
   357→    if (!result || result.duration < 1) return;
   358→    const blob = result.blob;
   359→    setIsSendingVoice(true);
   360→    try {
   361→      const fd = new FormData();
   362→      fd.append("file", blob, "voice-note.webm");
   363→      fd.append("conversation_id", conversationId);
   364→      fd.append("voice_duration", String(result.duration));
   365→      fd.append("waveform_data", JSON.stringify(result.amplitudes));
   366→      if (ephemeralSeconds) fd.append("ephemeral_seconds", String(ephemeralSeconds));
   367→
   368→      const res = await fetch("/api/messages/voice", { method: "POST", body: fd });
   369→      if (!res.ok) {
   370→        const data = await res.json().catch(() => ({ error: "Voice upload failed" }));
   371→        throw new Error(data.error || `Voice upload failed: ${res.status}`);
   372→      }
   373→      /* Message is created server-side and broadcast via realtime. */
   374→    } catch (err) {
   375→      console.error("[ChatLayout] voice upload error:", err);
   376→      toast({
   377→        title: "Voice note failed",
   378→        description: err instanceof Error ? err.message : "Could not send the voice note.",
   379→        variant: "destructive",
   380→      });
   381→    } finally {
   382→      setIsSendingVoice(false);
   383→    }
   384→  }, [voice, conversationId, ephemeralSeconds]);
   385→
   386→  const handleVoiceCancel = useCallback(async () => {
   387→    await voice.stopRecording();
   388→    /* showVoiceWaveform is set to false by the useEffect watching voice.isRecording */
   389→  }, [voice]);
   390→
   391→  const handleKeyDown = useCallback((e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   392→    if (e.key === "Enter" && !e.shiftKey) { e.preventDefault(); const form = e.currentTarget.closest("form"); if (form) form.requestSubmit(); }
   393→  }, []);
   394→
   395→  const handleInputChange = useCallback((e: React.ChangeEvent<HTMLTextAreaElement>) => { setDraft(e.target.value); onTyping(); }, [onTyping]);
   396→  const handleInput = useCallback((e: React.FormEvent<HTMLTextAreaElement>) => { const el = e.currentTarget; el.style.height = "auto"; el.style.height = `${Math.min(el.scrollHeight, 128)}px`; }, []);
   397→
   398→  const canSend = hasText || hasAttachment;
   399→
   400→  /* ---- Attachment handlers ---------------------------------------- */
   401→  const handleFileSelect = useCallback((e: ChangeEvent<HTMLInputElement>) => {
   402→    const file = e.target.files?.[0] ?? null;
   403→    if (!file) return;
   404→    if (file.size > MAX_ATTACHMENT_SIZE) {
   405→      toast({ title: "File too large", description: "Maximum size is 10 MB.", variant: "destructive" });
   406→      e.target.value = "";
   407→      return;
   408→    }
   409→    setAttachmentFile(file);
   410→    /* Generate thumbnail preview for images */
   411→    if (IMAGE_MIME_TYPES.has(file.type)) {
   412→      const url = URL.createObjectURL(file);
   413→      setAttachmentPreview(url);
   414→    } else {
   415→      setAttachmentPreview(null);
   416→    }
   417→    /* Reset input so the same file can be re-selected */
   418→    e.target.value = "";
   419→    requestAnimationFrame(() => inputRef.current?.focus());
   420→  }, []);
   421→
   422→  const handlePaperclipClick = useCallback(() => {
   423→    fileInputRef.current?.click();
   424→  }, []);
   425→
   426→  /* Status text for header */
   427→  const statusText = isPartnerTyping
   428→    ? "typing\u2026"
   429→    : partnerStatus.online
   430→      ? "Online"
   431→      : formatLastSeen(partnerStatus.last_seen);
   432→
   433→  /* ---- Render --------------------------------------------------- */
   434→
   435→  return (
   436→    <div className="flex w-full flex-col bg-background" style={containerStyle}>
   437→      <div style={kbOffset} className="flex min-h-0 flex-1 flex-col">
   438→        {/* Header - sticky with glassmorphic blur + shadow */}
   439→        <header className="sticky top-0 z-30 flex h-14 shrink-0 items-center gap-3 border-b border-[var(--app-accent-subtle)] bg-background/80 shadow-[0_1px_3px_rgba(0,0,0,0.04)] backdrop-blur-2xl supports-[backdrop-filter]:bg-background/60 px-4">
   440→          <button
   441→            type="button"
   442→            onClick={() => setInfoPanelOpen(true)}
   443→            className="flex items-center gap-3 rounded-lg p-1 -ml-1 transition-colors hover:bg-muted/50 focus-visible:outline-none"
   444→            aria-label="Open user info"
   445→          >
   446→            <Avatar className="size-9 ring-2 ring-[var(--app-accent-lighter)]/30">
   447→              {partner.avatar_url && <AvatarImage src={partner.avatar_url} alt={partner.name} />}
   448→              <AvatarFallback
   449→                className="text-xs font-semibold text-white"
   450→                style={{ background: "linear-gradient(to bottom right, var(--app-accent-from), var(--app-accent-to))" }}
   451→              >{partner.name.slice(0, 2).toUpperCase()}</AvatarFallback>
   452→            </Avatar>
   453→            <div className="flex flex-col">
   454→              <span className="text-sm font-semibold leading-tight">{partner.name}</span>
   455→              <div className="flex items-center gap-1.5">
   456→                <StatusDot online={partnerStatus.online} />
   457→                <span
   458→                  className={cn("text-xs", isPartnerTyping ? "font-medium" : "text-muted-foreground")}
   459→                  style={isPartnerTyping ? { color: "var(--app-accent)" } : undefined}
   460→                >{statusText}</span>
   461→              </div>
   462→            </div>
   463→          </button>
   464→
   465→          {/* Right-side header actions */}
   466→          <div className="ml-auto flex items-center gap-1">
   467→            <ChatBackgroundPicker themeId={themeId} onSelect={selectTheme} wallpaper={wallpaper} onSetWallpaper={setWallpaper} />
   468→          </div>
   469→        </header>
   470→
   471→        {/* Feed area – pb-28 so messages never get covered by the floating input bar */}
   472→        <div className="relative min-h-0 flex-1 pb-28">
   473→          <MessageFeed
   474→            messages={messages}
   475→            isLoading={isLoading}
   476→            isPartnerTyping={isPartnerTyping}
   477→            currentUserId={currentUserId}
   478→            partnerName={partner.name}
   479→            partnerAvatar={partner.avatar_url}
   480→            backgroundStyle={bgStyle}
   481→            onMarkAsRead={markAsRead}
   482→            onVanishMessage={vanishMessage}
   483→            onDeleteMessage={deleteMessage}
   484→            onReplyTo={setReplyTo}
   485→            onReact={sendReaction}
   486→            loadMore={loadMore}
   487→            hasMore={hasMore}
   488→            loadingMore={loadingMore}
   489→          />
   490→        </div>
   491→      </div>
   492→
   493→      {/* Hidden file input for attachments */}
   494→      <input
   495→        ref={fileInputRef}
   496→        type="file"
   497→        accept={ACCEPTED_TYPES}
   498→        onChange={handleFileSelect}
   499→        className="hidden"
   500→        aria-hidden="true"
   501→      />
   502→
   503→      {/* Floating preview bars above the pill */
   504→      {(replyTo || attachmentFile) && (
   505→        <div
   506→          className="fixed left-4 right-4 max-w-2xl mx-auto z-30 flex flex-col gap-1.5"
   507→          style={{ bottom: 80 }}
   508→        >
   509→          {/* Attachment preview bar */}

   514→                animate={{ height: "auto", opacity: 1 }}
   515→                exit={{ height: 0, opacity: 0 }}
   516→                transition={{ duration: 0.15, ease: "easeOut" }}
   517→                className="overflow-hidden rounded-2xl border backdrop-blur-sm"
   518→                style={{ backgroundColor: "var(--app-accent-subtle)", borderColor: "var(--app-accent-subtle)" }}
   519→              >
   520→                <div className="flex items-center gap-3 px-4 py-2.5">
   521→                  {attachmentPreview ? (
   522→                    <div className="relative size-12 shrink-0 overflow-hidden rounded-lg" style={{ boxShadow: `0 0 0 2px var(--app-accent-subtle)` }}>
   523→                      {/* eslint-disable-next-line @next/next/no-img-element */}
   524→                      <img src={attachmentPreview} alt={attachmentFile.name} className="size-full object-cover" />
   525→                    </div>
   526→                  ) : (
   527→                    <div className="flex size-12 shrink-0 items-center justify-center rounded-xl" style={{ backgroundColor: "var(--app-accent-subtle)", color: "var(--app-accent)" }}>
   528→                      {attachmentFile.type === "application/pdf" ? (
   529→                        <FileText className="size-5" />
   530→                      ) : attachmentFile.type.startsWith("video/") ? (
   531→                        <svg className="size-5" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round"><polygon points="23 7 16 12 23 17 23 7" /><rect x="1" y="5" width="15" height="14" rx="2" ry="2" /></svg>
   532→                      ) : (
   533→                        <FileText className="size-5" />
   534→                      )}
   535→                    </div>
   536→                  )}
   537→                  <div className="min-w-0 flex-1">
   538→                    <p className="truncate text-sm font-medium text-foreground">{attachmentFile.name}</p>
   539→                    <p className="text-xs text-muted-foreground">{formatFileSize(attachmentFile.size)}</p>
   540→                  </div>
   541→                  {isUploading ? (
   542→                    <Loader2 className="size-5 shrink-0 animate-spin" style={{ color: "var(--app-accent)" }} />
   543→                  ) : (
   544→                    <button
   545→                      type="button"
   546→                      onClick={clearAttachment}
   547→                      className="flex size-8 shrink-0 items-center justify-center rounded-full text-muted-foreground transition-colors hover:bg-muted hover:text-foreground active:scale-95"
   548→                      aria-label="Remove attachment"
   549→                    >
   550→                      <X className="size-4" />
   551→                    </button>
   552→                  )}
   553→                </div>
   554→              </m.div>
   555→            )}
   556→          </AnimatePresence>
   557→
   558→          {/* Reply preview bar */}
   559→          <AnimatePresence>
   560→            {replyTo && (
   561→              <m.div
   562→                initial={{ height: 0, opacity: 0 }}
   563→                animate={{ height: "auto", opacity: 1 }}
   564→                exit={{ height: 0, opacity: 0 }}
   565→                transition={{ duration: 0.15, ease: "easeOut" }}
   566→                className="overflow-hidden backdrop-blur-sm rounded-2xl"
   567→                style={{ backgroundColor: "var(--app-accent-subtle)", borderLeft: "2px solid var(--app-accent)" }}
   568→              >
   569→                <div className="flex items-center gap-2 px-4 py-2">
   570→                  <div className="w-1 h-8 rounded-full" style={{ background: "linear-gradient(to bottom, var(--app-accent-from), var(--app-accent-to))" }} />
   571→                  <div className="min-w-0 flex-1">
   572→                    <p className="text-[11px] font-semibold" style={{ color: "var(--app-accent)" }}>{replyTo.sender_name || "Unknown"}</p>
   573→                    <p className="truncate text-xs text-muted-foreground">{replyTo.content}</p>
   574→                  </div>
   575→                  <div className="flex items-center gap-1.5">
   576→                    <button
   577→                      type="button"
   578→                      onClick={() => setReplyTo(null)}
   579→                      className="flex size-6 items-center justify-center rounded-full text-muted-foreground hover:bg-muted active:scale-95"
   580→                    >
   581→                      <X className="size-3.5" />
   582→                    </button>
   583→                    <kbd className="hidden sm:inline-flex h-5 items-center rounded border bg-muted px-1.5 text-[10px] font-medium text-muted-foreground">Esc</kbd>
   584→                  </div>
   585→                </div>
   586→              </m.div>
   587→            )}
   588→          </AnimatePresence>
   589→        </div>
   590→      )}
   591→
   592→      {/* Floating glassmorphic input pill */}
   593→      <form onSubmit={handleSubmit}>
   594→        <div
   595→          className={cn(
   596→            "fixed bottom-4 left-4 right-4 max-w-2xl mx-auto z-30 flex items-center gap-1.5",
   597→            "rounded-full",
   598→            "bg-white/80 dark:bg-neutral-900/80",
   599→            "border border-black/5 dark:border-[var(--app-accent-subtle)]",
   600→            "backdrop-blur-md shadow-2xl",
   601→            "transition-all duration-200",
   602→            "focus-within:border-[var(--app-accent-light)]",
   603→            "focus-within:shadow-[0_0_20px_var(--app-accent-glow)]",
   604→            /* Use a high-specificity outline-none for when textarea inside is focused */
   605→          )}
   606→        >
   607→          {/* Paperclip button — left side of pill */}
   608→          {!showVoiceWaveform && (
   609→            <button
   610→              type="button"
   611→              onClick={handlePaperclipClick}
   612→              className={cn(
   613→                "flex size-9 shrink-0 items-center justify-center rounded-full transition-colors duration-200 active:scale-95",
   614→                hasAttachment ? "text-[var(--app-accent)]" : "text-muted-foreground",
   615→              )}
   616→              aria-label="Attach file"
   617→            >
   618→              <Paperclip className="size-4" />
   619→            </button>
   620→          )}
   621→
   622→          {/* Emoji button — left side of pill, next to paperclip */}
   623→          {!showVoiceWaveform && (
   624→            <EmojiPicker
   625→              onSelect={(emoji) => {
   626→                const ta = inputRef.current;
   627→                if (!ta) return;
   628→                const start = ta.selectionStart ?? draft.length;
   629→                const end = ta.selectionEnd ?? draft.length;
   630→                const before = draft.slice(0, start);
   631→                const after = draft.slice(end);
   632→                setDraft(before + emoji + after);
   633→                requestAnimationFrame(() => {
   634→                  const pos = start + emoji.length;
   635→                  ta.focus();
   636→                  ta.setSelectionRange(pos, pos);
   637→                });
   638→              }}
   639→            >
   640→              <button
   641→                type="button"
   642→                className="flex size-9 shrink-0 items-center justify-center rounded-full text-muted-foreground transition-colors duration-200 hover:text-foreground active:scale-95"
   643→                aria-label="Emoji"
   644→              >
   645→                <Smile className="size-4" />
   646→              </button>
   647→            </EmojiPicker>
   648→          )}
   649→
   650→          {/* Textarea — middle of pill (hidden when recording) */}
   651→          {!showVoiceWaveform && (
   652→            <textarea
   653→              ref={inputRef}
   654→              value={draft}
   655→              onChange={handleInputChange}
   656→              onKeyDown={handleKeyDown}
   657→              onInput={handleInput}
   658→              placeholder={replyTo ? `Reply to ${replyTo.sender_name || "message"}...` : "Type a message..."}
   659→              rows={1}
   660→              className={cn(
   661→                "flex-1 resize-none bg-transparent py-3 px-1 text-sm",
   662→                "placeholder:text-muted-foreground",
   663→                "focus-visible:outline-none",
   664→                "max-h-32 overflow-y-auto",
   665→                "scrollbar-none",
   666→              )}
   667→            />
   668→          )}
   669→
   670→          {/* Voice waveform overlay — replaces textarea inside the pill */}
   671→          {showVoiceWaveform && (
   672→            <div className="flex flex-1 items-center gap-3 px-3 py-2.5">
   673→              {/* Red recording dot */}
   674→              <div className="flex size-8 shrink-0 items-center justify-center rounded-full bg-red-500/15 text-red-500">
   675→                {voice.isRecording && (
   676→                  <span className="relative flex size-3">
   677→                    <span className="absolute inline-flex size-full animate-ping rounded-full bg-red-400 opacity-75" />
   678→                    <span className="relative inline-flex size-3 rounded-full bg-red-500" />
   679→                  </span>
   680→                )}
   681→              </div>
   682→
   683→              {/* Waveform bars */}
   684→              <div className="flex flex-1 items-end justify-center gap-px" style={{ height: 32 }}>
   685→                {voice.amplitudes.slice(-40).map((amp, i) => (
   686→                  <div
   687→                    key={i}
   688→                    className="w-[2px] rounded-full transition-all duration-75"
   689→                    style={{
   690→                      height: `${Math.max(3, amp * 32)}px`,
   691→                      backgroundColor: "var(--app-accent)",
   692→                      opacity: 0.7,
   693→                    }}
   694→                  />
   695→                ))}
   696→              </div>
   697→
   698→              {/* Timer */}
   699→              <span className="shrink-0 text-xs font-medium tabular-nums text-muted-foreground">
   700→                {Math.floor(voice.duration / 60)}:{String(voice.duration % 60).padStart(2, "0")}
   701→              </span>
   702→            </div>
   703→          )}
   704→
   705→          {/* Right side of pill: Send / Mic / Voice controls */}
   706→
   707→          {/* Cancel recording button */}
   708→          {showVoiceWaveform && (
   709→            <button
   710→              type="button"
   711→              onClick={handleVoiceCancel}
   712→              className="flex size-9 shrink-0 items-center justify-center rounded-full text-red-500 transition-colors hover:bg-red-500/10 active:scale-95"
   713→              aria-label="Cancel recording"
   714→            >
   715→              <MicOff className="size-4" />
   716→            </button>
   717→          )}
   718→
   719→          {/* Voice send button */}
   720→          {showVoiceWaveform && (
   721→            <button
   722→              type="button"
   723→              onClick={handleVoiceSend}
   724→              disabled={isSendingVoice}
   725→              className="flex size-9 shrink-0 items-center justify-center rounded-full text-white transition-opacity active:scale-95 disabled:opacity-50"
   726→              style={{ background: "linear-gradient(to right, var(--app-accent-from), var(--app-accent-to))" }}
   727→              aria-label="Send voice"
   728→            >
   729→              {isSendingVoice ? <Loader2 className="size-4 animate-spin" /> : <ArrowUp className="size-4" />}
   730→            </button>
   731→          )}
   732→
   733→          {/* Send button (shown when there's text or attachment) */}
   734→          {!showVoiceWaveform && canSend && (
   735→            <AnimatePresence>
   736→              <m.div
   737→                key={sendBtnKey}
   738→                initial={{ scale: 0, opacity: 0, rotate: -90 }}
   739→                animate={{ scale: 1, opacity: 1, rotate: 0 }}
   740→                exit={{ scale: 0, opacity: 0 }}
   741→                transition={{ duration: 0.15, ease: "easeOut" }}
   742→              >
   743→                <Button
   744→                  type="submit"
   745→                  size="icon"
   746→                  disabled={isSending || isUploading}
   747→                  className="size-9 shrink-0 rounded-full text-white transition-all duration-200 hover:opacity-80 hover:scale-105 active:scale-95"
   748→                  style={{
   749→                    background: "linear-gradient(to right, var(--app-accent-from), var(--app-accent-to))",
   750→                    boxShadow: `0 10px 15px -3px var(--app-accent-glow)`,
   751→                  }}
   752→                  aria-label="Send message"
   753→                >
   754→                  <ArrowUp className="size-4" />
   755→                </Button>
   756→              </m.div>
   757→            </AnimatePresence>
   758→          )}
   759→
   760→          {/* Mic button (shown when input is empty and no attachment) */}
   761→          {!showVoiceWaveform && !canSend && (
   762→            <AnimatePresence>
   763→              <m.div
   764→                initial={{ scale: 0, opacity: 0 }}
   765→                animate={{ scale: 1, opacity: 1 }}
   766→                exit={{ scale: 0, opacity: 0 }}
   767→                transition={{ duration: 0.15, ease: "easeOut" }}
   768→              >
   769→                <button
   770→                  type="button"
   771→                  onClick={() => voice.startRecording()}
   772→                  className="flex size-9 shrink-0 items-center justify-center rounded-full text-muted-foreground transition-colors duration-200 hover:bg-muted active:scale-95"
   773→                  aria-label="Record voice"
   774→                >
   775→                  <Mic className="size-4" />
   776→                </button>
   777→              </m.div>
   778→            </AnimatePresence>
   779→          )}
   780→        </div>
   781→      </form>
   782→
   783→      {/* Partner Info Panel */}
   784→      <PartnerInfoPanel
   785→        partner={partner}
   786→        partnerStatus={partnerStatus}
   787→        open={infoPanelOpen}
   788→        onClose={() => setInfoPanelOpen(false)}
   789→      />
   790→    </div>
   791→  );
