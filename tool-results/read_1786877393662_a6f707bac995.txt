    87→/* ------------------------------------------------------------------ */
    88→
    89→function EphemeralTimer({ seconds, onExpire }: { seconds: number; onExpire: () => void }) {
    90→  const [remaining, setRemaining] = useState(seconds);
    91→  const intervalRef = useRef<ReturnType<typeof setInterval> | undefined>(undefined);
    92→
    93→  useEffect(() => {
    94→    intervalRef.current = setInterval(() => {
    95→      setRemaining((r) => {
    96→        if (r <= 1) { clearInterval(intervalRef.current); onExpire(); return 0; }
    97→        return r - 1;
    98→      });
    99→    }, 1000);
   100→    return () => clearInterval(intervalRef.current);
   101→  }, [seconds, onExpire]);
   102→
   103→  const pct = remaining / seconds;
   104→  const circ = 2 * Math.PI * 7;
   105→  const offset = circ * (1 - pct);
   106→
   107→  return (
   108→    <svg className="size-4 shrink-0 ephemeral-timer-pulse" viewBox="0 0 18 18" aria-label={`${remaining}s remaining`}>
   109→      <circle cx="9" cy="9" r="7" fill="none" stroke="currentColor" strokeWidth="1.5" className="text-white/20" />
   110→      <circle cx="9" cy="9" r="7" fill="none" stroke="currentColor" strokeWidth="1.5" strokeDasharray={circ} strokeDashoffset={offset} strokeLinecap="round" className="text-violet-300" style={{ transform: "rotate(-90deg)", transformOrigin: "50% 50%", transition: "stroke-dashoffset 1s linear" }} />
   111→    </svg>
   112→  );
   113→}

   196→function VoiceBubble({ message, isOwn, waveform }: {
   197→  message: ChatMessage; isOwn: boolean; waveform: number[];
   198→}) {
   199→  const audioRef = useRef<HTMLAudioElement>(null);
   200→  const [playing, setPlaying] = useState(false);
   201→  const [progress, setProgress] = useState(0);
   202→  const [speed, setSpeed] = useState(1);
   203→  const speeds = [1, 1.5, 2];
   204→  const duration = message.voice_duration ?? 0;
   205→
   206→  useEffect(() => {
   207→    const audio = audioRef.current;
   208→    if (!audio) return;
   209→    const onTime = () => { if (audio.duration) setProgress(audio.currentTime / audio.duration); };
   210→    const onEnd = () => setPlaying(false);
   211→    audio.addEventListener("timeupdate", onTime);
   212→    audio.addEventListener("ended", onEnd);
   213→    return () => { audio.removeEventListener("timeupdate", onTime); audio.removeEventListener("ended", onEnd); };
   214→  }, []);
   215→
   216→  const toggle = useCallback(() => {
   217→    const audio = audioRef.current;
   218→    if (!audio) return;
   219→    if (playing) { audio.pause(); setPlaying(false); }
   220→    else { audio.playbackRate = speed; audio.play().catch(() => {}); setPlaying(true); }
   221→  }, [playing, speed]);
   222→
   223→  const cycleSpeed = useCallback(() => {
   224→    setSpeed((s) => {
   225→      const idx = speeds.indexOf(s);
   226→      const next = speeds[(idx + 1) % speeds.length];
   227→      if (audioRef.current) audioRef.current.playbackRate = next;
   228→      return next;
   229→    });
   230→  }, []);
   231→
   232→  /* Build waveform bars */
   233→  const bars = useMemo(() => {
   234→    const count = Math.min(waveform.length, 40);
   235→    const step = Math.max(1, Math.floor(waveform.length / count));
   236→    return Array.from({ length: count }, (_, i) => waveform[i * step] ?? 0.1);
   237→  }, [waveform]);
   238→
   239→  const barColor = isOwn ? "bg-white/50" : "bg-zinc-400/70";
   240→  const progressColor = isOwn ? "bg-white" : "bg-violet-400";
   241→  const progressIdx = Math.floor(progress * bars.length);
   242→
   243→  return (
   244→    <div className={cn("flex items-center gap-3", isOwn ? "flex-row-reverse" : "")}>
   245→      <button type="button" onClick={toggle} className={cn("flex size-9 shrink-0 items-center justify-center rounded-full transition-all duration-150", isOwn ? "bg-white/20 hover:bg-white/30 hover:scale-105" : "bg-zinc-600 hover:bg-zinc-500 hover:scale-105")} aria-label={playing ? "Pause" : "Play"}>
   246→        {playing ? <Pause className="size-3.5 text-white" /> : <Play className="size-3.5 text-white ml-0.5" />}
   247→      </button>
   248→      <div className="flex items-center gap-[2px]" style={{ height: 32 }}>
   249→        {bars.map((amp, i) => (
   250→          <div key={i} className={cn("w-[2.5px] rounded-full transition-all duration-100", i < progressIdx ? progressColor : barColor)} style={{ height: `${Math.max(4, amp * 32)}px` }} />
   251→        ))}
   252→      </div>
   253→      <button type="button" onClick={cycleSpeed} className={cn("shrink-0 text-[11px] font-semibold tabular-nums transition-colors hover:opacity-80", isOwn ? "text-white/70" : "text-zinc-400")}>{speed}x</button>
   254→      <span className={cn("shrink-0 text-[11px] tabular-nums", isOwn ? "text-white/50" : "text-zinc-500")}>{Math.floor(duration / 60)}:{String(duration % 60).padStart(2, "0")}</span>
   255→      <audio ref={audioRef} src={message.content} preload="metadata" />
   256→    </div>
   257→  );
   258→}

   260→/* ------------------------------------------------------------------ */
   261→/*  LinkifiedText – detects & renders clickable URLs                   */
   262→/* ------------------------------------------------------------------ */
   263→
   264→const URL_REGEX = /(https?:\/\/[^\s<]+)/g;
   265→
   266→type TextSegment = { type: "text"; value: string } | { type: "link"; value: string };
   267→
   268→function parseUrls(text: string): TextSegment[] {
   269→  const segments: TextSegment[] = [];
   270→  let lastIndex = 0;
   271→  let match: RegExpExecArray | null;
   272→  while ((match = URL_REGEX.exec(text)) !== null) {
   273→    if (match.index > lastIndex) {
   274→      segments.push({ type: "text", value: text.slice(lastIndex, match.index) });
   275→    }
   276→    segments.push({ type: "link", value: match[1] });
   277→    lastIndex = URL_REGEX.lastIndex;
   278→  }
   279→  if (lastIndex < text.length) {
   280→    segments.push({ type: "text", value: text.slice(lastIndex) });
   281→  }
   282→  return segments;
   283→}
   284→
   285→const LinkifiedText = React.memo(function LinkifiedText({ text, className }: { text: string; className?: string }) {
   286→  const segments = useMemo(() => parseUrls(text), [text]);
   287→  return (
   288→    <span className={className}>
   289→      {segments.map((seg, i) =>
   290→        seg.type === "link" ? (
   291→          <a
   292→            key={i}
   293→            href={seg.value}
   294→            target="_blank"
   295→            rel="noopener noreferrer"
   296→            className="underline decoration-violet-400/50 underline-offset-2 hover:decoration-violet-400 transition-colors"
   297→            onClick={(e) => e.stopPropagation()}
   298→          >
   299→            {seg.value}
   300→          </a>
   301→        ) : (
   302→          <span key={i}>{seg.value}</span>
   303→        ),
   304→      )}
   305→    </span>
   306→  );
   307→});

   407→function MessageBubble({ message, isOwn, partnerName, partnerAvatar, onReplyTo, onReact, onDeleteMessage }: {
   408→  message: ChatMessage; isOwn: boolean; partnerName: string; partnerAvatar: string | null;
   409→  onReplyTo?: (msg: ChatMessage) => void; onReact?: (id: string, emoji: string, add: boolean) => void; onDeleteMessage?: (id: string) => void;
   410→}) {
   411→  const [showHeart, setShowHeart] = useState(false);
   412→  const [showTapback, setShowTapback] = useState(false);
   413→  const [showLightbox, setShowLightbox] = useState<string | null>(null);
   414→  const lastTapRef = useRef(0);
   415→  const longPressRef = useRef<ReturnType<typeof setTimeout> | undefined>(undefined);
   416→  const bubbleRef = useRef<HTMLDivElement>(null);
   417→
   418→  const triggerHeart = useCallback(() => {
   419→    setShowHeart(true); setTimeout(() => setShowHeart(false), 800);
   420→    if (typeof navigator !== "undefined" && "vibrate" in navigator) navigator.vibrate(10);
   421→  }, []);
   422→
   423→  const handleClick = useCallback(() => {
   424→    clearTimeout(longPressRef.current);
   425→    const now = Date.now();
   426→    if (now - lastTapRef.current < DOUBLE_TAP_MS) triggerHeart();
   427→    lastTapRef.current = now;
   428→  }, [triggerHeart]);
   429→
   430→  const handlePressStart = useCallback(() => {
   431→    longPressRef.current = setTimeout(() => {
   432→      if (typeof navigator !== "undefined" && "vibrate" in navigator) navigator.vibrate(20);
   433→      setShowTapback(true);
   434→    }, LONG_PRESS_MS);
   435→  }, []);
   436→
   437→  const handlePressEnd = useCallback(() => { clearTimeout(longPressRef.current); }, []);
   438→
   439→  const handleContextMenu = useCallback((e: React.MouseEvent) => {
   440→    e.preventDefault(); setShowTapback(true);
   441→  }, []);
   442→
   443→  const handleCopy = useCallback(() => {
   444→    navigator.clipboard?.writeText(message.content).then(() => {
   445→      toast({ title: "Copied to clipboard", description: message.content.length > 50 ? message.content.slice(0, 50) + "..." : message.content });
   446→    }).catch(() => {
   447→      toast({ title: "Failed to copy", description: "Could not access clipboard", variant: "destructive" });
   448→    });
   449→    setShowTapback(false);
   450→  }, [message.content]);
   451→
   452→  const handleDelete = useCallback(() => {
   453→    onDeleteMessage?.(message.id);
   454→    setShowTapback(false);
   455→  }, [message.id, onDeleteMessage]);
   456→
   457→  const handleSwipeEnd = useCallback((_: MouseEvent | TouchEvent | PointerEvent, info: PanInfo) => {
   458→    if (info.offset.x > SWIPE_REPLY_THRESHOLD && info.velocity.x > 200) {
   459→      if (typeof navigator !== "undefined" && "vibrate" in navigator) navigator.vibrate(10);
   460→      onReplyTo?.(message);
   461→    }
   462→  }, [message, onReplyTo]);
   463→
   464→  const isFailed = message.status === "failed";
   465→  const isVoice = message.type === "voice";
   466→  const isNew = message.status === "sending" || Date.now() - new Date(message.created_at).getTime() < 2000;
   467→
   468→  return (
   469→    <>
   470→    <m.div
   471→      className={cn("relative flex w-full gap-2.5 px-4", isOwn ? "justify-end" : "justify-start")}
   472→      initial={isNew ? { opacity: 0, y: 16, scale: 0.97 } : false}
   473→      animate={{ opacity: 1, y: 0, scale: 1 }}
   474→      transition={{ duration: 0.3, ease: [0.22, 1, 0.36, 1] }}
   475→    >
   476→      <HeartBurst show={showHeart} />
   477→      <AnimatePresence>{showTapback && onReact && (
   478→        <TapbackDock message={message} isOwn={isOwn} onReact={(emoji, add) => { onReact(message.id, emoji, add); setShowTapback(false); }} onReply={() => { onReplyTo?.(message); setShowTapback(false); }} onCopy={handleCopy} onDelete={isOwn ? handleDelete : undefined} />
   479→      )}</AnimatePresence>
   480→
   481→      {!isOwn && (
   482→        <Avatar className="mt-auto size-8 shrink-0 ring-1 ring-muted">
   483→          {partnerAvatar && <AvatarImage src={partnerAvatar} alt={partnerName} />}
   484→          <AvatarFallback className="bg-gradient-to-br from-violet-100 to-purple-200 text-xs font-medium text-violet-700 dark:from-violet-500 dark:to-purple-600 dark:text-white">{partnerName.slice(0, 2).toUpperCase()}</AvatarFallback>
   485→        </Avatar>
   486→      )}
   487→
   488→      <div className={cn("flex max-w-[75%] flex-col gap-1", isOwn ? "items-end" : "items-start")}>
   489→        <m.div
   490→          ref={bubbleRef}
   491→          drag="x"
   492→          dragConstraints={{ left: 0, right: 0 }}
   493→          dragElastic={{ left: 0, right: 0.35 }}
   494→          onDragEnd={handleSwipeEnd}
   495→          onClick={handleClick}
   496→          onPointerDown={handlePressStart}
   497→          onPointerUp={handlePressEnd}
   498→          onPointerLeave={handlePressEnd}
   499→          onContextMenu={handleContextMenu}
   500→          className={cn(
   501→            "relative cursor-default select-none rounded-2xl px-4 py-2.5 text-sm leading-relaxed transition-all duration-200 active:scale-[0.98]",
   502→            // Own messages: vibrant violet-purple gradient with subtle shadow
   503→            isOwn && "bg-gradient-to-r from-violet-500 to-purple-600 text-white rounded-br-md shadow-md shadow-violet-500/15",
   504→            // Partner messages: softer background with shadow
   505→            !isOwn && "bg-muted text-foreground rounded-bl-md shadow-sm dark:bg-zinc-800 dark:text-zinc-100 dark:shadow-violet-950/20",
   506→            isFailed && "ring-2 ring-red-400/50",
   507→            isVoice && "py-2",
   508→            !isVoice && message.type !== "image" && "md:hover:shadow-md md:hover:-translate-y-0.5"
   509→          )}
   510→          style={{ transform: "translate3d(0,0,0)" }}
   511→        >
   512→          {/* Reply quote */}
   513→          {message.reply_to_id && message.reply_to_content && (
   514→            <ReplyPreview senderName={message.reply_to_sender_name ?? ""} content={message.reply_to_content} isOwn={isOwn} />
   515→          )}
   516→
   517→          {/* Content */}
   518→          {isVoice ? (
   519→            message.waveform_data ? (
   520→              <VoiceBubble message={message} isOwn={isOwn} waveform={message.waveform_data} />
   521→            ) : <p className="text-white/60 italic">Voice message</p>
   522→          ) : message.type === "image" ? (
   523→            <div className="group/img relative cursor-zoom-in" onClick={() => setShowLightbox(message.content)}>
   524→              <img src={message.content} alt="Shared image" className="max-h-64 rounded-lg object-contain transition-opacity duration-150 group-hover/img:opacity-90" />
   525→              <div className="absolute inset-0 flex items-center justify-center rounded-lg bg-black/0 transition-colors duration-150 group-hover/img:bg-black/20">
   526→                <ZoomIn className="size-6 text-white opacity-0 transition-opacity duration-150 group-hover/img:opacity-80" />
   527→              </div>
   528→            </div>
   529→          ) : (
   530→            <LinkifiedText text={message.content} className="whitespace-pre-wrap break-words" />
   531→          )}
   532→
   533→          {/* Vanish indicator */}
   534→          {message.vanish_mode && (
   535→            <div className="mt-1 flex items-center gap-1 text-[10px] text-white/50"><EyeOff className="size-3" /><span>View once</span></div>
   536→          )}
   537→        </m.div>
   538→
   539→        {/* Timestamp + receipt + ephemeral timer */}
   540→        <div className={cn("flex items-center gap-1.5 px-1 text-[11px] text-muted-foreground", isOwn && "flex-row-reverse")}>
   541→          {message.ephemeral_seconds && message.ephemeral_seconds > 0 && (
   542→            <EphemeralTimer seconds={message.ephemeral_seconds} onExpire={() => { /* trigger vanish */ }} />
   543→          )}
   544→          <span className={cn(isNew && "animate-[timestamp-fade_0.4s_ease_0.3s_both]")}>{formatDistanceToNow(new Date(message.created_at), { addSuffix: true })}</span>
   545→          {isOwn && <ReceiptIcon status={message.status} />}
   546→        </div>
   547→
   548→        {/* Reactions */}
   549→        {onReact && <ReactionPills reactions={message.reactions} isOwn={isOwn} onReact={(e, a) => onReact(message.id, e, a)} currentUserId={""} />}
   550→
   551→        {isFailed && (
   552→          <button
   553→            type="button"
   554→            onClick={(e) => { e.stopPropagation(); onDeleteMessage?.(message.id); }}
   555→            className="mt-1 flex items-center gap-1 px-1 text-[10px] text-red-400 transition-colors hover:text-red-300"
   556→          >
   557→            <AlertCircle className="size-3" />
   558→            <span>Failed to send &middot; Tap to retry</span>
   559→          </button>
   560→        )}
   561→      </div>
   562→
   563→      {isOwn && <div className="w-8 shrink-0" />}
   564→    </m.div>
   565→
   566→    {/* Image Lightbox */}
   567→    <AnimatePresence>
   568→      {showLightbox && (
   569→        <ImageLightbox src={showLightbox} onClose={() => setShowLightbox(null)} />
   570→      )}
   571→    </AnimatePresence>
   572→    </>
   573→  );
   574→}

   580→function formatDateSeparator(dateStr: string): string {
   581→  const d = new Date(dateStr);
   582→  const now = new Date();
   583→  const today = new Date(now.getFullYear(), now.getMonth(), now.getDate());
   584→  const yesterday = new Date(today);
   585→  yesterday.setDate(yesterday.getDate() - 1);
   586→  const msgDay = new Date(d.getFullYear(), d.getMonth(), d.getDate());
   587→
   588→  if (msgDay.getTime() === today.getTime()) return "Today";
   589→  if (msgDay.getTime() === yesterday.getTime()) return "Yesterday";
   590→  return d.toLocaleDateString(undefined, { month: "short", day: "numeric" });
   591→}

   702→export function MessageFeed({ messages, isLoading, isPartnerTyping, currentUserId, partnerName, partnerAvatar, backgroundStyle, onMarkAsRead, onVanishMessage, onReplyTo, onDeleteMessage, onReact, scrollToMessageId, onScrolledToMessage }: MessageFeedProps) {
   703→  const parentRef = useRef<HTMLDivElement>(null);
   704→  const isAutoScrollRef = useRef(true);
   705→  const [showNewBtn, setShowNewBtn] = useState(false);
   706→  const unreadRef = useRef(0);
   707→  const prevCountRef = useRef(messages.length);
   708→
   709→  const totalItems = useMemo(() => messages.length + (isPartnerTyping ? 1 : 0), [messages.length, isPartnerTyping]);
   710→
   711→  const virtualizer = useVirtualizer({ count: totalItems, getScrollElement: () => parentRef.current, estimateSize: () => ESTIMATED_ROW_HEIGHT, overscan: OVERSCAN, paddingStart: 16, paddingEnd: 16, gap: 12 });
   712→
   713→  const scrollToBottom = useCallback((behavior: ScrollBehavior = "smooth") => {
   714→    const el = parentRef.current; if (!el) return;
   715→    el.scrollTo({ top: el.scrollHeight, behavior });
   716→    isAutoScrollRef.current = true; setShowNewBtn(false); unreadRef.current = 0;
   717→  }, []);
   718→
   719→  const handleScroll = useCallback((e: UIEvent<HTMLDivElement>) => {
   720→    const el = e.currentTarget;
   721→    const dist = el.scrollHeight - el.scrollTop - el.clientHeight;
   722→    if (dist <= AUTO_SCROLL_THRESHOLD) { isAutoScrollRef.current = true; setShowNewBtn(false); unreadRef.current = 0; }
   723→    else { isAutoScrollRef.current = false; }
   724→  }, []);
   725→
   726→  useEffect(() => { if (messages.length > 0) requestAnimationFrame(() => scrollToBottom("instant")); }, [isLoading, scrollToBottom]);
   727→
   728→  useEffect(() => { if (!onMarkAsRead) return; const unread = messages.filter((m) => m.sender_id !== currentUserId && m.status !== "read"); if (unread.length > 0 && isAutoScrollRef.current) onMarkAsRead(unread.map((m) => m.id)); }, [messages, currentUserId, onMarkAsRead]);
   729→
   730→  /* ---- Scroll to a specific message ----------------------------------- */
   731→  useEffect(() => {
   732→    if (!scrollToMessageId) return;
   733→    const idx = messages.findIndex((m) => m.id === scrollToMessageId);
   734→    if (idx === -1) { onScrolledToMessage?.(); return; }
   735→    virtualizer.scrollToIndex(idx, { align: "center" });
   736→    onScrolledToMessage?.();
   737→  }, [scrollToMessageId, messages, virtualizer, onScrolledToMessage]);
   738→
   739→  useEffect(() => {
   740→    const prev = prevCountRef.current; const next = messages.length; prevCountRef.current = next;
   741→    if (next <= prev) return;
   742→    const last = messages[next - 1]; const own = last && last.sender_id === currentUserId;
   743→    if (own) { requestAnimationFrame(() => scrollToBottom("instant")); return; }
   744→    if (isAutoScrollRef.current) requestAnimationFrame(() => scrollToBottom("smooth"));
   745→    else { unreadRef.current += next - prev; setShowNewBtn(true); }
   746→  }, [messages.length, currentUserId, scrollToBottom]);
   747→
   748→  if (isLoading) return <MessageListSkeleton />;
   749→  if (messages.length === 0 && !isPartnerTyping) return <EmptyState partnerName={partnerName} partnerAvatar={partnerAvatar} />;
   750→
   751→  const virtualItems = virtualizer.getVirtualItems();
   752→
   753→  return (
   754→    <div className="relative h-full w-full overflow-hidden" style={backgroundStyle}>
   755→      <div ref={parentRef} onScroll={handleScroll} className="h-full w-full overflow-y-auto overscroll-contain" style={{ WebkitOverflowScrolling: "touch" }}>
   756→        <div style={{ height: `${virtualizer.getTotalSize()}px`, width: "100%", position: "relative" }}>
   757→          {virtualItems.map((vi) => {
   758→            const isTyping = vi.index === messages.length;
   759→            const msg = isTyping ? null : messages[vi.index];
   760→            // Check if we need a date separator before this message
   761→            const showDateSep = msg && vi.index > 0 && messages[vi.index - 1] && (() => {
   762→              const prev = new Date(messages[vi.index - 1].created_at);
   763→              const curr = new Date(msg.created_at);
   764→              return prev.toDateString() !== curr.toDateString();
   765→            })();
   766→            const showFirstDateSep = msg && vi.index === 0;
   767→            return (
   768→              <div key={vi.key} data-index={vi.index} ref={virtualizer.measureElement} style={{ position: "absolute", top: 0, left: 0, width: "100%", transform: `translateY(${vi.start}px)` }}>
   769→                {isTyping ? <TypingIndicator partnerName={partnerName} partnerAvatar={partnerAvatar} /> : msg ? (
   770→                  <>
   771→                    {(showDateSep || showFirstDateSep) && <DateSeparator date={msg.created_at} />}
   772→                    <MessageBubble message={msg} isOwn={msg.sender_id === currentUserId} partnerName={partnerName} partnerAvatar={partnerAvatar} onReplyTo={onReplyTo} onReact={onReact} onDeleteMessage={onDeleteMessage} />
   773→                  </>
   774→                ) : null}
   775→              </div>
   776→            );
   777→          })}
   778→        </div>
   779→      </div>
   780→      {showNewBtn && unreadRef.current > 0 && <NewMessagesButton count={unreadRef.current} onClick={() => scrollToBottom("smooth")} />}
   781→    </div>
   782→  );
   783→}
