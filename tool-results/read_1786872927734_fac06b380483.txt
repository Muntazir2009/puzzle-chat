    84→/*  Ephemeral Timer (circular SVG countdown)                           */
    85→/* ------------------------------------------------------------------ */
    86→
    87→function EphemeralTimer({ seconds, onExpire }: { seconds: number; onExpire: () => void }) {
    88→  const [remaining, setRemaining] = useState(seconds);
    89→  const intervalRef = useRef<ReturnType<typeof setInterval> | undefined>(undefined);
    90→
    91→  useEffect(() => {
    92→    intervalRef.current = setInterval(() => {
    93→      setRemaining((r) => {
    94→        if (r <= 1) { clearInterval(intervalRef.current); onExpire(); return 0; }
    95→        return r - 1;
    96→      });
    97→    }, 1000);
    98→    return () => clearInterval(intervalRef.current);
    99→  }, [seconds, onExpire]);
   100→
   101→  const pct = remaining / seconds;
   102→  const circ = 2 * Math.PI * 7;
   103→  const offset = circ * (1 - pct);
   104→
   105→  return (
   106→    <svg className="size-4 shrink-0 ephemeral-timer-pulse" viewBox="0 0 18 18" aria-label={`${remaining}s remaining`}>
   107→      <circle cx="9" cy="9" r="7" fill="none" stroke="currentColor" strokeWidth="1.5" className="text-white/20" />
   108→      <circle cx="9" cy="9" r="7" fill="none" stroke="currentColor" strokeWidth="1.5" strokeDasharray={circ} strokeDashoffset={offset} strokeLinecap="round" className="text-violet-300" style={{ transform: "rotate(-90deg)", transformOrigin: "50% 50%", transition: "stroke-dashoffset 1s linear" }} />
   109→    </svg>
   110→  );
   111→}

   194→function VoiceBubble({ message, isOwn, waveform }: {
   195→  message: ChatMessage; isOwn: boolean; waveform: number[];
   196→}) {
   197→  const audioRef = useRef<HTMLAudioElement>(null);
   198→  const [playing, setPlaying] = useState(false);
   199→  const [progress, setProgress] = useState(0);
   200→  const [speed, setSpeed] = useState(1);
   201→  const speeds = [1, 1.5, 2];
   202→  const duration = message.voice_duration ?? 0;
   203→
   204→  useEffect(() => {
   205→    const audio = audioRef.current;
   206→    if (!audio) return;
   207→    const onTime = () => { if (audio.duration) setProgress(audio.currentTime / audio.duration); };
   208→    const onEnd = () => setPlaying(false);
   209→    audio.addEventListener("timeupdate", onTime);
   210→    audio.addEventListener("ended", onEnd);
   211→    return () => { audio.removeEventListener("timeupdate", onTime); audio.removeEventListener("ended", onEnd); };
   212→  }, []);
   213→
   214→  const toggle = useCallback(() => {
   215→    const audio = audioRef.current;
   216→    if (!audio) return;
   217→    if (playing) { audio.pause(); setPlaying(false); }
   218→    else { audio.playbackRate = speed; audio.play().catch(() => {}); setPlaying(true); }
   219→  }, [playing, speed]);
   220→
   221→  const cycleSpeed = useCallback(() => {
   222→    setSpeed((s) => {
   223→      const idx = speeds.indexOf(s);
   224→      const next = speeds[(idx + 1) % speeds.length];
   225→      if (audioRef.current) audioRef.current.playbackRate = next;
   226→      return next;
   227→    });
   228→  }, []);
   229→
   230→  /* Build waveform bars */
   231→  const bars = useMemo(() => {
   232→    const count = Math.min(waveform.length, 40);
   233→    const step = Math.max(1, Math.floor(waveform.length / count));
   234→    return Array.from({ length: count }, (_, i) => waveform[i * step] ?? 0.1);
   235→  }, [waveform]);
   236→
   237→  const barColor = isOwn ? "bg-white/50" : "bg-zinc-400/70";
   238→  const progressColor = isOwn ? "bg-white" : "bg-violet-400";
   239→  const progressIdx = Math.floor(progress * bars.length);
   240→
   241→  return (
   242→    <div className={cn("flex items-center gap-3", isOwn ? "flex-row-reverse" : "")}>
   243→      <button type="button" onClick={toggle} className={cn("flex size-9 shrink-0 items-center justify-center rounded-full transition-all duration-150", isOwn ? "bg-white/20 hover:bg-white/30 hover:scale-105" : "bg-zinc-600 hover:bg-zinc-500 hover:scale-105")} aria-label={playing ? "Pause" : "Play"}>
   244→        {playing ? <Pause className="size-3.5 text-white" /> : <Play className="size-3.5 text-white ml-0.5" />}
   245→      </button>
   246→      <div className="flex items-center gap-[2px]" style={{ height: 32 }}>
   247→        {bars.map((amp, i) => (
   248→          <div key={i} className={cn("w-[2.5px] rounded-full transition-all duration-100", i < progressIdx ? progressColor : barColor)} style={{ height: `${Math.max(4, amp * 32)}px` }} />
   249→        ))}
   250→      </div>
   251→      <button type="button" onClick={cycleSpeed} className={cn("shrink-0 text-[11px] font-semibold tabular-nums transition-colors hover:opacity-80", isOwn ? "text-white/70" : "text-zinc-400")}>{speed}x</button>
   252→      <span className={cn("shrink-0 text-[11px] tabular-nums", isOwn ? "text-white/50" : "text-zinc-500")}>{Math.floor(duration / 60)}:{String(duration % 60).padStart(2, "0")}</span>
   253→      <audio ref={audioRef} src={message.content} preload="metadata" />
   254→    </div>
   255→  );
   256→}

   309→
   310→function MessageBubble({ message, isOwn, partnerName, partnerAvatar, onReplyTo, onReact, onDeleteMessage }: {
   311→  message: ChatMessage; isOwn: boolean; partnerName: string; partnerAvatar: string | null;
   312→  onReplyTo?: (msg: ChatMessage) => void; onReact?: (id: string, emoji: string, add: boolean) => void; onDeleteMessage?: (id: string) => void;
   313→}) {
   314→  const [showHeart, setShowHeart] = useState(false);
   315→  const [showTapback, setShowTapback] = useState(false);
   316→  const lastTapRef = useRef(0);
   317→  const longPressRef = useRef<ReturnType<typeof setTimeout> | undefined>(undefined);
   318→  const bubbleRef = useRef<HTMLDivElement>(null);
   319→
   320→  const triggerHeart = useCallback(() => {
   321→    setShowHeart(true); setTimeout(() => setShowHeart(false), 800);
   322→    if (typeof navigator !== "undefined" && "vibrate" in navigator) navigator.vibrate(10);
   323→  }, []);
   324→
   325→  const handleClick = useCallback(() => {
   326→    clearTimeout(longPressRef.current);
   327→    const now = Date.now();
   328→    if (now - lastTapRef.current < DOUBLE_TAP_MS) triggerHeart();
   329→    lastTapRef.current = now;
   330→  }, [triggerHeart]);
   331→
   332→  const handlePressStart = useCallback(() => {
   333→    longPressRef.current = setTimeout(() => {
   334→      if (typeof navigator !== "undefined" && "vibrate" in navigator) navigator.vibrate(20);
   335→      setShowTapback(true);
   336→    }, LONG_PRESS_MS);
   337→  }, []);
   338→
   339→  const handlePressEnd = useCallback(() => { clearTimeout(longPressRef.current); }, []);
   340→
   341→  const handleContextMenu = useCallback((e: React.MouseEvent) => {
   342→    e.preventDefault(); setShowTapback(true);
   343→  }, []);
   344→
   345→  const handleCopy = useCallback(() => {
   346→    navigator.clipboard?.writeText(message.content).catch(() => {});
   347→    setShowTapback(false);
   348→  }, [message.content]);
   349→
   350→  const handleDelete = useCallback(() => {
   351→    onDeleteMessage?.(message.id);
   352→    setShowTapback(false);
   353→  }, [message.id, onDeleteMessage]);
   354→
   355→  const handleSwipeEnd = useCallback((_: MouseEvent | TouchEvent | PointerEvent, info: PanInfo) => {
   356→    if (info.offset.x > SWIPE_REPLY_THRESHOLD && info.velocity.x > 200) {
   357→      if (typeof navigator !== "undefined" && "vibrate" in navigator) navigator.vibrate(10);
   358→      onReplyTo?.(message);
   359→    }
   360→  }, [message, onReplyTo]);
   361→
   362→  const isFailed = message.status === "failed";
   363→  const isVoice = message.type === "voice";
   364→  const isNew = message.status === "sending" || Date.now() - new Date(message.created_at).getTime() < 2000;
   365→
   366→  return (
   367→    <m.div
   368→      className={cn("relative flex w-full gap-2.5 px-4", isOwn ? "justify-end" : "justify-start")}
   369→      initial={isNew ? { opacity: 0, y: 16, scale: 0.97 } : false}
   370→      animate={{ opacity: 1, y: 0, scale: 1 }}
   371→      transition={{ duration: 0.3, ease: [0.22, 1, 0.36, 1] }}
   372→    >
   373→      <HeartBurst show={showHeart} />
   374→      <AnimatePresence>{showTapback && onReact && (
   375→        <TapbackDock message={message} isOwn={isOwn} onReact={(emoji, add) => { onReact(message.id, emoji, add); setShowTapback(false); }} onReply={() => { onReplyTo?.(message); setShowTapback(false); }} onCopy={handleCopy} onDelete={isOwn ? handleDelete : undefined} />
   376→      )}</AnimatePresence>
   377→
   378→      {!isOwn && (
   379→        <Avatar className="mt-auto size-8 shrink-0 ring-1 ring-muted">
   380→          {partnerAvatar && <AvatarImage src={partnerAvatar} alt={partnerName} />}
   381→          <AvatarFallback className="bg-gradient-to-br from-violet-100 to-purple-200 text-xs font-medium text-violet-700 dark:from-violet-500 dark:to-purple-600 dark:text-white">{partnerName.slice(0, 2).toUpperCase()}</AvatarFallback>
   382→        </Avatar>
   383→      )}
   384→
   385→      <div className={cn("flex max-w-[75%] flex-col gap-1", isOwn ? "items-end" : "items-start")}>
   386→        <m.div
   387→          ref={bubbleRef}
   388→          drag="x"
   389→          dragConstraints={{ left: 0, right: 0 }}
   390→          dragElastic={{ left: 0, right: 0.35 }}
   391→          onDragEnd={handleSwipeEnd}
   392→          onClick={handleClick}
   393→          onPointerDown={handlePressStart}
   394→          onPointerUp={handlePressEnd}
   395→          onPointerLeave={handlePressEnd}
   396→          onContextMenu={handleContextMenu}
   397→          className={cn(
   398→            "relative cursor-default select-none rounded-2xl px-4 py-2.5 text-sm leading-relaxed transition-transform duration-100 active:scale-[0.98]",
   399→            // Own messages: vibrant violet-purple gradient with subtle shadow
   400→            isOwn && "bg-gradient-to-r from-violet-500 to-purple-600 text-white rounded-br-md shadow-md shadow-violet-500/15",
   401→            // Partner messages: softer background with shadow
   402→            !isOwn && "bg-muted text-foreground rounded-bl-md shadow-sm dark:bg-zinc-800 dark:text-zinc-100 dark:shadow-violet-950/20",
   403→            isFailed && "ring-2 ring-red-400/50",
   404→            isVoice && "py-2"
   405→          )}
   406→          style={{ transform: "translate3d(0,0,0)" }}
   407→        >
   408→          {/* Reply quote */}
   409→          {message.reply_to_id && message.reply_to_content && (
   410→            <ReplyPreview senderName={message.reply_to_sender_name ?? ""} content={message.reply_to_content} isOwn={isOwn} />
   411→          )}
   412→
   413→          {/* Content */}
   414→          {isVoice ? (
   415→            message.waveform_data ? (
   416→              <VoiceBubble message={message} isOwn={isOwn} waveform={message.waveform_data} />
   417→            ) : <p className="text-white/60 italic">Voice message</p>
   418→          ) : message.type === "image" ? (
   419→            <img src={message.content} alt="Shared image" className="max-h-64 rounded-lg object-contain" />
   420→          ) : (
   421→            <p className="whitespace-pre-wrap break-words">{message.content}</p>
   422→          )}
   423→
   424→          {/* Vanish indicator */}
   425→          {message.vanish_mode && (
   426→            <div className="mt-1 flex items-center gap-1 text-[10px] text-white/50"><EyeOff className="size-3" /><span>View once</span></div>
   427→          )}
   428→        </m.div>
   429→
   430→        {/* Timestamp + receipt + ephemeral timer */}
   431→        <div className={cn("flex items-center gap-1.5 px-1 text-[11px] text-muted-foreground", isOwn && "flex-row-reverse")}>
   432→          {message.ephemeral_seconds && message.ephemeral_seconds > 0 && (
   433→            <EphemeralTimer seconds={message.ephemeral_seconds} onExpire={() => { /* trigger vanish */ }} />
   434→          )}
   435→          <span>{formatDistanceToNow(new Date(message.created_at), { addSuffix: true })}</span>
   436→          {isOwn && <ReceiptIcon status={message.status} />}
   437→        </div>
   438→
   439→        {/* Reactions */}
   440→        {onReact && <ReactionPills reactions={message.reactions} isOwn={isOwn} onReact={(e, a) => onReact(message.id, e, a)} currentUserId={""} />}
   441→
   442→        {isFailed && <span className="px-1 text-[10px] text-red-500">Failed to send. Retry.</span>}
   443→      </div>
   444→
   445→      {isOwn && <div className="w-8 shrink-0" />}
   446→    </m.div>
   447→  );
   448→}

   454→function formatDateSeparator(dateStr: string): string {
   455→  const d = new Date(dateStr);
   456→  const now = new Date();
   457→  const today = new Date(now.getFullYear(), now.getMonth(), now.getDate());
   458→  const yesterday = new Date(today);
   459→  yesterday.setDate(yesterday.getDate() - 1);
   460→  const msgDay = new Date(d.getFullYear(), d.getMonth(), d.getDate());
   461→
   462→  if (msgDay.getTime() === today.getTime()) return "Today";
   463→  if (msgDay.getTime() === yesterday.getTime()) return "Yesterday";
   464→  return d.toLocaleDateString(undefined, { month: "short", day: "numeric" });
   465→}

   563→export function MessageFeed({ messages, isLoading, isPartnerTyping, currentUserId, partnerName, partnerAvatar, onMarkAsRead, onVanishMessage, onReplyTo, onDeleteMessage, onReact, scrollToMessageId, onScrolledToMessage }: MessageFeedProps) {
   564→  const parentRef = useRef<HTMLDivElement>(null);
   565→  const isAutoScrollRef = useRef(true);
   566→  const [showNewBtn, setShowNewBtn] = useState(false);
   567→  const unreadRef = useRef(0);
   568→  const prevCountRef = useRef(messages.length);
   569→
   570→  const totalItems = useMemo(() => messages.length + (isPartnerTyping ? 1 : 0), [messages.length, isPartnerTyping]);
   571→
   572→  const virtualizer = useVirtualizer({ count: totalItems, getScrollElement: () => parentRef.current, estimateSize: () => ESTIMATED_ROW_HEIGHT, overscan: OVERSCAN, paddingStart: 16, paddingEnd: 16, gap: 12 });
   573→
   574→  const scrollToBottom = useCallback((behavior: ScrollBehavior = "smooth") => {
   575→    const el = parentRef.current; if (!el) return;
   576→    el.scrollTo({ top: el.scrollHeight, behavior });
   577→    isAutoScrollRef.current = true; setShowNewBtn(false); unreadRef.current = 0;
   578→  }, []);
   579→
   580→  const handleScroll = useCallback((e: UIEvent<HTMLDivElement>) => {
   581→    const el = e.currentTarget;
   582→    const dist = el.scrollHeight - el.scrollTop - el.clientHeight;
   583→    if (dist <= AUTO_SCROLL_THRESHOLD) { isAutoScrollRef.current = true; setShowNewBtn(false); unreadRef.current = 0; }
   584→    else { isAutoScrollRef.current = false; }
   585→  }, []);
   586→
   587→  useEffect(() => { if (messages.length > 0) requestAnimationFrame(() => scrollToBottom("instant")); }, [isLoading, scrollToBottom]);
   588→
   589→  useEffect(() => { if (!onMarkAsRead) return; const unread = messages.filter((m) => m.sender_id !== currentUserId && m.status !== "read"); if (unread.length > 0 && isAutoScrollRef.current) onMarkAsRead(unread.map((m) => m.id)); }, [messages, currentUserId, onMarkAsRead]);
   590→
   591→  /* ---- Scroll to a specific message ----------------------------------- */
   592→  useEffect(() => {
   593→    if (!scrollToMessageId) return;
   594→    const idx = messages.findIndex((m) => m.id === scrollToMessageId);
   595→    if (idx === -1) { onScrolledToMessage?.(); return; }
   596→    virtualizer.scrollToIndex(idx, { align: "center" });
   597→    onScrolledToMessage?.();
   598→  }, [scrollToMessageId, messages, virtualizer, onScrolledToMessage]);
   599→
   600→  useEffect(() => {
   601→    const prev = prevCountRef.current; const next = messages.length; prevCountRef.current = next;
   602→    if (next <= prev) return;
   603→    const last = messages[next - 1]; const own = last && last.sender_id === currentUserId;
   604→    if (own) { requestAnimationFrame(() => scrollToBottom("instant")); return; }
   605→    if (isAutoScrollRef.current) requestAnimationFrame(() => scrollToBottom("smooth"));
   606→    else { unreadRef.current += next - prev; setShowNewBtn(true); }
   607→  }, [messages.length, currentUserId, scrollToBottom]);
   608→
   609→  if (isLoading) return <MessageListSkeleton />;
   610→  if (messages.length === 0 && !isPartnerTyping) return <EmptyState partnerName={partnerName} partnerAvatar={partnerAvatar} />;
   611→
   612→  const virtualItems = virtualizer.getVirtualItems();
   613→
   614→  return (
   615→    <div className="relative h-full w-full overflow-hidden">
   616→      <div ref={parentRef} onScroll={handleScroll} className="h-full w-full overflow-y-auto overscroll-contain" style={{ WebkitOverflowScrolling: "touch" }}>
   617→        <div style={{ height: `${virtualizer.getTotalSize()}px`, width: "100%", position: "relative" }}>
   618→          {virtualItems.map((vi) => {
   619→            const isTyping = vi.index === messages.length;
   620→            const msg = isTyping ? null : messages[vi.index];
   621→            // Check if we need a date separator before this message
   622→            const showDateSep = msg && vi.index > 0 && messages[vi.index - 1] && (() => {
   623→              const prev = new Date(messages[vi.index - 1].created_at);
   624→              const curr = new Date(msg.created_at);
   625→              return prev.toDateString() !== curr.toDateString();
   626→            })();
   627→            const showFirstDateSep = msg && vi.index === 0;
   628→            return (
   629→              <div key={vi.key} data-index={vi.index} ref={virtualizer.measureElement} style={{ position: "absolute", top: 0, left: 0, width: "100%", transform: `translateY(${vi.start}px)` }}>
   630→                {isTyping ? <TypingIndicator partnerName={partnerName} /> : msg ? (
   631→                  <>
   632→                    {(showDateSep || showFirstDateSep) && <DateSeparator date={msg.created_at} />}
   633→                    <MessageBubble message={msg} isOwn={msg.sender_id === currentUserId} partnerName={partnerName} partnerAvatar={partnerAvatar} onReplyTo={onReplyTo} onReact={onReact} onDeleteMessage={onDeleteMessage} />
   634→                  </>
   635→                ) : null}
   636→              </div>
   637→            );
   638→          })}
   639→        </div>
   640→      </div>
   641→      {showNewBtn && unreadRef.current > 0 && <NewMessagesButton count={unreadRef.current} onClick={() => scrollToBottom("smooth")} />}
   642→    </div>
   643→  );
   644→}
