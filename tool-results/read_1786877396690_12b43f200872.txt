    87→    87→/* ------------------------------------------------------------------ */
    88→    88→
    89→    89→function EphemeralTimer({ seconds, onExpire }: { seconds: number; onExpire: () => void }) {
    90→    90→  const [remaining, setRemaining] = useState(seconds);
    91→    91→  const intervalRef = useRef<ReturnType<typeof setInterval> | undefined>(undefined);
    92→    92→
    93→    93→  useEffect(() => {
    94→    94→    intervalRef.current = setInterval(() => {
    95→    95→      setRemaining((r) => {
    96→    96→        if (r <= 1) { clearInterval(intervalRef.current); onExpire(); return 0; }
    97→    97→        return r - 1;
    98→    98→      });
    99→    99→    }, 1000);
   100→   100→    return () => clearInterval(intervalRef.current);
   101→   101→  }, [seconds, onExpire]);
   102→   102→
   103→   103→  const pct = remaining / seconds;
   104→   104→  const circ = 2 * Math.PI * 7;
   105→   105→  const offset = circ * (1 - pct);
   106→   106→
   107→   107→  return (
   108→   108→    <svg className="size-4 shrink-0 ephemeral-timer-pulse" viewBox="0 0 18 18" aria-label={`${remaining}s remaining`}>
   109→   109→      <circle cx="9" cy="9" r="7" fill="none" stroke="currentColor" strokeWidth="1.5" className="text-white/20" />
   110→   110→      <circle cx="9" cy="9" r="7" fill="none" stroke="currentColor" strokeWidth="1.5" strokeDasharray={circ} strokeDashoffset={offset} strokeLinecap="round" className="text-violet-300" style={{ transform: "rotate(-90deg)", transformOrigin: "50% 50%", transition: "stroke-dashoffset 1s linear" }} />
   111→   111→    </svg>
   112→   112→  );
   113→   113→}

   196→   196→function VoiceBubble({ message, isOwn, waveform }: {
   197→   197→  message: ChatMessage; isOwn: boolean; waveform: number[];
   198→   198→}) {
   199→   199→  const audioRef = useRef<HTMLAudioElement>(null);
   200→   200→  const [playing, setPlaying] = useState(false);
   201→   201→  const [progress, setProgress] = useState(0);
   202→   202→  const [speed, setSpeed] = useState(1);
   203→   203→  const speeds = [1, 1.5, 2];
   204→   204→  const duration = message.voice_duration ?? 0;
   205→   205→
   206→   206→  useEffect(() => {
   207→   207→    const audio = audioRef.current;
   208→   208→    if (!audio) return;
   209→   209→    const onTime = () => { if (audio.duration) setProgress(audio.currentTime / audio.duration); };
   210→   210→    const onEnd = () => setPlaying(false);
   211→   211→    audio.addEventListener("timeupdate", onTime);
   212→   212→    audio.addEventListener("ended", onEnd);
   213→   213→    return () => { audio.removeEventListener("timeupdate", onTime); audio.removeEventListener("ended", onEnd); };
   214→   214→  }, []);
   215→   215→
   216→   216→  const toggle = useCallback(() => {
   217→   217→    const audio = audioRef.current;
   218→   218→    if (!audio) return;
   219→   219→    if (playing) { audio.pause(); setPlaying(false); }
   220→   220→    else { audio.playbackRate = speed; audio.play().catch(() => {}); setPlaying(true); }
   221→   221→  }, [playing, speed]);
   222→   222→
   223→   223→  const cycleSpeed = useCallback(() => {
   224→   224→    setSpeed((s) => {
   225→   225→      const idx = speeds.indexOf(s);
   226→   226→      const next = speeds[(idx + 1) % speeds.length];
   227→   227→      if (audioRef.current) audioRef.current.playbackRate = next;
   228→   228→      return next;
   229→   229→    });
   230→   230→  }, []);
   231→   231→
   232→   232→  /* Build waveform bars */
   233→   233→  const bars = useMemo(() => {
   234→   234→    const count = Math.min(waveform.length, 40);
   235→   235→    const step = Math.max(1, Math.floor(waveform.length / count));
   236→   236→    return Array.from({ length: count }, (_, i) => waveform[i * step] ?? 0.1);
   237→   237→  }, [waveform]);
   238→   238→
   239→   239→  const barColor = isOwn ? "bg-white/50" : "bg-zinc-400/70";
   240→   240→  const progressColor = isOwn ? "bg-white" : "bg-violet-400";
   241→   241→  const progressIdx = Math.floor(progress * bars.length);
   242→   242→
   243→   243→  return (
   244→   244→    <div className={cn("flex items-center gap-3", isOwn ? "flex-row-reverse" : "")}>
   245→   245→      <button type="button" onClick={toggle} className={cn("flex size-9 shrink-0 items-center justify-center rounded-full transition-all duration-150", isOwn ? "bg-white/20 hover:bg-white/30 hover:scale-105" : "bg-zinc-600 hover:bg-zinc-500 hover:scale-105")} aria-label={playing ? "Pause" : "Play"}>
   246→   246→        {playing ? <Pause className="size-3.5 text-white" /> : <Play className="size-3.5 text-white ml-0.5" />}
   247→   247→      </button>
   248→   248→      <div className="flex items-center gap-[2px]" style={{ height: 32 }}>
   249→   249→        {bars.map((amp, i) => (
   250→   250→          <div key={i} className={cn("w-[2.5px] rounded-full transition-all duration-100", i < progressIdx ? progressColor : barColor)} style={{ height: `${Math.max(4, amp * 32)}px` }} />
   251→   251→        ))}
   252→   252→      </div>
   253→   253→      <button type="button" onClick={cycleSpeed} className={cn("shrink-0 text-[11px] font-semibold tabular-nums transition-colors hover:opacity-80", isOwn ? "text-white/70" : "text-zinc-400")}>{speed}x</button>
   254→   254→      <span className={cn("shrink-0 text-[11px] tabular-nums", isOwn ? "text-white/50" : "text-zinc-500")}>{Math.floor(duration / 60)}:{String(duration % 60).padStart(2, "0")}</span>
   255→   255→      <audio ref={audioRef} src={message.content} preload="metadata" />
   256→   256→    </div>
   257→   257→  );
   258→   258→}

   260→   260→/* ------------------------------------------------------------------ */
   261→   261→/*  LinkifiedText – detects & renders clickable URLs                   */
   262→   262→/* ------------------------------------------------------------------ */
   263→   263→
   264→   264→const URL_REGEX = /(https?:\/\/[^\s<]+)/g;
   265→   265→
   266→   266→type TextSegment = { type: "text"; value: string } | { type: "link"; value: string };
   267→   267→
   268→   268→function parseUrls(text: string): TextSegment[] {
   269→   269→  const segments: TextSegment[] = [];
   270→   270→  let lastIndex = 0;
   271→   271→  let match: RegExpExecArray | null;
   272→   272→  while ((match = URL_REGEX.exec(text)) !== null) {
   273→   273→    if (match.index > lastIndex) {
   274→   274→      segments.push({ type: "text", value: text.slice(lastIndex, match.index) });
   275→   275→    }
   276→   276→    segments.push({ type: "link", value: match[1] });
   277→   277→    lastIndex = URL_REGEX.lastIndex;
   278→   278→  }
   279→   279→  if (lastIndex < text.length) {
   280→   280→    segments.push({ type: "text", value: text.slice(lastIndex) });
   281→   281→  }
   282→   282→  return segments;
   283→   283→}
   284→   284→
   285→   285→const LinkifiedText = React.memo(function LinkifiedText({ text, className }: { text: string; className?: string }) {
   286→   286→  const segments = useMemo(() => parseUrls(text), [text]);
   287→   287→  return (
   288→   288→    <span className={className}>
   289→   289→      {segments.map((seg, i) =>
   290→   290→        seg.type === "link" ? (
   291→   291→          <a
   292→   292→            key={i}
   293→   293→            href={seg.value}
   294→   294→            target="_blank"
   295→   295→            rel="noopener noreferrer"
   296→   296→            className="underline decoration-violet-400/50 underline-offset-2 hover:decoration-violet-400 transition-colors"
   297→   297→            onClick={(e) => e.stopPropagation()}
   298→   298→          >
   299→   299→            {seg.value}
   300→   300→          </a>
   301→   301→        ) : (
   302→   302→          <span key={i}>{seg.value}</span>
   303→   303→        ),
   304→   304→      )}
   305→   305→    </span>
   306→   306→  );
   307→   307→});

   580→   580→function formatDateSeparator(dateStr: string): string {
   581→   581→  const d = new Date(dateStr);
   582→   582→  const now = new Date();
   583→   583→  const today = new Date(now.getFullYear(), now.getMonth(), now.getDate());
   584→   584→  const yesterday = new Date(today);
   585→   585→  yesterday.setDate(yesterday.getDate() - 1);
   586→   586→  const msgDay = new Date(d.getFullYear(), d.getMonth(), d.getDate());
   587→   587→
   588→   588→  if (msgDay.getTime() === today.getTime()) return "Today";
   589→   589→  if (msgDay.getTime() === yesterday.getTime()) return "Yesterday";
   590→   590→  return d.toLocaleDateString(undefined, { month: "short", day: "numeric" });
   591→   591→}

   702→   702→export function MessageFeed({ messages, isLoading, isPartnerTyping, currentUserId, partnerName, partnerAvatar, backgroundStyle, onMarkAsRead, onVanishMessage, onReplyTo, onDeleteMessage, onReact, scrollToMessageId, onScrolledToMessage }: MessageFeedProps) {
   703→   703→  const parentRef = useRef<HTMLDivElement>(null);
   704→   704→  const isAutoScrollRef = useRef(true);
   705→   705→  const [showNewBtn, setShowNewBtn] = useState(false);
   706→   706→  const unreadRef = useRef(0);
   707→   707→  const prevCountRef = useRef(messages.length);
   708→   708→
   709→   709→  const totalItems = useMemo(() => messages.length + (isPartnerTyping ? 1 : 0), [messages.length, isPartnerTyping]);
   710→   710→
   711→   711→  const virtualizer = useVirtualizer({ count: totalItems, getScrollElement: () => parentRef.current, estimateSize: () => ESTIMATED_ROW_HEIGHT, overscan: OVERSCAN, paddingStart: 16, paddingEnd: 16, gap: 12 });
   712→   712→
   713→   713→  const scrollToBottom = useCallback((behavior: ScrollBehavior = "smooth") => {
   714→   714→    const el = parentRef.current; if (!el) return;
   715→   715→    el.scrollTo({ top: el.scrollHeight, behavior });
   716→   716→    isAutoScrollRef.current = true; setShowNewBtn(false); unreadRef.current = 0;
   717→   717→  }, []);
   718→   718→
   719→   719→  const handleScroll = useCallback((e: UIEvent<HTMLDivElement>) => {
   720→   720→    const el = e.currentTarget;
   721→   721→    const dist = el.scrollHeight - el.scrollTop - el.clientHeight;
   722→   722→    if (dist <= AUTO_SCROLL_THRESHOLD) { isAutoScrollRef.current = true; setShowNewBtn(false); unreadRef.current = 0; }
   723→   723→    else { isAutoScrollRef.current = false; }
   724→   724→  }, []);
   725→   725→
   726→   726→  useEffect(() => { if (messages.length > 0) requestAnimationFrame(() => scrollToBottom("instant")); }, [isLoading, scrollToBottom]);
   727→   727→
   728→   728→  useEffect(() => { if (!onMarkAsRead) return; const unread = messages.filter((m) => m.sender_id !== currentUserId && m.status !== "read"); if (unread.length > 0 && isAutoScrollRef.current) onMarkAsRead(unread.map((m) => m.id)); }, [messages, currentUserId, onMarkAsRead]);
   729→   729→
   730→   730→  /* ---- Scroll to a specific message ----------------------------------- */
   731→   731→  useEffect(() => {
   732→   732→    if (!scrollToMessageId) return;
   733→   733→    const idx = messages.findIndex((m) => m.id === scrollToMessageId);
   734→   734→    if (idx === -1) { onScrolledToMessage?.(); return; }
   735→   735→    virtualizer.scrollToIndex(idx, { align: "center" });
   736→   736→    onScrolledToMessage?.();
   737→   737→  }, [scrollToMessageId, messages, virtualizer, onScrolledToMessage]);
   738→   738→
   739→   739→  useEffect(() => {
   740→   740→    const prev = prevCountRef.current; const next = messages.length; prevCountRef.current = next;
   741→   741→    if (next <= prev) return;
   742→   742→    const last = messages[next - 1]; const own = last && last.sender_id === currentUserId;
   743→   743→    if (own) { requestAnimationFrame(() => scrollToBottom("instant")); return; }
   744→   744→    if (isAutoScrollRef.current) requestAnimationFrame(() => scrollToBottom("smooth"));
   745→   745→    else { unreadRef.current += next - prev; setShowNewBtn(true); }
   746→   746→  }, [messages.length, currentUserId, scrollToBottom]);
