    85→/*  Ephemeral Timer (circular SVG countdown)                           */
    86→/* ------------------------------------------------------------------ */
    87→
    88→function EphemeralTimer({ seconds, onExpire }: { seconds: number; onExpire: () => void }) {
    89→  const [remaining, setRemaining] = useState(seconds);
    90→  const intervalRef = useRef<ReturnType<typeof setInterval> | undefined>(undefined);
    91→
    92→  useEffect(() => {
    93→    intervalRef.current = setInterval(() => {
    94→      setRemaining((r) => {
    95→        if (r <= 1) { clearInterval(intervalRef.current); onExpire(); return 0; }
    96→        return r - 1;
    97→      });
    98→    }, 1000);
    99→    return () => clearInterval(intervalRef.current);
   100→  }, [seconds, onExpire]);
   101→
   102→  const pct = remaining / seconds;
   103→  const circ = 2 * Math.PI * 7;
   104→  const offset = circ * (1 - pct);
   105→
   106→  return (
   107→    <svg className="size-4 shrink-0 ephemeral-timer-pulse" viewBox="0 0 18 18" aria-label={`${remaining}s remaining`}>
   108→      <circle cx="9" cy="9" r="7" fill="none" stroke="currentColor" strokeWidth="1.5" className="text-white/20" />
   109→      <circle cx="9" cy="9" r="7" fill="none" stroke="currentColor" strokeWidth="1.5" strokeDasharray={circ} strokeDashoffset={offset} strokeLinecap="round" className="text-violet-300" style={{ transform: "rotate(-90deg)", transformOrigin: "50% 50%", transition: "stroke-dashoffset 1s linear" }} />
   110→    </svg>
   111→  );
   112→}

   195→function VoiceBubble({ message, isOwn, waveform }: {
   196→  message: ChatMessage; isOwn: boolean; waveform: number[];
   197→}) {
   198→  const audioRef = useRef<HTMLAudioElement>(null);
   199→  const [playing, setPlaying] = useState(false);
   200→  const [progress, setProgress] = useState(0);
   201→  const [speed, setSpeed] = useState(1);
   202→  const speeds = [1, 1.5, 2];
   203→  const duration = message.voice_duration ?? 0;
   204→
   205→  useEffect(() => {
   206→    const audio = audioRef.current;
   207→    if (!audio) return;
   208→    const onTime = () => { if (audio.duration) setProgress(audio.currentTime / audio.duration); };
   209→    const onEnd = () => setPlaying(false);
   210→    audio.addEventListener("timeupdate", onTime);
   211→    audio.addEventListener("ended", onEnd);
   212→    return () => { audio.removeEventListener("timeupdate", onTime); audio.removeEventListener("ended", onEnd); };
   213→  }, []);
   214→
   215→  const toggle = useCallback(() => {
   216→    const audio = audioRef.current;
   217→    if (!audio) return;
   218→    if (playing) { audio.pause(); setPlaying(false); }
   219→    else { audio.playbackRate = speed; audio.play().catch(() => {}); setPlaying(true); }
   220→  }, [playing, speed]);
   221→
   222→  const cycleSpeed = useCallback(() => {
   223→    setSpeed((s) => {
   224→      const idx = speeds.indexOf(s);
   225→      const next = speeds[(idx + 1) % speeds.length];
   226→      if (audioRef.current) audioRef.current.playbackRate = next;
   227→      return next;
   228→    });
   229→  }, []);
   230→
   231→  /* Build waveform bars */
   232→  const bars = useMemo(() => {
   233→    const count = Math.min(waveform.length, 40);
   234→    const step = Math.max(1, Math.floor(waveform.length / count));
   235→    return Array.from({ length: count }, (_, i) => waveform[i * step] ?? 0.1);
   236→  }, [waveform]);
   237→
   238→  const barColor = isOwn ? "bg-white/50" : "bg-zinc-400/70";
   239→  const progressColor = isOwn ? "bg-white" : "bg-violet-400";
   240→  const progressIdx = Math.floor(progress * bars.length);
   241→
   242→  return (
   243→    <div className={cn("flex items-center gap-3", isOwn ? "flex-row-reverse" : "")}>
   244→      <button type="button" onClick={toggle} className={cn("flex size-9 shrink-0 items-center justify-center rounded-full transition-all duration-150", isOwn ? "bg-white/20 hover:bg-white/30 hover:scale-105" : "bg-zinc-600 hover:bg-zinc-500 hover:scale-105")} aria-label={playing ? "Pause" : "Play"}>
   245→        {playing ? <Pause className="size-3.5 text-white" /> : <Play className="size-3.5 text-white ml-0.5" />}
   246→      </button>
   247→      <div className="flex items-center gap-[2px]" style={{ height: 32 }}>
   248→        {bars.map((amp, i) => (
   249→          <div key={i} className={cn("w-[2.5px] rounded-full transition-all duration-100", i < progressIdx ? progressColor : barColor)} style={{ height: `${Math.max(4, amp * 32)}px` }} />
   250→        ))}
   251→      </div>
   252→      <button type="button" onClick={cycleSpeed} className={cn("shrink-0 text-[11px] font-semibold tabular-nums transition-colors hover:opacity-80", isOwn ? "text-white/70" : "text-zinc-400")}>{speed}x</button>
   253→      <span className={cn("shrink-0 text-[11px] tabular-nums", isOwn ? "text-white/50" : "text-zinc-500")}>{Math.floor(duration / 60)}:{String(duration % 60).padStart(2, "0")}</span>
   254→      <audio ref={audioRef} src={message.content} preload="metadata" />
   255→    </div>
   256→  );
   257→}

   259→/* ------------------------------------------------------------------ */
   260→/*  LinkifiedText – detects & renders clickable URLs                   */
   261→/* ------------------------------------------------------------------ */
   262→
   263→const URL_REGEX = /(https?:\/\/[^\s<]+)/g;
   264→
   265→type TextSegment = { type: "text"; value: string } | { type: "link"; value: string };
   266→
   267→function parseUrls(text: string): TextSegment[] {
   268→  const segments: TextSegment[] = [];
   269→  let lastIndex = 0;
   270→  let match: RegExpExecArray | null;
   271→  while ((match = URL_REGEX.exec(text)) !== null) {
   272→    if (match.index > lastIndex) {
   273→      segments.push({ type: "text", value: text.slice(lastIndex, match.index) });
   274→    }
   275→    segments.push({ type: "link", value: match[1] });
   276→    lastIndex = URL_REGEX.lastIndex;
   277→  }
   278→  if (lastIndex < text.length) {
   279→    segments.push({ type: "text", value: text.slice(lastIndex) });
   280→  }
   281→  return segments;
   282→}
   283→
   284→const LinkifiedText = React.memo(function LinkifiedText({ text, className }: { text: string; className?: string }) {
   285→  const segments = useMemo(() => parseUrls(text), [text]);
   286→  return (
   287→    <span className={className}>
   288→      {segments.map((seg, i) =>
   289→        seg.type === "link" ? (
   290→          <a
   291→            key={i}
   292→            href={seg.value}
   293→            target="_blank"
   294→            rel="noopener noreferrer"
   295→            className="underline decoration-violet-400/50 underline-offset-2 hover:decoration-violet-400 transition-colors"
   296→            onClick={(e) => e.stopPropagation()}
   297→          >
   298→            {seg.value}
   299→          </a>
   300→        ) : (
   301→          <span key={i}>{seg.value}</span>
   302→        ),
   303→      )}
   304→    </span>
   305→  );
   306→});

   405→function MessageBubble({ message, isOwn, partnerName, partnerAvatar, onReplyTo, onReact, onDeleteMessage }: {
   406→  message: ChatMessage; isOwn: boolean; partnerName: string; partnerAvatar: string | null;
   407→  onReplyTo?: (msg: ChatMessage) => void; onReact?: (id: string, emoji: string, add: boolean) => void; onDeleteMessage?: (id: string) => void;
   408→}) {
   409→  const [showHeart, setShowHeart] = useState(false);
   410→  const [showTapback, setShowTapback] = useState(false);
   411→  const [showLightbox, setShowLightbox] = useState<string | null>(null);
   412→  const lastTapRef = useRef(0);
   413→  const longPressRef = useRef<ReturnType<typeof setTimeout> | undefined>(undefined);
   414→  const bubbleRef = useRef<HTMLDivElement>(null);
   415→
   416→  const triggerHeart = useCallback(() => {
   417→    setShowHeart(true); setTimeout(() => setShowHeart(false), 800);
   418→    if (typeof navigator !== "undefined" && "vibrate" in navigator) navigator.vibrate(10);
   419→  }, []);
   420→
   421→  const handleClick = useCallback(() => {
   422→    clearTimeout(longPressRef.current);
   423→    const now = Date.now();
   424→    if (now - lastTapRef.current < DOUBLE_TAP_MS) triggerHeart();
   425→    lastTapRef.current = now;
   426→  }, [triggerHeart]);
   427→
   428→  const handlePressStart = useCallback(() => {
   429→    longPressRef.current = setTimeout(() => {
   430→      if (typeof navigator !== "undefined" && "vibrate" in navigator) navigator.vibrate(20);
   431→      setShowTapback(true);
   432→    }, LONG_PRESS_MS);
   433→  }, []);
   434→
   435→  const handlePressEnd = useCallback(() => { clearTimeout(longPressRef.current); }, []);
   436→
   437→  const handleContextMenu = useCallback((e: React.MouseEvent) => {
   438→    e.preventDefault(); setShowTapback(true);
   439→  }, []);
   440→
   441→  const handleCopy = useCallback(() => {
   442→    navigator.clipboard?.writeText(message.content).then(() => {
   443→      toast({ title: "Copied to clipboard", description: message.content.length > 50 ? message.content.slice(0, 50) + "..." : message.content });
   444→    }).catch(() => {
   445→      toast({ title: "Failed to copy", description: "Could not access clipboard", variant: "destructive" });
   446→    });
   447→    setShowTapback(false);
   448→  }, [message.content]);
   449→
   450→  const handleDelete = useCallback(() => {
   451→    onDeleteMessage?.(message.id);
   452→    setShowTapback(false);
   453→  }, [message.id, onDeleteMessage]);
   454→
   455→  const handleSwipeEnd = useCallback((_: MouseEvent | TouchEvent | PointerEvent, info: PanInfo) => {
   456→    if (info.offset.x > SWIPE_REPLY_THRESHOLD && info.velocity.x > 200) {
   457→      if (typeof navigator !== "undefined" && "vibrate" in navigator) navigator.vibrate(10);
   458→      onReplyTo?.(message);
   459→    }
   460→  }, [message, onReplyTo]);
   461→
   462→  const isFailed = message.status === "failed";
   463→  const isVoice = message.type === "voice";
   464→  const isNew = message.status === "sending" || Date.now() - new Date(message.created_at).getTime() < 2000;

   466→  return (

   472→    >
   473→      <HeartBurst show={showHeart} />
   474→      <AnimatePresence>{showTapback && onReact && (
   475→        <TapbackDock message={message} isOwn={isOwn} onReact={(emoji, add) => { onReact(message.id, emoji, add); setShowTapback(false); }} onReply={() => { onReplyTo?.(message); setShowTapback(false); }} onCopy={handleCopy} onDelete={isOwn ? handleDelete : undefined} />
   476→      )}</AnimatePresence>
   477→
   478→      {!isOwn && (
   479→        <Avatar className="mt-auto size-8 shrink-0 ring-1 ring-muted">
   480→          {partnerAvatar && <AvatarImage src={partnerAvatar} alt={partnerName} />}
   481→          <AvatarFallback className="bg-gradient-to-br from-violet-100 to-purple-200 text-xs font-medium text-violet-700 dark:from-violet-500 dark:to-purple-600 dark:text-white">{partnerName.slice(0, 2).toUpperCase()}</AvatarFallback>
   482→        </Avatar>
   483→      )}
   484→
   485→      <div className={cn("flex max-w-[75%] flex-col gap-1", isOwn ? "items-end" : "items-start")}>
   486→        <m.div
   487→          ref={bubbleRef}
   488→          drag="x"
   489→          dragConstraints={{ left: 0, right: 0 }}
   490→          dragElastic={{ left: 0, right: 0.35 }}
   491→          onDragEnd={handleSwipeEnd}
   492→          onClick={handleClick}
   493→          onPointerDown={handlePressStart}
   494→          onPointerUp={handlePressEnd}
   495→          onPointerLeave={handlePressEnd}
   496→          onContextMenu={handleContextMenu}
   497→          className={cn(
   498→            "relative cursor-default select-none rounded-2xl px-4 py-2.5 text-sm leading-relaxed transition-transform duration-100 active:scale-[0.98]",
   499→            // Own messages: vibrant violet-purple gradient with subtle shadow
   500→            isOwn && "bg-gradient-to-r from-violet-500 to-purple-600 text-white rounded-br-md shadow-md shadow-violet-500/15",
   501→            // Partner messages: softer background with shadow
   502→            !isOwn && "bg-muted text-foreground rounded-bl-md shadow-sm dark:bg-zinc-800 dark:text-zinc-100 dark:shadow-violet-950/20",
   503→            isFailed && "ring-2 ring-red-400/50",
   504→            isVoice && "py-2"
   505→          )}
   506→          style={{ transform: "translate3d(0,0,0)" }}
   507→        >
   508→          {/* Reply quote */}
   509→          {message.reply_to_id && message.reply_to_content && (
   510→            <ReplyPreview senderName={message.reply_to_sender_name ?? ""} content={message.reply_to_content} isOwn={isOwn} />
   511→          )}
   512→
   513→          {/* Content */}
   514→          {isVoice ? (
   515→            message.waveform_data ? (
   516→              <VoiceBubble message={message} isOwn={isOwn} waveform={message.waveform_data} />
   517→            ) : <p className="text-white/60 italic">Voice message</p>
   518→          ) : message.type === "image" ? (
   519→            <div className="group/img relative cursor-zoom-in" onClick={() => setShowLightbox(message.content)}>
   520→              <img src={message.content} alt="Shared image" className="max-h-64 rounded-lg object-contain transition-opacity duration-150 group-hover/img:opacity-90" />
   521→              <div className="absolute inset-0 flex items-center justify-center rounded-lg bg-black/0 transition-colors duration-150 group-hover/img:bg-black/20">
   522→                <ZoomIn className="size-6 text-white opacity-0 transition-opacity duration-150 group-hover/img:opacity-80" />
   523→              </div>
   524→            </div>
   525→          ) : (
   526→            <LinkifiedText text={message.content} className="whitespace-pre-wrap break-words" />
   527→          )}
   528→
   529→          {/* Vanish indicator */}
   530→          {message.vanish_mode && (
   531→            <div className="mt-1 flex items-center gap-1 text-[10px] text-white/50"><EyeOff className="size-3" /><span>View once</span></div>
   532→          )}
   533→        </m.div>
   534→
   535→        {/* Timestamp + receipt + ephemeral timer */}
   536→        <div className={cn("flex items-center gap-1.5 px-1 text-[11px] text-muted-foreground", isOwn && "flex-row-reverse")}>
   537→          {message.ephemeral_seconds && message.ephemeral_seconds > 0 && (
   538→            <EphemeralTimer seconds={message.ephemeral_seconds} onExpire={() => { /* trigger vanish */ }} />
   539→          )}
   540→          <span>{formatDistanceToNow(new Date(message.created_at), { addSuffix: true })}</span>
   541→          {isOwn && <ReceiptIcon status={message.status} />}
   542→        </div>
   543→
   544→        {/* Reactions */}
   545→        {onReact && <ReactionPills reactions={message.reactions} isOwn={isOwn} onReact={(e, a) => onReact(message.id, e, a)} currentUserId={""} />}
   546→
   547→        {isFailed && (
   548→          <button
   549→            type="button"
   550→            onClick={(e) => { e.stopPropagation(); onDeleteMessage?.(message.id); }}
   551→            className="mt-1 flex items-center gap-1 px-1 text-[10px] text-red-400 transition-colors hover:text-red-300"
   552→          >
   553→            <AlertCircle className="size-3" />
   554→            <span>Failed to send &middot; Tap to retry</span>
   555→          </button>
   556→        )}
   557→      </div>
   558→
   559→      {isOwn && <div className="w-8 shrink-0" />}
   560→    </m.div>
   561→
   562→    {/* Image Lightbox */}
   563→    <AnimatePresence>
   564→      {showLightbox && (
   565→        <ImageLightbox src={showLightbox} onClose={() => setShowLightbox(null)} />
   566→      )}
   567→    </AnimatePresence>
   568→  );
   569→}

   575→function formatDateSeparator(dateStr: string): string {
   576→  const d = new Date(dateStr);
   577→  const now = new Date();
   578→  const today = new Date(now.getFullYear(), now.getMonth(), now.getDate());
   579→  const yesterday = new Date(today);
   580→  yesterday.setDate(yesterday.getDate() - 1);
   581→  const msgDay = new Date(d.getFullYear(), d.getMonth(), d.getDate());
   582→
   583→  if (msgDay.getTime() === today.getTime()) return "Today";
   584→  if (msgDay.getTime() === yesterday.getTime()) return "Yesterday";
   585→  return d.toLocaleDateString(undefined, { month: "short", day: "numeric" });
   586→}

   697→export function MessageFeed({ messages, isLoading, isPartnerTyping, currentUserId, partnerName, partnerAvatar, onMarkAsRead, onVanishMessage, onReplyTo, onDeleteMessage, onReact, scrollToMessageId, onScrolledToMessage }: MessageFeedProps) {
   698→  const parentRef = useRef<HTMLDivElement>(null);
   699→  const isAutoScrollRef = useRef(true);
   700→  const [showNewBtn, setShowNewBtn] = useState(false);
   701→  const unreadRef = useRef(0);
   702→  const prevCountRef = useRef(messages.length);
   703→
   704→  const totalItems = useMemo(() => messages.length + (isPartnerTyping ? 1 : 0), [messages.length, isPartnerTyping]);
   705→
   706→  const virtualizer = useVirtualizer({ count: totalItems, getScrollElement: () => parentRef.current, estimateSize: () => ESTIMATED_ROW_HEIGHT, overscan: OVERSCAN, paddingStart: 16, paddingEnd: 16, gap: 12 });
   707→
   708→  const scrollToBottom = useCallback((behavior: ScrollBehavior = "smooth") => {
   709→    const el = parentRef.current; if (!el) return;
   710→    el.scrollTo({ top: el.scrollHeight, behavior });
   711→    isAutoScrollRef.current = true; setShowNewBtn(false); unreadRef.current = 0;
   712→  }, []);
   713→
   714→  const handleScroll = useCallback((e: UIEvent<HTMLDivElement>) => {
   715→    const el = e.currentTarget;
   716→    const dist = el.scrollHeight - el.scrollTop - el.clientHeight;
   717→    if (dist <= AUTO_SCROLL_THRESHOLD) { isAutoScrollRef.current = true; setShowNewBtn(false); unreadRef.current = 0; }
   718→    else { isAutoScrollRef.current = false; }
   719→  }, []);
   720→
   721→  useEffect(() => { if (messages.length > 0) requestAnimationFrame(() => scrollToBottom("instant")); }, [isLoading, scrollToBottom]);
   722→
   723→  useEffect(() => { if (!onMarkAsRead) return; const unread = messages.filter((m) => m.sender_id !== currentUserId && m.status !== "read"); if (unread.length > 0 && isAutoScrollRef.current) onMarkAsRead(unread.map((m) => m.id)); }, [messages, currentUserId, onMarkAsRead]);
   724→
   725→  /* ---- Scroll to a specific message ----------------------------------- */
   726→  useEffect(() => {
   727→    if (!scrollToMessageId) return;
   728→    const idx = messages.findIndex((m) => m.id === scrollToMessageId);
   729→    if (idx === -1) { onScrolledToMessage?.(); return; }
   730→    virtualizer.scrollToIndex(idx, { align: "center" });
   731→    onScrolledToMessage?.();
   732→  }, [scrollToMessageId, messages, virtualizer, onScrolledToMessage]);
   733→
   734→  useEffect(() => {
   735→    const prev = prevCountRef.current; const next = messages.length; prevCountRef.current = next;
   736→    if (next <= prev) return;
   737→    const last = messages[next - 1]; const own = last && last.sender_id === currentUserId;
   738→    if (own) { requestAnimationFrame(() => scrollToBottom("instant")); return; }
   739→    if (isAutoScrollRef.current) requestAnimationFrame(() => scrollToBottom("smooth"));
   740→    else { unreadRef.current += next - prev; setShowNewBtn(true); }
   741→  }, [messages.length, currentUserId, scrollToBottom]);
   742→
   743→  if (isLoading) return <MessageListSkeleton />;
   744→  if (messages.length === 0 && !isPartnerTyping) return <EmptyState partnerName={partnerName} partnerAvatar={partnerAvatar} />;
   745→
   746→  const virtualItems = virtualizer.getVirtualItems();
   747→
   748→  return (
   749→    <div className="relative h-full w-full overflow-hidden">
   750→      <div ref={parentRef} onScroll={handleScroll} className="h-full w-full overflow-y-auto overscroll-contain" style={{ WebkitOverflowScrolling: "touch" }}>
   751→        <div style={{ height: `${virtualizer.getTotalSize()}px`, width: "100%", position: "relative" }}>
   752→          {virtualItems.map((vi) => {
   753→            const isTyping = vi.index === messages.length;
   754→            const msg = isTyping ? null : messages[vi.index];
   755→            // Check if we need a date separator before this message
   756→            const showDateSep = msg && vi.index > 0 && messages[vi.index - 1] && (() => {
   757→              const prev = new Date(messages[vi.index - 1].created_at);
   758→              const curr = new Date(msg.created_at);
   759→              return prev.toDateString() !== curr.toDateString();
   760→            })();
   761→            const showFirstDateSep = msg && vi.index === 0;
   762→            return (
   763→              <div key={vi.key} data-index={vi.index} ref={virtualizer.measureElement} style={{ position: "absolute", top: 0, left: 0, width: "100%", transform: `translateY(${vi.start}px)` }}>
   764→                {isTyping ? <TypingIndicator partnerName={partnerName} partnerAvatar={partnerAvatar} /> : msg ? (
   765→                  <>
   766→                    {(showDateSep || showFirstDateSep) && <DateSeparator date={msg.created_at} />}
   767→                    <MessageBubble message={msg} isOwn={msg.sender_id === currentUserId} partnerName={partnerName} partnerAvatar={partnerAvatar} onReplyTo={onReplyTo} onReact={onReact} onDeleteMessage={onDeleteMessage} />
   768→                  </>
   769→                ) : null}
   770→              </div>
   771→            );
   772→          })}
   773→        </div>
   774→      </div>
   775→      {showNewBtn && unreadRef.current > 0 && <NewMessagesButton count={unreadRef.current} onClick={() => scrollToBottom("smooth")} />}
   776→    </div>
   777→  );
   778→}
