    37→    37→function formatLastSeen(lastSeen: string | null): string {
    38→    38→  if (!lastSeen) return "Offline";
    39→    39→  const diffMs = Date.now() - new Date(lastSeen).getTime();
    40→    40→  const diffSec = Math.floor(diffMs / 1000);
    41→    41→  if (diffSec < 60) return "Seen just now";
    42→    42→  const diffMin = Math.floor(diffSec / 60);
    43→    43→  if (diffMin < 60) return `Seen ${diffMin}m ago`;
    44→    44→  const diffHr = Math.floor(diffMin / 60);
    45→    45→  if (diffHr < 24) return `Seen ${diffHr}h ago`;
    46→    46→  const diffDay = Math.floor(diffHr / 24);
    47→    47→  if (diffDay < 7) return `Seen ${diffDay}d ago`;
    48→    48→  return `Seen ${new Date(lastSeen).toLocaleDateString()}`;
    49→    49→}
