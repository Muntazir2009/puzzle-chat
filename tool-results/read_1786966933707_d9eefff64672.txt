   121→/*  Ephemeral Timer (circular SVG countdown)                           */
   122→/* ------------------------------------------------------------------ */
   123→
   124→function EphemeralTimer({
   125→  seconds,
   126→  onExpire,
   127→}: {
   128→  seconds: number;
   129→  onExpire: () => void;
   130→}) {
   131→  const [remaining, setRemaining] = useState(seconds);
   132→  const intervalRef = useRef<ReturnType<typeof setInterval> | undefined>(
   133→    undefined,
   134→  );
   135→
   136→  useEffect(() => {
   137→    intervalRef.current = setInterval(() => {
   138→      setRemaining((r) => {
   139→        if (r <= 1) {
   140→          clearInterval(intervalRef.current);
   141→          onExpire();
   142→          return 0;
   143→        }
   144→        return r - 1;
   145→      });
   146→    }, 1000);
   147→    return () => clearInterval(intervalRef.current);
   148→  }, [seconds, onExpire]);
   149→
   150→  const pct = remaining / seconds;
   151→  const circ = 2 * Math.PI * 7;
   152→  const offset = circ * (1 - pct);
   153→
   154→  return (
   155→    <svg
   156→      className="size-3.5 shrink-0 ephemeral-timer-pulse"
   157→      viewBox="0 0 18 18"
   158→      aria-label={`${remaining}s remaining`}
   159→    >
   160→      <circle
   161→        cx="9"
   162→        cy="9"
   163→        r="7"
   164→        fill="none"
   165→        stroke="currentColor"
   166→        strokeWidth="1.5"
   167→        className="text-white/20"
   168→      />
   169→      <circle
   170→        cx="9"
   171→        cy="9"
   172→        r="7"
   173→        fill="none"
   174→        stroke="currentColor"
   175→        strokeWidth="1.5"
   176→        strokeDasharray={circ}
   177→        strokeDashoffset={offset}
   178→        strokeLinecap="round"
   179→        style={{
   180→          color: "var(--app-accent-lighter)",
   181→          transform: "rotate(-90deg)",
   182→          transformOrigin: "50% 50%",
   183→          transition: "stroke-dashoffset 1s linear",
   184→        }}
   185→      />
   186→    </svg>
   187→  );
   188→}

   397→/*  Voice Bubble                                                       */
   398→/* ------------------------------------------------------------------ */
   399→
   400→function VoiceBubble({
   401→  message,
   402→  isOwn,
   403→  waveform,
   404→}: {
   405→  message: ChatMessage;
   406→  isOwn: boolean;
   407→  waveform: number[];
   408→}) {
   409→  const audioRef = useRef<HTMLAudioElement>(null);
   410→  const [playing, setPlaying] = useState(false);
   411→  const [progress, setProgress] = useState(0);
   412→  const [speed, setSpeed] = useState(1);
   413→  const speeds = [1, 1.5, 2];
   414→  const duration = message.voice_duration ?? 0;
   415→
   416→  useEffect(() => {
   417→    const audio = audioRef.current;
   418→    if (!audio) return;
   419→    const onTime = () => {
   420→      if (audio.duration) setProgress(audio.currentTime / audio.duration);
   421→    };
   422→    const onEnd = () => setPlaying(false);
   423→    audio.addEventListener("timeupdate", onTime);
   424→    audio.addEventListener("ended", onEnd);
   425→    return () => {
   426→      audio.removeEventListener("timeupdate", onTime);
   427→      audio.removeEventListener("ended", onEnd);
   428→    };
   429→  }, []);
   430→
   431→  const toggle = useCallback(() => {
   432→    const audio = audioRef.current;
   433→    if (!audio) return;
   434→    if (playing) {
   435→      audio.pause();
   436→      setPlaying(false);
   437→    } else {
   438→      audio.playbackRate = speed;
   439→      audio.play().catch(() => {});
   440→      setPlaying(true);
   441→    }
   442→  }, [playing, speed]);
   443→
   444→  const cycleSpeed = useCallback(() => {
   445→    setSpeed((s) => {
   446→      const idx = speeds.indexOf(s);
   447→      const next = speeds[(idx + 1) % speeds.length];
   448→      if (audioRef.current) audioRef.current.playbackRate = next;
   449→      return next;
   450→    });
   451→  }, []);
   452→
   453→  /* Build waveform bars */
   454→  const bars = useMemo(() => {
   455→    const count = Math.min(waveform.length, 40);
   456→    const step = Math.max(1, Math.floor(waveform.length / count));
   457→    return Array.from({ length: count }, (_, i) => waveform[i * step] ?? 0.1);
   458→  }, [waveform]);
   459→
   460→  const progressIdx = Math.floor(progress * bars.length);
   461→
   462→  return (
   463→    <div className={cn("flex items-center gap-3", isOwn ? "flex-row-reverse" : "")}>
   464→      <button
   465→        type="button"
   466→        onClick={toggle}
   467→        className={cn(
   468→          "flex size-9 shrink-0 items-center justify-center rounded-full transition-all duration-150 active:scale-95",
   469→          isOwn
   470→            ? "bg-white/20 hover:bg-white/30"
   471→            : "bg-zinc-600 hover:bg-[var(--app-accent-subtle)]",
   472→        )}
   473→        aria-label={playing ? "Pause" : "Play"}
   474→      >
   475→        {playing ? (
   476→          <Pause className="size-3.5 text-white" />
   477→        ) : (
   478→          <Play className="size-3.5 text-white ml-0.5" />
   479→        )}
   480→      </button>
   481→      <div className="flex items-center gap-[2px]" style={{ height: 32 }}>
   482→        {bars.map((amp, i) => (
   483→          <div
   484→            key={i}
   485→            className="w-[2.5px] rounded-full transition-all duration-100"
   486→            style={{
   487→              height: `${Math.max(4, amp * 32)}px`,
   488→              backgroundColor:
   489→                i < progressIdx
   490→                  ? isOwn
   491→                    ? "rgba(255,255,255,1)"
   492→                    : "var(--app-accent-light)"
   493→                  : isOwn
   494→                    ? "rgba(255,255,255,0.5)"
   495→                    : "rgba(161,161,170,0.7)",
   496→            }}
   497→          />
   498→        ))}
   499→      </div>
   500→      <button
   501→        type="button"
   502→        onClick={cycleSpeed}
   503→        className={cn(
   504→          "shrink-0 text-[11px] font-semibold tabular-nums transition-colors hover:opacity-80 active:scale-95",
   505→          isOwn ? "text-white/70" : "text-zinc-400",
   506→        )}
   507→      >
   508→        {speed}x
   509→      </button>
   510→      <span
   511→        className={cn(
   512→          "shrink-0 text-[11px] tabular-nums",
   513→          isOwn ? "text-white/50" : "text-zinc-500",
   514→        )}
   515→      >
   516→        {Math.floor(duration / 60)}:{String(duration % 60).padStart(2, "0")}
   517→      </span>
   518→      <audio ref={audioRef} src={message.content} preload="metadata" />
   519→    </div>
   520→  );
   521→}

   523→/* ------------------------------------------------------------------ */
   524→/*  LinkifiedText – detects & renders clickable URLs                   */
   525→/* ------------------------------------------------------------------ */
   526→
   527→const URL_REGEX = /(https?:\/\/[^\s<]+)/g;
   528→
   529→type TextSegment =
   530→  | { type: "text"; value: string }
   531→  | { type: "link"; value: string };
   532→
   533→function parseUrls(text: string): TextSegment[] {
   534→  const segments: TextSegment[] = [];
   535→  let lastIndex = 0;
   536→  let match: RegExpExecArray | null;
   537→  while ((match = URL_REGEX.exec(text)) !== null) {
   538→    if (match.index > lastIndex) {
   539→      segments.push({ type: "text", value: text.slice(lastIndex, match.index) });
   540→    }
   541→    segments.push({ type: "link", value: match[1] });
   542→    lastIndex = URL_REGEX.lastIndex;
   543→  }
   544→  if (lastIndex < text.length) {
   545→    segments.push({ type: "text", value: text.slice(lastIndex) });
   546→  }
   547→  return segments;
   548→}
   549→
   550→/** Split a plain-text string into highlighted / non-highlighted fragments */
   551→function highlightText(text: string, query: string): React.ReactNode[] {
   552→  if (!query) return [text];
   553→  const escaped = query.replace(/[.*+?^${}()|[\]\\]/g, "\\$");
   554→  const re = new RegExp(`(${escaped})`, "gi");
   555→  const parts: React.ReactNode[] = [];
   556→  let lastIndex = 0;
   557→  let match: RegExpExecArray | null;
   558→  while ((match = re.exec(text)) !== null) {
   559→    if (match.index > lastIndex) {
   560→      parts.push(text.slice(lastIndex, match.index));
   561→    }
   562→    parts.push(
   563→      <mark
   564→        key={match.index}
   565→        className="bg-yellow-300/60 dark:bg-yellow-400/40 rounded-sm px-0.5"
   566→      >
   567→        {match[1]}
   568→      </mark>,
   569→    );
   570→    lastIndex = re.lastIndex;
   571→  }
   572→  if (lastIndex < text.length) {
   573→    parts.push(text.slice(lastIndex));
   574→  }
   575→  return parts;
   576→}

   578→export const LinkifiedText = React.memo(function LinkifiedText({
   579→  text,
   580→  className,
   581→  highlight,
   582→}: {
   583→  text: string;
   584→  className?: string;
   585→  highlight?: string;
   586→}) {
   587→  const segments = useMemo(() => parseUrls(text), [text]);
   588→  return (
   589→    <span className={className}>
   590→      {segments.map((seg, i) =>
   591→        seg.type === "link" ? (
   592→          <a
   593→            key={i}
   594→            href={seg.value}
   595→            target="_blank"
   596→            rel="noopener noreferrer"
   597→            className="underline underline-offset-2 transition-colors"
   598→            style={{ textDecorationColor: "var(--app-accent-light)" }}
   599→            onClick={(e) => e.stopPropagation()}
   600→          >
   601→            {seg.value}
   602→          </a>
   603→        ) : (
   604→          <span key={i}>
   605→            {highlight ? highlightText(seg.value, highlight) : seg.value}
   606→          </span>
   607→        ),

   666→function ImageLightbox({ src, onClose }: { src: string; onClose: () => void }) {
   667→  useEffect(() => {

   686→      onClick={onClose}
   687→    >
   688→      <button
   689→        type="button"
   690→        onClick={onClose}
   691→        className="absolute right-4 top-4 z-10 flex size-10 items-center justify-center rounded-full bg-white/10 text-white transition-colors hover:bg-white/20 active:scale-95"
   692→        aria-label="Close"
   693→      >
   694→        <X className="size-5" />
   695→      </button>

   697→        initial={{ opacity: 0, scale: 0.9 }}
   698→        animate={{ opacity: 1, scale: 1 }}
   699→        exit={{ opacity: 0, scale: 0.95 }}
   700→        transition={{ duration: 0.15, ease: "easeOut" }}
   701→        src={src}
   702→        alt="Full size image"
   703→        className="max-h-[90vh] max-w-[90vw] rounded-lg object-contain shadow-2xl"
   704→        onClick={(e) => e.stopPropagation()}
   705→      />

   716→  isOwn,
   717→  onReact,
   718→  currentUserId,

   754→                      backgroundColor: "rgba(255,255,255,0.2)",

   767→              className="text-[10px] font-medium tabular-nums"

   769→                color: isActive
   770→                  ? isOwn
   771→                    ? "rgba(255,255,255,0.7)"

   789→function MessageBubble({
   790→  message,
   791→  isOwn,
   792→  partnerName,
   793→  partnerAvatar,
   794→  currentUserId,
   795→  onReplyTo,
   796→  onReact,
   797→  onDeleteMessage,
   798→  searchHighlight,
   799→}: {
   800→  message: ChatMessage;
   801→  isOwn: boolean;
   802→  partnerName: string;
   803→  partnerAvatar: string | null;
   804→  currentUserId: string;
   805→  onReplyTo?: (msg: ChatMessage) => void;
   806→  onReact?: (id: string, emoji: string, add: boolean) => void;
   807→  onDeleteMessage?: (id: string) => void;
   808→  searchHighlight?: string;
   809→}) {
   810→  const [showTapback, setShowTapback] = useState(false);
   811→  const [showActionSheet, setShowActionSheet] = useState(false);
   812→  const [showLightbox, setShowLightbox] = useState<string | null>(null);
   813→  const [showReplyArrow, setShowReplyArrow] = useState(false);
   814→  const longPressRef = useRef<ReturnType<typeof setTimeout> | undefined>(
   815→    undefined,
   816→  );
   817→  const bubbleRef = useRef<HTMLDivElement>(null);
   818→  const wasDraggedRef = useRef(false);
   819→
   820→  const handleClick = useCallback(() => {
   821→    clearTimeout(longPressRef.current);
   822→    if (wasDraggedRef.current) {
   823→      wasDraggedRef.current = false;
   824→      return;
   825→    }
   826→    setShowActionSheet((prev) => {
   827→      if (!prev) setShowTapback(false);
   828→      return !prev;
   829→    });
   830→  }, []);
   831→
   832→  const handlePressStart = useCallback(() => {
   833→    setShowActionSheet(false);
   834→    longPressRef.current = setTimeout(() => {
   835→      if (typeof navigator !== "undefined" && "vibrate" in navigator)
   836→        navigator.vibrate(20);
   837→      setShowTapback(true);
   838→    }, LONG_PRESS_MS);
   839→  }, []);
   840→
   841→  const handlePressEnd = useCallback(() => {
   842→    clearTimeout(longPressRef.current);
   843→  }, []);
   844→
   845→  const handleContextMenu = useCallback((e: React.MouseEvent) => {
   846→    e.preventDefault();
   847→    setShowActionSheet(false);
   848→    setShowTapback(true);
   849→  }, []);
   850→
   851→  const handleCopy = useCallback(() => {
   852→    navigator.clipboard
   853→      ?.writeText(message.content)
   854→      .then(() => {
   855→        toast({
   856→          title: "Copied to clipboard",
   857→          description:
   858→            message.content.length > 50
   859→              ? message.content.slice(0, 50) + "..."
   860→              : message.content,
   861→        });
   862→      })
   863→      .catch(() => {
   864→        toast({
   865→          title: "Failed to copy",
   866→          description: "Could not access clipboard",
   867→          variant: "destructive",
   868→        });
   869→      });
   870→    setShowTapback(false);
   871→  }, [message.content]);
   872→
   873→  const handleDelete = useCallback(() => {
   874→    onDeleteMessage?.(message.id);
   875→    setShowTapback(false);
   876→    setShowActionSheet(false);
   877→  }, [message.id, onDeleteMessage]);
   878→
   879→  const handleActionSheetCopy = useCallback(() => {
   880→    navigator.clipboard
   881→      ?.writeText(message.content)
   882→      .then(() => {
   883→        toast({
   884→          title: "Copied to clipboard",
   885→          description:
   886→            message.content.length > 50
   887→              ? message.content.slice(0, 50) + "..."
   888→              : message.content,
   889→        });
   890→      })
   891→      .catch(() => {
   892→        toast({
   893→          title: "Failed to copy",
   894→          description: "Could not access clipboard",
   895→          variant: "destructive",
   896→        });
   897→      });
   898→    setShowActionSheet(false);
   899→  }, [message.content]);
   900→
   901→  const handleActionSheetReply = useCallback(() => {
   902→    onReplyTo?.(message);
   903→    setShowActionSheet(false);
   904→  }, [message, onReplyTo]);
   905→
   906→  const handleActionSheetForward = useCallback(() => {
   907→    toast({
   908→      title: "Forward",
   909→      description: "Forward coming soon!",
   910→    });
   911→    setShowActionSheet(false);
   912→  }, []);
   913→
   914→  const isFailed = message.status === "failed";
   915→  const isVoice = message.type === "voice";
   916→  const isText = !isVoice && message.type !== "image";
   917→  const isNew =
   918→    message.status === "sending" ||
   919→    Date.now() - new Date(message.created_at).getTime() < 2000;
   920→
   921→  const timestampText = formatDistanceToNow(
   922→    new Date(message.created_at || Date.now()),
   923→    { addSuffix: true },
   924→  );
   925→
   926→  return (
   927→    <>
   928→      <m.div
   929→        className={cn(
   930→          "relative flex w-full select-none gap-2.5 px-1.5 sm:px-3",
   931→          isOwn ? "justify-end" : "justify-start",
   932→        )}
   933→        initial={isNew ? { opacity: 0, y: 16, scale: 0.97 } : false}
   934→        animate={{ opacity: 1, y: 0, scale: 1 }}
   935→        transition={{ duration: 0.3, ease: [0.22, 1, 0.36, 1] }}
   936→      >
   937→        <AnimatePresence>
   938→          {showTapback && onReact && (
   939→            <TapbackDock
   940→              message={message}
   941→              isOwn={isOwn}
   942→              onReact={(emoji, add) => {
   943→                onReact(message.id, emoji, add);
   944→                setShowTapback(false);
   945→              }}
   946→              onReply={() => {
   947→                onReplyTo?.(message);
   948→                setShowTapback(false);
   949→              }}
   950→              onCopy={handleCopy}
   951→              onDelete={isOwn ? handleDelete : undefined}
   952→            />
   953→          )}
   954→        </AnimatePresence>
   955→
   956→        {/* Reply arrow indicator for swipe-to-reply */}
   957→        <m.div
   958→          className={cn(
   959→            "absolute top-1/2 z-10 -translate-y-1/2",
   960→            isOwn ? "right-1.5 sm:right-3" : "left-1.5 sm:left-3",
   961→          )}
   962→          animate={
   963→            showReplyArrow
   964→              ? { scale: 1.2, opacity: 1 }
   965→              : { scale: 0.5, opacity: 0 }
   966→          }
   967→          transition={{ duration: 0.2, ease: "easeOut" }}
   968→          style={{ pointerEvents: "none" }}
   969→        >
   970→          <div
   971→            className="flex items-center justify-center rounded-full p-1.5"
   972→            style={{ backgroundColor: "var(--app-accent)" }}
   973→          >
   974→            <Reply className="size-3.5 text-white" />
   975→          </div>
   976→        </m.div>
   977→
   978→        {!isOwn && (
   979→          <Avatar className="mt-auto size-8 shrink-0 ring-1 ring-muted">
   980→            {partnerAvatar && (
   981→              <AvatarImage src={partnerAvatar} alt={partnerName} />
   982→            )}
   983→            <AvatarFallback
   984→              className="text-xs font-medium"
   985→              style={{
   986→                background:
   987→                  "linear-gradient(to bottom right, var(--app-accent-lighter), var(--app-accent-light))",
   988→                color: "var(--app-accent-dark)",
   989→              }}
   990→            >
   991→              {partnerName.slice(0, 2).toUpperCase()}
   992→            </AvatarFallback>
   993→          </Avatar>
   994→        )}
   995→
   996→        <div
   997→          className={cn(
   998→            "flex max-w-[75%] flex-col gap-1 select-none",
   999→            isOwn
  1000→              ? "items-end mr-1.5 sm:mr-3"
  1001→              : "items-start ml-1.5 sm:ml-3",
  1002→          )}
  1003→        >
  1004→          <m.div
  1005→            ref={bubbleRef}
  1006→            drag="x"
  1007→            dragConstraints={{ left: 0, right: 120 }}
  1008→            dragElastic={0.3}
  1009→            transition={{ type: "spring", stiffness: 500, damping: 35 }}

  1015→            onDragEnd={(_, info) => {

  1023→            onPointerDown={handlePressStart}
  1024→            onPointerUp={handlePressEnd}
  1025→            onPointerLeave={handlePressEnd}
  1026→            onContextMenu={handleContextMenu}
  1027→            className={cn(
  1028→              "relative cursor-default select-none rounded-2xl px-4 py-2.5 text-sm leading-relaxed transition-shadow duration-200 active:scale-95",
  1029→              isOwn && "rounded-br-sm text-white",
  1030→              !isOwn &&
  1031→                "bg-muted text-foreground rounded-bl-sm shadow-sm dark:bg-zinc-800 dark:text-zinc-100",
  1032→              isFailed && "ring-2 ring-red-400/50",
  1033→              isVoice && "py-2",
  1034→              !isVoice &&
  1035→                message.type !== "image" &&
  1036→                "md:hover:shadow-md",

  1076→                <div className="min-w-0 flex-1">

  1080→                    highlight={searchHighlight}

  1088→                        onExpire={() => {

  1096→                      isNew &&
  1097→                        "animate-[timestamp-fade_0.4s_ease_0.3s_both]",

  1147→                "flex items-center gap-1.5 px-1 text-[10px] text-muted-foreground/60",

  1162→                  isNew && "animate-[timestamp-fade_0.4s_ease_0.3s_both]",

  1164→              >
  1165→                {timestampText}
  1166→              </span>
  1167→              {isOwn && <ReceiptIcon status={message.status} />}

  1170→
  1171→          {/* Reactions – overlapping bubble bottom edge */}

  1180→
  1181→          {isFailed && (

  1218→function formatDateSeparator(dateStr: string): string {
  1219→  const d = new Date(dateStr);
  1220→  const now = new Date();
  1221→  const today = new Date(now.getFullYear(), now.getMonth(), now.getDate());
  1222→  const yesterday = new Date(today);
  1223→  yesterday.setDate(yesterday.getDate() - 1);
  1224→  const msgDay = new Date(d.getFullYear(), d.getMonth(), d.getDate());
  1225→
  1226→  if (msgDay.getTime() === today.getTime()) return "Today";
  1227→  if (msgDay.getTime() === yesterday.getTime()) return "Yesterday";
  1228→  return d.toLocaleDateString(undefined, { month: "short", day: "numeric" });
  1229→}

  1437→
  1438→export function MessageFeed({
  1439→  messages,
  1440→  isLoading,
  1441→  isPartnerTyping,
  1442→  currentUserId,
  1443→  partnerName,
  1444→  partnerAvatar,
  1445→  backgroundStyle,
  1446→  onMarkAsRead,
  1447→  onVanishMessage,
  1448→  onReplyTo,
  1449→  onDeleteMessage,
  1450→  onReact,
  1451→  scrollToMessageId,
  1452→  onScrolledToMessage,
  1453→  searchHighlight,
  1454→  onClearSearchHighlight,
  1455→  loadMore,
  1456→  hasMore,
  1457→  loadingMore,
  1458→}: MessageFeedProps) {
  1459→  const parentRef = useRef<HTMLDivElement>(null);
  1460→  const isAutoScrollRef = useRef(true);
  1461→  const [showNewBtn, setShowNewBtn] = useState(false);
  1462→  const unreadRef = useRef(0);
  1463→  const prevCountRef = useRef(messages.length);
  1464→  const prevFirstIdRef = useRef(
  1465→    messages.length > 0 ? messages[0].id : null,
  1466→  );
  1467→  const prevScrollHeightRef = useRef(0);
  1468→
  1469→  const totalItems = useMemo(
  1470→    () => messages.length + (isPartnerTyping ? 1 : 0),
  1471→    [messages.length, isPartnerTyping],
  1472→  );
  1473→  const loadMoreRef = useRef(loadMore);
  1474→  loadMoreRef.current = loadMore;
  1475→
  1476→  const virtualizer = useVirtualizer({
  1477→    count: totalItems,
  1478→    getScrollElement: () => parentRef.current,
  1479→    estimateSize: () => ESTIMATED_ROW_HEIGHT,
  1480→    overscan: OVERSCAN,
  1481→    paddingStart: 16,
  1482→    paddingEnd: 16,
  1483→    gap: 12,
  1484→  });
  1485→
  1486→  const handleScroll = useCallback(
  1487→    (e: UIEvent<HTMLDivElement>) => {
  1488→      const el = e.currentTarget;
  1489→      const dist = el.scrollHeight - el.scrollTop - el.clientHeight;
  1490→      if (dist <= AUTO_SCROLL_THRESHOLD) {
  1491→        isAutoScrollRef.current = true;
  1492→        setShowNewBtn(false);
  1493→        unreadRef.current = 0;
  1494→      } else {
  1495→        isAutoScrollRef.current = false;
  1496→      }
  1497→      /* Trigger loadMore when scrolled near top */
  1498→      if (el.scrollTop < 300 && hasMore && !loadingMore) {
  1499→        loadMoreRef.current?.();
  1500→      }
  1501→    },
  1502→    [hasMore, loadingMore],
  1503→  );
  1504→
  1505→  const scrollToBottom = useCallback(
  1506→    (behavior: ScrollBehavior = "smooth") => {
  1507→      const el = parentRef.current;
  1508→      if (!el) return;
  1509→      el.scrollTo({ top: el.scrollHeight, behavior });
  1510→      isAutoScrollRef.current = true;
  1511→      setShowNewBtn(false);
  1512→      unreadRef.current = 0;
  1513→    },
  1514→    [],
  1515→  );
  1516→
  1517→  useEffect(() => {
  1518→    if (messages.length > 0)
  1519→      requestAnimationFrame(() => scrollToBottom("instant"));
  1520→  }, [isLoading, scrollToBottom]);
  1521→
  1522→  useEffect(() => {
  1523→    if (!onMarkAsRead) return;
  1524→    const unread = messages.filter(
  1525→      (m) => m.sender_id !== currentUserId && m.status !== "read",
  1526→    );
  1527→    if (unread.length > 0 && isAutoScrollRef.current)
  1528→      onMarkAsRead(unread.map((m) => m.id));
  1529→  }, [messages, currentUserId, onMarkAsRead]);
  1530→
  1531→  /* ---- Scroll to a specific message ----------------------------------- */
  1532→  useEffect(() => {
  1533→    if (!scrollToMessageId) return;
  1534→    const idx = messages.findIndex((m) => m.id === scrollToMessageId);
  1535→    if (idx === -1) {
  1536→      onScrolledToMessage?.();
  1537→      return;
  1538→    }
  1539→    virtualizer.scrollToIndex(idx, { align: "center" });
  1540→    onScrolledToMessage?.();
  1541→  }, [scrollToMessageId, messages, virtualizer, onScrolledToMessage]);
  1542→
  1543→  /* ---- Handle message count changes (new or prepended) ------------ */
  1544→  useEffect(() => {
  1545→    const prev = prevCountRef.current;
  1546→    const prevFirstId = prevFirstIdRef.current;
  1547→    prevCountRef.current = messages.length;
  1548→    prevFirstIdRef.current =
  1549→      messages.length > 0 ? messages[0].id : null;
  1550→
  1551→    if (messages.length <= prev) return;
  1552→
  1553→    /* Check if older messages were prepended (first message changed) */
  1554→    const firstId = messages[0]?.id;
  1555→    const wasPrepended = firstId !== prevFirstId && prevFirstId !== null;
  1556→
  1557→    if (wasPrepended) {
  1558→      /* Older messages prepended – preserve scroll position */
  1559→      const el = parentRef.current;
  1560→      if (el) {
  1561→        const oldHeight = prevScrollHeightRef.current;
  1562→        prevScrollHeightRef.current = el.scrollHeight;
  1563→        requestAnimationFrame(() => {
  1564→          const newHeight = el.scrollHeight;
  1565→          el.scrollTop += newHeight - oldHeight;
  1566→        });
  1567→      }
  1568→      return;
  1569→    }
  1570→
  1571→    /* New messages appended at the end */
  1572→    const last = messages[messages.length - 1];
  1573→    const own = last && last.sender_id === currentUserId;
  1574→    if (own) {
  1575→      requestAnimationFrame(() => scrollToBottom("instant"));
  1576→      return;
  1577→    }
  1578→    if (isAutoScrollRef.current)
  1579→      requestAnimationFrame(() => scrollToBottom("smooth"));
  1580→    else {
  1581→      unreadRef.current += messages.length - prev;
  1582→      setShowNewBtn(true);
  1583→    }
  1584→  }, [messages.length, currentUserId, scrollToBottom]);
  1585→
  1586→  /* Track scroll height for scroll preservation */
  1587→  useEffect(() => {
  1588→    const el = parentRef.current;
  1589→    if (!el) return;
  1590→    prevScrollHeightRef.current = el.scrollHeight;
  1591→  }, [messages.length]);
  1592→
  1593→  if (isLoading) return <MessageListSkeleton />;
  1594→  if (messages.length === 0 && !isPartnerTyping)
  1595→    return (
  1596→      <EmptyState partnerName={partnerName} partnerAvatar={partnerAvatar} />
  1597→    );
  1598→
  1599→  const virtualItems = virtualizer.getVirtualItems();
  1600→
  1601→  return (
  1602→    <div
  1603→      className="relative h-full w-full overflow-hidden"
  1604→      style={backgroundStyle}
  1605→    >
  1606→      <div
  1607→        ref={parentRef}
  1608→        onScroll={handleScroll}
  1609→        onClick={() => onClearSearchHighlight?.()}
  1610→        className="h-full w-full select-none overflow-y-auto overscroll-contain"
  1611→        style={{ WebkitOverflowScrolling: "touch" }}
  1612→      >
  1613→        <div
  1614→          style={{
  1615→            height: `${virtualizer.getTotalSize()}px`,
  1616→            width: "100%",
  1617→            position: "relative",
  1618→          }}
  1619→        >
  1620→          {/* Loading more spinner */}
  1621→          {loadingMore && (
  1622→            <div
  1623→              style={{
  1624→                position: "absolute",
  1625→                top: 0,
  1626→                left: 0,
  1627→                width: "100%",
  1628→                height: "48px",
  1629→                zIndex: 5,
  1630→              }}
  1631→            >
  1632→              <div className="flex items-center justify-center py-3">
  1633→                <svg
  1634→                  className="size-5 animate-spin"
  1635→                  viewBox="0 0 24 24"
  1636→                  fill="none"
  1637→                  aria-label="Loading more messages"
  1638→                  style={{ color: "var(--app-accent-light)" }}
  1639→                >
  1640→                  <circle
  1641→                    className="opacity-25"
  1642→                    cx="12"
  1643→                    cy="12"
  1644→                    r="10"
  1645→                    stroke="currentColor"
  1646→                    strokeWidth="4"
  1647→                  />
  1648→                  <path
  1649→                    className="opacity-75"
  1650→                    fill="currentColor"
  1651→                    d="M4 12a8 8 0 018-8V0C5.373 0 0 5.373 0 12h4z"
  1652→                  />
  1653→                </svg>
  1654→              </div>
  1655→            </div>
  1656→          )}
  1657→          {virtualItems.map((vi) => {
  1658→            const isTyping = vi.index === messages.length;
  1659→            const msg = isTyping ? null : messages[vi.index];
  1660→            // Check if we need a date separator before this message
  1661→            const showDateSep =
  1662→              msg &&
  1663→              vi.index > 0 &&
  1664→              messages[vi.index - 1] &&
  1665→              (() => {
  1666→                const prev = new Date(messages[vi.index - 1].created_at);
  1667→                const curr = new Date(msg.created_at);
  1668→                return prev.toDateString() !== curr.toDateString();
  1669→              })();
  1670→            const showFirstDateSep = msg && vi.index === 0;
  1671→            return (
  1672→              <div
  1673→                key={vi.key}
  1674→                data-index={vi.index}
  1675→                ref={virtualizer.measureElement}
  1676→                style={{
  1677→                  position: "absolute",
  1678→                  top: 0,
  1679→                  left: 0,
  1680→                  width: "100%",
  1681→                  transform: `translateY(${vi.start}px)`,
  1682→                }}
  1683→              >
  1684→                {isTyping ? (
  1685→                  <TypingIndicator
  1686→                    partnerName={partnerName}
  1687→                    partnerAvatar={partnerAvatar}
  1688→                  />
  1689→                ) : msg ? (
  1690→                  <>
  1691→                    {(showDateSep || showFirstDateSep) && (
  1692→                      <DateSeparator date={msg.created_at} />
  1693→                    )}
  1694→                    <MessageBubble
  1695→                      message={msg}
  1696→                      isOwn={msg.sender_id === currentUserId}
  1697→                      partnerName={partnerName}
  1698→                      partnerAvatar={partnerAvatar}
  1699→                      currentUserId={currentUserId}
  1700→                      onReplyTo={onReplyTo}
  1701→                      onReact={onReact}
  1702→                      onDeleteMessage={onDeleteMessage}
  1703→                      searchHighlight={searchHighlight}
  1704→                    />
  1705→                  </>
  1706→                ) : null}
  1707→              </div>
  1708→            );
  1709→          })}
  1710→        </div>
  1711→      </div>
  1712→      {showNewBtn && unreadRef.current > 0 && (
  1713→        <NewMessagesButton
  1714→          count={unreadRef.current}
  1715→          onClick={() => scrollToBottom("smooth")}
  1716→        />
  1717→      )}
  1718→    </div>
  1719→  );
  1720→}
