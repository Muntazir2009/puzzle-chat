    46→function formatFileSize(bytes: number): string {
    47→  if (bytes < 1024) return `${bytes} B`;
    48→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
    49→  return `${(bytes / (1024 * 1024)).toFixed(1)} MB`;
    50→}
    51→
    52→function formatLastSeen(lastSeen: string | null): string {
    53→  if (!lastSeen) return "Offline";
    54→  const diffMs = Date.now() - new Date(lastSeen).getTime();
    55→  const diffSec = Math.floor(diffMs / 1000);
    56→  if (diffSec < 60) return "Seen just now";
    57→  const diffMin = Math.floor(diffSec / 60);
    58→  if (diffMin < 60) return `Seen ${diffMin}m ago`;
    59→  const diffHr = Math.floor(diffMin / 60);
    60→  if (diffHr < 24) return `Seen ${diffHr}h ago`;
    61→  const diffDay = Math.floor(diffHr / 24);
    62→  if (diffDay < 7) return `Seen ${diffDay}d ago`;
    63→  return `Seen ${new Date(lastSeen).toLocaleDateString()}`;
    64→}

    82→function SearchPanel({
    83→  conversationId,
    84→  onResultClick,
    85→  onClose,
    86→}: {
    87→  conversationId: string;
    88→  onResultClick: (messageId: string, query: string) => void;
    89→  onClose: () => void;
    90→}) {
    91→  const [query, setQuery] = useState("");
    92→  const [results, setResults] = useState<ChatMessage[]>([]);
    93→  const [isSearching, setIsSearching] = useState(false);
    94→  const inputRef = useRef<HTMLInputElement>(null);
    95→  const debounceRef = useRef<ReturnType<typeof setTimeout> | undefined>(undefined);
    96→
    97→  /* Auto-focus on open */
    98→  useEffect(() => {
    99→    requestAnimationFrame(() => inputRef.current?.focus());
   100→  }, []);
   101→
   102→  /* Escape key to close */
   103→  useEffect(() => {
   104→    function handleKeyDown(e: KeyboardEvent) {
   105→      if (e.key === "Escape") onClose();
   106→    }
   107→    window.addEventListener("keydown", handleKeyDown);
   108→    return () => window.removeEventListener("keydown", handleKeyDown);
   109→  }, [onClose]);
   110→
   111→  /* Debounced search */
   112→  useEffect(() => {
   113→    if (debounceRef.current) clearTimeout(debounceRef.current);
   114→    if (!query.trim()) { setResults([]); setIsSearching(false); return; }
   115→    setIsSearching(true);
   116→    debounceRef.current = setTimeout(async () => {
   117→      try {
   118→        const res = await fetch(`/api/messages/search?conversation_id=${conversationId}&q=${encodeURIComponent(query)}`);
   119→        if (res.ok) {
   120→          const data: ChatMessage[] = await res.json();
   121→          setResults(data);
   122→        }
   123→      } catch (_err) {
   124→        /* silent */
   125→      } finally {
   126→        setIsSearching(false);
   127→      }
   128→    }, 300);
   129→    return () => { if (debounceRef.current) clearTimeout(debounceRef.current); };
   130→  }, [query, conversationId]);
   131→
   132→  /** Highlight matching text in a snippet */
   133→  function highlightedSnippet(text: string, q: string, maxLen = 60): React.ReactNode {
   134→    if (!q) return text.length > maxLen ? text.slice(0, maxLen) + "..." : text;
   135→    const idx = text.toLowerCase().indexOf(q.toLowerCase());
   136→    if (idx === -1) return text.length > maxLen ? text.slice(0, maxLen) + "..." : text;
   137→    const before = text.slice(Math.max(0, idx - 20), idx);
   138→    const match = text.slice(idx, idx + q.length);
   139→    const after = text.slice(idx + q.length, idx + q.length + maxLen - 20 - q.length);
   140→    return (
   141→      <>
   142→        {before.length > 0 && idx > 20 && "..."}{before}
   143→        <mark className="bg-yellow-300/60 dark:bg-yellow-400/40 rounded-sm px-0.5">{match}</mark>
   144→        {after}{text.length > idx + q.length + maxLen - 20 - q.length && "..."}
   145→      </>
   146→    );
   147→  }
   148→
   149→  return (
   150→    <div
   151→      className="overflow-hidden border-b bg-background/80 backdrop-blur-xl"
   152→    >
   153→      <div className="flex items-center gap-2 px-4 py-2.5">
   154→        <Search className="size-4 shrink-0 text-muted-foreground" />
   155→        <input
   156→          ref={inputRef}
   157→          type="text"
   158→          value={query}
   159→          onChange={(e) => setQuery(e.target.value)}
   160→          placeholder="Search messages..."
   161→          className="flex-1 bg-transparent text-sm outline-none placeholder:text-muted-foreground"
   162→          aria-label="Search messages"
   163→        />
   164→        {isSearching ? (
   165→          <Loader2 className="size-4 shrink-0 animate-spin text-muted-foreground" />
   166→        ) : (
   167→          <button
   168→            type="button"
   169→            onClick={() => { setQuery(""); onClose(); }}
   170→            className="flex size-6 items-center justify-center rounded-full text-muted-foreground transition-colors hover:bg-muted"
   171→            aria-label="Close search"
   172→          >
   173→            <X className="size-3.5" />
   174→          </button>
   175→        )}
   176→      </div>
   177→
   178→      {/* Results list */}
   179→      {results.length > 0 && (
   180→        <div className="max-h-[200px] overflow-y-auto border-t px-2 pb-2">
   181→          {results.map((msg) => (
   182→            <button
   183→              key={msg.id}
   184→              type="button"
   185→              onClick={() => onResultClick(msg.id, query)}
   186→              className="flex w-full flex-col gap-0.5 rounded-lg px-3 py-2.5 text-left transition-colors hover:bg-muted/50"
   187→            >
   188→              <p className="truncate text-sm text-foreground">
   189→                {highlightedSnippet(msg.content, query)}
   190→              </p>
   191→              <span className="text-[11px] text-muted-foreground">
   192→                {formatDistanceToNow(new Date(msg.created_at), { addSuffix: true })}
   193→              </span>
   194→            </button>
   195→          ))}
   196→        </div>
   197→      )}
   198→
   199→      {query.trim() && !isSearching && results.length === 0 && (
   200→        <div className="border-t px-4 py-3 text-center text-xs text-muted-foreground">
   201→          No results found
   202→        </div>
   203→      )}
   204→    </div>
   205→  );
   206→}

   366→export function ChatLayout({ currentUserId, otherUserId, conversationId, partner, initialMessages }: ChatLayoutProps) {
   367→  const { messages, isLoading, isPartnerTyping, partnerStatus, sendMessage, onTyping, markAsRead, vanishMessage, deleteMessage, sendReaction, loadMore, hasMore, loadingMore } =
   368→    useChat({ currentUserId, otherUserId, conversationId, initialMessages });

   370→  const [draft, setDraft] = useState("");
   371→  const [sendBtnKey, setSendBtnKey] = useState(0);
   372→  const [isSending, setIsSending] = useState(false);
   373→  const [vanishMode, setVanishMode] = useState(false);
   374→  const [replyTo, setReplyTo] = useState<ChatMessage | null>(null);
   375→  const [ephemeralOpen, setEphemeralOpen] = useState(false);
   376→  const [ephemeralSeconds, setEphemeralSeconds] = useState<number | null>(null);
   377→  const [infoPanelOpen, setInfoPanelOpen] = useState(false);
   378→  const [searchOpen, setSearchOpen] = useState(false);
   379→  const [searchHighlight, setSearchHighlight] = useState<string | undefined>(undefined);
   380→  const [scrollToMessageId, setScrollToMessageId] = useState<string | null>(null);
   381→  const inputRef = useRef<HTMLTextAreaElement>(null);
   382→  const viewport = useVisualViewport();
   383→
   384→  /* Background theme */
   385→  const { themeId, theme, selectTheme } = useChatBackground();
   386→
   387→  /* Voice */
   388→  const voice = useVoiceRecorder();
   389→  const [showVoiceWaveform, setShowVoiceWaveform] = useState(false);
   390→
   391→  /* Attachment */
   392→  const [attachmentFile, setAttachmentFile] = useState<File | null>(null);
   393→  const [attachmentPreview, setAttachmentPreview] = useState<string | null>(null);
   394→  const [isUploading, setIsUploading] = useState(false);
   395→  const fileInputRef = useRef<HTMLInputElement>(null);
   396→
   397→  const containerStyle = viewport.height > 0 ? ({ height: `${viewport.height}px` } as React.CSSProperties) : ({ height: "100dvh" } as React.CSSProperties);
   398→  const kbOffset = viewport.isKeyboardVisible && viewport.offsetTop > 0 ? ({ marginTop: `-${viewport.offsetTop}px` } as React.CSSProperties) : undefined;
   399→
   400→  useEffect(() => { inputRef.current?.focus(); }, []);
   401→
   402→  /* Show voice waveform when recording */
   403→  useEffect(() => { setShowVoiceWaveform(voice.isRecording); }, [voice.isRecording]);
   404→
   405→  /* Global keyboard shortcuts */
   406→  useEffect(() => {
   407→    function handleGlobalKey(e: KeyboardEvent) {
   408→      if (e.key === "Escape") {
   409→        if (replyTo) { setReplyTo(null); e.preventDefault(); }
   410→        else if (ephemeralOpen) { setEphemeralOpen(false); e.preventDefault(); }
   411→        else if (infoPanelOpen) { setInfoPanelOpen(false); e.preventDefault(); }
   412→        else if (searchOpen) { setSearchOpen(false); e.preventDefault(); }
   413→      }
   414→    }
   415→    window.addEventListener("keydown", handleGlobalKey);
   416→    return () => window.removeEventListener("keydown", handleGlobalKey);
   417→  }, [replyTo, ephemeralOpen, infoPanelOpen, searchOpen]);
   418→
   419→  /* Derived values — MUST be declared before any useCallback that
   420→     references them in its dependency array to avoid TDZ errors
   421→     in the production webpack bundle. */
   422→  const hasAttachment = attachmentFile !== null;
   423→  const hasText = draft.trim().length > 0;
   424→
   425→  const handleSubmit = useCallback(async (e: FormEvent) => {
   426→    e.preventDefault();
   427→    /* If an attachment is selected, upload it instead of sending text */
   428→    if (hasAttachment && !hasText) {
   429→      await uploadAttachment();
   430→      return;
   431→    }
   432→    /* If there's both text and an attachment, send text first, then upload */
   433→    const trimmed = draft.trim();
   434→    if (!trimmed || isSending) return;
   435→    setIsSending(true);
   436→    const opts: SendMessageOptions = { vanish_mode: vanishMode, ephemeral_seconds: ephemeralSeconds, reply_to_id: replyTo?.id ?? null };
   437→    setDraft(""); setReplyTo(null);
   438→    if (inputRef.current) inputRef.current.style.height = "auto";
   439→    try {
   440→      await sendMessage(trimmed, opts);
   441→      setSendBtnKey((k) => k + 1);
   442→      /* Upload attachment after text is sent */
   443→      if (hasAttachment) await uploadAttachment();
   444→    }
   445→    finally { setIsSending(false); requestAnimationFrame(() => inputRef.current?.focus()); }
   446→  }, [draft, isSending, sendMessage, vanishMode, ephemeralSeconds, replyTo, hasAttachment, uploadAttachment]);
   447→
   448→  const handleVoiceSend = useCallback(async () => {
   449→    const result = await voice.stopRecording();
   450→    if (!result || result.duration < 1) return;
   451→    const blob = result.blob;
   452→    const url = URL.createObjectURL(blob);
   453→    await sendMessage(url, { type: "voice", voice_duration: result.duration, waveform_data: result.amplitudes, ephemeral_seconds: ephemeralSeconds });
   454→    URL.revokeObjectURL(url);
   455→  }, [voice, sendMessage, ephemeralSeconds]);
   456→
   457→  const handleKeyDown = useCallback((e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   458→    if (e.key === "Enter" && !e.shiftKey) { e.preventDefault(); const form = e.currentTarget.closest("form"); if (form) form.requestSubmit(); }
   459→  }, []);
   460→
   461→  const handleInputChange = useCallback((e: React.ChangeEvent<HTMLTextAreaElement>) => { setDraft(e.target.value); onTyping(); if (searchHighlight) setSearchHighlight(undefined); }, [onTyping, searchHighlight]);
   462→  const handleInput = useCallback((e: React.FormEvent<HTMLTextAreaElement>) => { const el = e.currentTarget; el.style.height = "auto"; el.style.height = `${Math.min(el.scrollHeight, 128)}px`; }, []);
   463→
   464→  const canSend = hasText || hasAttachment;
   465→
   466→  /* Search result click → scroll to message & apply highlight */
   467→  const handleSearchResultClick = useCallback((messageId: string, query: string) => {
   468→    setScrollToMessageId(messageId);
   469→    setSearchHighlight(query);
   470→    setSearchOpen(false);
   471→  }, []);
   472→
   473→  const handleClearSearchHighlight = useCallback(() => {
   474→    setSearchHighlight(undefined);
   475→  }, []);
   476→
   477→  const handleScrolledToMessage = useCallback(() => {
   478→    setScrollToMessageId(null);
   479→  }, []);
   480→
   481→  /* ---- Attachment handlers ---------------------------------------- */
   482→  const handleFileSelect = useCallback((e: ChangeEvent<HTMLInputElement>) => {
   483→    const file = e.target.files?.[0] ?? null;
   484→    if (!file) return;
   485→    if (file.size > MAX_ATTACHMENT_SIZE) {
   486→      toast({ title: "File too large", description: "Maximum size is 10 MB.", variant: "destructive" });
   487→      e.target.value = "";
   488→      return;
   489→    }
   490→    setAttachmentFile(file);
   491→    /* Generate thumbnail preview for images */
   492→    if (IMAGE_MIME_TYPES.has(file.type)) {
   493→      const url = URL.createObjectURL(file);
   494→      setAttachmentPreview(url);
   495→    } else {
   496→      setAttachmentPreview(null);
   497→    }
   498→    /* Reset input so the same file can be re-selected */
   499→    e.target.value = "";
   500→  }, []);
   501→
   502→  const clearAttachment = useCallback(() => {
   503→    if (attachmentPreview) URL.revokeObjectURL(attachmentPreview);
   504→    setAttachmentFile(null);
   505→    setAttachmentPreview(null);
   506→  }, [attachmentPreview]);
   507→
   508→  const uploadAttachment = useCallback(async () => {
   509→    if (!attachmentFile || isUploading) return;
   510→    setIsUploading(true);
   511→    try {
   512→      const fd = new FormData();
   513→      fd.append("file", attachmentFile);
   514→      fd.append("conversation_id", conversationId);
   515→      if (replyTo?.id) fd.append("reply_to_id", replyTo.id);
   516→      if (vanishMode) fd.append("vanish_mode", "true");
   517→      if (ephemeralSeconds) fd.append("ephemeral_seconds", String(ephemeralSeconds));
   518→
