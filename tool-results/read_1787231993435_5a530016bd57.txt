  1439→function formatDateSeparator(dateStr: string): string {
  1440→  const d = new Date(dateStr);
  1441→  const now = new Date();
  1442→  const today = new Date(now.getFullYear(), now.getMonth(), now.getDate());
  1443→  const yesterday = new Date(today);
  1444→  yesterday.setDate(yesterday.getDate() - 1);
  1445→  const msgDay = new Date(d.getFullYear(), d.getMonth(), d.getDate());
  1446→
  1447→  if (msgDay.getTime() === today.getTime()) return "Today";
  1448→  if (msgDay.getTime() === yesterday.getTime()) return "Yesterday";
  1449→  return d.toLocaleDateString(undefined, { month: "short", day: "numeric" });
  1450→}

  1733→      requestAnimationFrame(() => scrollToBottom("instant"));
  1734→  }, [isLoading, scrollToBottom]);
  1735→
  1736→  useEffect(() => {
  1737→    if (!onMarkAsRead) return;
  1738→    const unread = messages.filter(
  1739→      (m) => m.sender_id !== currentUserId && m.status !== "read",
  1740→    );
  1741→    if (unread.length > 0 && isAutoScrollRef.current)
  1742→      onMarkAsRead(unread.map((m) => m.id));
  1743→  }, [messages, currentUserId, onMarkAsRead]);
  1744→
  1745→  /* ---- Scroll to a specific message ----------------------------------- */
  1746→  useEffect(() => {
  1747→    if (!scrollToMessageId) return;
  1748→    // Find which group contains the target message
  1749→    const groupIdx = groups.findIndex((g) =>
  1750→      g.messages.some((m) => m.id === scrollToMessageId),
  1751→    );
  1752→    if (groupIdx === -1) {
  1753→      onScrolledToMessage?.();
  1754→      return;
  1755→    }
  1756→    virtualizer.scrollToIndex(groupIdx, { align: "center" });
  1757→    onScrolledToMessage?.();
  1758→  }, [scrollToMessageId, groups, virtualizer, onScrolledToMessage]);
  1759→
  1760→  /* ---- Handle message count changes (new or prepended) ------------ */
  1761→  useEffect(() => {
  1762→    const prev = prevCountRef.current;
  1763→    const prevFirstId = prevFirstIdRef.current;
  1764→    prevCountRef.current = messages.length;
  1765→    prevFirstIdRef.current =
  1766→      messages.length > 0 ? messages[0].id : null;
  1767→
  1768→    if (messages.length <= prev) return;
  1769→
  1770→    /* Check if older messages were prepended (first message changed) */
  1771→    const firstId = messages[0]?.id;
  1772→    const wasPrepended = firstId !== prevFirstId && prevFirstId !== null;
  1773→
  1774→    if (wasPrepended) {
  1775→      /* Older messages prepended – preserve scroll position */
  1776→      const el = parentRef.current;
  1777→      if (el) {
  1778→        const oldHeight = prevScrollHeightRef.current;
  1779→        prevScrollHeightRef.current = el.scrollHeight;
  1780→        requestAnimationFrame(() => {
  1781→          const newHeight = el.scrollHeight;
  1782→          el.scrollTop += newHeight - oldHeight;
  1783→        });
  1784→      }
  1785→      return;
  1786→    }
  1787→
  1788→    /* New messages appended at the end */
  1789→    const last = messages[messages.length - 1];
  1790→    const own = last && last.sender_id === currentUserId;
  1791→    if (own) {
  1792→      requestAnimationFrame(() => scrollToBottom("instant"));
  1793→      return;
  1794→    }
  1795→    if (isAutoScrollRef.current)
  1796→      requestAnimationFrame(() => scrollToBottom("smooth"));
  1797→    else {
  1798→      unreadRef.current += messages.length - prev;
  1799→      setShowNewBtn(true);
  1800→    }
  1801→  }, [messages.length, currentUserId, scrollToBottom]);
  1802→
  1803→  /* Track scroll height for scroll preservation */
  1804→  useEffect(() => {
  1805→    const el = parentRef.current;
  1806→    if (!el) return;
  1807→    prevScrollHeightRef.current = el.scrollHeight;
  1808→  }, [messages.length]);
  1809→
  1810→  if (isLoading) return <MessageListSkeleton />;
  1811→  if (messages.length === 0 && !isPartnerTyping)
  1812→    return (
  1813→      <EmptyState partnerName={partnerName} partnerAvatar={partnerAvatar} />
  1814→    );
  1815→
  1816→  const virtualItems = virtualizer.getVirtualItems();
  1817→
  1818→  return (
  1819→    <div
  1820→      className="relative h-full w-full overflow-hidden"
  1821→      style={backgroundStyle}
  1822→    >
  1823→      <div
  1824→        ref={parentRef}
  1825→        onScroll={handleScroll}
  1826→        onClick={(e) => {
  1827→          onClearSearchHighlight?.();
  1828→          // Close menus if click is directly on the scroll container or background
  1829→          const target = e.target as HTMLElement;
  1830→          if (target === e.currentTarget || target.closest('.message-bubble-row') === null) {
  1831→            setActiveActionId(null);
  1832→            setActiveTapbackId(null);
  1833→          }
  1834→        }}
  1835→        className="h-full w-full select-none overflow-y-auto overscroll-contain"
  1836→        style={{ WebkitOverflowScrolling: "touch" }}
  1837→      >
  1838→        <div
  1839→          style={{
  1840→            height: `${virtualizer.getTotalSize()}px`,
  1841→            width: "100%",
  1842→            position: "relative",
  1843→          }}
  1844→        >
  1845→          {/* Loading more spinner */}
  1846→          {loadingMore && (
  1847→            <div
  1848→              style={{
  1849→                position: "absolute",
  1850→                top: 0,
  1851→                left: 0,
  1852→                width: "100%",
  1853→                height: "48px",
  1854→                zIndex: 5,
  1855→              }}
  1856→            >
  1857→              <div className="flex items-center justify-center py-3">
  1858→                <svg
  1859→                  className="size-5 animate-spin"
  1860→                  viewBox="0 0 24 24"
  1861→                  fill="none"
  1862→                  aria-label="Loading more messages"
  1863→                  style={{ color: "var(--app-accent-light)" }}
  1864→                >
  1865→                  <circle
  1866→                    className="opacity-25"
  1867→                    cx="12"
  1868→                    cy="12"
  1869→                    r="10"
  1870→                    stroke="currentColor"
  1871→                    strokeWidth="4"
  1872→                  />
  1873→                  <path
  1874→                    className="opacity-75"
  1875→                    fill="currentColor"
  1876→                    d="M4 12a8 8 0 018-8V0C5.373 0 0 5.373 0 12h4z"
  1877→                  />
  1878→                </svg>
  1879→              </div>
  1880→            </div>
  1881→          )}
  1882→          {virtualItems.map((vi) => {
  1883→            const isTyping = vi.index === groups.length;
  1884→            const group = isTyping ? null : groups[vi.index];
  1885→
  1886→            // Determine if we need a date separator before this group
  1887→            let showDateSep = false;
  1888→            if (group) {
  1889→              const firstMsg = group.messages[0];
  1890→              if (vi.index === 0) {
  1891→                showDateSep = true;
  1892→              } else {
  1893→                const prevGroup = groups[vi.index - 1];
  1894→                const prevLastMsg = prevGroup.messages[prevGroup.messages.length - 1];
  1895→                const prevDate = new Date(prevLastMsg.created_at);
  1896→                const currDate = new Date(firstMsg.created_at);
  1897→                showDateSep = prevDate.toDateString() !== currDate.toDateString();
  1898→              }
  1899→            }
  1900→
  1901→            return (
  1902→              <div
  1903→                key={vi.key}
  1904→                data-index={vi.index}
  1905→                ref={virtualizer.measureElement}
  1906→                style={{
  1907→                  position: "absolute",
  1908→                  top: 0,
  1909→                  left: 0,
  1910→                  width: "100%",
  1911→                  transform: `translateY(${vi.start}px)`,
  1912→                }}
  1913→              >
  1914→                {isTyping ? (
  1915→                  <TypingIndicator
  1916→                    partnerName={partnerName}
  1917→                    partnerAvatar={partnerAvatar}
  1918→                  />
  1919→                ) : group ? (
  1920→                  <>
  1921→                    {showDateSep && (
  1922→                      <DateSeparator date={group.messages[0].created_at} />
  1923→                    )}
  1924→                    <MessageGroup
  1925→                      messages={group.messages}
  1926→                      currentUserId={currentUserId}
  1927→                      currentUserName={currentUserName}
  1928→                      currentUserAvatar={currentUserAvatar}
  1929→                      partnerName={partnerName}
  1930→                      partnerAvatar={partnerAvatar}
  1931→                      onReplyTo={onReplyTo}
  1932→                      onReact={onReact}
  1933→                      onDeleteMessage={onDeleteMessage}
  1934→                      searchHighlight={searchHighlight}
  1935→                      activeActionId={activeActionId}
  1936→                      onOpenAction={setActiveActionId}
  1937→                      activeTapbackId={activeTapbackId}
  1938→                      onOpenTapback={setActiveTapbackId}
  1939→                    />
  1940→                  </>
  1941→                ) : null}
  1942→              </div>
  1943→            );
  1944→          })}
  1945→        </div>
  1946→      </div>
  1947→      {showNewBtn && unreadRef.current > 0 && (
  1948→        <NewMessagesButton
  1949→          count={unreadRef.current}
  1950→          onClick={() => scrollToBottom("smooth")}
  1951→        />
  1952→      )}
  1953→    </div>
  1954→  );
  1955→}
