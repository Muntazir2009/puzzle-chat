    53→/*  Double-tap hook                                                      */
    54→/* ------------------------------------------------------------------ */
    55→
    56→function useDoubleTap(
    57→  onDoubleTap: () => void,
    58→  delay = 300,
    59→) {
    60→  const lastTapRef = useRef(0);
    61→  return useCallback((): boolean => {
    62→    const now = Date.now();
    63→    if (now - lastTapRef.current < delay) {
    64→      lastTapRef.current = 0;
    65→      // Haptic feedback on double-tap
    66→      if (typeof navigator !== "undefined" && "vibrate" in navigator) {
    67→        navigator.vibrate(10);
    68→      }
    69→      onDoubleTap();
    70→      return true; // is double tap
    71→    }
    72→    lastTapRef.current = now;
    73→    return false; // is single tap
    74→  }, [onDoubleTap, delay]);
    75→}

   148→/*  Ephemeral Timer (circular SVG countdown)                           */
   149→/* ------------------------------------------------------------------ */
   150→
   151→function EphemeralTimer({
   152→  seconds,
   153→  onExpire,
   154→}: {
   155→  seconds: number;
   156→  onExpire: () => void;
   157→}) {
   158→  const [remaining, setRemaining] = useState(seconds);
   159→  const intervalRef = useRef<ReturnType<typeof setInterval> | undefined>(
   160→    undefined,
   161→  );
   162→
   163→  useEffect(() => {
   164→    intervalRef.current = setInterval(() => {
   165→      setRemaining((r) => {
   166→        if (r <= 1) {
   167→          clearInterval(intervalRef.current);
   168→          onExpire();
   169→          return 0;
   170→        }
   171→        return r - 1;
   172→      });
   173→    }, 1000);
   174→    return () => clearInterval(intervalRef.current);
   175→  }, [seconds, onExpire]);
   176→
   177→  const pct = remaining / seconds;
   178→  const circ = 2 * Math.PI * 7;
   179→  const offset = circ * (1 - pct);
   180→
   181→  return (
   182→    <svg
   183→      className="size-3.5 shrink-0 ephemeral-timer-pulse"
   184→      viewBox="0 0 18 18"
   185→      aria-label={`${remaining}s remaining`}
   186→    >
   187→      <circle
   188→        cx="9"
   189→        cy="9"
   190→        r="7"
   191→        fill="none"
   192→        stroke="currentColor"
   193→        strokeWidth="1.5"
   194→        className="text-white/20"
   195→      />
   196→      <circle
   197→        cx="9"
   198→        cy="9"
   199→        r="7"
   200→        fill="none"
   201→        stroke="currentColor"
   202→        strokeWidth="1.5"
   203→        strokeDasharray={circ}
   204→        strokeDashoffset={offset}
   205→        strokeLinecap="round"
   206→        style={{
   207→          color: "var(--app-accent-lighter)",
   208→          transform: "rotate(-90deg)",
   209→          transformOrigin: "50% 50%",
   210→          transition: "stroke-dashoffset 1s linear",
   211→        }}
   212→      />
   213→    </svg>
   214→  );
   215→}

   427→
   428→function VoiceBubble({
   429→  message,
   430→  isOwn,
   431→  waveform,
   432→}: {
   433→  message: ChatMessage;
   434→  isOwn: boolean;
   435→  waveform: number[];
   436→}) {
   437→  const audioRef = useRef<HTMLAudioElement>(null);
   438→  const [playing, setPlaying] = useState(false);
   439→  const [progress, setProgress] = useState(0);
   440→  const [speed, setSpeed] = useState(1);
   441→  const speeds = [1, 1.5, 2];
   442→  const duration = message.voice_duration ?? 0;
   443→
   444→  useEffect(() => {
   445→    const audio = audioRef.current;
   446→    if (!audio) return;
   447→    const onTime = () => {
   448→      if (audio.duration) setProgress(audio.currentTime / audio.duration);
   449→    };
   450→    const onEnd = () => setPlaying(false);
   451→    audio.addEventListener("timeupdate", onTime);
   452→    audio.addEventListener("ended", onEnd);
   453→    return () => {
   454→      audio.removeEventListener("timeupdate", onTime);
   455→      audio.removeEventListener("ended", onEnd);
   456→    };
   457→  }, []);
   458→
   459→  const toggle = useCallback(() => {
   460→    const audio = audioRef.current;
   461→    if (!audio) return;
   462→    if (playing) {
   463→      audio.pause();
   464→      setPlaying(false);
   465→    } else {
   466→      audio.playbackRate = speed;
   467→      audio.play().catch(() => {});
   468→      setPlaying(true);
   469→    }
   470→  }, [playing, speed]);
   471→
   472→  const cycleSpeed = useCallback(() => {
   473→    setSpeed((s) => {
   474→      const idx = speeds.indexOf(s);
   475→      const next = speeds[(idx + 1) % speeds.length];
   476→      if (audioRef.current) audioRef.current.playbackRate = next;
   477→      return next;
   478→    });
   479→  }, []);
   480→
   481→  /* Build waveform bars */
   482→  const bars = useMemo(() => {
   483→    const count = Math.min(waveform.length, 40);
   484→    const step = Math.max(1, Math.floor(waveform.length / count));
   485→    return Array.from({ length: count }, (_, i) => waveform[i * step] ?? 0.1);
   486→  }, [waveform]);
   487→
   488→  const progressIdx = Math.floor(progress * bars.length);
   489→
   490→  return (
   491→    <div className={cn("flex items-center gap-3", isOwn ? "flex-row-reverse" : "")}>
   492→      <button
   493→        type="button"
   494→        onClick={toggle}
   495→        className={cn(
   496→          "flex size-9 shrink-0 items-center justify-center rounded-full transition-all duration-150 active:scale-95",
   497→          isOwn
   498→            ? "bg-white/20 hover:bg-white/30"
   499→            : "bg-zinc-600 hover:bg-[var(--app-accent-subtle)]",
   500→        )}
   501→        aria-label={playing ? "Pause" : "Play"}
   502→      >
   503→        {playing ? (
   504→          <Pause className="size-3.5 text-white" />
   505→        ) : (
   506→          <Play className="size-3.5 text-white ml-0.5" />
   507→        )}
   508→      </button>
   509→      <div className="flex items-center gap-[2px]" style={{ height: 32 }}>
   510→        {bars.map((amp, i) => (
   511→          <div
   512→            key={i}
   513→            className="w-[2.5px] rounded-full transition-all duration-100"
   514→            style={{
   515→              height: `${Math.max(4, amp * 32)}px`,
   516→              backgroundColor:
   517→                i < progressIdx
   518→                  ? isOwn
   519→                    ? "rgba(255,255,255,1)"
   520→                    : "var(--app-accent-light)"
   521→                  : isOwn
   522→                    ? "rgba(255,255,255,0.5)"
   523→                    : "rgba(161,161,170,0.7)",
   524→            }}
   525→          />
   526→        ))}
   527→      </div>
   528→      <button
   529→        type="button"
   530→        onClick={cycleSpeed}
   531→        className={cn(
   532→          "shrink-0 text-[11px] font-semibold tabular-nums transition-colors hover:opacity-80 active:scale-95",
   533→          isOwn ? "text-white/70" : "text-zinc-400",
   534→        )}
   535→      >
   536→        {speed}x
   537→      </button>
   538→      <span
   539→        className={cn(
   540→          "shrink-0 text-[11px] tabular-nums",
   541→          isOwn ? "text-white/50" : "text-zinc-500",
   542→        )}
   543→      >
   544→        {Math.floor(duration / 60)}:{String(duration % 60).padStart(2, "0")}
   545→      </span>
   546→      <audio ref={audioRef} src={message.content} preload="metadata" />
   547→    </div>
   548→  );
   549→}

   551→/* ------------------------------------------------------------------ */
   552→/*  LinkifiedText – detects & renders clickable URLs                   */
   553→/* ------------------------------------------------------------------ */
   554→
   555→const URL_REGEX = /(https?:\/\/[^\s<]+)/g;
   556→
   557→type TextSegment =
   558→  | { type: "text"; value: string }
   559→  | { type: "link"; value: string };
   560→
   561→function parseUrls(text: string): TextSegment[] {
   562→  const segments: TextSegment[] = [];
   563→  let lastIndex = 0;
   564→  let match: RegExpExecArray | null;
   565→  while ((match = URL_REGEX.exec(text)) !== null) {
   566→    if (match.index > lastIndex) {
   567→      segments.push({ type: "text", value: text.slice(lastIndex, match.index) });
   568→    }
   569→    segments.push({ type: "link", value: match[1] });
   570→    lastIndex = URL_REGEX.lastIndex;
   571→  }
   572→  if (lastIndex < text.length) {
   573→    segments.push({ type: "text", value: text.slice(lastIndex) });
   574→  }
   575→  return segments;
   576→}

   578→/** Split a plain-text string into highlighted / non-highlighted fragments */
   579→function highlightText(text: string, query: string): React.ReactNode[] {
   580→  if (!query) return [text];
   581→  const escaped = query.replace(/[.*+?^${}()|[\]\\]/g, "\\");
   582→  const re = new RegExp(`(${escaped})`, "gi");
   583→  const parts: React.ReactNode[] = [];
   584→  let lastIndex = 0;
   585→  let match: RegExpExecArray | null;
   586→  while ((match = re.exec(text)) !== null) {
   587→    if (match.index > lastIndex) {
   588→      parts.push(text.slice(lastIndex, match.index));
   589→    }
   590→    parts.push(
   591→      <mark
   592→        key={match.index}
   593→        className="bg-yellow-300/60 dark:bg-yellow-400/40 rounded-sm px-0.5"
   594→      >
   595→        {match[1]}
   596→      </mark>,
   597→    );
   598→    lastIndex = re.lastIndex;
   599→  }
   600→  if (lastIndex < text.length) {
   601→    parts.push(text.slice(lastIndex));
   602→  }
   603→  return parts;

   811→/* ------------------------------------------------------------------ */
   812→/*  Bubble rounding helper for grouped messages                        */

   814→
   815→function getBubbleRounding({
   816→  isFirst,
   817→  isLast,
   818→  isOwn,
   819→}: {
   820→  isFirst: boolean;
   821→  isLast: boolean;
   822→  isOwn: boolean;
   823→}): string {
   824→  if (isFirst && isLast) {
   825→    // Single message in group (or only message)
   826→    return isOwn
   827→      ? "rounded-2xl rounded-br-sm"
   828→      : "rounded-2xl rounded-bl-sm";
   829→  }
   830→  if (isFirst) {
   831→    // First message of a multi-message group
   832→    return isOwn
   833→      ? "rounded-tl-2xl rounded-bl-2xl rounded-tr-sm rounded-br-sm"
   834→      : "rounded-tl-2xl rounded-bl-sm rounded-tr-2xl rounded-br-sm";
   835→  }
   836→  if (isLast) {
   837→    // Last message of a multi-message group
   838→    return isOwn
   839→      ? "rounded-tl-2xl rounded-bl-2xl rounded-tr-sm rounded-br-2xl"
   840→      : "rounded-tl-2xl rounded-bl-2xl rounded-tr-2xl rounded-br-sm";
   841→  }
   842→  // Middle message
   843→  return isOwn
   844→    ? "rounded-tl-2xl rounded-bl-2xl rounded-tr-sm rounded-br-sm"
   845→    : "rounded-tl-2xl rounded-bl-sm rounded-tr-2xl rounded-br-sm";
   846→}

  1456→function formatDateSeparator(dateStr: string): string {
  1457→  const d = new Date(dateStr);
  1458→  const now = new Date();
  1459→  const today = new Date(now.getFullYear(), now.getMonth(), now.getDate());
  1460→  const yesterday = new Date(today);
  1461→  yesterday.setDate(yesterday.getDate() - 1);
  1462→  const msgDay = new Date(d.getFullYear(), d.getMonth(), d.getDate());
  1463→
  1464→  if (msgDay.getTime() === today.getTime()) return "Today";
  1465→  if (msgDay.getTime() === yesterday.getTime()) return "Yesterday";
  1466→  return d.toLocaleDateString(undefined, { month: "short", day: "numeric" });
  1467→}

  1753→  useEffect(() => {
  1754→    if (!onMarkAsRead) return;
  1755→    const unread = messages.filter(
  1756→      (m) => m.sender_id !== currentUserId && m.status !== "read",
  1757→    );
  1758→    if (unread.length > 0 && isAutoScrollRef.current)
  1759→      onMarkAsRead(unread.map((m) => m.id));
  1760→  }, [messages, currentUserId, onMarkAsRead]);
  1761→
  1762→  /* ---- Scroll to a specific message ----------------------------------- */
  1763→  useEffect(() => {
  1764→    if (!scrollToMessageId) return;
  1765→    // Find which group contains the target message
  1766→    const groupIdx = groups.findIndex((g) =>
  1767→      g.messages.some((m) => m.id === scrollToMessageId),
  1768→    );
  1769→    if (groupIdx === -1) {
  1770→      onScrolledToMessage?.();
  1771→      return;
  1772→    }
  1773→    virtualizer.scrollToIndex(groupIdx, { align: "center" });
  1774→    onScrolledToMessage?.();
  1775→  }, [scrollToMessageId, groups, virtualizer, onScrolledToMessage]);
  1776→
  1777→  /* ---- Handle message count changes (new or prepended) ------------ */
  1778→  useEffect(() => {
  1779→    const prev = prevCountRef.current;
  1780→    const prevFirstId = prevFirstIdRef.current;
  1781→    prevCountRef.current = messages.length;
  1782→    prevFirstIdRef.current =
  1783→      messages.length > 0 ? messages[0].id : null;
  1784→
  1785→    if (messages.length <= prev) return;
  1786→
  1787→    /* Check if older messages were prepended (first message changed) */
  1788→    const firstId = messages[0]?.id;
  1789→    const wasPrepended = firstId !== prevFirstId && prevFirstId !== null;
  1790→
  1791→    if (wasPrepended) {
  1792→      /* Older messages prepended – preserve scroll position */
  1793→      const el = parentRef.current;
  1794→      if (el) {
  1795→        const oldHeight = prevScrollHeightRef.current;
  1796→        prevScrollHeightRef.current = el.scrollHeight;
  1797→        requestAnimationFrame(() => {
  1798→          const newHeight = el.scrollHeight;
  1799→          el.scrollTop += newHeight - oldHeight;
  1800→        });
  1801→      }
  1802→      return;
  1803→    }
  1804→
  1805→    /* New messages appended at the end */
  1806→    const last = messages[messages.length - 1];
  1807→    const own = last && last.sender_id === currentUserId;
  1808→    if (own) {
  1809→      requestAnimationFrame(() => scrollToBottom("instant"));
  1810→      return;
  1811→    }
  1812→    if (isAutoScrollRef.current)
  1813→      requestAnimationFrame(() => scrollToBottom("smooth"));
  1814→    else {
  1815→      unreadRef.current += messages.length - prev;
  1816→      setShowNewBtn(true);
  1817→    }
  1818→  }, [messages.length, currentUserId, scrollToBottom]);
  1819→
  1820→  /* Track scroll height for scroll preservation */
  1821→  useEffect(() => {
  1822→    const el = parentRef.current;
  1823→    if (!el) return;
  1824→    prevScrollHeightRef.current = el.scrollHeight;
  1825→  }, [messages.length]);
  1826→
  1827→  if (isLoading) return <MessageListSkeleton />;
  1828→  if (messages.length === 0 && !isPartnerTyping)
  1829→    return (
  1830→      <EmptyState partnerName={partnerName} partnerAvatar={partnerAvatar} />
  1831→    );
  1832→
  1833→  const virtualItems = virtualizer.getVirtualItems();
  1834→
  1835→  return (
  1836→    <div
  1837→      className="relative h-full w-full overflow-hidden"
  1838→      style={backgroundStyle}
  1839→    >
  1840→      {/* Click-away overlay for action sheets / tapback docks */}
  1841→      {(activeActionId || activeTapbackId) && (
  1842→        <div
  1843→          className="fixed inset-0 z-40"
  1844→          onClick={() => { setActiveActionId(null); setActiveTapbackId(null); }}
  1845→        />
  1846→      )}
  1847→      <div
  1848→        ref={parentRef}
  1849→        onScroll={handleScroll}
  1850→        onClick={(e) => {
  1851→          onClearSearchHighlight?.();
  1852→          // Close menus if click is directly on the scroll container or background
  1853→          const target = e.target as HTMLElement;
  1854→          if (target === e.currentTarget || target.closest('.message-bubble-row') === null) {
  1855→            setActiveActionId(null);
  1856→            setActiveTapbackId(null);
  1857→          }
  1858→        }}
  1859→        className="h-full w-full select-none overflow-y-auto overscroll-contain"
  1860→        style={{ WebkitOverflowScrolling: "touch" }}
  1861→      >
  1862→        <div
  1863→          style={{
  1864→            height: `${virtualizer.getTotalSize()}px`,
  1865→            width: "100%",
  1866→            position: "relative",
  1867→          }}
  1868→        >
  1869→          {/* Loading more spinner */}
  1870→          {loadingMore && (
  1871→            <div
  1872→              style={{
  1873→                position: "absolute",
  1874→                top: 0,
  1875→                left: 0,
  1876→                width: "100%",
  1877→                height: "48px",
  1878→                zIndex: 5,
  1879→              }}
  1880→            >
  1881→              <div className="flex items-center justify-center py-3">
  1882→                <svg
  1883→                  className="size-5 animate-spin"
  1884→                  viewBox="0 0 24 24"
  1885→                  fill="none"
  1886→                  aria-label="Loading more messages"
  1887→                  style={{ color: "var(--app-accent-light)" }}
  1888→                >
  1889→                  <circle
  1890→                    className="opacity-25"
  1891→                    cx="12"
  1892→                    cy="12"
  1893→                    r="10"
  1894→                    stroke="currentColor"
  1895→                    strokeWidth="4"
  1896→                  />
  1897→                  <path
  1898→                    className="opacity-75"
  1899→                    fill="currentColor"
  1900→                    d="M4 12a8 8 0 018-8V0C5.373 0 0 5.373 0 12h4z"
  1901→                  />
  1902→                </svg>
  1903→              </div>
  1904→            </div>
  1905→          )}
  1906→          {virtualItems.map((vi) => {
  1907→            const isTyping = vi.index === groups.length;
  1908→            const group = isTyping ? null : groups[vi.index];
  1909→
  1910→            // Determine if we need a date separator before this group
  1911→            let showDateSep = false;
  1912→            if (group) {
  1913→              const firstMsg = group.messages[0];
  1914→              if (vi.index === 0) {
  1915→                showDateSep = true;
  1916→              } else {
  1917→                const prevGroup = groups[vi.index - 1];
  1918→                const prevLastMsg = prevGroup.messages[prevGroup.messages.length - 1];
  1919→                const prevDate = new Date(prevLastMsg.created_at);
  1920→                const currDate = new Date(firstMsg.created_at);
  1921→                showDateSep = prevDate.toDateString() !== currDate.toDateString();
  1922→              }
  1923→            }
  1924→
  1925→            return (
  1926→              <div
  1927→                key={vi.key}
  1928→                data-index={vi.index}
  1929→                ref={virtualizer.measureElement}
  1930→                style={{
  1931→                  position: "absolute",
  1932→                  top: 0,
  1933→                  left: 0,
  1934→                  width: "100%",
  1935→                  transform: `translateY(${vi.start}px)`,
  1936→                }}
  1937→              >
  1938→                {isTyping ? (
  1939→                  <TypingIndicator
  1940→                    partnerName={partnerName}
  1941→                    partnerAvatar={partnerAvatar}
  1942→                  />
  1943→                ) : group ? (
  1944→                  <>
  1945→                    {showDateSep && (
  1946→                      <DateSeparator date={group.messages[0].created_at} />
  1947→                    )}
  1948→                    <MessageGroup
  1949→                      messages={group.messages}
  1950→                      currentUserId={currentUserId}
  1951→                      currentUserName={currentUserName}
  1952→                      currentUserAvatar={currentUserAvatar}
  1953→                      partnerName={partnerName}
  1954→                      partnerAvatar={partnerAvatar}
  1955→                      onReplyTo={onReplyTo}
  1956→                      onReact={onReact}
  1957→                      onDeleteMessage={onDeleteMessage}
  1958→                      searchHighlight={searchHighlight}
  1959→                      activeActionId={activeActionId}
  1960→                      onOpenAction={setActiveActionId}
  1961→                      activeTapbackId={activeTapbackId}
  1962→                      onOpenTapback={setActiveTapbackId}
  1963→                    />
  1964→                  </>
  1965→                ) : null}
  1966→              </div>
  1967→            );
  1968→          })}
  1969→        </div>
  1970→      </div>
  1971→      {showNewBtn && unreadRef.current > 0 && (
  1972→        <NewMessagesButton
  1973→          count={unreadRef.current}
  1974→          onClick={() => scrollToBottom("smooth")}
  1975→        />
  1976→      )}
  1977→    </div>
  1978→  );
  1979→}
