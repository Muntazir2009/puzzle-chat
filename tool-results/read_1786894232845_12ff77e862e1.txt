    93→    93→
    94→    94→function EphemeralTimer({ seconds, onExpire }: { seconds: number; onExpire: () => void }) {
    95→    95→  const [remaining, setRemaining] = useState(seconds);
    96→    96→  const intervalRef = useRef<ReturnType<typeof setInterval> | undefined>(undefined);
    97→    97→
    98→    98→  useEffect(() => {
    99→    99→    intervalRef.current = setInterval(() => {
   100→   100→      setRemaining((r) => {
   101→   101→        if (r <= 1) { clearInterval(intervalRef.current); onExpire(); return 0; }
   102→   102→        return r - 1;
   103→   103→      });
   104→   104→    }, 1000);
   105→   105→    return () => clearInterval(intervalRef.current);
   106→   106→  }, [seconds, onExpire]);
   107→   107→
   108→   108→  const pct = remaining / seconds;
   109→   109→  const circ = 2 * Math.PI * 7;
   110→   110→  const offset = circ * (1 - pct);
   111→   111→
   112→   112→  return (
   113→   113→    <svg className="size-4 shrink-0 ephemeral-timer-pulse" viewBox="0 0 18 18" aria-label={`${remaining}s remaining`}>
   114→   114→      <circle cx="9" cy="9" r="7" fill="none" stroke="currentColor" strokeWidth="1.5" className="text-white/20" />
   115→   115→      <circle cx="9" cy="9" r="7" fill="none" stroke="currentColor" strokeWidth="1.5" strokeDasharray={circ} strokeDashoffset={offset} strokeLinecap="round" className="text-violet-300" style={{ transform: "rotate(-90deg)", transformOrigin: "50% 50%", transition: "stroke-dashoffset 1s linear" }} />
   116→   116→    </svg>
   117→   117→  );
   118→   118→}

   201→   201→function VoiceBubble({ message, isOwn, waveform }: {
   202→   202→  message: ChatMessage; isOwn: boolean; waveform: number[];
   203→   203→}) {
   204→   204→  const audioRef = useRef<HTMLAudioElement>(null);
   205→   205→  const [playing, setPlaying] = useState(false);
   206→   206→  const [progress, setProgress] = useState(0);
   207→   207→  const [speed, setSpeed] = useState(1);
   208→   208→  const speeds = [1, 1.5, 2];
   209→   209→  const duration = message.voice_duration ?? 0;
   210→   210→
   211→   211→  useEffect(() => {
   212→   212→    const audio = audioRef.current;
   213→   213→    if (!audio) return;
   214→   214→    const onTime = () => { if (audio.duration) setProgress(audio.currentTime / audio.duration); };
   215→   215→    const onEnd = () => setPlaying(false);
   216→   216→    audio.addEventListener("timeupdate", onTime);
   217→   217→    audio.addEventListener("ended", onEnd);
   218→   218→    return () => { audio.removeEventListener("timeupdate", onTime); audio.removeEventListener("ended", onEnd); };
   219→   219→  }, []);
   220→   220→
   221→   221→  const toggle = useCallback(() => {
   222→   222→    const audio = audioRef.current;
   223→   223→    if (!audio) return;
   224→   224→    if (playing) { audio.pause(); setPlaying(false); }
   225→   225→    else { audio.playbackRate = speed; audio.play().catch(() => {}); setPlaying(true); }
   226→   226→  }, [playing, speed]);
   227→   227→
   228→   228→  const cycleSpeed = useCallback(() => {
   229→   229→    setSpeed((s) => {
   230→   230→      const idx = speeds.indexOf(s);
   231→   231→      const next = speeds[(idx + 1) % speeds.length];
   232→   232→      if (audioRef.current) audioRef.current.playbackRate = next;
   233→   233→      return next;
   234→   234→    });
   235→   235→  }, []);
   236→   236→
   237→   237→  /* Build waveform bars */
   238→   238→  const bars = useMemo(() => {
   239→   239→    const count = Math.min(waveform.length, 40);
   240→   240→    const step = Math.max(1, Math.floor(waveform.length / count));
   241→   241→    return Array.from({ length: count }, (_, i) => waveform[i * step] ?? 0.1);
   242→   242→  }, [waveform]);
   243→   243→
   244→   244→  const barColor = isOwn ? "bg-white/50" : "bg-zinc-400/70";
   245→   245→  const progressColor = isOwn ? "bg-white" : "bg-violet-400";
   246→   246→  const progressIdx = Math.floor(progress * bars.length);
   247→   247→
   248→   248→  return (
   249→   249→    <div className={cn("flex items-center gap-3", isOwn ? "flex-row-reverse" : "")}>
   250→   250→      <button type="button" onClick={toggle} className={cn("flex size-9 shrink-0 items-center justify-center rounded-full transition-all duration-150", isOwn ? "bg-white/20 hover:bg-white/30 hover:scale-105" : "bg-zinc-600 hover:bg-zinc-500 hover:scale-105")} aria-label={playing ? "Pause" : "Play"}>
   251→   251→        {playing ? <Pause className="size-3.5 text-white" /> : <Play className="size-3.5 text-white ml-0.5" />}
   252→   252→      </button>
   253→   253→      <div className="flex items-center gap-[2px]" style={{ height: 32 }}>
   254→   254→        {bars.map((amp, i) => (
   255→   255→          <div key={i} className={cn("w-[2.5px] rounded-full transition-all duration-100", i < progressIdx ? progressColor : barColor)} style={{ height: `${Math.max(4, amp * 32)}px` }} />
   256→   256→        ))}
   257→   257→      </div>
   258→   258→      <button type="button" onClick={cycleSpeed} className={cn("shrink-0 text-[11px] font-semibold tabular-nums transition-colors hover:opacity-80", isOwn ? "text-white/70" : "text-zinc-400")}>{speed}x</button>
   259→   259→      <span className={cn("shrink-0 text-[11px] tabular-nums", isOwn ? "text-white/50" : "text-zinc-500")}>{Math.floor(duration / 60)}:{String(duration % 60).padStart(2, "0")}</span>
   260→   260→      <audio ref={audioRef} src={message.content} preload="metadata" />
   261→   261→    </div>
   262→   262→  );
   263→   263→}

   266→   266→/*  LinkifiedText – detects & renders clickable URLs                   */
   267→   267→/* ------------------------------------------------------------------ */
   268→   268→
   269→   269→const URL_REGEX = /(https?:\/\/[^\s<]+)/g;
   270→   270→
   271→   271→type TextSegment = { type: "text"; value: string } | { type: "link"; value: string };
   272→   272→
   273→   273→function parseUrls(text: string): TextSegment[] {
   274→   274→  const segments: TextSegment[] = [];
   275→   275→  let lastIndex = 0;
   276→   276→  let match: RegExpExecArray | null;
   277→   277→  while ((match = URL_REGEX.exec(text)) !== null) {
   278→   278→    if (match.index > lastIndex) {
   279→   279→      segments.push({ type: "text", value: text.slice(lastIndex, match.index) });
   280→   280→    }
   281→   281→    segments.push({ type: "link", value: match[1] });
   282→   282→    lastIndex = URL_REGEX.lastIndex;
   283→   283→  }
   284→   284→  if (lastIndex < text.length) {
   285→   285→    segments.push({ type: "text", value: text.slice(lastIndex) });
   286→   286→  }
   287→   287→  return segments;
   288→   288→}

   290→   290→/** Split a plain-text string into highlighted / non-highlighted fragments */
   291→   291→function highlightText(text: string, query: string): React.ReactNode[] {
   292→   292→  if (!query) return [text];
   293→   293→  const escaped = query.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
   294→   294→  const re = new RegExp(`(${escaped})`, "gi");
   295→   295→  const parts: React.ReactNode[] = [];
   296→   296→  let lastIndex = 0;
   297→   297→  let match: RegExpExecArray | null;
   298→   298→  while ((match = re.exec(text)) !== null) {
   299→   299→    if (match.index > lastIndex) {
   300→   300→      parts.push(text.slice(lastIndex, match.index));
   301→   301→    }
   302→   302→    parts.push(
   303→   303→      <mark key={match.index} className="bg-yellow-300/60 dark:bg-yellow-400/40 rounded-sm px-0.5">
   304→   304→        {match[1]}
   305→   305→      </mark>,
   306→   306→    );
   307→   307→    lastIndex = re.lastIndex;
   308→   308→  }
   309→   309→  if (lastIndex < text.length) {
   310→   310→    parts.push(text.slice(lastIndex));
   311→   311→  }
   312→   312→  return parts;
   313→   313→}

   315→   315→export const LinkifiedText = React.memo(function LinkifiedText({ text, className, highlight }: { text: string; className?: string; highlight?: string }) {
   316→   316→  const segments = useMemo(() => parseUrls(text), [text]);
   317→   317→  return (
   318→   318→    <span className={className}>
   319→   319→      {segments.map((seg, i) =>
   320→   320→        seg.type === "link" ? (
   321→   321→          <a
   322→   322→            key={i}
   323→   323→            href={seg.value}
   324→   324→            target="_blank"
   325→   325→            rel="noopener noreferrer"
   326→   326→            className="underline decoration-violet-400/50 underline-offset-2 hover:decoration-violet-400 transition-colors"
   327→   327→            onClick={(e) => e.stopPropagation()}
   328→   328→          >
   329→   329→            {seg.value}
   330→   330→          </a>
   331→   331→        ) : (
   332→   332→          <span key={i}>{highlight ? highlightText(seg.value, highlight) : seg.value}</span>
   333→   333→        ),
   334→   334→      )}
   335→   335→    </span>
   336→   336→  );

   610→   610→function formatDateSeparator(dateStr: string): string {
   611→   611→  const d = new Date(dateStr);
   612→   612→  const now = new Date();
   613→   613→  const today = new Date(now.getFullYear(), now.getMonth(), now.getDate());
   614→   614→  const yesterday = new Date(today);
   615→   615→  yesterday.setDate(yesterday.getDate() - 1);
   616→   616→  const msgDay = new Date(d.getFullYear(), d.getMonth(), d.getDate());
   617→   617→
   618→   618→  if (msgDay.getTime() === today.getTime()) return "Today";
   619→   619→  if (msgDay.getTime() === yesterday.getTime()) return "Yesterday";
   620→   620→  return d.toLocaleDateString(undefined, { month: "short", day: "numeric" });
   621→   621→}

   760→   760→    el.scrollTo({ top: el.scrollHeight, behavior });
   761→   761→    isAutoScrollRef.current = true; setShowNewBtn(false); unreadRef.current = 0;
   762→   762→  }, []);
   763→   763→
   764→   764→  useEffect(() => { if (messages.length > 0) requestAnimationFrame(() => scrollToBottom("instant")); }, [isLoading, scrollToBottom]);
   765→   765→
   766→   766→  useEffect(() => { if (!onMarkAsRead) return; const unread = messages.filter((m) => m.sender_id !== currentUserId && m.status !== "read"); if (unread.length > 0 && isAutoScrollRef.current) onMarkAsRead(unread.map((m) => m.id)); }, [messages, currentUserId, onMarkAsRead]);
   767→   767→
   768→   768→  /* ---- Scroll to a specific message ----------------------------------- */
   769→   769→  useEffect(() => {
   770→   770→    if (!scrollToMessageId) return;
   771→   771→    const idx = messages.findIndex((m) => m.id === scrollToMessageId);
   772→   772→    if (idx === -1) { onScrolledToMessage?.(); return; }
   773→   773→    virtualizer.scrollToIndex(idx, { align: "center" });
   774→   774→    onScrolledToMessage?.();
   775→   775→  }, [scrollToMessageId, messages, virtualizer, onScrolledToMessage]);
   776→   776→
   777→   777→  /* ---- Handle message count changes (new or prepended) ------------ */
   778→   778→  useEffect(() => {
   779→   779→    const prev = prevCountRef.current;
   780→   780→    const prevFirstId = prevFirstIdRef.current;
   781→   781→    prevCountRef.current = messages.length;
   782→   782→    prevFirstIdRef.current = messages.length > 0 ? messages[0].id : null;
   783→   783→
   784→   784→    if (messages.length <= prev) return;
   785→   785→
   786→   786→    /* Check if older messages were prepended (first message changed) */
   787→   787→    const firstId = messages[0]?.id;
   788→   788→    const wasPrepended = firstId !== prevFirstId && prevFirstId !== null;
   789→   789→
   790→   790→    if (wasPrepended) {
   791→   791→      /* Older messages prepended – preserve scroll position */
   792→   792→      const el = parentRef.current;
   793→   793→      if (el) {
   794→   794→        const oldHeight = prevScrollHeightRef.current;
   795→   795→        prevScrollHeightRef.current = el.scrollHeight;
   796→   796→        requestAnimationFrame(() => {
   797→   797→          const newHeight = el.scrollHeight;
   798→   798→          el.scrollTop += newHeight - oldHeight;
   799→   799→        });
   800→   800→      }
   801→   801→      return;
   802→   802→    }
   803→   803→
   804→   804→    /* New messages appended at the end */
   805→   805→    const last = messages[messages.length - 1];
   806→   806→    const own = last && last.sender_id === currentUserId;
   807→   807→    if (own) { requestAnimationFrame(() => scrollToBottom("instant")); return; }
   808→   808→    if (isAutoScrollRef.current) requestAnimationFrame(() => scrollToBottom("smooth"));
   809→   809→    else { unreadRef.current += messages.length - prev; setShowNewBtn(true); }
   810→   810→  }, [messages.length, currentUserId, scrollToBottom]);
   811→   811→
   812→   812→  /* Track scroll height for scroll preservation */
   813→   813→  useEffect(() => {
   814→   814→    const el = parentRef.current;
   815→   815→    if (!el) return;
   816→   816→    prevScrollHeightRef.current = el.scrollHeight;
   817→   817→  }, [messages.length]);
   818→   818→
   819→   819→  if (isLoading) return <MessageListSkeleton />;
   820→   820→  if (messages.length === 0 && !isPartnerTyping) return <EmptyState partnerName={partnerName} partnerAvatar={partnerAvatar} />;
   821→   821→
   822→   822→  const virtualItems = virtualizer.getVirtualItems();
   823→   823→
   824→   824→  return (
   825→   825→    <div className="relative h-full w-full overflow-hidden" style={backgroundStyle}>
   826→   826→      <div ref={parentRef} onScroll={handleScroll} onClick={() => onClearSearchHighlight?.()} className="h-full w-full overflow-y-auto overscroll-contain" style={{ WebkitOverflowScrolling: "touch" }}>
   827→   827→        <div style={{ height: `${virtualizer.getTotalSize()}px`, width: "100%", position: "relative" }}>
   828→   828→          {/* Loading more spinner */}
   829→   829→          {loadingMore && (
   830→   830→            <div style={{ position: "absolute", top: 0, left: 0, width: "100%", height: "48px", zIndex: 5 }}>
   831→   831→              <div className="flex items-center justify-center py-3">
   832→   832→                <svg className="size-5 animate-spin text-violet-400" viewBox="0 0 24 24" fill="none" aria-label="Loading more messages">
   833→   833→                  <circle className="opacity-25" cx="12" cy="12" r="10" stroke="currentColor" strokeWidth="4" />
   834→   834→                  <path className="opacity-75" fill="currentColor" d="M4 12a8 8 0 018-8V0C5.373 0 0 5.373 0 12h4z" />
   835→   835→                </svg>
   836→   836→              </div>
   837→   837→            </div>
   838→   838→          )}
   839→   839→          {virtualItems.map((vi) => {
   840→   840→            const isTyping = vi.index === messages.length;
   841→   841→            const msg = isTyping ? null : messages[vi.index];
   842→   842→            // Check if we need a date separator before this message
   843→   843→            const showDateSep = msg && vi.index > 0 && messages[vi.index - 1] && (() => {
   844→   844→              const prev = new Date(messages[vi.index - 1].created_at);
   845→   845→              const curr = new Date(msg.created_at);
   846→   846→              return prev.toDateString() !== curr.toDateString();
   847→   847→            })();
   848→   848→            const showFirstDateSep = msg && vi.index === 0;
   849→   849→            return (
   850→   850→              <div key={vi.key} data-index={vi.index} ref={virtualizer.measureElement} style={{ position: "absolute", top: 0, left: 0, width: "100%", transform: `translateY(${vi.start}px)` }}>
   851→   851→                {isTyping ? <TypingIndicator partnerName={partnerName} partnerAvatar={partnerAvatar} /> : msg ? (
   852→   852→                  <>
   853→   853→                    {(showDateSep || showFirstDateSep) && <DateSeparator date={msg.created_at} />}
   854→   854→                    <MessageBubble message={msg} isOwn={msg.sender_id === currentUserId} partnerName={partnerName} partnerAvatar={partnerAvatar} onReplyTo={onReplyTo} onReact={onReact} onDeleteMessage={onDeleteMessage} searchHighlight={searchHighlight} />
   855→   855→                  </>
   856→   856→                ) : null}
   857→   857→              </div>
   858→   858→            );
   859→   859→          })}
   860→   860→        </div>
   861→   861→      </div>
   862→   862→      {showNewBtn && unreadRef.current > 0 && <NewMessagesButton count={unreadRef.current} onClick={() => scrollToBottom("smooth")} />}
   863→   863→    </div>
   864→   864→  );
   865→   865→}
