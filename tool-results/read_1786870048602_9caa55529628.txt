    85→    85→
    86→    86→function EphemeralTimer({ seconds, onExpire }: { seconds: number; onExpire: () => void }) {
    87→    87→  const [remaining, setRemaining] = useState(seconds);
    88→    88→  const intervalRef = useRef<ReturnType<typeof setInterval> | undefined>(undefined);
    89→    89→
    90→    90→  useEffect(() => {
    91→    91→    intervalRef.current = setInterval(() => {
    92→    92→      setRemaining((r) => {
    93→    93→        if (r <= 1) { clearInterval(intervalRef.current); onExpire(); return 0; }
    94→    94→        return r - 1;
    95→    95→      });
    96→    96→    }, 1000);
    97→    97→    return () => clearInterval(intervalRef.current);
    98→    98→  }, [seconds, onExpire]);
    99→    99→
   100→   100→  const pct = remaining / seconds;
   101→   101→  const circ = 2 * Math.PI * 7;
   102→   102→  const offset = circ * (1 - pct);
   103→   103→
   104→   104→  return (
   105→   105→    <svg className="size-4 shrink-0" viewBox="0 0 18 18" aria-label={`${remaining}s remaining`}>
   106→   106→      <circle cx="9" cy="9" r="7" fill="none" stroke="currentColor" strokeWidth="1.5" className="text-white/20" />
   107→   107→      <circle cx="9" cy="9" r="7" fill="none" stroke="currentColor" strokeWidth="1.5" strokeDasharray={circ} strokeDashoffset={offset} strokeLinecap="round" className="text-white/60" style={{ transform: "rotate(-90deg)", transformOrigin: "50% 50%", transition: "stroke-dashoffset 1s linear" }} />
   108→   108→    </svg>
   109→   109→  );
   110→   110→}

   153→   153→function VoiceBubble({ message, isOwn, waveform }: {
   154→   154→  message: ChatMessage; isOwn: boolean; waveform: number[];
   155→   155→}) {
   156→   156→  const audioRef = useRef<HTMLAudioElement>(null);
   157→   157→  const [playing, setPlaying] = useState(false);
   158→   158→  const [progress, setProgress] = useState(0);
   159→   159→  const [speed, setSpeed] = useState(1);
   160→   160→  const speeds = [1, 1.5, 2];
   161→   161→  const duration = message.voice_duration ?? 0;
   162→   162→
   163→   163→  useEffect(() => {
   164→   164→    const audio = audioRef.current;
   165→   165→    if (!audio) return;
   166→   166→    const onTime = () => { if (audio.duration) setProgress(audio.currentTime / audio.duration); };
   167→   167→    const onEnd = () => setPlaying(false);
   168→   168→    audio.addEventListener("timeupdate", onTime);
   169→   169→    audio.addEventListener("ended", onEnd);
   170→   170→    return () => { audio.removeEventListener("timeupdate", onTime); audio.removeEventListener("ended", onEnd); };
   171→   171→  }, []);
   172→   172→
   173→   173→  const toggle = useCallback(() => {
   174→   174→    const audio = audioRef.current;
   175→   175→    if (!audio) return;
   176→   176→    if (playing) { audio.pause(); setPlaying(false); }
   177→   177→    else { audio.playbackRate = speed; audio.play().catch(() => {}); setPlaying(true); }
   178→   178→  }, [playing, speed]);
   179→   179→
   180→   180→  const cycleSpeed = useCallback(() => {
   181→   181→    setSpeed((s) => {
   182→   182→      const idx = speeds.indexOf(s);
   183→   183→      const next = speeds[(idx + 1) % speeds.length];
   184→   184→      if (audioRef.current) audioRef.current.playbackRate = next;
   185→   185→      return next;
   186→   186→    });
   187→   187→  }, []);
   188→   188→
   189→   189→  /* Build waveform bars */
   190→   190→  const bars = useMemo(() => {
   191→   191→    const count = Math.min(waveform.length, 40);
   192→   192→    const step = Math.max(1, Math.floor(waveform.length / count));
   193→   193→    return Array.from({ length: count }, (_, i) => waveform[i * step] ?? 0.1);
   194→   194→  }, [waveform]);
   195→   195→
   196→   196→  const barColor = isOwn ? "bg-white/70" : "bg-zinc-400";
   197→   197→  const progressColor = isOwn ? "bg-white" : "bg-violet-400";
   198→   198→  const progressIdx = Math.floor(progress * bars.length);
   199→   199→
   200→   200→  return (
   201→   201→    <div className={cn("flex items-center gap-2.5", isOwn ? "flex-row-reverse" : "")}>
   202→   202→      <button type="button" onClick={toggle} className={cn("flex size-9 shrink-0 items-center justify-center rounded-full transition-colors", isOwn ? "bg-white/20 hover:bg-white/30" : "bg-zinc-600 hover:bg-zinc-500")} aria-label={playing ? "Pause" : "Play"}>
   203→   203→        {playing ? <Pause className="size-3.5 text-white" /> : <Play className="size-3.5 text-white ml-0.5" />}
   204→   204→      </button>
   205→   205→      <div className="flex items-center gap-px" style={{ height: 28 }}>
   206→   206→        {bars.map((amp, i) => (
   207→   207→          <div key={i} className={cn("w-[3px] rounded-full transition-all duration-75", i < progressIdx ? progressColor : barColor)} style={{ height: `${Math.max(4, amp * 28)}px` }} />
   208→   208→        ))}
   209→   209→      </div>
   210→   210→      <button type="button" onClick={cycleSpeed} className={cn("shrink-0 text-[11px] font-medium tabular-nums", isOwn ? "text-white/70" : "text-zinc-400")}>{speed}x</button>
   211→   211→      <span className={cn("shrink-0 text-[11px] tabular-nums", isOwn ? "text-white/60" : "text-zinc-500")}>{Math.floor(duration / 60)}:{String(duration % 60).padStart(2, "0")}</span>
   212→   212→      <audio ref={audioRef} src={message.content} preload="metadata" />
   213→   213→    </div>
   214→   214→  );
   215→   215→}

   257→   257→
   258→   258→function MessageBubble({ message, isOwn, partnerName, partnerAvatar, onReplyTo, onReact }: {
   259→   259→  message: ChatMessage; isOwn: boolean; partnerName: string; partnerAvatar: string | null;
   260→   260→  onReplyTo?: (msg: ChatMessage) => void; onReact?: (id: string, emoji: string, add: boolean) => void;
   261→   261→}) {
   262→   262→  const [showHeart, setShowHeart] = useState(false);
   263→   263→  const [showTapback, setShowTapback] = useState(false);
   264→   264→  const lastTapRef = useRef(0);
   265→   265→  const longPressRef = useRef<ReturnType<typeof setTimeout> | undefined>(undefined);
   266→   266→  const bubbleRef = useRef<HTMLDivElement>(null);
   267→   267→
   268→   268→  const triggerHeart = useCallback(() => {
   269→   269→    setShowHeart(true); setTimeout(() => setShowHeart(false), 800);
   270→   270→    if (typeof navigator !== "undefined" && "vibrate" in navigator) navigator.vibrate(10);
   271→   271→  }, []);
   272→   272→
   273→   273→  const handleClick = useCallback(() => {
   274→   274→    clearTimeout(longPressRef.current);
   275→   275→    const now = Date.now();
   276→   276→    if (now - lastTapRef.current < DOUBLE_TAP_MS) triggerHeart();
   277→   277→    lastTapRef.current = now;
   278→   278→  }, [triggerHeart]);
   279→   279→
   280→   280→  const handlePressStart = useCallback(() => {
   281→   281→    longPressRef.current = setTimeout(() => {
   282→   282→      if (typeof navigator !== "undefined" && "vibrate" in navigator) navigator.vibrate(20);
   283→   283→      setShowTapback(true);
   284→   284→    }, LONG_PRESS_MS);
   285→   285→  }, []);
   286→   286→
   287→   287→  const handlePressEnd = useCallback(() => { clearTimeout(longPressRef.current); }, []);
   288→   288→
   289→   289→  const handleContextMenu = useCallback((e: React.MouseEvent) => {
   290→   290→    e.preventDefault(); setShowTapback(true);
   291→   291→  }, []);
   292→   292→
   293→   293→  const handleSwipeEnd = useCallback((_: MouseEvent | TouchEvent | PointerEvent, info: PanInfo) => {
   294→   294→    if (info.offset.x > SWIPE_REPLY_THRESHOLD && info.velocity.x > 200) {
   295→   295→      if (typeof navigator !== "undefined" && "vibrate" in navigator) navigator.vibrate(10);
   296→   296→      onReplyTo?.(message);
   297→   297→    }
   298→   298→  }, [message, onReplyTo]);
   299→   299→
   300→   300→  const isFailed = message.status === "failed";
   301→   301→  const isVoice = message.type === "voice";
   302→   302→  const isNew = message.status === "sending" || Date.now() - new Date(message.created_at).getTime() < 2000;
   303→   303→
   304→   304→  return (
   305→   305→    <m.div
   306→   306→      className={cn("relative flex w-full gap-2.5 px-4", isOwn ? "justify-end" : "justify-start")}
   307→   307→      initial={isNew ? { opacity: 0, y: 12 } : false}
   308→   308→      animate={{ opacity: 1, y: 0 }}
   309→   309→      transition={{ duration: 0.25, ease: "easeOut" }}
   310→   310→    >
   311→   311→      <HeartBurst show={showHeart} />
   312→   312→      <AnimatePresence>{showTapback && onReact && (
   313→   313→        <TapbackDock message={message} isOwn={isOwn} onReact={(emoji, add) => { onReact(message.id, emoji, add); setShowTapback(false); }} onReply={() => { onReplyTo?.(message); setShowTapback(false); }} />
   314→   314→      )}</AnimatePresence>
   315→   315→
   316→   316→      {!isOwn && (
   317→   317→        <Avatar className="mt-auto size-8 shrink-0 ring-1 ring-muted">
   318→   318→          {partnerAvatar && <AvatarImage src={partnerAvatar} alt={partnerName} />}
   319→   319→          <AvatarFallback className="bg-gradient-to-br from-violet-100 to-purple-200 text-xs font-medium text-violet-700 dark:from-violet-500 dark:to-purple-600 dark:text-white">{partnerName.slice(0, 2).toUpperCase()}</AvatarFallback>
   320→   320→        </Avatar>
   321→   321→      )}
   322→   322→
   323→   323→      <div className={cn("flex max-w-[75%] flex-col gap-1", isOwn ? "items-end" : "items-start")}>
   324→   324→        <m.div
   325→   325→          ref={bubbleRef}
   326→   326→          drag="x"
   327→   327→          dragConstraints={{ left: 0, right: 0 }}
   328→   328→          dragElastic={{ left: 0, right: 0.35 }}
   329→   329→          onDragEnd={handleSwipeEnd}
   330→   330→          onClick={handleClick}
   331→   331→          onPointerDown={handlePressStart}
   332→   332→          onPointerUp={handlePressEnd}
   333→   333→          onPointerLeave={handlePressEnd}
   334→   334→          onContextMenu={handleContextMenu}
   335→   335→          className={cn(
   336→   336→            "relative cursor-default select-none rounded-2xl px-4 py-2.5 text-sm leading-relaxed transition-transform duration-100 active:scale-[0.98]",
   337→   337→            // Own messages: vibrant violet-purple gradient with subtle shadow
   338→   338→            isOwn && "bg-gradient-to-r from-violet-500 to-purple-600 text-white rounded-br-md shadow-md shadow-violet-500/15",
   339→   339→            // Partner messages: softer background with shadow
   340→   340→            !isOwn && "bg-muted text-foreground rounded-bl-md shadow-sm dark:bg-zinc-800 dark:text-zinc-100 dark:shadow-violet-950/20",
   341→   341→            isFailed && "ring-2 ring-red-400/50",
   342→   342→            isVoice && "py-2"
   343→   343→          )}
   344→   344→          style={{ transform: "translate3d(0,0,0)" }}
   345→   345→        >
   346→   346→          {/* Reply quote */}
   347→   347→          {message.reply_to_id && message.reply_to_content && (
   348→   348→            <ReplyPreview senderName={message.reply_to_sender_name ?? ""} content={message.reply_to_content} isOwn={isOwn} />
   349→   349→          )}
   350→   350→
   351→   351→          {/* Content */}
   352→   352→          {isVoice ? (
   353→   353→            message.waveform_data ? (
   354→   354→              <VoiceBubble message={message} isOwn={isOwn} waveform={message.waveform_data} />
   355→   355→            ) : <p className="text-white/60 italic">Voice message</p>
   356→   356→          ) : message.type === "image" ? (
   357→   357→            <img src={message.content} alt="Shared image" className="max-h-64 rounded-lg object-contain" />
   358→   358→          ) : (
   359→   359→            <p className="whitespace-pre-wrap break-words">{message.content}</p>
   360→   360→          )}
   361→   361→
   362→   362→          {/* Vanish indicator */}
   363→   363→          {message.vanish_mode && (
   364→   364→            <div className="mt-1 flex items-center gap-1 text-[10px] text-white/50"><EyeOff className="size-3" /><span>View once</span></div>
   365→   365→          )}
   366→   366→        </m.div>
   367→   367→
   368→   368→        {/* Timestamp + receipt + ephemeral timer */}
   369→   369→        <div className={cn("flex items-center gap-1.5 px-1 text-[11px] text-muted-foreground", isOwn && "flex-row-reverse")}>
   370→   370→          {message.ephemeral_seconds && message.ephemeral_seconds > 0 && (
   371→   371→            <EphemeralTimer seconds={message.ephemeral_seconds} onExpire={() => { /* trigger vanish */ }} />
   372→   372→          )}
   373→   373→          <span>{formatDistanceToNow(new Date(message.created_at), { addSuffix: true })}</span>
   374→   374→          {isOwn && <ReceiptIcon status={message.status} />}
   375→   375→        </div>
   376→   376→
   377→   377→        {/* Reactions */}

   474→   474→export function MessageFeed({ messages, isLoading, isPartnerTyping, currentUserId, partnerName, partnerAvatar, onMarkAsRead, onVanishMessage, onReplyTo, onReact, scrollToMessageId, onScrolledToMessage }: MessageFeedProps) {
   475→   475→  const parentRef = useRef<HTMLDivElement>(null);
   476→   476→  const isAutoScrollRef = useRef(true);
   477→   477→  const [showNewBtn, setShowNewBtn] = useState(false);
   478→   478→  const unreadRef = useRef(0);
   479→   479→  const prevCountRef = useRef(messages.length);
   480→   480→
   481→   481→  const totalItems = useMemo(() => messages.length + (isPartnerTyping ? 1 : 0), [messages.length, isPartnerTyping]);
   482→   482→
   483→   483→  const virtualizer = useVirtualizer({ count: totalItems, getScrollElement: () => parentRef.current, estimateSize: () => ESTIMATED_ROW_HEIGHT, overscan: OVERSCAN, paddingStart: 16, paddingEnd: 16, gap: 12 });
   484→   484→
   485→   485→  const scrollToBottom = useCallback((behavior: ScrollBehavior = "smooth") => {
   486→   486→    const el = parentRef.current; if (!el) return;
   487→   487→    el.scrollTo({ top: el.scrollHeight, behavior });
   488→   488→    isAutoScrollRef.current = true; setShowNewBtn(false); unreadRef.current = 0;
   489→   489→  }, []);
   490→   490→
   491→   491→  const handleScroll = useCallback((e: UIEvent<HTMLDivElement>) => {
   492→   492→    const el = e.currentTarget;
   493→   493→    const dist = el.scrollHeight - el.scrollTop - el.clientHeight;
   494→   494→    if (dist <= AUTO_SCROLL_THRESHOLD) { isAutoScrollRef.current = true; setShowNewBtn(false); unreadRef.current = 0; }
   495→   495→    else { isAutoScrollRef.current = false; }
   496→   496→  }, []);
   497→   497→
   498→   498→  useEffect(() => { if (messages.length > 0) requestAnimationFrame(() => scrollToBottom("instant")); }, [isLoading, scrollToBottom]);
   499→   499→
   500→   500→  useEffect(() => { if (!onMarkAsRead) return; const unread = messages.filter((m) => m.sender_id !== currentUserId && m.status !== "read"); if (unread.length > 0 && isAutoScrollRef.current) onMarkAsRead(unread.map((m) => m.id)); }, [messages, currentUserId, onMarkAsRead]);
   501→   501→
   502→   502→  /* ---- Scroll to a specific message ----------------------------------- */
   503→   503→  useEffect(() => {
   504→   504→    if (!scrollToMessageId) return;
   505→   505→    const idx = messages.findIndex((m) => m.id === scrollToMessageId);
   506→   506→    if (idx === -1) { onScrolledToMessage?.(); return; }
   507→   507→    virtualizer.scrollToIndex(idx, { align: "center" });
   508→   508→    onScrolledToMessage?.();
   509→   509→  }, [scrollToMessageId, messages, virtualizer, onScrolledToMessage]);
   510→   510→
   511→   511→  useEffect(() => {
   512→   512→    const prev = prevCountRef.current; const next = messages.length; prevCountRef.current = next;
   513→   513→    if (next <= prev) return;
   514→   514→    const last = messages[next - 1]; const own = last && last.sender_id === currentUserId;
   515→   515→    if (own) { requestAnimationFrame(() => scrollToBottom("instant")); return; }
   516→   516→    if (isAutoScrollRef.current) requestAnimationFrame(() => scrollToBottom("smooth"));
   517→   517→    else { unreadRef.current += next - prev; setShowNewBtn(true); }
   518→   518→  }, [messages.length, currentUserId, scrollToBottom]);
   519→   519→
   520→   520→  if (isLoading) return <MessageListSkeleton />;
   521→   521→  if (messages.length === 0 && !isPartnerTyping) return <EmptyState partnerName={partnerName} partnerAvatar={partnerAvatar} />;
   522→   522→
   523→   523→  const virtualItems = virtualizer.getVirtualItems();
   524→   524→
   525→   525→  return (
   526→   526→    <div className="relative h-full w-full overflow-hidden">
   527→   527→      <div ref={parentRef} onScroll={handleScroll} className="h-full w-full overflow-y-auto overscroll-contain" style={{ WebkitOverflowScrolling: "touch" }}>
   528→   528→        <div style={{ height: `${virtualizer.getTotalSize()}px`, width: "100%", position: "relative" }}>
   529→   529→          {virtualItems.map((vi) => {
   530→   530→            const isTyping = vi.index === messages.length;
   531→   531→            const msg = isTyping ? null : messages[vi.index];
   532→   532→            return (
   533→   533→              <div key={vi.key} data-index={vi.index} ref={virtualizer.measureElement} style={{ position: "absolute", top: 0, left: 0, width: "100%", transform: `translateY(${vi.start}px)` }}>
   534→   534→                {isTyping ? <TypingIndicator partnerName={partnerName} /> : msg ? (
   535→   535→                  <MessageBubble message={msg} isOwn={msg.sender_id === currentUserId} partnerName={partnerName} partnerAvatar={partnerAvatar} onReplyTo={onReplyTo} onReact={onReact} />
   536→   536→                ) : null}
   537→   537→              </div>
   538→   538→            );
   539→   539→          })}
   540→   540→        </div>
   541→   541→      </div>
   542→   542→      {showNewBtn && unreadRef.current > 0 && <NewMessagesButton count={unreadRef.current} onClick={() => scrollToBottom("smooth")} />}
   543→   543→    </div>
   544→   544→  );
   545→   545→}
