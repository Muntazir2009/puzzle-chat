   122→/* ------------------------------------------------------------------ */
   123→
   124→function EphemeralTimer({
   125→  seconds,
   126→  onExpire,
   127→}: {
   128→  seconds: number;
   129→  onExpire: () => void;
   130→}) {
   131→  const [remaining, setRemaining] = useState(seconds);
   132→  const intervalRef = useRef<ReturnType<typeof setInterval> | undefined>(
   133→    undefined,
   134→  );
   135→
   136→  useEffect(() => {
   137→    intervalRef.current = setInterval(() => {
   138→      setRemaining((r) => {
   139→        if (r <= 1) {
   140→          clearInterval(intervalRef.current);
   141→          onExpire();
   142→          return 0;
   143→        }
   144→        return r - 1;
   145→      });
   146→    }, 1000);
   147→    return () => clearInterval(intervalRef.current);
   148→  }, [seconds, onExpire]);
   149→
   150→  const pct = remaining / seconds;
   151→  const circ = 2 * Math.PI * 7;
   152→  const offset = circ * (1 - pct);
   153→
   154→  return (
   155→    <svg
   156→      className="size-3.5 shrink-0 ephemeral-timer-pulse"
   157→      viewBox="0 0 18 18"
   158→      aria-label={`${remaining}s remaining`}
   159→    >
   160→      <circle
   161→        cx="9"
   162→        cy="9"
   163→        r="7"
   164→        fill="none"
   165→        stroke="currentColor"
   166→        strokeWidth="1.5"
   167→        className="text-white/20"
   168→      />
   169→      <circle
   170→        cx="9"
   171→        cy="9"
   172→        r="7"
   173→        fill="none"
   174→        stroke="currentColor"
   175→        strokeWidth="1.5"
   176→        strokeDasharray={circ}
   177→        strokeDashoffset={offset}
   178→        strokeLinecap="round"
   179→        style={{
   180→          color: "var(--app-accent-lighter)",
   181→          transform: "rotate(-90deg)",
   182→          transformOrigin: "50% 50%",
   183→          transition: "stroke-dashoffset 1s linear",
   184→        }}
   185→      />
   186→    </svg>
   187→  );
   188→}

   397→/* ------------------------------------------------------------------ */
   398→/*  Voice Bubble                                                       */
   399→/* ------------------------------------------------------------------ */
   400→
   401→function VoiceBubble({
   402→  message,
   403→  isOwn,
   404→  waveform,
   405→}: {
   406→  message: ChatMessage;
   407→  isOwn: boolean;
   408→  waveform: number[];
   409→}) {
   410→  const audioRef = useRef<HTMLAudioElement>(null);
   411→  const [playing, setPlaying] = useState(false);
   412→  const [progress, setProgress] = useState(0);
   413→  const [speed, setSpeed] = useState(1);
   414→  const speeds = [1, 1.5, 2];
   415→  const duration = message.voice_duration ?? 0;
   416→
   417→  useEffect(() => {
   418→    const audio = audioRef.current;
   419→    if (!audio) return;
   420→    const onTime = () => {
   421→      if (audio.duration) setProgress(audio.currentTime / audio.duration);
   422→    };
   423→    const onEnd = () => setPlaying(false);
   424→    audio.addEventListener("timeupdate", onTime);
   425→    audio.addEventListener("ended", onEnd);
   426→    return () => {
   427→      audio.removeEventListener("timeupdate", onTime);
   428→      audio.removeEventListener("ended", onEnd);
   429→    };
   430→  }, []);
   431→
   432→  const toggle = useCallback(() => {
   433→    const audio = audioRef.current;
   434→    if (!audio) return;
   435→    if (playing) {
   436→      audio.pause();
   437→      setPlaying(false);
   438→    } else {
   439→      audio.playbackRate = speed;
   440→      audio.play().catch(() => {});
   441→      setPlaying(true);
   442→    }
   443→  }, [playing, speed]);
   444→
   445→  const cycleSpeed = useCallback(() => {
   446→    setSpeed((s) => {
   447→      const idx = speeds.indexOf(s);
   448→      const next = speeds[(idx + 1) % speeds.length];
   449→      if (audioRef.current) audioRef.current.playbackRate = next;
   450→      return next;
   451→    });
   452→  }, []);
   453→
   454→  /* Build waveform bars */
   455→  const bars = useMemo(() => {
   456→    const count = Math.min(waveform.length, 40);
   457→    const step = Math.max(1, Math.floor(waveform.length / count));
   458→    return Array.from({ length: count }, (_, i) => waveform[i * step] ?? 0.1);
   459→  }, [waveform]);
   460→
   461→  const progressIdx = Math.floor(progress * bars.length);
   462→
   463→  return (
   464→    <div className={cn("flex items-center gap-3", isOwn ? "flex-row-reverse" : "")}>
   465→      <button
   466→        type="button"
   467→        onClick={toggle}
   468→        className={cn(
   469→          "flex size-9 shrink-0 items-center justify-center rounded-full transition-all duration-150 active:scale-95",
   470→          isOwn
   471→            ? "bg-white/20 hover:bg-white/30"
   472→            : "bg-zinc-600 hover:bg-[var(--app-accent-subtle)]",
   473→        )}
   474→        aria-label={playing ? "Pause" : "Play"}
   475→      >
   476→        {playing ? (
   477→          <Pause className="size-3.5 text-white" />
   478→        ) : (
   479→          <Play className="size-3.5 text-white ml-0.5" />
   480→        )}
   481→      </button>
   482→      <div className="flex items-center gap-[2px]" style={{ height: 32 }}>
   483→        {bars.map((amp, i) => (
   484→          <div
   485→            key={i}
   486→            className="w-[2.5px] rounded-full transition-all duration-100"
   487→            style={{
   488→              height: `${Math.max(4, amp * 32)}px`,
   489→              backgroundColor:
   490→                i < progressIdx
   491→                  ? isOwn
   492→                    ? "rgba(255,255,255,1)"
   493→                    : "var(--app-accent-light)"
   494→                  : isOwn
   495→                    ? "rgba(255,255,255,0.5)"
   496→                    : "rgba(161,161,170,0.7)",
   497→            }}
   498→          />
   499→        ))}
   500→      </div>
   501→      <button
   502→        type="button"
   503→        onClick={cycleSpeed}
   504→        className={cn(
   505→          "shrink-0 text-[11px] font-semibold tabular-nums transition-colors hover:opacity-80 active:scale-95",
   506→          isOwn ? "text-white/70" : "text-zinc-400",
   507→        )}
   508→      >
   509→        {speed}x
   510→      </button>
   511→      <span
   512→        className={cn(
   513→          "shrink-0 text-[11px] tabular-nums",
   514→          isOwn ? "text-white/50" : "text-zinc-500",
   515→        )}
   516→      >
   517→        {Math.floor(duration / 60)}:{String(duration % 60).padStart(2, "0")}
   518→      </span>
   519→      <audio ref={audioRef} src={message.content} preload="metadata" />
   520→    </div>
   521→  );
   522→}

   524→/* ------------------------------------------------------------------ */
   525→/*  LinkifiedText – detects & renders clickable URLs                   */
   526→/* ------------------------------------------------------------------ */
   527→
   528→const URL_REGEX = /(https?:\/\/[^\s<]+)/g;
   529→
   530→type TextSegment =
   531→  | { type: "text"; value: string }
   532→  | { type: "link"; value: string };
   533→
   534→function parseUrls(text: string): TextSegment[] {
   535→  const segments: TextSegment[] = [];
   536→  let lastIndex = 0;
   537→  let match: RegExpExecArray | null;
   538→  while ((match = URL_REGEX.exec(text)) !== null) {
   539→    if (match.index > lastIndex) {
   540→      segments.push({ type: "text", value: text.slice(lastIndex, match.index) });
   541→    }
   542→    segments.push({ type: "link", value: match[1] });
   543→    lastIndex = URL_REGEX.lastIndex;
   544→  }
   545→  if (lastIndex < text.length) {
   546→    segments.push({ type: "text", value: text.slice(lastIndex) });
   547→  }
   548→  return segments;
   549→}
   550→
   551→/** Split a plain-text string into highlighted / non-highlighted fragments */
   552→function highlightText(text: string, query: string): React.ReactNode[] {
   553→  if (!query) return [text];
   554→  const escaped = query.replace(/[.*+?^${}()|[\]\\]/g, "\\");
   555→  const re = new RegExp(`(${escaped})`, "gi");
   556→  const parts: React.ReactNode[] = [];
   557→  let lastIndex = 0;
   558→  let match: RegExpExecArray | null;
   559→  while ((match = re.exec(text)) !== null) {
   560→    if (match.index > lastIndex) {
   561→      parts.push(text.slice(lastIndex, match.index));
   562→    }
   563→    parts.push(
   564→      <mark
   565→        key={match.index}
   566→        className="bg-yellow-300/60 dark:bg-yellow-400/40 rounded-sm px-0.5"
   567→      >
   568→        {match[1]}
   569→      </mark>,
   570→    );
   571→    lastIndex = re.lastIndex;
   572→  }
   573→  if (lastIndex < text.length) {
   574→    parts.push(text.slice(lastIndex));
   575→  }
   576→  return parts;
   577→}

   579→export const LinkifiedText = React.memo(function LinkifiedText({
   580→  text,
   581→  className,
   582→  highlight,
   583→}: {
   584→  text: string;
   585→  className?: string;
   586→  highlight?: string;
   587→}) {
   588→  const segments = useMemo(() => parseUrls(text), [text]);
   589→  return (
   590→    <span className={className}>
   591→      {segments.map((seg, i) =>
   592→        seg.type === "link" ? (
   593→          <a
   594→            key={i}
   595→            href={seg.value}
   596→            target="_blank"
   597→            rel="noopener noreferrer"
   598→            className="underline underline-offset-2 transition-colors"
   599→            style={{ textDecorationColor: "var(--app-accent-light)" }}
   600→            onClick={(e) => e.stopPropagation()}
   601→          >
   602→            {seg.value}

   784→/* ------------------------------------------------------------------ */
   785→/*  Bubble rounding helper for grouped messages                        */
   786→/* ------------------------------------------------------------------ */
   787→
   788→function getBubbleRounding({
   789→  isFirst,
   790→  isLast,
   791→  isOwn,
   792→}: {
   793→  isFirst: boolean;
   794→  isLast: boolean;
   795→  isOwn: boolean;
   796→}): string {
   797→  if (isFirst && isLast) {
   798→    // Single message in group (or only message)
   799→    return isOwn
   800→      ? "rounded-2xl rounded-br-sm"
   801→      : "rounded-2xl rounded-bl-sm";
   802→  }
   803→  if (isFirst) {
   804→    // First message of a multi-message group
   805→    return isOwn
   806→      ? "rounded-tl-2xl rounded-bl-2xl rounded-tr-sm rounded-br-sm"
   807→      : "rounded-tl-2xl rounded-bl-sm rounded-tr-2xl rounded-br-sm";
   808→  }
   809→  if (isLast) {
   810→    // Last message of a multi-message group
   811→    return isOwn
   812→      ? "rounded-tl-2xl rounded-bl-2xl rounded-tr-sm rounded-br-2xl"
   813→      : "rounded-tl-2xl rounded-bl-2xl rounded-tr-2xl rounded-br-sm";
   814→  }
   815→  // Middle message
   816→  return isOwn
   817→    ? "rounded-tl-2xl rounded-bl-2xl rounded-tr-sm rounded-br-sm"
   818→    : "rounded-tl-2xl rounded-bl-sm rounded-tr-2xl rounded-br-sm";
   819→}

   833→  onReplyTo,
   834→  onReact,

  1314→function formatDateSeparator(dateStr: string): string {
  1315→  const d = new Date(dateStr);
  1316→  const now = new Date();
  1317→  const today = new Date(now.getFullYear(), now.getMonth(), now.getDate());
  1318→  const yesterday = new Date(today);
  1319→  yesterday.setDate(yesterday.getDate() - 1);
  1320→  const msgDay = new Date(d.getFullYear(), d.getMonth(), d.getDate());
  1321→
  1322→  if (msgDay.getTime() === today.getTime()) return "Today";
  1323→  if (msgDay.getTime() === yesterday.getTime()) return "Yesterday";
  1324→  return d.toLocaleDateString(undefined, { month: "short", day: "numeric" });
  1325→}

  1506→type MessageGroupData = { senderId: string; messages: ChatMessage[] };

  1508→export function MessageFeed({
  1509→  messages,
  1510→  isLoading,
  1511→  isPartnerTyping,
  1512→  currentUserId,
  1513→  partnerName,
  1514→  partnerAvatar,
  1515→  backgroundStyle,
  1516→  onMarkAsRead,
  1517→  onVanishMessage,
  1518→  onReplyTo,
  1519→  onDeleteMessage,
  1520→  onReact,
  1521→  scrollToMessageId,
  1522→  onScrolledToMessage,
  1523→  searchHighlight,
  1524→  onClearSearchHighlight,
  1525→  loadMore,
  1526→  hasMore,
  1527→  loadingMore,
  1528→}: MessageFeedProps) {
  1529→  const parentRef = useRef<HTMLDivElement>(null);
  1530→  const isAutoScrollRef = useRef(true);
  1531→  const [activeActionId, setActiveActionId] = useState<string | null>(null);
  1532→  const [showNewBtn, setShowNewBtn] = useState(false);
  1533→  const unreadRef = useRef(0);
  1534→  const prevCountRef = useRef(messages.length);
  1535→  const prevFirstIdRef = useRef(
  1536→    messages.length > 0 ? messages[0].id : null,
  1537→  );
  1538→  const prevScrollHeightRef = useRef(0);
  1539→
  1540→  /* ---- Build message groups ---------------------------------------- */
  1541→  const groups = useMemo(() => {
  1542→    const result: MessageGroupData[] = [];
  1543→    for (const msg of messages) {
  1544→      const last = result[result.length - 1];
  1545→      if (last && last.senderId === msg.sender_id) {
  1546→        last.messages.push(msg);
  1547→      } else {
  1548→        result.push({ senderId: msg.sender_id, messages: [msg] });
  1549→      }
  1550→    }
  1551→    return result;
  1552→  }, [messages]);
  1553→
  1554→  const totalItems = useMemo(
  1555→    () => groups.length + (isPartnerTyping ? 1 : 0),
  1556→    [groups.length, isPartnerTyping],
  1557→  );
  1558→
  1559→  const loadMoreRef = useRef(loadMore);
  1560→  loadMoreRef.current = loadMore;
  1561→
  1562→  const virtualizer = useVirtualizer({
  1563→    count: totalItems,
  1564→    getScrollElement: () => parentRef.current,
  1565→    estimateSize: () => ESTIMATED_ROW_HEIGHT,
  1566→    overscan: OVERSCAN,
  1567→    paddingStart: 16,
  1568→    paddingEnd: 16,
  1569→    gap: 12,
  1570→  });
  1571→
  1572→  const handleScroll = useCallback(
  1573→    (e: UIEvent<HTMLDivElement>) => {
  1574→      const el = e.currentTarget;
  1575→      const dist = el.scrollHeight - el.scrollTop - el.clientHeight;
  1576→      if (dist <= AUTO_SCROLL_THRESHOLD) {
  1577→        isAutoScrollRef.current = true;
  1578→        setShowNewBtn(false);
  1579→        unreadRef.current = 0;
  1580→      } else {
  1581→        isAutoScrollRef.current = false;
  1582→      }
  1583→      /* Trigger loadMore when scrolled near top */
  1584→      if (el.scrollTop < 300 && hasMore && !loadingMore) {
  1585→        loadMoreRef.current?.();
  1586→      }
  1587→    },
  1588→    [hasMore, loadingMore],
  1589→  );
  1590→
  1591→  const scrollToBottom = useCallback(
  1592→    (behavior: ScrollBehavior = "smooth") => {
  1593→      const el = parentRef.current;
  1594→      if (!el) return;
  1595→      el.scrollTo({ top: el.scrollHeight, behavior });
  1596→      isAutoScrollRef.current = true;
  1597→      setShowNewBtn(false);
  1598→      unreadRef.current = 0;
  1599→    },
  1600→    [],
  1601→  );
  1602→
  1603→  useEffect(() => {
  1604→    if (messages.length > 0)
  1605→      requestAnimationFrame(() => scrollToBottom("instant"));
  1606→  }, [isLoading, scrollToBottom]);
  1607→
  1608→  useEffect(() => {
  1609→    if (!onMarkAsRead) return;
  1610→    const unread = messages.filter(
  1611→      (m) => m.sender_id !== currentUserId && m.status !== "read",
  1612→    );
  1613→    if (unread.length > 0 && isAutoScrollRef.current)
  1614→      onMarkAsRead(unread.map((m) => m.id));
  1615→  }, [messages, currentUserId, onMarkAsRead]);
  1616→
  1617→  /* ---- Scroll to a specific message ----------------------------------- */
  1618→  useEffect(() => {
  1619→    if (!scrollToMessageId) return;
  1620→    // Find which group contains the target message
  1621→    const groupIdx = groups.findIndex((g) =>
  1622→      g.messages.some((m) => m.id === scrollToMessageId),
  1623→    );
  1624→    if (groupIdx === -1) {
  1625→      onScrolledToMessage?.();
  1626→      return;
  1627→    }
  1628→    virtualizer.scrollToIndex(groupIdx, { align: "center" });
  1629→    onScrolledToMessage?.();
  1630→  }, [scrollToMessageId, groups, virtualizer, onScrolledToMessage]);
  1631→
  1632→  /* ---- Handle message count changes (new or prepended) ------------ */
  1633→  useEffect(() => {
  1634→    const prev = prevCountRef.current;
  1635→    const prevFirstId = prevFirstIdRef.current;
  1636→    prevCountRef.current = messages.length;
  1637→    prevFirstIdRef.current =
  1638→      messages.length > 0 ? messages[0].id : null;
  1639→
  1640→    if (messages.length <= prev) return;
  1641→
  1642→    /* Check if older messages were prepended (first message changed) */
  1643→    const firstId = messages[0]?.id;
  1644→    const wasPrepended = firstId !== prevFirstId && prevFirstId !== null;
  1645→
  1646→    if (wasPrepended) {
  1647→      /* Older messages prepended – preserve scroll position */
  1648→      const el = parentRef.current;
  1649→      if (el) {
  1650→        const oldHeight = prevScrollHeightRef.current;
  1651→        prevScrollHeightRef.current = el.scrollHeight;
  1652→        requestAnimationFrame(() => {
  1653→          const newHeight = el.scrollHeight;
  1654→          el.scrollTop += newHeight - oldHeight;
  1655→        });
  1656→      }
  1657→      return;
  1658→    }
  1659→
  1660→    /* New messages appended at the end */
  1661→    const last = messages[messages.length - 1];
  1662→    const own = last && last.sender_id === currentUserId;
  1663→    if (own) {
  1664→      requestAnimationFrame(() => scrollToBottom("instant"));
  1665→      return;
  1666→    }
  1667→    if (isAutoScrollRef.current)
  1668→      requestAnimationFrame(() => scrollToBottom("smooth"));
  1669→    else {
  1670→      unreadRef.current += messages.length - prev;
  1671→      setShowNewBtn(true);
  1672→    }
  1673→  }, [messages.length, currentUserId, scrollToBottom]);
  1674→
  1675→  /* Track scroll height for scroll preservation */
  1676→  useEffect(() => {
  1677→    const el = parentRef.current;
  1678→    if (!el) return;
  1679→    prevScrollHeightRef.current = el.scrollHeight;
  1680→  }, [messages.length]);
  1681→
  1682→  if (isLoading) return <MessageListSkeleton />;
  1683→  if (messages.length === 0 && !isPartnerTyping)
  1684→    return (
  1685→      <EmptyState partnerName={partnerName} partnerAvatar={partnerAvatar} />
  1686→    );
  1687→
  1688→  const virtualItems = virtualizer.getVirtualItems();
  1689→
  1690→  return (
  1691→    <div
  1692→      className="relative h-full w-full overflow-hidden"
  1693→      style={backgroundStyle}
  1694→    >
  1695→      <div
  1696→        ref={parentRef}
  1697→        onScroll={handleScroll}
  1698→        onClick={() => {
  1699→          onClearSearchHighlight?.();
  1700→          setActiveActionId(null);
  1701→        }}
  1702→        className="h-full w-full select-none overflow-y-auto overscroll-contain"
  1703→        style={{ WebkitOverflowScrolling: "touch" }}
  1704→      >
  1705→        <div
  1706→          style={{
  1707→            height: `${virtualizer.getTotalSize()}px`,
  1708→            width: "100%",
  1709→            position: "relative",
  1710→          }}
  1711→        >
  1712→          {/* Loading more spinner */}
  1713→          {loadingMore && (
  1714→            <div
  1715→              style={{
  1716→                position: "absolute",
  1717→                top: 0,
  1718→                left: 0,
  1719→                width: "100%",
  1720→                height: "48px",
  1721→                zIndex: 5,
  1722→              }}
  1723→            >
  1724→              <div className="flex items-center justify-center py-3">
  1725→                <svg
  1726→                  className="size-5 animate-spin"
  1727→                  viewBox="0 0 24 24"
  1728→                  fill="none"
  1729→                  aria-label="Loading more messages"
  1730→                  style={{ color: "var(--app-accent-light)" }}
  1731→                >
  1732→                  <circle
  1733→                    className="opacity-25"
  1734→                    cx="12"
  1735→                    cy="12"
  1736→                    r="10"
  1737→                    stroke="currentColor"
  1738→                    strokeWidth="4"
  1739→                  />
  1740→                  <path
  1741→                    className="opacity-75"
  1742→                    fill="currentColor"
  1743→                    d="M4 12a8 8 0 018-8V0C5.373 0 0 5.373 0 12h4z"
  1744→                  />
  1745→                </svg>
  1746→              </div>
  1747→            </div>
  1748→          )}
  1749→          {virtualItems.map((vi) => {
  1750→            const isTyping = vi.index === groups.length;
  1751→            const group = isTyping ? null : groups[vi.index];
  1752→
  1753→            // Determine if we need a date separator before this group
  1754→            let showDateSep = false;
  1755→            if (group) {
  1756→              const firstMsg = group.messages[0];
  1757→              if (vi.index === 0) {
  1758→                showDateSep = true;
  1759→              } else {
  1760→                const prevGroup = groups[vi.index - 1];
  1761→                const prevLastMsg = prevGroup.messages[prevGroup.messages.length - 1];
  1762→                const prevDate = new Date(prevLastMsg.created_at);
  1763→                const currDate = new Date(firstMsg.created_at);
  1764→                showDateSep = prevDate.toDateString() !== currDate.toDateString();
  1765→              }
  1766→            }
  1767→
  1768→            return (
  1769→              <div
  1770→                key={vi.key}
  1771→                data-index={vi.index}
  1772→                ref={virtualizer.measureElement}
  1773→                style={{
  1774→                  position: "absolute",
  1775→                  top: 0,
  1776→                  left: 0,
  1777→                  width: "100%",
  1778→                  transform: `translateY(${vi.start}px)`,
  1779→                }}
  1780→              >
  1781→                {isTyping ? (
  1782→                  <TypingIndicator
  1783→                    partnerName={partnerName}
  1784→                    partnerAvatar={partnerAvatar}
  1785→                  />
  1786→                ) : group ? (
