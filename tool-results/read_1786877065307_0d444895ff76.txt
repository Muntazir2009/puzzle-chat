    46→function formatFileSize(bytes: number): string {
    47→  if (bytes < 1024) return `${bytes} B`;
    48→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
    49→  return `${(bytes / (1024 * 1024)).toFixed(1)} MB`;
    50→}
    51→
    52→function formatLastSeen(lastSeen: string | null): string {
    53→  if (!lastSeen) return "Offline";
    54→  const diffMs = Date.now() - new Date(lastSeen).getTime();
    55→  const diffSec = Math.floor(diffMs / 1000);
    56→  if (diffSec < 60) return "Seen just now";
    57→  const diffMin = Math.floor(diffSec / 60);
    58→  if (diffMin < 60) return `Seen ${diffMin}m ago`;
    59→  const diffHr = Math.floor(diffMin / 60);
    60→  if (diffHr < 24) return `Seen ${diffHr}h ago`;
    61→  const diffDay = Math.floor(diffHr / 24);
    62→  if (diffDay < 7) return `Seen ${diffDay}d ago`;
    63→  return `Seen ${new Date(lastSeen).toLocaleDateString()}`;
    64→}

    82→function SearchPanel({
    83→  conversationId,
    84→  onResultClick,
    85→  onClose,
    86→}: {
    87→  conversationId: string;
    88→  onResultClick: (messageId: string) => void;
    89→  onClose: () => void;
    90→}) {
    91→  const [query, setQuery] = useState("");
    92→  const [results, setResults] = useState<ChatMessage[]>([]);
    93→  const [isSearching, setIsSearching] = useState(false);
    94→  const inputRef = useRef<HTMLInputElement>(null);
    95→  const debounceRef = useRef<ReturnType<typeof setTimeout> | undefined>(undefined);
    96→
    97→  /* Auto-focus on open */
    98→  useEffect(() => {
    99→    requestAnimationFrame(() => inputRef.current?.focus());
   100→  }, []);
   101→
   102→  /* Escape key to close */
   103→  useEffect(() => {
   104→    function handleKeyDown(e: KeyboardEvent) {
   105→      if (e.key === "Escape") onClose();
   106→    }
   107→    window.addEventListener("keydown", handleKeyDown);
   108→    return () => window.removeEventListener("keydown", handleKeyDown);
   109→  }, [onClose]);
   110→
   111→  /* Debounced search */
   112→  useEffect(() => {
   113→    if (debounceRef.current) clearTimeout(debounceRef.current);
   114→    if (!query.trim()) { setResults([]); setIsSearching(false); return; }
   115→    setIsSearching(true);
   116→    debounceRef.current = setTimeout(async () => {
   117→      try {
   118→        const res = await fetch(`/api/messages/search?conversation_id=${conversationId}&q=${encodeURIComponent(query)}`);
   119→        if (res.ok) {
   120→          const data: ChatMessage[] = await res.json();
   121→          setResults(data);
   122→        }
   123→      } catch (_err) {
   124→        /* silent */
   125→      } finally {
   126→        setIsSearching(false);
   127→      }
   128→    }, 300);
   129→    return () => { if (debounceRef.current) clearTimeout(debounceRef.current); };
   130→  }, [query, conversationId]);
   131→
   132→  return (
   133→    <div
   134→      className="overflow-hidden border-b bg-background/80 backdrop-blur-xl"
   135→    >
   136→      <div className="flex items-center gap-2 px-4 py-2.5">
   137→        <Search className="size-4 shrink-0 text-muted-foreground" />
   138→        <input
   139→          ref={inputRef}
   140→          type="text"
   141→          value={query}
   142→          onChange={(e) => setQuery(e.target.value)}
   143→          placeholder="Search messages..."
   144→          className="flex-1 bg-transparent text-sm outline-none placeholder:text-muted-foreground"
   145→          aria-label="Search messages"
   146→        />
   147→        {isSearching ? (
   148→          <Loader2 className="size-4 shrink-0 animate-spin text-muted-foreground" />
   149→        ) : (
   150→          <button
   151→            type="button"
   152→            onClick={() => { setQuery(""); onClose(); }}
   153→            className="flex size-6 items-center justify-center rounded-full text-muted-foreground transition-colors hover:bg-muted"
   154→            aria-label="Close search"
   155→          >
   156→            <X className="size-3.5" />
   157→          </button>
   158→        )}
   159→      </div>
   160→
   161→      {/* Results list */}
   162→      {results.length > 0 && (
   163→        <div className="max-h-[200px] overflow-y-auto border-t px-2 pb-2">
   164→          {results.map((msg) => (
   165→            <button
   166→              key={msg.id}
   167→              type="button"
   168→              onClick={() => onResultClick(msg.id)}
   169→              className="flex w-full flex-col gap-0.5 rounded-lg px-3 py-2.5 text-left transition-colors hover:bg-muted/50"
   170→            >
   171→              <p className="truncate text-sm text-foreground">
   172→                {msg.content.length > 60 ? msg.content.slice(0, 60) + "..." : msg.content}
   173→              </p>
   174→              <span className="text-[11px] text-muted-foreground">
   175→                {formatDistanceToNow(new Date(msg.created_at), { addSuffix: true })}
   176→              </span>
   177→            </button>
   178→          ))}
   179→        </div>
   180→      )}
   181→
   182→      {query.trim() && !isSearching && results.length === 0 && (
   183→        <div className="border-t px-4 py-3 text-center text-xs text-muted-foreground">
   184→          No results found
   185→        </div>
   186→      )}
   187→    </div>
   188→  );
   189→}

   349→export function ChatLayout({ currentUserId, otherUserId, conversationId, partner, initialMessages }: ChatLayoutProps) {
   350→  const { messages, isLoading, isPartnerTyping, partnerStatus, sendMessage, onTyping, markAsRead, vanishMessage, deleteMessage, sendReaction } =
   351→    useChat({ currentUserId, otherUserId, conversationId, initialMessages });
   352→
   353→  const [draft, setDraft] = useState("");
   354→  const [sendBtnKey, setSendBtnKey] = useState(0);
   355→  const [isSending, setIsSending] = useState(false);
   356→  const [vanishMode, setVanishMode] = useState(false);
   357→  const [replyTo, setReplyTo] = useState<ChatMessage | null>(null);
   358→  const [ephemeralOpen, setEphemeralOpen] = useState(false);
   359→  const [ephemeralSeconds, setEphemeralSeconds] = useState<number | null>(null);
   360→  const [infoPanelOpen, setInfoPanelOpen] = useState(false);
   361→  const [searchOpen, setSearchOpen] = useState(false);
   362→  const [scrollToMessageId, setScrollToMessageId] = useState<string | null>(null);
   363→  const inputRef = useRef<HTMLTextAreaElement>(null);
   364→  const viewport = useVisualViewport();
   365→
   366→  /* Background theme */
   367→  const { themeId, theme, selectTheme } = useChatBackground();
   368→
   369→  /* Voice */
   370→  const voice = useVoiceRecorder();
   371→  const [showVoiceWaveform, setShowVoiceWaveform] = useState(false);
   372→
   373→  /* Attachment */
   374→  const [attachmentFile, setAttachmentFile] = useState<File | null>(null);
   375→  const [attachmentPreview, setAttachmentPreview] = useState<string | null>(null);
   376→  const [isUploading, setIsUploading] = useState(false);
   377→  const fileInputRef = useRef<HTMLInputElement>(null);
   378→
   379→  const containerStyle = viewport.height > 0 ? ({ height: `${viewport.height}px` } as React.CSSProperties) : ({ height: "100dvh" } as React.CSSProperties);
   380→  const kbOffset = viewport.isKeyboardVisible && viewport.offsetTop > 0 ? ({ marginTop: `-${viewport.offsetTop}px` } as React.CSSProperties) : undefined;
   381→
   382→  useEffect(() => { inputRef.current?.focus(); }, []);
   383→
   384→  /* Show voice waveform when recording */
   385→  useEffect(() => { setShowVoiceWaveform(voice.isRecording); }, [voice.isRecording]);
   386→
   387→  /* Global keyboard shortcuts */
   388→  useEffect(() => {
   389→    function handleGlobalKey(e: KeyboardEvent) {
   390→      if (e.key === "Escape") {
   391→        if (replyTo) { setReplyTo(null); e.preventDefault(); }
   392→        else if (ephemeralOpen) { setEphemeralOpen(false); e.preventDefault(); }
   393→        else if (infoPanelOpen) { setInfoPanelOpen(false); e.preventDefault(); }
   394→        else if (searchOpen) { setSearchOpen(false); e.preventDefault(); }
   395→      }
   396→    }
   397→    window.addEventListener("keydown", handleGlobalKey);
   398→    return () => window.removeEventListener("keydown", handleGlobalKey);
   399→  }, [replyTo, ephemeralOpen, infoPanelOpen, searchOpen]);
   400→
   401→  const handleSubmit = useCallback(async (e: FormEvent) => {
   402→    e.preventDefault();
   403→    /* If an attachment is selected, upload it instead of sending text */
   404→    if (hasAttachment && !hasText) {
   405→      await uploadAttachment();
   406→      return;
   407→    }
   408→    /* If there's both text and an attachment, send text first, then upload */
   409→    const trimmed = draft.trim();
   410→    if (!trimmed || isSending) return;
   411→    setIsSending(true);
   412→    const opts: SendMessageOptions = { vanish_mode: vanishMode, ephemeral_seconds: ephemeralSeconds, reply_to_id: replyTo?.id ?? null };
   413→    setDraft(""); setReplyTo(null);
   414→    if (inputRef.current) inputRef.current.style.height = "auto";
   415→    try {
   416→      await sendMessage(trimmed, opts);
   417→      setSendBtnKey((k) => k + 1);
   418→      /* Upload attachment after text is sent */
   419→      if (hasAttachment) await uploadAttachment();
   420→    }
   421→    finally { setIsSending(false); requestAnimationFrame(() => inputRef.current?.focus()); }
   422→  }, [draft, isSending, sendMessage, vanishMode, ephemeralSeconds, replyTo, hasAttachment, uploadAttachment]);
   423→
   424→  const handleVoiceSend = useCallback(async () => {
   425→    const result = await voice.stopRecording();
   426→    if (!result || result.duration < 1) return;
   427→    const blob = result.blob;
   428→    const url = URL.createObjectURL(blob);
   429→    await sendMessage(url, { type: "voice", voice_duration: result.duration, waveform_data: result.amplitudes, ephemeral_seconds: ephemeralSeconds });
   430→    URL.revokeObjectURL(url);
   431→  }, [voice, sendMessage, ephemeralSeconds]);
   432→
   433→  const handleKeyDown = useCallback((e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   434→    if (e.key === "Enter" && !e.shiftKey) { e.preventDefault(); const form = e.currentTarget.closest("form"); if (form) form.requestSubmit(); }
   435→  }, []);
   436→
   437→  const handleInputChange = useCallback((e: React.ChangeEvent<HTMLTextAreaElement>) => { setDraft(e.target.value); onTyping(); }, [onTyping]);
   438→  const handleInput = useCallback((e: React.FormEvent<HTMLTextAreaElement>) => { const el = e.currentTarget; el.style.height = "auto"; el.style.height = `${Math.min(el.scrollHeight, 128)}px`; }, []);
   439→
   440→  const hasText = draft.trim().length > 0;
   441→  const canSend = hasText || hasAttachment;
   442→
   443→  /* Search result click → scroll to message */
   444→  const handleSearchResultClick = useCallback((messageId: string) => {
   445→    setScrollToMessageId(messageId);
   446→    setSearchOpen(false);
   447→  }, []);
   448→
   449→  const handleScrolledToMessage = useCallback(() => {
   450→    setScrollToMessageId(null);
   451→  }, []);
   452→
   453→  /* ---- Attachment handlers ---------------------------------------- */
   454→  const handleFileSelect = useCallback((e: ChangeEvent<HTMLInputElement>) => {
   455→    const file = e.target.files?.[0] ?? null;
   456→    if (!file) return;
   457→    if (file.size > MAX_ATTACHMENT_SIZE) {
   458→      toast({ title: "File too large", description: "Maximum size is 10 MB.", variant: "destructive" });
   459→      e.target.value = "";
   460→      return;
   461→    }
   462→    setAttachmentFile(file);
   463→    /* Generate thumbnail preview for images */
   464→    if (IMAGE_MIME_TYPES.has(file.type)) {
   465→      const url = URL.createObjectURL(file);
   466→      setAttachmentPreview(url);
   467→    } else {
   468→      setAttachmentPreview(null);
   469→    }
   470→    /* Reset input so the same file can be re-selected */
   471→    e.target.value = "";
   472→  }, []);
   473→
   474→  const clearAttachment = useCallback(() => {
   475→    if (attachmentPreview) URL.revokeObjectURL(attachmentPreview);
   476→    setAttachmentFile(null);
   477→    setAttachmentPreview(null);
   478→  }, [attachmentPreview]);
   479→
   480→  const uploadAttachment = useCallback(async () => {
   481→    if (!attachmentFile || isUploading) return;
   482→    setIsUploading(true);
   483→    try {
   484→      const fd = new FormData();
   485→      fd.append("file", attachmentFile);
   486→      fd.append("conversation_id", conversationId);
   487→      if (replyTo?.id) fd.append("reply_to_id", replyTo.id);
   488→      if (vanishMode) fd.append("vanish_mode", "true");
   489→      if (ephemeralSeconds) fd.append("ephemeral_seconds", String(ephemeralSeconds));
   490→
   491→      const res = await fetch("/api/messages/upload", { method: "POST", body: fd });
   492→      if (!res.ok) {
   493→        const data = await res.json().catch(() => ({ error: "Upload failed" }));
   494→        throw new Error(data.error || `Upload failed: ${res.status}`);
   495→      }
   496→      /* Message is inserted server-side and pushed via Pusher;
   497→         the useChat hook will pick it up automatically. */
   498→      clearAttachment();
   499→      setReplyTo(null);
   500→    } catch (err) {
   501→      console.error("[ChatLayout] upload error:", err);
   502→      toast({
   503→        title: "Upload failed",
   504→        description: err instanceof Error ? err.message : "Could not send the file.",
   505→        variant: "destructive",
   506→      });
   507→    } finally {
   508→      setIsUploading(false);
   509→    }
   510→  }, [attachmentFile, isUploading, conversationId, replyTo, vanishMode, ephemeralSeconds, clearAttachment]);
   511→
   512→  const handlePaperclipClick = useCallback(() => {
   513→    fileInputRef.current?.click();
   514→  }, []);
   515→
   516→  const hasAttachment = attachmentFile !== null;
   517→
   518→  /* Status text for header */
   519→  const statusText = isPartnerTyping
   520→    ? "typing\u2026"
   521→    : partnerStatus.online
   522→      ? "Online"
   523→      : formatLastSeen(partnerStatus.last_seen);
   524→
   525→  /* ---- Render --------------------------------------------------- */
   526→
   527→  return (
   528→    <div className="flex w-full flex-col bg-background" style={containerStyle}>
