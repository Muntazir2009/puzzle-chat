   123→   123→
   124→   124→function EphemeralTimer({
   125→   125→  seconds,
   126→   126→  onExpire,
   127→   127→}: {
   128→   128→  seconds: number;
   129→   129→  onExpire: () => void;
   130→   130→}) {
   131→   131→  const [remaining, setRemaining] = useState(seconds);
   132→   132→  const intervalRef = useRef<ReturnType<typeof setInterval> | undefined>(
   133→   133→    undefined,
   134→   134→  );
   135→   135→
   136→   136→  useEffect(() => {
   137→   137→    intervalRef.current = setInterval(() => {
   138→   138→      setRemaining((r) => {
   139→   139→        if (r <= 1) {
   140→   140→          clearInterval(intervalRef.current);
   141→   141→          onExpire();
   142→   142→          return 0;
   143→   143→        }
   144→   144→        return r - 1;
   145→   145→      });
   146→   146→    }, 1000);
   147→   147→    return () => clearInterval(intervalRef.current);
   148→   148→  }, [seconds, onExpire]);
   149→   149→
   150→   150→  const pct = remaining / seconds;
   151→   151→  const circ = 2 * Math.PI * 7;
   152→   152→  const offset = circ * (1 - pct);
   153→   153→
   154→   154→  return (
   155→   155→    <svg
   156→   156→      className="size-3.5 shrink-0 ephemeral-timer-pulse"
   157→   157→      viewBox="0 0 18 18"
   158→   158→      aria-label={`${remaining}s remaining`}
   159→   159→    >
   160→   160→      <circle
   161→   161→        cx="9"
   162→   162→        cy="9"
   163→   163→        r="7"
   164→   164→        fill="none"
   165→   165→        stroke="currentColor"
   166→   166→        strokeWidth="1.5"
   167→   167→        className="text-white/20"
   168→   168→      />
   169→   169→      <circle
   170→   170→        cx="9"
   171→   171→        cy="9"
   172→   172→        r="7"
   173→   173→        fill="none"
   174→   174→        stroke="currentColor"
   175→   175→        strokeWidth="1.5"
   176→   176→        strokeDasharray={circ}
   177→   177→        strokeDashoffset={offset}
   178→   178→        strokeLinecap="round"
   179→   179→        style={{
   180→   180→          color: "var(--app-accent-lighter)",
   181→   181→          transform: "rotate(-90deg)",
   182→   182→          transformOrigin: "50% 50%",
   183→   183→          transition: "stroke-dashoffset 1s linear",
   184→   184→        }}
   185→   185→      />
   186→   186→    </svg>
   187→   187→  );
   188→   188→}

   398→   398→/* ------------------------------------------------------------------ */
   399→   399→
   400→   400→function VoiceBubble({
   401→   401→  message,
   402→   402→  isOwn,
   403→   403→  waveform,
   404→   404→}: {
   405→   405→  message: ChatMessage;
   406→   406→  isOwn: boolean;
   407→   407→  waveform: number[];
   408→   408→}) {
   409→   409→  const audioRef = useRef<HTMLAudioElement>(null);
   410→   410→  const [playing, setPlaying] = useState(false);
   411→   411→  const [progress, setProgress] = useState(0);
   412→   412→  const [speed, setSpeed] = useState(1);
   413→   413→  const speeds = [1, 1.5, 2];
   414→   414→  const duration = message.voice_duration ?? 0;
   415→   415→
   416→   416→  useEffect(() => {
   417→   417→    const audio = audioRef.current;
   418→   418→    if (!audio) return;
   419→   419→    const onTime = () => {
   420→   420→      if (audio.duration) setProgress(audio.currentTime / audio.duration);
   421→   421→    };
   422→   422→    const onEnd = () => setPlaying(false);
   423→   423→    audio.addEventListener("timeupdate", onTime);
   424→   424→    audio.addEventListener("ended", onEnd);
   425→   425→    return () => {
   426→   426→      audio.removeEventListener("timeupdate", onTime);
   427→   427→      audio.removeEventListener("ended", onEnd);
   428→   428→    };
   429→   429→  }, []);
   430→   430→
   431→   431→  const toggle = useCallback(() => {
   432→   432→    const audio = audioRef.current;
   433→   433→    if (!audio) return;
   434→   434→    if (playing) {
   435→   435→      audio.pause();
   436→   436→      setPlaying(false);
   437→   437→    } else {
   438→   438→      audio.playbackRate = speed;
   439→   439→      audio.play().catch(() => {});
   440→   440→      setPlaying(true);
   441→   441→    }
   442→   442→  }, [playing, speed]);
   443→   443→
   444→   444→  const cycleSpeed = useCallback(() => {
   445→   445→    setSpeed((s) => {
   446→   446→      const idx = speeds.indexOf(s);
   447→   447→      const next = speeds[(idx + 1) % speeds.length];
   448→   448→      if (audioRef.current) audioRef.current.playbackRate = next;
   449→   449→      return next;
   450→   450→    });
   451→   451→  }, []);
   452→   452→
   453→   453→  /* Build waveform bars */
   454→   454→  const bars = useMemo(() => {
   455→   455→    const count = Math.min(waveform.length, 40);
   456→   456→    const step = Math.max(1, Math.floor(waveform.length / count));
   457→   457→    return Array.from({ length: count }, (_, i) => waveform[i * step] ?? 0.1);
   458→   458→  }, [waveform]);
   459→   459→
   460→   460→  const progressIdx = Math.floor(progress * bars.length);
   461→   461→
   462→   462→  return (
   463→   463→    <div className={cn("flex items-center gap-3", isOwn ? "flex-row-reverse" : "")}>
   464→   464→      <button
   465→   465→        type="button"
   466→   466→        onClick={toggle}
   467→   467→        className={cn(
   468→   468→          "flex size-9 shrink-0 items-center justify-center rounded-full transition-all duration-150 active:scale-95",
   469→   469→          isOwn
   470→   470→            ? "bg-white/20 hover:bg-white/30"
   471→   471→            : "bg-zinc-600 hover:bg-[var(--app-accent-subtle)]",
   472→   472→        )}
   473→   473→        aria-label={playing ? "Pause" : "Play"}
   474→   474→      >
   475→   475→        {playing ? (
   476→   476→          <Pause className="size-3.5 text-white" />
   477→   477→        ) : (
   478→   478→          <Play className="size-3.5 text-white ml-0.5" />
   479→   479→        )}
   480→   480→      </button>
   481→   481→      <div className="flex items-center gap-[2px]" style={{ height: 32 }}>
   482→   482→        {bars.map((amp, i) => (
   483→   483→          <div
   484→   484→            key={i}
   485→   485→            className="w-[2.5px] rounded-full transition-all duration-100"
   486→   486→            style={{
   487→   487→              height: `${Math.max(4, amp * 32)}px`,
   488→   488→              backgroundColor:
   489→   489→                i < progressIdx
   490→   490→                  ? isOwn
   491→   491→                    ? "rgba(255,255,255,1)"
   492→   492→                    : "var(--app-accent-light)"
   493→   493→                  : isOwn
   494→   494→                    ? "rgba(255,255,255,0.5)"
   495→   495→                    : "rgba(161,161,170,0.7)",
   496→   496→            }}
   497→   497→          />

   525→   525→/* ------------------------------------------------------------------ */
   526→   526→
   527→   527→const URL_REGEX = /(https?:\/\/[^\s<]+)/g;
   528→   528→
   529→   529→type TextSegment =
   530→   530→  | { type: "text"; value: string }
   531→   531→  | { type: "link"; value: string };
   532→   532→
   533→   533→function parseUrls(text: string): TextSegment[] {
   534→   534→  const segments: TextSegment[] = [];
   535→   535→  let lastIndex = 0;
   536→   536→  let match: RegExpExecArray | null;
   537→   537→  while ((match = URL_REGEX.exec(text)) !== null) {
   538→   538→    if (match.index > lastIndex) {
   539→   539→      segments.push({ type: "text", value: text.slice(lastIndex, match.index) });
   540→   540→    }
   541→   541→    segments.push({ type: "link", value: match[1] });
   542→   542→    lastIndex = URL_REGEX.lastIndex;
   543→   543→  }
   544→   544→  if (lastIndex < text.length) {
   545→   545→    segments.push({ type: "text", value: text.slice(lastIndex) });
   546→   546→  }
   547→   547→  return segments;
   548→   548→}
   549→   549→
   550→   550→/** Split a plain-text string into highlighted / non-highlighted fragments */
   551→   551→function highlightText(text: string, query: string): React.ReactNode[] {
   552→   552→  if (!query) return [text];
   553→   553→  const escaped = query.replace(/[.*+?^${}()|[\]\\]/g, "\\$");
   554→   554→  const re = new RegExp(`(${escaped})`, "gi");
   555→   555→  const parts: React.ReactNode[] = [];
   556→   556→  let lastIndex = 0;
   557→   557→  let match: RegExpExecArray | null;
   558→   558→  while ((match = re.exec(text)) !== null) {
   559→   559→    if (match.index > lastIndex) {
   560→   560→      parts.push(text.slice(lastIndex, match.index));
   561→   561→    }
   562→   562→    parts.push(
   563→   563→      <mark
   564→   564→        key={match.index}
   565→   565→        className="bg-yellow-300/60 dark:bg-yellow-400/40 rounded-sm px-0.5"
   566→   566→      >
   567→   567→        {match[1]}
   568→   568→      </mark>,
   569→   569→    );
   570→   570→    lastIndex = re.lastIndex;
   571→   571→  }
   572→   572→  if (lastIndex < text.length) {
   573→   573→    parts.push(text.slice(lastIndex));
   574→   574→  }
   575→   575→  return parts;
   576→   576→}

  1432→  1432→}

  1436→  1436→/* ------------------------------------------------------------------ */
  1437→  1437→
  1438→  1438→export function MessageFeed({
  1439→  1439→  messages,
  1440→  1440→  isLoading,
  1441→  1441→  isPartnerTyping,
  1442→  1442→  currentUserId,
  1443→  1443→  partnerName,
  1444→  1444→  partnerAvatar,
  1445→  1445→  backgroundStyle,
  1446→  1446→  onMarkAsRead,
  1447→  1447→  onVanishMessage,
  1448→  1448→  onReplyTo,
  1449→  1449→  onDeleteMessage,
  1450→  1450→  onReact,
  1451→  1451→  scrollToMessageId,
  1452→  1452→  onScrolledToMessage,
  1453→  1453→  searchHighlight,
  1454→  1454→  onClearSearchHighlight,
  1455→  1455→  loadMore,
  1456→  1456→  hasMore,
  1457→  1457→  loadingMore,
  1458→  1458→}: MessageFeedProps) {
  1459→  1459→  const parentRef = useRef<HTMLDivElement>(null);
  1460→  1460→  const isAutoScrollRef = useRef(true);
  1461→  1461→  const [activeActionId, setActiveActionId] = useState<string | null>(null);
  1462→  1462→  const [showNewBtn, setShowNewBtn] = useState(false);
  1463→  1463→  const unreadRef = useRef(0);
  1464→  1464→  const prevCountRef = useRef(messages.length);
  1465→  1465→  const prevFirstIdRef = useRef(
  1466→  1466→    messages.length > 0 ? messages[0].id : null,
  1467→  1467→  );
  1468→  1468→  const prevScrollHeightRef = useRef(0);
  1469→  1469→
  1470→  1470→  const totalItems = useMemo(
  1471→  1471→    () => messages.length + (isPartnerTyping ? 1 : 0),
  1472→  1472→    [messages.length, isPartnerTyping],
  1473→  1473→  );
  1474→  1474→  const loadMoreRef = useRef(loadMore);
  1475→  1475→  loadMoreRef.current = loadMore;
  1476→  1476→
  1477→  1477→  const virtualizer = useVirtualizer({
  1478→  1478→    count: totalItems,
  1479→  1479→    getScrollElement: () => parentRef.current,
  1480→  1480→    estimateSize: () => ESTIMATED_ROW_HEIGHT,
  1481→  1481→    overscan: OVERSCAN,
  1482→  1482→    paddingStart: 16,
  1483→  1483→    paddingEnd: 16,
  1484→  1484→    gap: 12,
  1485→  1485→  });
  1486→  1486→
  1487→  1487→  const handleScroll = useCallback(
  1488→  1488→    (e: UIEvent<HTMLDivElement>) => {
  1489→  1489→      const el = e.currentTarget;
  1490→  1490→      const dist = el.scrollHeight - el.scrollTop - el.clientHeight;
  1491→  1491→      if (dist <= AUTO_SCROLL_THRESHOLD) {
  1492→  1492→        isAutoScrollRef.current = true;
  1493→  1493→        setShowNewBtn(false);
  1494→  1494→        unreadRef.current = 0;
  1495→  1495→      } else {
  1496→  1496→        isAutoScrollRef.current = false;
  1497→  1497→      }
  1498→  1498→      /* Trigger loadMore when scrolled near top */
  1499→  1499→      if (el.scrollTop < 300 && hasMore && !loadingMore) {
  1500→  1500→        loadMoreRef.current?.();
  1501→  1501→      }
  1502→  1502→    },
  1503→  1503→    [hasMore, loadingMore],
  1504→  1504→  );
  1505→  1505→
  1506→  1506→  const scrollToBottom = useCallback(
  1507→  1507→    (behavior: ScrollBehavior = "smooth") => {
  1508→  1508→      const el = parentRef.current;
  1509→  1509→      if (!el) return;
  1510→  1510→      el.scrollTo({ top: el.scrollHeight, behavior });
  1511→  1511→      isAutoScrollRef.current = true;
  1512→  1512→      setShowNewBtn(false);
  1513→  1513→      unreadRef.current = 0;
  1514→  1514→    },
  1515→  1515→    [],
  1516→  1516→  );
  1517→  1517→
  1518→  1518→  useEffect(() => {
  1519→  1519→    if (messages.length > 0)
  1520→  1520→      requestAnimationFrame(() => scrollToBottom("instant"));
  1521→  1521→  }, [isLoading, scrollToBottom]);
  1522→  1522→
  1523→  1523→  useEffect(() => {
  1524→  1524→    if (!onMarkAsRead) return;
  1525→  1525→    const unread = messages.filter(
  1526→  1526→      (m) => m.sender_id !== currentUserId && m.status !== "read",
  1527→  1527→    );
  1528→  1528→    if (unread.length > 0 && isAutoScrollRef.current)
  1529→  1529→      onMarkAsRead(unread.map((m) => m.id));
  1530→  1530→  }, [messages, currentUserId, onMarkAsRead]);
  1531→  1531→
  1532→  1532→  /* ---- Scroll to a specific message ----------------------------------- */
  1533→  1533→  useEffect(() => {
  1534→  1534→    if (!scrollToMessageId) return;
  1535→  1535→    const idx = messages.findIndex((m) => m.id === scrollToMessageId);
  1536→  1536→    if (idx === -1) {
  1537→  1537→      onScrolledToMessage?.();
  1538→  1538→      return;
  1539→  1539→    }
  1540→  1540→    virtualizer.scrollToIndex(idx, { align: "center" });
  1541→  1541→    onScrolledToMessage?.();
  1542→  1542→  }, [scrollToMessageId, messages, virtualizer, onScrolledToMessage]);
  1543→  1543→
  1544→  1544→  /* ---- Handle message count changes (new or prepended) ------------ */
  1545→  1545→  useEffect(() => {
  1546→  1546→    const prev = prevCountRef.current;
  1547→  1547→    const prevFirstId = prevFirstIdRef.current;
  1548→  1548→    prevCountRef.current = messages.length;
  1549→  1549→    prevFirstIdRef.current =
  1550→  1550→      messages.length > 0 ? messages[0].id : null;
  1551→  1551→
  1552→  1552→    if (messages.length <= prev) return;
  1553→  1553→
  1554→  1554→    /* Check if older messages were prepended (first message changed) */
  1555→  1555→    const firstId = messages[0]?.id;
  1556→  1556→    const wasPrepended = firstId !== prevFirstId && prevFirstId !== null;
  1557→  1557→
  1558→  1558→    if (wasPrepended) {
  1559→  1559→      /* Older messages prepended – preserve scroll position */
  1560→  1560→      const el = parentRef.current;
  1561→  1561→      if (el) {
  1562→  1562→        const oldHeight = prevScrollHeightRef.current;
  1563→  1563→        prevScrollHeightRef.current = el.scrollHeight;
  1564→  1564→        requestAnimationFrame(() => {
  1565→  1565→          const newHeight = el.scrollHeight;
  1566→  1566→          el.scrollTop += newHeight - oldHeight;
  1567→  1567→        });
  1568→  1568→      }
  1569→  1569→      return;
  1570→  1570→    }
  1571→  1571→
  1572→  1572→    /* New messages appended at the end */
  1573→  1573→    const last = messages[messages.length - 1];
  1574→  1574→    const own = last && last.sender_id === currentUserId;
  1575→  1575→    if (own) {
  1576→  1576→      requestAnimationFrame(() => scrollToBottom("instant"));
  1577→  1577→      return;
  1578→  1578→    }
  1579→  1579→    if (isAutoScrollRef.current)
  1580→  1580→      requestAnimationFrame(() => scrollToBottom("smooth"));
  1581→  1581→    else {
  1582→  1582→      unreadRef.current += messages.length - prev;
  1583→  1583→      setShowNewBtn(true);
  1584→  1584→    }
  1585→  1585→  }, [messages.length, currentUserId, scrollToBottom]);
  1586→  1586→
  1587→  1587→  /* Track scroll height for scroll preservation */
  1588→  1588→  useEffect(() => {
  1589→  1589→    const el = parentRef.current;
  1590→  1590→    if (!el) return;
  1591→  1591→    prevScrollHeightRef.current = el.scrollHeight;
  1592→  1592→  }, [messages.length]);
  1593→  1593→
  1594→  1594→  if (isLoading) return <MessageListSkeleton />;
  1595→  1595→  if (messages.length === 0 && !isPartnerTyping)
  1596→  1596→    return (
  1597→  1597→      <EmptyState partnerName={partnerName} partnerAvatar={partnerAvatar} />
  1598→  1598→    );
  1599→  1599→
  1600→  1600→  const virtualItems = virtualizer.getVirtualItems();
  1601→  1601→
  1602→  1602→  return (
  1603→  1603→    <div
  1604→  1604→      className="relative h-full w-full overflow-hidden"
  1605→  1605→      style={backgroundStyle}
  1606→  1606→    >
  1607→  1607→      <div
  1608→  1608→        ref={parentRef}
  1609→  1609→        onScroll={handleScroll}
  1610→  1610→        onClick={() => { onClearSearchHighlight?.(); setActiveActionId(null); }}
  1611→  1611→        className="h-full w-full select-none overflow-y-auto overscroll-contain"
  1612→  1612→        style={{ WebkitOverflowScrolling: "touch" }}
  1613→  1613→      >
  1614→  1614→        <div
  1615→  1615→          style={{
  1616→  1616→            height: `${virtualizer.getTotalSize()}px`,
  1617→  1617→            width: "100%",
  1618→  1618→            position: "relative",
  1619→  1619→          }}
  1620→  1620→        >
  1621→  1621→          {/* Loading more spinner */}
  1622→  1622→          {loadingMore && (
  1623→  1623→            <div
  1624→  1624→              style={{
  1625→  1625→                position: "absolute",
  1626→  1626→                top: 0,
  1627→  1627→                left: 0,
  1628→  1628→                width: "100%",
  1629→  1629→                height: "48px",
  1630→  1630→                zIndex: 5,
  1631→  1631→              }}
  1632→  1632→            >
  1633→  1633→              <div className="flex items-center justify-center py-3">
  1634→  1634→                <svg
  1635→  1635→                  className="size-5 animate-spin"
  1636→  1636→                  viewBox="0 0 24 24"
  1637→  1637→                  fill="none"
  1638→  1638→                  aria-label="Loading more messages"
  1639→  1639→                  style={{ color: "var(--app-accent-light)" }}
  1640→  1640→                >
  1641→  1641→                  <circle
  1642→  1642→                    className="opacity-25"
  1643→  1643→                    cx="12"
  1644→  1644→                    cy="12"
  1645→  1645→                    r="10"
  1646→  1646→                    stroke="currentColor"
  1647→  1647→                    strokeWidth="4"
  1648→  1648→                  />
  1649→  1649→                  <path
  1650→  1650→                    className="opacity-75"
  1651→  1651→                    fill="currentColor"
  1652→  1652→                    d="M4 12a8 8 0 018-8V0C5.373 0 0 5.373 0 12h4z"
  1653→  1653→                  />
  1654→  1654→                </svg>
  1655→  1655→              </div>
  1656→  1656→            </div>
  1657→  1657→          )}
  1658→  1658→          {virtualItems.map((vi) => {
  1659→  1659→            const isTyping = vi.index === messages.length;
  1660→  1660→            const msg = isTyping ? null : messages[vi.index];
  1661→  1661→            // Check if we need a date separator before this message
  1662→  1662→            const showDateSep =
  1663→  1663→              msg &&
  1664→  1664→              vi.index > 0 &&
  1665→  1665→              messages[vi.index - 1] &&
  1666→  1666→              (() => {
  1667→  1667→                const prev = new Date(messages[vi.index - 1].created_at);
  1668→  1668→                const curr = new Date(msg.created_at);
  1669→  1669→                return prev.toDateString() !== curr.toDateString();
  1670→  1670→              })();
  1671→  1671→            const showFirstDateSep = msg && vi.index === 0;
  1672→  1672→            return (
  1673→  1673→              <div
  1674→  1674→                key={vi.key}
  1675→  1675→                data-index={vi.index}
  1676→  1676→                ref={virtualizer.measureElement}
  1677→  1677→                style={{
  1678→  1678→                  position: "absolute",
  1679→  1679→                  top: 0,
  1680→  1680→                  left: 0,
  1681→  1681→                  width: "100%",
  1682→  1682→                  transform: `translateY(${vi.start}px)`,
  1683→  1683→                }}
  1684→  1684→              >
  1685→  1685→                {isTyping ? (
  1686→  1686→                  <TypingIndicator
  1687→  1687→                    partnerName={partnerName}
  1688→  1688→                    partnerAvatar={partnerAvatar}
  1689→  1689→                  />
  1690→  1690→                ) : msg ? (
  1691→  1691→                  <>
  1692→  1692→                    {(showDateSep || showFirstDateSep) && (
  1693→  1693→                      <DateSeparator date={msg.created_at} />
  1694→  1694→                    )}
  1695→  1695→                    <MessageBubble
  1696→  1696→                      message={msg}
  1697→  1697→                      isOwn={msg.sender_id === currentUserId}
  1698→  1698→                      partnerName={partnerName}
  1699→  1699→                      partnerAvatar={partnerAvatar}
  1700→  1700→                      currentUserId={currentUserId}
  1701→  1701→                      onReplyTo={onReplyTo}
  1702→  1702→                      onReact={onReact}
  1703→  1703→                      onDeleteMessage={onDeleteMessage}
  1704→  1704→                      searchHighlight={searchHighlight}
  1705→  1705→                      activeActionId={activeActionId}
  1706→  1706→                      onOpenAction={setActiveActionId}
  1707→  1707→                    />
  1708→  1708→                  </>

  1712→  1712→          })}
  1713→  1713→        </div>
  1714→  1714→      </div>
  1715→  1715→      {showNewBtn && unreadRef.current > 0 && (
  1716→  1716→        <NewMessagesButton
  1717→  1717→          count={unreadRef.current}
  1718→  1718→          onClick={() => scrollToBottom("smooth")}
  1719→  1719→        />
  1720→  1720→      )}
  1721→  1721→    </div>
  1722→  1722→  );
  1723→  1723→}
