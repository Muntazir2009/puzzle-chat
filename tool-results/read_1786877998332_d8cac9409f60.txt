    90→
    91→function EphemeralTimer({ seconds, onExpire }: { seconds: number; onExpire: () => void }) {
    92→  const [remaining, setRemaining] = useState(seconds);
    93→  const intervalRef = useRef<ReturnType<typeof setInterval> | undefined>(undefined);
    94→
    95→  useEffect(() => {
    96→    intervalRef.current = setInterval(() => {
    97→      setRemaining((r) => {
    98→        if (r <= 1) { clearInterval(intervalRef.current); onExpire(); return 0; }
    99→        return r - 1;
   100→      });
   101→    }, 1000);
   102→    return () => clearInterval(intervalRef.current);
   103→  }, [seconds, onExpire]);
   104→
   105→  const pct = remaining / seconds;
   106→  const circ = 2 * Math.PI * 7;
   107→  const offset = circ * (1 - pct);
   108→
   109→  return (
   110→    <svg className="size-4 shrink-0 ephemeral-timer-pulse" viewBox="0 0 18 18" aria-label={`${remaining}s remaining`}>
   111→      <circle cx="9" cy="9" r="7" fill="none" stroke="currentColor" strokeWidth="1.5" className="text-white/20" />
   112→      <circle cx="9" cy="9" r="7" fill="none" stroke="currentColor" strokeWidth="1.5" strokeDasharray={circ} strokeDashoffset={offset} strokeLinecap="round" className="text-violet-300" style={{ transform: "rotate(-90deg)", transformOrigin: "50% 50%", transition: "stroke-dashoffset 1s linear" }} />
   113→    </svg>
   114→  );
   115→}

   198→function VoiceBubble({ message, isOwn, waveform }: {
   199→  message: ChatMessage; isOwn: boolean; waveform: number[];
   200→}) {
   201→  const audioRef = useRef<HTMLAudioElement>(null);
   202→  const [playing, setPlaying] = useState(false);
   203→  const [progress, setProgress] = useState(0);
   204→  const [speed, setSpeed] = useState(1);
   205→  const speeds = [1, 1.5, 2];
   206→  const duration = message.voice_duration ?? 0;
   207→
   208→  useEffect(() => {
   209→    const audio = audioRef.current;
   210→    if (!audio) return;
   211→    const onTime = () => { if (audio.duration) setProgress(audio.currentTime / audio.duration); };
   212→    const onEnd = () => setPlaying(false);
   213→    audio.addEventListener("timeupdate", onTime);
   214→    audio.addEventListener("ended", onEnd);
   215→    return () => { audio.removeEventListener("timeupdate", onTime); audio.removeEventListener("ended", onEnd); };
   216→  }, []);
   217→
   218→  const toggle = useCallback(() => {
   219→    const audio = audioRef.current;
   220→    if (!audio) return;
   221→    if (playing) { audio.pause(); setPlaying(false); }
   222→    else { audio.playbackRate = speed; audio.play().catch(() => {}); setPlaying(true); }
   223→  }, [playing, speed]);
   224→
   225→  const cycleSpeed = useCallback(() => {
   226→    setSpeed((s) => {
   227→      const idx = speeds.indexOf(s);
   228→      const next = speeds[(idx + 1) % speeds.length];
   229→      if (audioRef.current) audioRef.current.playbackRate = next;
   230→      return next;
   231→    });
   232→  }, []);
   233→
   234→  /* Build waveform bars */
   235→  const bars = useMemo(() => {
   236→    const count = Math.min(waveform.length, 40);
   237→    const step = Math.max(1, Math.floor(waveform.length / count));
   238→    return Array.from({ length: count }, (_, i) => waveform[i * step] ?? 0.1);
   239→  }, [waveform]);
   240→
   241→  const barColor = isOwn ? "bg-white/50" : "bg-zinc-400/70";
   242→  const progressColor = isOwn ? "bg-white" : "bg-violet-400";
   243→  const progressIdx = Math.floor(progress * bars.length);
   244→
   245→  return (
   246→    <div className={cn("flex items-center gap-3", isOwn ? "flex-row-reverse" : "")}>
   247→      <button type="button" onClick={toggle} className={cn("flex size-9 shrink-0 items-center justify-center rounded-full transition-all duration-150", isOwn ? "bg-white/20 hover:bg-white/30 hover:scale-105" : "bg-zinc-600 hover:bg-zinc-500 hover:scale-105")} aria-label={playing ? "Pause" : "Play"}>
   248→        {playing ? <Pause className="size-3.5 text-white" /> : <Play className="size-3.5 text-white ml-0.5" />}
   249→      </button>
   250→      <div className="flex items-center gap-[2px]" style={{ height: 32 }}>
   251→        {bars.map((amp, i) => (
   252→          <div key={i} className={cn("w-[2.5px] rounded-full transition-all duration-100", i < progressIdx ? progressColor : barColor)} style={{ height: `${Math.max(4, amp * 32)}px` }} />
   253→        ))}
   254→      </div>
   255→      <button type="button" onClick={cycleSpeed} className={cn("shrink-0 text-[11px] font-semibold tabular-nums transition-colors hover:opacity-80", isOwn ? "text-white/70" : "text-zinc-400")}>{speed}x</button>
   256→      <span className={cn("shrink-0 text-[11px] tabular-nums", isOwn ? "text-white/50" : "text-zinc-500")}>{Math.floor(duration / 60)}:{String(duration % 60).padStart(2, "0")}</span>
   257→      <audio ref={audioRef} src={message.content} preload="metadata" />
   258→    </div>
   259→  );
   260→}

   262→/* ------------------------------------------------------------------ */
   263→/*  LinkifiedText – detects & renders clickable URLs                   */
   264→/* ------------------------------------------------------------------ */
   265→
   266→const URL_REGEX = /(https?:\/\/[^\s<]+)/g;
   267→
   268→type TextSegment = { type: "text"; value: string } | { type: "link"; value: string };
   269→
   270→function parseUrls(text: string): TextSegment[] {
   271→  const segments: TextSegment[] = [];
   272→  let lastIndex = 0;
   273→  let match: RegExpExecArray | null;
   274→  while ((match = URL_REGEX.exec(text)) !== null) {
   275→    if (match.index > lastIndex) {
   276→      segments.push({ type: "text", value: text.slice(lastIndex, match.index) });
   277→    }
   278→    segments.push({ type: "link", value: match[1] });
   279→    lastIndex = URL_REGEX.lastIndex;
   280→  }
   281→  if (lastIndex < text.length) {
   282→    segments.push({ type: "text", value: text.slice(lastIndex) });
   283→  }
   284→  return segments;
   285→}
   286→
   287→/** Split a plain-text string into highlighted / non-highlighted fragments */
   288→function highlightText(text: string, query: string): React.ReactNode[] {
   289→  if (!query) return [text];
   290→  const escaped = query.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
   291→  const re = new RegExp(`(${escaped})`, "gi");
   292→  const parts: React.ReactNode[] = [];
   293→  let lastIndex = 0;
   294→  let match: RegExpExecArray | null;
   295→  while ((match = re.exec(text)) !== null) {
   296→    if (match.index > lastIndex) {
   297→      parts.push(text.slice(lastIndex, match.index));
   298→    }
   299→    parts.push(
   300→      <mark key={match.index} className="bg-yellow-300/60 dark:bg-yellow-400/40 rounded-sm px-0.5">
   301→        {match[1]}
   302→      </mark>,
   303→    );
   304→    lastIndex = re.lastIndex;
   305→  }
   306→  if (lastIndex < text.length) {
   307→    parts.push(text.slice(lastIndex));
   308→  }
   309→  return parts;
   310→}
   311→
   312→export const LinkifiedText = React.memo(function LinkifiedText({ text, className, highlight }: { text: string; className?: string; highlight?: string }) {
   313→  const segments = useMemo(() => parseUrls(text), [text]);
   314→  return (
   315→    <span className={className}>
   316→      {segments.map((seg, i) =>
   317→        seg.type === "link" ? (
   318→          <a
   319→            key={i}
   320→            href={seg.value}
   321→            target="_blank"
   322→            rel="noopener noreferrer"
   323→            className="underline decoration-violet-400/50 underline-offset-2 hover:decoration-violet-400 transition-colors"
   324→            onClick={(e) => e.stopPropagation()}
   325→          >
   326→            {seg.value}
   327→          </a>
   328→        ) : (
   329→          <span key={i}>{highlight ? highlightText(seg.value, highlight) : seg.value}</span>
   330→        ),
   331→      )}
   332→    </span>
   333→  );
   334→});

   462→  }, []);
   463→
   464→  const handlePressEnd = useCallback(() => { clearTimeout(longPressRef.current); }, []);
   465→
   466→  const handleContextMenu = useCallback((e: React.MouseEvent) => {
   467→    e.preventDefault(); setShowTapback(true);
   468→  }, []);
   469→
   470→  const handleCopy = useCallback(() => {
   471→    navigator.clipboard?.writeText(message.content).then(() => {
   472→      toast({ title: "Copied to clipboard", description: message.content.length > 50 ? message.content.slice(0, 50) + "..." : message.content });
   473→    }).catch(() => {
   474→      toast({ title: "Failed to copy", description: "Could not access clipboard", variant: "destructive" });
   475→    });
   476→    setShowTapback(false);
   477→  }, [message.content]);
   478→
   479→  const handleDelete = useCallback(() => {
   480→    onDeleteMessage?.(message.id);
   481→    setShowTapback(false);
   482→  }, [message.id, onDeleteMessage]);
   483→
   484→  const handleSwipeEnd = useCallback((_: MouseEvent | TouchEvent | PointerEvent, info: PanInfo) => {
   485→    if (info.offset.x > SWIPE_REPLY_THRESHOLD && info.velocity.x > 200) {
   486→      if (typeof navigator !== "undefined" && "vibrate" in navigator) navigator.vibrate(10);
   487→      onReplyTo?.(message);
   488→    }
   489→  }, [message, onReplyTo]);
   490→
   491→  const isFailed = message.status === "failed";
   492→  const isVoice = message.type === "voice";
   493→  const isNew = message.status === "sending" || Date.now() - new Date(message.created_at).getTime() < 2000;
   494→
   495→  return (
   496→    <>
   497→    <m.div
   498→      className={cn("relative flex w-full gap-2.5 px-4", isOwn ? "justify-end" : "justify-start")}
   499→      initial={isNew ? { opacity: 0, y: 16, scale: 0.97 } : false}
   500→      animate={{ opacity: 1, y: 0, scale: 1 }}
   501→      transition={{ duration: 0.3, ease: [0.22, 1, 0.36, 1] }}
   502→    >
   503→      <HeartBurst show={showHeart} />
   504→      <AnimatePresence>{showTapback && onReact && (
   505→        <TapbackDock message={message} isOwn={isOwn} onReact={(emoji, add) => { onReact(message.id, emoji, add); setShowTapback(false); }} onReply={() => { onReplyTo?.(message); setShowTapback(false); }} onCopy={handleCopy} onDelete={isOwn ? handleDelete : undefined} />
   506→      )}</AnimatePresence>
   507→
   508→      {!isOwn && (
   509→        <Avatar className="mt-auto size-8 shrink-0 ring-1 ring-muted">
   510→          {partnerAvatar && <AvatarImage src={partnerAvatar} alt={partnerName} />}
   511→          <AvatarFallback className="bg-gradient-to-br from-violet-100 to-purple-200 text-xs font-medium text-violet-700 dark:from-violet-500 dark:to-purple-600 dark:text-white">{partnerName.slice(0, 2).toUpperCase()}</AvatarFallback>
   512→        </Avatar>
   513→      )}
   514→
   515→      <div className={cn("flex max-w-[75%] flex-col gap-1", isOwn ? "items-end" : "items-start")}>
   516→        <m.div
   517→          ref={bubbleRef}
   518→          drag="x"
   519→          dragConstraints={{ left: 0, right: 0 }}
   520→          dragElastic={{ left: 0, right: 0.35 }}
   521→          onDragEnd={handleSwipeEnd}
   522→          onClick={handleClick}
   523→          onPointerDown={handlePressStart}
   524→          onPointerUp={handlePressEnd}
   525→          onPointerLeave={handlePressEnd}
   526→          onContextMenu={handleContextMenu}
   527→          className={cn(
   528→            "relative cursor-default select-none rounded-2xl px-4 py-2.5 text-sm leading-relaxed transition-all duration-200 active:scale-[0.98]",
   529→            // Own messages: vibrant violet-purple gradient with subtle shadow
   530→            isOwn && "bg-gradient-to-r from-violet-500 to-purple-600 text-white rounded-br-md shadow-md shadow-violet-500/15",
   531→            // Partner messages: softer background with shadow
   532→            !isOwn && "bg-muted text-foreground rounded-bl-md shadow-sm dark:bg-zinc-800 dark:text-zinc-100 dark:shadow-violet-950/20",
   533→            isFailed && "ring-2 ring-red-400/50",
   534→            isVoice && "py-2",
   535→            !isVoice && message.type !== "image" && "md:hover:shadow-md md:hover:-translate-y-0.5"
   536→          )}
   537→          style={{ transform: "translate3d(0,0,0)" }}
   538→        >
   539→          {/* Reply quote */}
   540→          {message.reply_to_id && message.reply_to_content && (
   541→            <ReplyPreview senderName={message.reply_to_sender_name ?? ""} content={message.reply_to_content} isOwn={isOwn} />
   542→          )}
   543→
   544→          {/* Content */}
   545→          {isVoice ? (
   546→            message.waveform_data ? (
   547→              <VoiceBubble message={message} isOwn={isOwn} waveform={message.waveform_data} />
   548→            ) : <p className="text-white/60 italic">Voice message</p>
   549→          ) : message.type === "image" ? (
   550→            <div className="group/img relative cursor-zoom-in" onClick={() => setShowLightbox(message.content)}>
   551→              <img src={message.content} alt="Shared image" className="max-h-64 rounded-lg object-contain transition-opacity duration-150 group-hover/img:opacity-90" />
   552→              <div className="absolute inset-0 flex items-center justify-center rounded-lg bg-black/0 transition-colors duration-150 group-hover/img:bg-black/20">
   553→                <ZoomIn className="size-6 text-white opacity-0 transition-opacity duration-150 group-hover/img:opacity-80" />
   554→              </div>
   555→            </div>
   556→          ) : (
   557→            <LinkifiedText text={message.content} className="whitespace-pre-wrap break-words" highlight={searchHighlight} />
   558→          )}
   559→
   560→          {/* Vanish indicator */}
   561→          {message.vanish_mode && (
   562→            <div className="mt-1 flex items-center gap-1 text-[10px] text-white/50"><EyeOff className="size-3" /><span>View once</span></div>
   563→          )}
   564→        </m.div>
   565→
   566→        {/* Timestamp + receipt + ephemeral timer */}
   567→        <div className={cn("flex items-center gap-1.5 px-1 text-[11px] text-muted-foreground", isOwn && "flex-row-reverse")}>
   568→          {message.ephemeral_seconds && message.ephemeral_seconds > 0 && (
   569→            <EphemeralTimer seconds={message.ephemeral_seconds} onExpire={() => { /* trigger vanish */ }} />
   570→          )}
   571→          <span className={cn(isNew && "animate-[timestamp-fade_0.4s_ease_0.3s_both]")}>{formatDistanceToNow(new Date(message.created_at), { addSuffix: true })}</span>
   572→          {isOwn && <ReceiptIcon status={message.status} />}
   573→        </div>
   574→
   575→        {/* Reactions */}
   576→        {onReact && <ReactionPills reactions={message.reactions} isOwn={isOwn} onReact={(e, a) => onReact(message.id, e, a)} currentUserId={""} />}
   577→
   578→        {isFailed && (
   579→          <button
   580→            type="button"
   581→            onClick={(e) => { e.stopPropagation(); onDeleteMessage?.(message.id); }}
   582→            className="mt-1 flex items-center gap-1 px-1 text-[10px] text-red-400 transition-colors hover:text-red-300"
   583→          >
   584→            <AlertCircle className="size-3" />
   585→            <span>Failed to send &middot; Tap to retry</span>
   586→          </button>
   587→        )}
   588→      </div>
   589→
   590→      {isOwn && <div className="w-8 shrink-0" />}
   591→    </m.div>
   592→
   593→    {/* Image Lightbox */}
   594→    <AnimatePresence>
   595→      {showLightbox && (
   596→        <ImageLightbox src={showLightbox} onClose={() => setShowLightbox(null)} />
   597→      )}
   598→    </AnimatePresence>
   599→    </>
   600→  );
   601→}

   607→function formatDateSeparator(dateStr: string): string {
   608→  const d = new Date(dateStr);
   609→  const now = new Date();
   610→  const today = new Date(now.getFullYear(), now.getMonth(), now.getDate());
   611→  const yesterday = new Date(today);
   612→  yesterday.setDate(yesterday.getDate() - 1);
   613→  const msgDay = new Date(d.getFullYear(), d.getMonth(), d.getDate());
   614→
   615→  if (msgDay.getTime() === today.getTime()) return "Today";
   616→  if (msgDay.getTime() === yesterday.getTime()) return "Yesterday";
   617→  return d.toLocaleDateString(undefined, { month: "short", day: "numeric" });
   618→}

   729→export function MessageFeed({ messages, isLoading, isPartnerTyping, currentUserId, partnerName, partnerAvatar, backgroundStyle, onMarkAsRead, onVanishMessage, onReplyTo, onDeleteMessage, onReact, scrollToMessageId, onScrolledToMessage, searchHighlight, onClearSearchHighlight }: MessageFeedProps) {
   730→  const parentRef = useRef<HTMLDivElement>(null);
   731→  const isAutoScrollRef = useRef(true);
   732→  const [showNewBtn, setShowNewBtn] = useState(false);
   733→  const unreadRef = useRef(0);
   734→  const prevCountRef = useRef(messages.length);
   735→
   736→  const totalItems = useMemo(() => messages.length + (isPartnerTyping ? 1 : 0), [messages.length, isPartnerTyping]);
   737→
   738→  const virtualizer = useVirtualizer({ count: totalItems, getScrollElement: () => parentRef.current, estimateSize: () => ESTIMATED_ROW_HEIGHT, overscan: OVERSCAN, paddingStart: 16, paddingEnd: 16, gap: 12 });
   739→
   740→  const scrollToBottom = useCallback((behavior: ScrollBehavior = "smooth") => {
   741→    const el = parentRef.current; if (!el) return;
   742→    el.scrollTo({ top: el.scrollHeight, behavior });
   743→    isAutoScrollRef.current = true; setShowNewBtn(false); unreadRef.current = 0;
   744→  }, []);
   745→
   746→  const handleScroll = useCallback((e: UIEvent<HTMLDivElement>) => {
   747→    const el = e.currentTarget;
   748→    const dist = el.scrollHeight - el.scrollTop - el.clientHeight;
   749→    if (dist <= AUTO_SCROLL_THRESHOLD) { isAutoScrollRef.current = true; setShowNewBtn(false); unreadRef.current = 0; }
   750→    else { isAutoScrollRef.current = false; }
   751→  }, []);
   752→
   753→  useEffect(() => { if (messages.length > 0) requestAnimationFrame(() => scrollToBottom("instant")); }, [isLoading, scrollToBottom]);
   754→
   755→  useEffect(() => { if (!onMarkAsRead) return; const unread = messages.filter((m) => m.sender_id !== currentUserId && m.status !== "read"); if (unread.length > 0 && isAutoScrollRef.current) onMarkAsRead(unread.map((m) => m.id)); }, [messages, currentUserId, onMarkAsRead]);
   756→
   757→  /* ---- Scroll to a specific message ----------------------------------- */
   758→  useEffect(() => {
   759→    if (!scrollToMessageId) return;
   760→    const idx = messages.findIndex((m) => m.id === scrollToMessageId);
   761→    if (idx === -1) { onScrolledToMessage?.(); return; }
   762→    virtualizer.scrollToIndex(idx, { align: "center" });
   763→    onScrolledToMessage?.();
   764→  }, [scrollToMessageId, messages, virtualizer, onScrolledToMessage]);
   765→
   766→  useEffect(() => {
   767→    const prev = prevCountRef.current; const next = messages.length; prevCountRef.current = next;
   768→    if (next <= prev) return;
   769→    const last = messages[next - 1]; const own = last && last.sender_id === currentUserId;
   770→    if (own) { requestAnimationFrame(() => scrollToBottom("instant")); return; }
   771→    if (isAutoScrollRef.current) requestAnimationFrame(() => scrollToBottom("smooth"));
   772→    else { unreadRef.current += next - prev; setShowNewBtn(true); }
   773→  }, [messages.length, currentUserId, scrollToBottom]);
   774→
   775→  if (isLoading) return <MessageListSkeleton />;
   776→  if (messages.length === 0 && !isPartnerTyping) return <EmptyState partnerName={partnerName} partnerAvatar={partnerAvatar} />;
   777→
   778→  const virtualItems = virtualizer.getVirtualItems();
   779→
   780→  return (
   781→    <div className="relative h-full w-full overflow-hidden" style={backgroundStyle}>
   782→      <div ref={parentRef} onScroll={handleScroll} onClick={() => onClearSearchHighlight?.()} className="h-full w-full overflow-y-auto overscroll-contain" style={{ WebkitOverflowScrolling: "touch" }}>
   783→        <div style={{ height: `${virtualizer.getTotalSize()}px`, width: "100%", position: "relative" }}>
   784→          {virtualItems.map((vi) => {
   785→            const isTyping = vi.index === messages.length;
   786→            const msg = isTyping ? null : messages[vi.index];
   787→            // Check if we need a date separator before this message
   788→            const showDateSep = msg && vi.index > 0 && messages[vi.index - 1] && (() => {
   789→              const prev = new Date(messages[vi.index - 1].created_at);
   790→              const curr = new Date(msg.created_at);
   791→              return prev.toDateString() !== curr.toDateString();
   792→            })();
   793→            const showFirstDateSep = msg && vi.index === 0;
   794→            return (
   795→              <div key={vi.key} data-index={vi.index} ref={virtualizer.measureElement} style={{ position: "absolute", top: 0, left: 0, width: "100%", transform: `translateY(${vi.start}px)` }}>
   796→                {isTyping ? <TypingIndicator partnerName={partnerName} partnerAvatar={partnerAvatar} /> : msg ? (
   797→                  <>
   798→                    {(showDateSep || showFirstDateSep) && <DateSeparator date={msg.created_at} />}
   799→                    <MessageBubble message={msg} isOwn={msg.sender_id === currentUserId} partnerName={partnerName} partnerAvatar={partnerAvatar} onReplyTo={onReplyTo} onReact={onReact} onDeleteMessage={onDeleteMessage} searchHighlight={searchHighlight} />
   800→                  </>
   801→                ) : null}
   802→              </div>
   803→            );
   804→          })}
   805→        </div>
   806→      </div>
   807→      {showNewBtn && unreadRef.current > 0 && <NewMessagesButton count={unreadRef.current} onClick={() => scrollToBottom("smooth")} />}
   808→    </div>
   809→  );
   810→}
