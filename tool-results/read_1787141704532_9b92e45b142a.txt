  1428→function formatDateSeparator(dateStr: string): string {
  1429→  const d = new Date(dateStr);
  1430→  const now = new Date();
  1431→  const today = new Date(now.getFullYear(), now.getMonth(), now.getDate());
  1432→  const yesterday = new Date(today);
  1433→  yesterday.setDate(yesterday.getDate() - 1);
  1434→  const msgDay = new Date(d.getFullYear(), d.getMonth(), d.getDate());
  1435→
  1436→  if (msgDay.getTime() === today.getTime()) return "Today";
  1437→  if (msgDay.getTime() === yesterday.getTime()) return "Yesterday";
  1438→  return d.toLocaleDateString(undefined, { month: "short", day: "numeric" });
  1439→}

  1722→      requestAnimationFrame(() => scrollToBottom("instant"));

  1724→
  1725→  useEffect(() => {
  1726→    if (!onMarkAsRead) return;
  1727→    const unread = messages.filter(
  1728→      (m) => m.sender_id !== currentUserId && m.status !== "read",
  1729→    );
  1730→    if (unread.length > 0 && isAutoScrollRef.current)
  1731→      onMarkAsRead(unread.map((m) => m.id));
  1732→  }, [messages, currentUserId, onMarkAsRead]);
  1733→
  1734→  /* ---- Scroll to a specific message ----------------------------------- */
  1735→  useEffect(() => {
  1736→    if (!scrollToMessageId) return;
  1737→    // Find which group contains the target message
  1738→    const groupIdx = groups.findIndex((g) =>
  1739→      g.messages.some((m) => m.id === scrollToMessageId),
  1740→    );
  1741→    if (groupIdx === -1) {
  1742→      onScrolledToMessage?.();
  1743→      return;
  1744→    }
  1745→    virtualizer.scrollToIndex(groupIdx, { align: "center" });
  1746→    onScrolledToMessage?.();
  1747→  }, [scrollToMessageId, groups, virtualizer, onScrolledToMessage]);
  1748→
  1749→  /* ---- Handle message count changes (new or prepended) ------------ */
  1750→  useEffect(() => {
  1751→    const prev = prevCountRef.current;
  1752→    const prevFirstId = prevFirstIdRef.current;
  1753→    prevCountRef.current = messages.length;
  1754→    prevFirstIdRef.current =
  1755→      messages.length > 0 ? messages[0].id : null;
  1756→
  1757→    if (messages.length <= prev) return;
  1758→
  1759→    /* Check if older messages were prepended (first message changed) */
  1760→    const firstId = messages[0]?.id;
  1761→    const wasPrepended = firstId !== prevFirstId && prevFirstId !== null;
  1762→
  1763→    if (wasPrepended) {
  1764→      /* Older messages prepended – preserve scroll position */
  1765→      const el = parentRef.current;
  1766→      if (el) {
  1767→        const oldHeight = prevScrollHeightRef.current;
  1768→        prevScrollHeightRef.current = el.scrollHeight;
  1769→        requestAnimationFrame(() => {
  1770→          const newHeight = el.scrollHeight;
  1771→          el.scrollTop += newHeight - oldHeight;
  1772→        });
  1773→      }
  1774→      return;
  1775→    }
  1776→
  1777→    /* New messages appended at the end */
  1778→    const last = messages[messages.length - 1];
  1779→    const own = last && last.sender_id === currentUserId;
  1780→    if (own) {
  1781→      requestAnimationFrame(() => scrollToBottom("instant"));
  1782→      return;
  1783→    }
  1784→    if (isAutoScrollRef.current)
  1785→      requestAnimationFrame(() => scrollToBottom("smooth"));
  1786→    else {
  1787→      unreadRef.current += messages.length - prev;
  1788→      setShowNewBtn(true);
  1789→    }
  1790→  }, [messages.length, currentUserId, scrollToBottom]);
  1791→
  1792→  /* Track scroll height for scroll preservation */
  1793→  useEffect(() => {
  1794→    const el = parentRef.current;
  1795→    if (!el) return;
  1796→    prevScrollHeightRef.current = el.scrollHeight;
  1797→  }, [messages.length]);
  1798→
  1799→  if (isLoading) return <MessageListSkeleton />;
  1800→  if (messages.length === 0 && !isPartnerTyping)
  1801→    return (
  1802→      <EmptyState partnerName={partnerName} partnerAvatar={partnerAvatar} />
  1803→    );
  1804→
  1805→  const virtualItems = virtualizer.getVirtualItems();
  1806→
  1807→  return (
  1808→    <div
  1809→      className="relative h-full w-full overflow-hidden"
  1810→      style={backgroundStyle}
  1811→    >
  1812→      <div
  1813→        ref={parentRef}
  1814→        onScroll={handleScroll}
  1815→        onClick={(e) => {
  1816→          onClearSearchHighlight?.();
  1817→          // Close menus if click is directly on the scroll container or background
  1818→          const target = e.target as HTMLElement;
  1819→          if (target === e.currentTarget || target.closest('.message-bubble-row') === null) {
  1820→            setActiveActionId(null);
  1821→            setActiveTapbackId(null);
  1822→          }
  1823→        }}
  1824→        className="h-full w-full select-none overflow-y-auto overscroll-contain"
  1825→        style={{ WebkitOverflowScrolling: "touch" }}
  1826→      >
  1827→        <div
  1828→          style={{
  1829→            height: `${virtualizer.getTotalSize()}px`,
  1830→            width: "100%",
  1831→            position: "relative",
  1832→          }}
  1833→        >
  1834→          {/* Loading more spinner */}
  1835→          {loadingMore && (
  1836→            <div
  1837→              style={{
  1838→                position: "absolute",
  1839→                top: 0,
  1840→                left: 0,
  1841→                width: "100%",
  1842→                height: "48px",
  1843→                zIndex: 5,
  1844→              }}
  1845→            >
  1846→              <div className="flex items-center justify-center py-3">
  1847→                <svg
  1848→                  className="size-5 animate-spin"
  1849→                  viewBox="0 0 24 24"
  1850→                  fill="none"
  1851→                  aria-label="Loading more messages"
  1852→                  style={{ color: "var(--app-accent-light)" }}
  1853→                >
  1854→                  <circle
  1855→                    className="opacity-25"
  1856→                    cx="12"
  1857→                    cy="12"
  1858→                    r="10"
  1859→                    stroke="currentColor"
  1860→                    strokeWidth="4"
  1861→                  />
  1862→                  <path
  1863→                    className="opacity-75"
  1864→                    fill="currentColor"
  1865→                    d="M4 12a8 8 0 018-8V0C5.373 0 0 5.373 0 12h4z"
  1866→                  />
  1867→                </svg>
  1868→              </div>
  1869→            </div>
  1870→          )}
  1871→          {virtualItems.map((vi) => {
  1872→            const isTyping = vi.index === groups.length;
  1873→            const group = isTyping ? null : groups[vi.index];
  1874→
  1875→            // Determine if we need a date separator before this group
  1876→            let showDateSep = false;
  1877→            if (group) {
  1878→              const firstMsg = group.messages[0];
  1879→              if (vi.index === 0) {
  1880→                showDateSep = true;
  1881→              } else {
  1882→                const prevGroup = groups[vi.index - 1];
  1883→                const prevLastMsg = prevGroup.messages[prevGroup.messages.length - 1];
  1884→                const prevDate = new Date(prevLastMsg.created_at);
  1885→                const currDate = new Date(firstMsg.created_at);
  1886→                showDateSep = prevDate.toDateString() !== currDate.toDateString();
  1887→              }
  1888→            }
  1889→
  1890→            return (
  1891→              <div
  1892→                key={vi.key}
  1893→                data-index={vi.index}
  1894→                ref={virtualizer.measureElement}
  1895→                style={{
  1896→                  position: "absolute",
  1897→                  top: 0,
  1898→                  left: 0,
  1899→                  width: "100%",
  1900→                  transform: `translateY(${vi.start}px)`,
  1901→                }}
  1902→              >
  1903→                {isTyping ? (
  1904→                  <TypingIndicator
  1905→                    partnerName={partnerName}
  1906→                    partnerAvatar={partnerAvatar}
  1907→                  />
  1908→                ) : group ? (
  1909→                  <>
  1910→                    {showDateSep && (
  1911→                      <DateSeparator date={group.messages[0].created_at} />
  1912→                    )}
  1913→                    <MessageGroup
  1914→                      messages={group.messages}
  1915→                      currentUserId={currentUserId}
  1916→                      currentUserName={currentUserName}
  1917→                      currentUserAvatar={currentUserAvatar}
  1918→                      partnerName={partnerName}
  1919→                      partnerAvatar={partnerAvatar}
  1920→                      onReplyTo={onReplyTo}
  1921→                      onReact={onReact}
  1922→                      onDeleteMessage={onDeleteMessage}
  1923→                      searchHighlight={searchHighlight}
  1924→                      activeActionId={activeActionId}
  1925→                      onOpenAction={setActiveActionId}
  1926→                      activeTapbackId={activeTapbackId}
  1927→                      onOpenTapback={setActiveTapbackId}
  1928→                    />
  1929→                  </>
  1930→                ) : null}
  1931→              </div>
  1932→            );
  1933→          })}
  1934→        </div>
  1935→      </div>
  1936→      {showNewBtn && unreadRef.current > 0 && (
  1937→        <NewMessagesButton
  1938→          count={unreadRef.current}
  1939→          onClick={() => scrollToBottom("smooth")}
  1940→        />
  1941→      )}
  1942→    </div>
  1943→  );
