   124→/* ------------------------------------------------------------------ */
   125→
   126→function EphemeralTimer({
   127→  seconds,
   128→  onExpire,
   129→}: {
   130→  seconds: number;
   131→  onExpire: () => void;
   132→}) {
   133→  const [remaining, setRemaining] = useState(seconds);
   134→  const intervalRef = useRef<ReturnType<typeof setInterval> | undefined>(
   135→    undefined,
   136→  );
   137→
   138→  useEffect(() => {
   139→    intervalRef.current = setInterval(() => {
   140→      setRemaining((r) => {
   141→        if (r <= 1) {
   142→          clearInterval(intervalRef.current);
   143→          onExpire();
   144→          return 0;
   145→        }
   146→        return r - 1;
   147→      });
   148→    }, 1000);
   149→    return () => clearInterval(intervalRef.current);
   150→  }, [seconds, onExpire]);
   151→
   152→  const pct = remaining / seconds;
   153→  const circ = 2 * Math.PI * 7;
   154→  const offset = circ * (1 - pct);
   155→
   156→  return (
   157→    <svg
   158→      className="size-3.5 shrink-0 ephemeral-timer-pulse"
   159→      viewBox="0 0 18 18"
   160→      aria-label={`${remaining}s remaining`}
   161→    >
   162→      <circle
   163→        cx="9"
   164→        cy="9"
   165→        r="7"
   166→        fill="none"
   167→        stroke="currentColor"
   168→        strokeWidth="1.5"
   169→        className="text-white/20"
   170→      />
   171→      <circle
   172→        cx="9"
   173→        cy="9"
   174→        r="7"
   175→        fill="none"
   176→        stroke="currentColor"
   177→        strokeWidth="1.5"
   178→        strokeDasharray={circ}
   179→        strokeDashoffset={offset}
   180→        strokeLinecap="round"
   181→        style={{
   182→          color: "var(--app-accent-lighter)",
   183→          transform: "rotate(-90deg)",
   184→          transformOrigin: "50% 50%",
   185→          transition: "stroke-dashoffset 1s linear",
   186→        }}
   187→      />
   188→    </svg>
   189→  );
   190→}

   399→/* ------------------------------------------------------------------ */
   400→/*  Voice Bubble                                                       */
   401→/* ------------------------------------------------------------------ */
   402→
   403→function VoiceBubble({
   404→  message,
   405→  isOwn,
   406→  waveform,
   407→}: {
   408→  message: ChatMessage;
   409→  isOwn: boolean;
   410→  waveform: number[];
   411→}) {
   412→  const audioRef = useRef<HTMLAudioElement>(null);
   413→  const [playing, setPlaying] = useState(false);
   414→  const [progress, setProgress] = useState(0);
   415→  const [speed, setSpeed] = useState(1);
   416→  const speeds = [1, 1.5, 2];
   417→  const duration = message.voice_duration ?? 0;
   418→
   419→  useEffect(() => {
   420→    const audio = audioRef.current;
   421→    if (!audio) return;
   422→    const onTime = () => {
   423→      if (audio.duration) setProgress(audio.currentTime / audio.duration);
   424→    };
   425→    const onEnd = () => setPlaying(false);
   426→    audio.addEventListener("timeupdate", onTime);
   427→    audio.addEventListener("ended", onEnd);
   428→    return () => {
   429→      audio.removeEventListener("timeupdate", onTime);
   430→      audio.removeEventListener("ended", onEnd);
   431→    };
   432→  }, []);
   433→
   434→  const toggle = useCallback(() => {
   435→    const audio = audioRef.current;
   436→    if (!audio) return;
   437→    if (playing) {
   438→      audio.pause();
   439→      setPlaying(false);
   440→    } else {
   441→      audio.playbackRate = speed;
   442→      audio.play().catch(() => {});
   443→      setPlaying(true);
   444→    }
   445→  }, [playing, speed]);
   446→
   447→  const cycleSpeed = useCallback(() => {
   448→    setSpeed((s) => {
   449→      const idx = speeds.indexOf(s);
   450→      const next = speeds[(idx + 1) % speeds.length];
   451→      if (audioRef.current) audioRef.current.playbackRate = next;
   452→      return next;
   453→    });
   454→  }, []);
   455→
   456→  /* Build waveform bars */
   457→  const bars = useMemo(() => {
   458→    const count = Math.min(waveform.length, 40);
   459→    const step = Math.max(1, Math.floor(waveform.length / count));
   460→    return Array.from({ length: count }, (_, i) => waveform[i * step] ?? 0.1);
   461→  }, [waveform]);
   462→
   463→  const progressIdx = Math.floor(progress * bars.length);
   464→
   465→  return (
   466→    <div className={cn("flex items-center gap-3", isOwn ? "flex-row-reverse" : "")}>
   467→      <button
   468→        type="button"
   469→        onClick={toggle}
   470→        className={cn(
   471→          "flex size-9 shrink-0 items-center justify-center rounded-full transition-all duration-150 active:scale-95",
   472→          isOwn
   473→            ? "bg-white/20 hover:bg-white/30"
   474→            : "bg-zinc-600 hover:bg-[var(--app-accent-subtle)]",
   475→        )}
   476→        aria-label={playing ? "Pause" : "Play"}
   477→      >
   478→        {playing ? (
   479→          <Pause className="size-3.5 text-white" />
   480→        ) : (
   481→          <Play className="size-3.5 text-white ml-0.5" />
   482→        )}
   483→      </button>
   484→      <div className="flex items-center gap-[2px]" style={{ height: 32 }}>
   485→        {bars.map((amp, i) => (
   486→          <div
   487→            key={i}
   488→            className="w-[2.5px] rounded-full transition-all duration-100"
   489→            style={{
   490→              height: `${Math.max(4, amp * 32)}px`,
   491→              backgroundColor:
   492→                i < progressIdx
   493→                  ? isOwn
   494→                    ? "rgba(255,255,255,1)"
   495→                    : "var(--app-accent-light)"
   496→                  : isOwn
   497→                    ? "rgba(255,255,255,0.5)"
   498→                    : "rgba(161,161,170,0.7)",
   499→            }}
   500→          />
   501→        ))}
   502→      </div>
   503→      <button
   504→        type="button"
   505→        onClick={cycleSpeed}
   506→        className={cn(
   507→          "shrink-0 text-[11px] font-semibold tabular-nums transition-colors hover:opacity-80 active:scale-95",
   508→          isOwn ? "text-white/70" : "text-zinc-400",
   509→        )}
   510→      >
   511→        {speed}x
   512→      </button>
   513→      <span
   514→        className={cn(
   515→          "shrink-0 text-[11px] tabular-nums",
   516→          isOwn ? "text-white/50" : "text-zinc-500",
   517→        )}
   518→      >
   519→        {Math.floor(duration / 60)}:{String(duration % 60).padStart(2, "0")}
   520→      </span>
   521→      <audio ref={audioRef} src={message.content} preload="metadata" />
   522→    </div>
   523→  );
   524→}

   526→/* ------------------------------------------------------------------ */
   527→/*  LinkifiedText – detects & renders clickable URLs                   */
   528→/* ------------------------------------------------------------------ */
   529→
   530→const URL_REGEX = /(https?:\/\/[^\s<]+)/g;
   531→
   532→type TextSegment =
   533→  | { type: "text"; value: string }
   534→  | { type: "link"; value: string };
   535→
   536→function parseUrls(text: string): TextSegment[] {
   537→  const segments: TextSegment[] = [];
   538→  let lastIndex = 0;
   539→  let match: RegExpExecArray | null;
   540→  while ((match = URL_REGEX.exec(text)) !== null) {
   541→    if (match.index > lastIndex) {
   542→      segments.push({ type: "text", value: text.slice(lastIndex, match.index) });
   543→    }
   544→    segments.push({ type: "link", value: match[1] });
   545→    lastIndex = URL_REGEX.lastIndex;
   546→  }
   547→  if (lastIndex < text.length) {
   548→    segments.push({ type: "text", value: text.slice(lastIndex) });
   549→  }
   550→  return segments;
   551→}
   552→
   553→/** Split a plain-text string into highlighted / non-highlighted fragments */
   554→function highlightText(text: string, query: string): React.ReactNode[] {
   555→  if (!query) return [text];
   556→  const escaped = query.replace(/[.*+?^${}()|[\]\\]/g, "\\");
   557→  const re = new RegExp(`(${escaped})`, "gi");
   558→  const parts: React.ReactNode[] = [];
   559→  let lastIndex = 0;
   560→  let match: RegExpExecArray | null;
   561→  while ((match = re.exec(text)) !== null) {
   562→    if (match.index > lastIndex) {
   563→      parts.push(text.slice(lastIndex, match.index));
   564→    }
   565→    parts.push(
   566→      <mark
   567→        key={match.index}
   568→        className="bg-yellow-300/60 dark:bg-yellow-400/40 rounded-sm px-0.5"
   569→      >
   570→        {match[1]}
   571→      </mark>,
   572→    );
   573→    lastIndex = re.lastIndex;
   574→  }
   575→  if (lastIndex < text.length) {
   576→    parts.push(text.slice(lastIndex));
   577→  }
   578→  return parts;
   579→}

   581→export const LinkifiedText = React.memo(function LinkifiedText({
   582→  text,
   583→  className,
   584→  highlight,
   585→}: {
   586→  text: string;
   587→  className?: string;
   588→  highlight?: string;
   589→}) {
   590→  const segments = useMemo(() => parseUrls(text), [text]);
   591→  return (
   592→    <span className={className}>
   593→      {segments.map((seg, i) =>
   594→        seg.type === "link" ? (
   595→          <a
   596→            key={i}
   597→            href={seg.value}
   598→            target="_blank"
   599→            rel="noopener noreferrer"
   600→            className="underline underline-offset-2 transition-colors"
   601→            style={{ textDecorationColor: "var(--app-accent-light)" }}
   602→            onClick={(e) => e.stopPropagation()}

   786→/* ------------------------------------------------------------------ */
   787→/*  Bubble rounding helper for grouped messages                        */
   788→/* ------------------------------------------------------------------ */
   789→
   790→function getBubbleRounding({
   791→  isFirst,
   792→  isLast,
   793→  isOwn,
   794→}: {
   795→  isFirst: boolean;
   796→  isLast: boolean;
   797→  isOwn: boolean;
   798→}): string {
   799→  if (isFirst && isLast) {
   800→    // Single message in group (or only message)
   801→    return isOwn
   802→      ? "rounded-2xl rounded-br-sm"
   803→      : "rounded-2xl rounded-bl-sm";
   804→  }
   805→  if (isFirst) {
   806→    // First message of a multi-message group
   807→    return isOwn
   808→      ? "rounded-tl-2xl rounded-bl-2xl rounded-tr-sm rounded-br-sm"
   809→      : "rounded-tl-2xl rounded-bl-sm rounded-tr-2xl rounded-br-sm";
   810→  }
   811→  if (isLast) {
   812→    // Last message of a multi-message group
   813→    return isOwn
   814→      ? "rounded-tl-2xl rounded-bl-2xl rounded-tr-sm rounded-br-2xl"
   815→      : "rounded-tl-2xl rounded-bl-2xl rounded-tr-2xl rounded-br-sm";
   816→  }
   817→  // Middle message
   818→  return isOwn
   819→    ? "rounded-tl-2xl rounded-bl-2xl rounded-tr-sm rounded-br-sm"
   820→    : "rounded-tl-2xl rounded-bl-sm rounded-tr-2xl rounded-br-sm";
   821→}

  1343→function formatDateSeparator(dateStr: string): string {
  1344→  const d = new Date(dateStr);
  1345→  const now = new Date();
  1346→  const today = new Date(now.getFullYear(), now.getMonth(), now.getDate());
  1347→  const yesterday = new Date(today);
  1348→  yesterday.setDate(yesterday.getDate() - 1);
  1349→  const msgDay = new Date(d.getFullYear(), d.getMonth(), d.getDate());
  1350→
  1351→  if (msgDay.getTime() === today.getTime()) return "Today";
  1352→  if (msgDay.getTime() === yesterday.getTime()) return "Yesterday";
  1353→  return d.toLocaleDateString(undefined, { month: "short", day: "numeric" });
  1354→}

  1537→export function MessageFeed({
  1538→  messages,
  1539→  isLoading,
  1540→  isPartnerTyping,
  1541→  currentUserId,
  1542→  currentUserName,
  1543→  currentUserAvatar,
  1544→  partnerName,
  1545→  partnerAvatar,
  1546→  backgroundStyle,
  1547→  onMarkAsRead,
  1548→  onVanishMessage,
  1549→  onReplyTo,
  1550→  onDeleteMessage,
  1551→  onReact,
  1552→  scrollToMessageId,
  1553→  onScrolledToMessage,
  1554→  searchHighlight,
  1555→  onClearSearchHighlight,
  1556→  loadMore,
  1557→  hasMore,
  1558→  loadingMore,
  1559→}: MessageFeedProps) {
  1560→  const parentRef = useRef<HTMLDivElement>(null);
  1561→  const isAutoScrollRef = useRef(true);
  1562→  const [activeActionId, setActiveActionId] = useState<string | null>(null);
  1563→  const [showNewBtn, setShowNewBtn] = useState(false);
  1564→  const unreadRef = useRef(0);
  1565→  const prevCountRef = useRef(messages.length);
  1566→  const prevFirstIdRef = useRef(
  1567→    messages.length > 0 ? messages[0].id : null,
  1568→  );
  1569→  const prevScrollHeightRef = useRef(0);
  1570→
  1571→  /* ---- Build message groups ---------------------------------------- */
  1572→  const groups = useMemo(() => {
  1573→    const result: MessageGroupData[] = [];
  1574→    for (const msg of messages) {
  1575→      const last = result[result.length - 1];
  1576→      if (last && last.senderId === msg.sender_id) {
  1577→        last.messages.push(msg);
  1578→      } else {
  1579→        result.push({ senderId: msg.sender_id, messages: [msg] });
  1580→      }
  1581→    }
  1582→    return result;
  1583→  }, [messages]);
  1584→
  1585→  const totalItems = useMemo(
  1586→    () => groups.length + (isPartnerTyping ? 1 : 0),
  1587→    [groups.length, isPartnerTyping],
  1588→  );
  1589→
  1590→  const loadMoreRef = useRef(loadMore);
  1591→  loadMoreRef.current = loadMore;
  1592→
  1593→  const virtualizer = useVirtualizer({
  1594→    count: totalItems,
  1595→    getScrollElement: () => parentRef.current,
  1596→    estimateSize: () => ESTIMATED_ROW_HEIGHT,
  1597→    overscan: OVERSCAN,
  1598→    paddingStart: 16,
  1599→    paddingEnd: 16,
  1600→    gap: 12,
  1601→  });
  1602→
  1603→  const handleScroll = useCallback(
  1604→    (e: UIEvent<HTMLDivElement>) => {
  1605→      const el = e.currentTarget;
  1606→      const dist = el.scrollHeight - el.scrollTop - el.clientHeight;
  1607→      if (dist <= AUTO_SCROLL_THRESHOLD) {
  1608→        isAutoScrollRef.current = true;
  1609→        setShowNewBtn(false);
  1610→        unreadRef.current = 0;
  1611→      } else {
  1612→        isAutoScrollRef.current = false;
  1613→      }
  1614→      /* Trigger loadMore when scrolled near top */
  1615→      if (el.scrollTop < 300 && hasMore && !loadingMore) {
  1616→        loadMoreRef.current?.();
  1617→      }
  1618→    },
  1619→    [hasMore, loadingMore],
  1620→  );
  1621→
  1622→  const scrollToBottom = useCallback(
  1623→    (behavior: ScrollBehavior = "smooth") => {
  1624→      const el = parentRef.current;
  1625→      if (!el) return;
  1626→      el.scrollTo({ top: el.scrollHeight, behavior });
  1627→      isAutoScrollRef.current = true;
  1628→      setShowNewBtn(false);
  1629→      unreadRef.current = 0;
  1630→    },
  1631→    [],
  1632→  );
  1633→
  1634→  useEffect(() => {
  1635→    if (messages.length > 0)
  1636→      requestAnimationFrame(() => scrollToBottom("instant"));
  1637→  }, [isLoading, scrollToBottom]);
  1638→
  1639→  useEffect(() => {
  1640→    if (!onMarkAsRead) return;
  1641→    const unread = messages.filter(
  1642→      (m) => m.sender_id !== currentUserId && m.status !== "read",
  1643→    );
  1644→    if (unread.length > 0 && isAutoScrollRef.current)
  1645→      onMarkAsRead(unread.map((m) => m.id));
  1646→  }, [messages, currentUserId, onMarkAsRead]);
  1647→
  1648→  /* ---- Scroll to a specific message ----------------------------------- */
  1649→  useEffect(() => {
  1650→    if (!scrollToMessageId) return;
  1651→    // Find which group contains the target message
  1652→    const groupIdx = groups.findIndex((g) =>
  1653→      g.messages.some((m) => m.id === scrollToMessageId),
  1654→    );
  1655→    if (groupIdx === -1) {
  1656→      onScrolledToMessage?.();
  1657→      return;
  1658→    }
  1659→    virtualizer.scrollToIndex(groupIdx, { align: "center" });
  1660→    onScrolledToMessage?.();
  1661→  }, [scrollToMessageId, groups, virtualizer, onScrolledToMessage]);
  1662→
  1663→  /* ---- Handle message count changes (new or prepended) ------------ */
  1664→  useEffect(() => {
  1665→    const prev = prevCountRef.current;
  1666→    const prevFirstId = prevFirstIdRef.current;
  1667→    prevCountRef.current = messages.length;
  1668→    prevFirstIdRef.current =
  1669→      messages.length > 0 ? messages[0].id : null;
  1670→
  1671→    if (messages.length <= prev) return;
  1672→
  1673→    /* Check if older messages were prepended (first message changed) */
  1674→    const firstId = messages[0]?.id;
  1675→    const wasPrepended = firstId !== prevFirstId && prevFirstId !== null;
  1676→
  1677→    if (wasPrepended) {
  1678→      /* Older messages prepended – preserve scroll position */
  1679→      const el = parentRef.current;
  1680→      if (el) {
  1681→        const oldHeight = prevScrollHeightRef.current;
  1682→        prevScrollHeightRef.current = el.scrollHeight;
  1683→        requestAnimationFrame(() => {
  1684→          const newHeight = el.scrollHeight;
  1685→          el.scrollTop += newHeight - oldHeight;
  1686→        });
  1687→      }
  1688→      return;
  1689→    }
  1690→
  1691→    /* New messages appended at the end */
  1692→    const last = messages[messages.length - 1];
  1693→    const own = last && last.sender_id === currentUserId;
  1694→    if (own) {
  1695→      requestAnimationFrame(() => scrollToBottom("instant"));
  1696→      return;
  1697→    }
  1698→    if (isAutoScrollRef.current)
  1699→      requestAnimationFrame(() => scrollToBottom("smooth"));
  1700→    else {
  1701→      unreadRef.current += messages.length - prev;
  1702→      setShowNewBtn(true);
  1703→    }
  1704→  }, [messages.length, currentUserId, scrollToBottom]);
  1705→
  1706→  /* Track scroll height for scroll preservation */
  1707→  useEffect(() => {
  1708→    const el = parentRef.current;
  1709→    if (!el) return;
  1710→    prevScrollHeightRef.current = el.scrollHeight;
  1711→  }, [messages.length]);
  1712→
  1713→  if (isLoading) return <MessageListSkeleton />;
  1714→  if (messages.length === 0 && !isPartnerTyping)
  1715→    return (
  1716→      <EmptyState partnerName={partnerName} partnerAvatar={partnerAvatar} />
  1717→    );
  1718→
  1719→  const virtualItems = virtualizer.getVirtualItems();
  1720→
  1721→  return (
  1722→    <div
  1723→      className="relative h-full w-full overflow-hidden"
  1724→      style={backgroundStyle}
  1725→    >
  1726→      <div
  1727→        ref={parentRef}
  1728→        onScroll={handleScroll}
  1729→        onClick={(e) => {
  1730→          onClearSearchHighlight?.();
  1731→          // Only close action sheet if click is directly on the scroll container or background
  1732→          const target = e.target as HTMLElement;
  1733→          if (target === e.currentTarget || target.closest('.message-bubble-row') === null) {
  1734→            setActiveActionId(null);
  1735→          }
  1736→        }}
  1737→        className="h-full w-full select-none overflow-y-auto overscroll-contain"
  1738→        style={{ WebkitOverflowScrolling: "touch" }}
  1739→      >
  1740→        <div
  1741→          style={{
  1742→            height: `${virtualizer.getTotalSize()}px`,
  1743→            width: "100%",
  1744→            position: "relative",
  1745→          }}
  1746→        >
  1747→          {/* Loading more spinner */}
  1748→          {loadingMore && (
  1749→            <div
  1750→              style={{
  1751→                position: "absolute",
  1752→                top: 0,
  1753→                left: 0,
  1754→                width: "100%",
  1755→                height: "48px",
  1756→                zIndex: 5,
  1757→              }}
  1758→            >
  1759→              <div className="flex items-center justify-center py-3">
  1760→                <svg
  1761→                  className="size-5 animate-spin"
  1762→                  viewBox="0 0 24 24"
  1763→                  fill="none"
  1764→                  aria-label="Loading more messages"
  1765→                  style={{ color: "var(--app-accent-light)" }}
  1766→                >
  1767→                  <circle
  1768→                    className="opacity-25"
  1769→                    cx="12"
  1770→                    cy="12"
  1771→                    r="10"
  1772→                    stroke="currentColor"
  1773→                    strokeWidth="4"

  1775→                  <path

  1778→                    d="M4 12a8 8 0 018-8V0C5.373 0 0 5.373 0 12h4z"
  1779→                  />
  1780→                </svg>
  1781→              </div>

  1783→          )}
