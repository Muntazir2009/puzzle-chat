   121→/*  Ephemeral Timer (circular SVG countdown)                           */
   122→/* ------------------------------------------------------------------ */
   123→
   124→function EphemeralTimer({
   125→  seconds,
   126→  onExpire,
   127→}: {
   128→  seconds: number;
   129→  onExpire: () => void;
   130→}) {
   131→  const [remaining, setRemaining] = useState(seconds);
   132→  const intervalRef = useRef<ReturnType<typeof setInterval> | undefined>(
   133→    undefined,
   134→  );
   135→
   136→  useEffect(() => {
   137→    intervalRef.current = setInterval(() => {
   138→      setRemaining((r) => {
   139→        if (r <= 1) {
   140→          clearInterval(intervalRef.current);
   141→          onExpire();
   142→          return 0;
   143→        }
   144→        return r - 1;
   145→      });
   146→    }, 1000);
   147→    return () => clearInterval(intervalRef.current);
   148→  }, [seconds, onExpire]);
   149→
   150→  const pct = remaining / seconds;
   151→  const circ = 2 * Math.PI * 7;
   152→  const offset = circ * (1 - pct);
   153→
   154→  return (
   155→    <svg
   156→      className="size-3.5 shrink-0 ephemeral-timer-pulse"
   157→      viewBox="0 0 18 18"
   158→      aria-label={`${remaining}s remaining`}
   159→    >
   160→      <circle
   161→        cx="9"
   162→        cy="9"
   163→        r="7"
   164→        fill="none"
   165→        stroke="currentColor"
   166→        strokeWidth="1.5"
   167→        className="text-white/20"
   168→      />
   169→      <circle
   170→        cx="9"
   171→        cy="9"
   172→        r="7"
   173→        fill="none"
   174→        stroke="currentColor"
   175→        strokeWidth="1.5"
   176→        strokeDasharray={circ}
   177→        strokeDashoffset={offset}
   178→        strokeLinecap="round"
   179→        style={{
   180→          color: "var(--app-accent-lighter)",
   181→          transform: "rotate(-90deg)",
   182→          transformOrigin: "50% 50%",
   183→          transition: "stroke-dashoffset 1s linear",
   184→        }}
   185→      />
   186→    </svg>
   187→  );
   188→}

   397→/*  Voice Bubble                                                       */
   398→/* ------------------------------------------------------------------ */
   399→
   400→function VoiceBubble({
   401→  message,
   402→  isOwn,
   403→  waveform,
   404→}: {
   405→  message: ChatMessage;
   406→  isOwn: boolean;
   407→  waveform: number[];
   408→}) {
   409→  const audioRef = useRef<HTMLAudioElement>(null);
   410→  const [playing, setPlaying] = useState(false);
   411→  const [progress, setProgress] = useState(0);
   412→  const [speed, setSpeed] = useState(1);
   413→  const speeds = [1, 1.5, 2];
   414→  const duration = message.voice_duration ?? 0;
   415→
   416→  useEffect(() => {
   417→    const audio = audioRef.current;
   418→    if (!audio) return;
   419→    const onTime = () => {
   420→      if (audio.duration) setProgress(audio.currentTime / audio.duration);
   421→    };
   422→    const onEnd = () => setPlaying(false);
   423→    audio.addEventListener("timeupdate", onTime);
   424→    audio.addEventListener("ended", onEnd);
   425→    return () => {
   426→      audio.removeEventListener("timeupdate", onTime);
   427→      audio.removeEventListener("ended", onEnd);
   428→    };
   429→  }, []);
   430→
   431→  const toggle = useCallback(() => {
   432→    const audio = audioRef.current;
   433→    if (!audio) return;
   434→    if (playing) {
   435→      audio.pause();
   436→      setPlaying(false);
   437→    } else {
   438→      audio.playbackRate = speed;
   439→      audio.play().catch(() => {});
   440→      setPlaying(true);
   441→    }
   442→  }, [playing, speed]);
   443→
   444→  const cycleSpeed = useCallback(() => {
   445→    setSpeed((s) => {
   446→      const idx = speeds.indexOf(s);
   447→      const next = speeds[(idx + 1) % speeds.length];
   448→      if (audioRef.current) audioRef.current.playbackRate = next;
   449→      return next;
   450→    });
   451→  }, []);
   452→
   453→  /* Build waveform bars */
   454→  const bars = useMemo(() => {
   455→    const count = Math.min(waveform.length, 40);
   456→    const step = Math.max(1, Math.floor(waveform.length / count));
   457→    return Array.from({ length: count }, (_, i) => waveform[i * step] ?? 0.1);
   458→  }, [waveform]);
   459→
   460→  const progressIdx = Math.floor(progress * bars.length);
   461→
   462→  return (
   463→    <div className={cn("flex items-center gap-3", isOwn ? "flex-row-reverse" : "")}>
   464→      <button
   465→        type="button"
   466→        onClick={toggle}
   467→        className={cn(
   468→          "flex size-9 shrink-0 items-center justify-center rounded-full transition-all duration-150 active:scale-95",
   469→          isOwn
   470→            ? "bg-white/20 hover:bg-white/30"
   471→            : "bg-zinc-600 hover:bg-[var(--app-accent-subtle)]",
   472→        )}
   473→        aria-label={playing ? "Pause" : "Play"}
   474→      >
   475→        {playing ? (
   476→          <Pause className="size-3.5 text-white" />
   477→        ) : (
   478→          <Play className="size-3.5 text-white ml-0.5" />
   479→        )}
   480→      </button>
   481→      <div className="flex items-center gap-[2px]" style={{ height: 32 }}>
   482→        {bars.map((amp, i) => (
   483→          <div
   484→            key={i}
   485→            className="w-[2.5px] rounded-full transition-all duration-100"
   486→            style={{
   487→              height: `${Math.max(4, amp * 32)}px`,
   488→              backgroundColor:
   489→                i < progressIdx
   490→                  ? isOwn
   491→                    ? "rgba(255,255,255,1)"
   492→                    : "var(--app-accent-light)"
   493→                  : isOwn
   494→                    ? "rgba(255,255,255,0.5)"
   495→                    : "rgba(161,161,170,0.7)",
   496→            }}
   497→          />
   498→        ))}
   499→      </div>
   500→      <button
   501→        type="button"
   502→        onClick={cycleSpeed}
   503→        className={cn(
   504→          "shrink-0 text-[11px] font-semibold tabular-nums transition-colors hover:opacity-80 active:scale-95",
   505→          isOwn ? "text-white/70" : "text-zinc-400",
   506→        )}
   507→      >
   508→        {speed}x
   509→      </button>
   510→      <span
   511→        className={cn(
   512→          "shrink-0 text-[11px] tabular-nums",
   513→          isOwn ? "text-white/50" : "text-zinc-500",
   514→        )}
   515→      >
   516→        {Math.floor(duration / 60)}:{String(duration % 60).padStart(2, "0")}
   517→      </span>
   518→      <audio ref={audioRef} src={message.content} preload="metadata" />
   519→    </div>
   520→  );
   521→}

   523→/* ------------------------------------------------------------------ */
   524→/*  LinkifiedText – detects & renders clickable URLs                   */
   525→/* ------------------------------------------------------------------ */
   526→
   527→const URL_REGEX = /(https?:\/\/[^\s<]+)/g;
   528→
   529→type TextSegment =
   530→  | { type: "text"; value: string }
   531→  | { type: "link"; value: string };
   532→
   533→function parseUrls(text: string): TextSegment[] {
   534→  const segments: TextSegment[] = [];
   535→  let lastIndex = 0;
   536→  let match: RegExpExecArray | null;
   537→  while ((match = URL_REGEX.exec(text)) !== null) {
   538→    if (match.index > lastIndex) {
   539→      segments.push({ type: "text", value: text.slice(lastIndex, match.index) });
   540→    }
   541→    segments.push({ type: "link", value: match[1] });
   542→    lastIndex = URL_REGEX.lastIndex;
   543→  }
   544→  if (lastIndex < text.length) {
   545→    segments.push({ type: "text", value: text.slice(lastIndex) });
   546→  }
   547→  return segments;
   548→}
   549→
   550→/** Split a plain-text string into highlighted / non-highlighted fragments */
   551→function highlightText(text: string, query: string): React.ReactNode[] {
   552→  if (!query) return [text];
   553→  const escaped = query.replace(/[.*+?^${}()|[\]\\]/g, "\\");
   554→  const re = new RegExp(`(${escaped})`, "gi");
   555→  const parts: React.ReactNode[] = [];
   556→  let lastIndex = 0;
   557→  let match: RegExpExecArray | null;
   558→  while ((match = re.exec(text)) !== null) {
   559→    if (match.index > lastIndex) {
   560→      parts.push(text.slice(lastIndex, match.index));
   561→    }
   562→    parts.push(
   563→      <mark
   564→        key={match.index}
   565→        className="bg-yellow-300/60 dark:bg-yellow-400/40 rounded-sm px-0.5"
   566→      >
   567→        {match[1]}
   568→      </mark>,
   569→    );
   570→    lastIndex = re.lastIndex;
   571→  }
   572→  if (lastIndex < text.length) {
   573→    parts.push(text.slice(lastIndex));
   574→  }
   575→  return parts;
   576→}

   578→export const LinkifiedText = React.memo(function LinkifiedText({
   579→  text,
   580→  className,
   581→  highlight,
   582→}: {
   583→  text: string;
   584→  className?: string;
   585→  highlight?: string;
   586→}) {
   587→  const segments = useMemo(() => parseUrls(text), [text]);
   588→  return (
   589→    <span className={className}>
   590→      {segments.map((seg, i) =>
   591→        seg.type === "link" ? (
   592→          <a
   593→            key={i}
   594→            href={seg.value}
   595→            target="_blank"
   596→            rel="noopener noreferrer"
   597→            className="underline underline-offset-2 transition-colors"
   598→            style={{ textDecorationColor: "var(--app-accent-light)" }}
   599→            onClick={(e) => e.stopPropagation()}
   600→          >
   601→            {seg.value}
   602→          </a>
   603→        ) : (
   604→          <span key={i}>
   605→            {highlight ? highlightText(seg.value, highlight) : seg.value}
   606→          </span>
   607→        ),

   783→/* ------------------------------------------------------------------ */
   784→/*  Bubble rounding helper for grouped messages                        */
   785→/* ------------------------------------------------------------------ */
   786→
   787→function getBubbleRounding({
   788→  isFirst,
   789→  isLast,
   790→  isOwn,
   791→}: {
   792→  isFirst: boolean;
   793→  isLast: boolean;
   794→  isOwn: boolean;
   795→}): string {
   796→  if (isFirst && isLast) {
   797→    // Single message in group (or only message)
   798→    return isOwn
   799→      ? "rounded-2xl rounded-br-sm"
   800→      : "rounded-2xl rounded-bl-sm";
   801→  }
   802→  if (isFirst) {
   803→    // First message of a multi-message group
   804→    return isOwn
   805→      ? "rounded-tl-2xl rounded-bl-2xl rounded-tr-sm rounded-br-sm"
   806→      : "rounded-tl-2xl rounded-bl-sm rounded-tr-2xl rounded-br-sm";
   807→  }
   808→  if (isLast) {
   809→    // Last message of a multi-message group
   810→    return isOwn
   811→      ? "rounded-tl-2xl rounded-bl-2xl rounded-tr-sm rounded-br-2xl"
   812→      : "rounded-tl-2xl rounded-bl-2xl rounded-tr-2xl rounded-br-sm";
   813→  }
   814→  // Middle message
   815→  return isOwn
   816→    ? "rounded-tl-2xl rounded-bl-2xl rounded-tr-sm rounded-br-sm"
   817→    : "rounded-tl-2xl rounded-bl-sm rounded-tr-2xl rounded-br-sm";
   818→}

  1313→function formatDateSeparator(dateStr: string): string {
  1314→  const d = new Date(dateStr);
  1315→  const now = new Date();
  1316→  const today = new Date(now.getFullYear(), now.getMonth(), now.getDate());
  1317→  const yesterday = new Date(today);
  1318→  yesterday.setDate(yesterday.getDate() - 1);
  1319→  const msgDay = new Date(d.getFullYear(), d.getMonth(), d.getDate());
  1320→
  1321→  if (msgDay.getTime() === today.getTime()) return "Today";
  1322→  if (msgDay.getTime() === yesterday.getTime()) return "Yesterday";
  1323→  return d.toLocaleDateString(undefined, { month: "short", day: "numeric" });
  1324→}

  1507→export function MessageFeed({
  1508→  messages,
  1509→  isLoading,
  1510→  isPartnerTyping,
  1511→  currentUserId,
  1512→  partnerName,
  1513→  partnerAvatar,
  1514→  backgroundStyle,
  1515→  onMarkAsRead,
  1516→  onVanishMessage,
  1517→  onReplyTo,
  1518→  onDeleteMessage,
  1519→  onReact,
  1520→  scrollToMessageId,
  1521→  onScrolledToMessage,
  1522→  searchHighlight,
  1523→  onClearSearchHighlight,
  1524→  loadMore,
  1525→  hasMore,
  1526→  loadingMore,
  1527→}: MessageFeedProps) {
  1528→  const parentRef = useRef<HTMLDivElement>(null);
  1529→  const isAutoScrollRef = useRef(true);
  1530→  const [activeActionId, setActiveActionId] = useState<string | null>(null);
  1531→  const [showNewBtn, setShowNewBtn] = useState(false);
  1532→  const unreadRef = useRef(0);
  1533→  const prevCountRef = useRef(messages.length);
  1534→  const prevFirstIdRef = useRef(
  1535→    messages.length > 0 ? messages[0].id : null,
  1536→  );
  1537→  const prevScrollHeightRef = useRef(0);
  1538→
  1539→  /* ---- Build message groups ---------------------------------------- */
  1540→  const groups = useMemo(() => {
  1541→    const result: MessageGroupData[] = [];
  1542→    for (const msg of messages) {
  1543→      const last = result[result.length - 1];
  1544→      if (last && last.senderId === msg.sender_id) {
  1545→        last.messages.push(msg);
  1546→      } else {
  1547→        result.push({ senderId: msg.sender_id, messages: [msg] });
  1548→      }
  1549→    }
  1550→    return result;
  1551→  }, [messages]);
  1552→
  1553→  const totalItems = useMemo(
  1554→    () => groups.length + (isPartnerTyping ? 1 : 0),
  1555→    [groups.length, isPartnerTyping],
  1556→  );
  1557→
  1558→  const loadMoreRef = useRef(loadMore);
  1559→  loadMoreRef.current = loadMore;
  1560→
  1561→  const virtualizer = useVirtualizer({
  1562→    count: totalItems,
  1563→    getScrollElement: () => parentRef.current,
  1564→    estimateSize: () => ESTIMATED_ROW_HEIGHT,
  1565→    overscan: OVERSCAN,
  1566→    paddingStart: 16,
  1567→    paddingEnd: 16,
  1568→    gap: 12,
  1569→  });
  1570→
  1571→  const handleScroll = useCallback(
  1572→    (e: UIEvent<HTMLDivElement>) => {
  1573→      const el = e.currentTarget;
  1574→      const dist = el.scrollHeight - el.scrollTop - el.clientHeight;
  1575→      if (dist <= AUTO_SCROLL_THRESHOLD) {
  1576→        isAutoScrollRef.current = true;
  1577→        setShowNewBtn(false);
  1578→        unreadRef.current = 0;
  1579→      } else {
  1580→        isAutoScrollRef.current = false;
  1581→      }
  1582→      /* Trigger loadMore when scrolled near top */
  1583→      if (el.scrollTop < 300 && hasMore && !loadingMore) {
  1584→        loadMoreRef.current?.();
  1585→      }
  1586→    },
  1587→    [hasMore, loadingMore],
  1588→  );
  1589→
  1590→  const scrollToBottom = useCallback(
  1591→    (behavior: ScrollBehavior = "smooth") => {
  1592→      const el = parentRef.current;
  1593→      if (!el) return;
  1594→      el.scrollTo({ top: el.scrollHeight, behavior });
  1595→      isAutoScrollRef.current = true;
  1596→      setShowNewBtn(false);
  1597→      unreadRef.current = 0;
  1598→    },
  1599→    [],
  1600→  );
  1601→
  1602→  useEffect(() => {
  1603→    if (messages.length > 0)
  1604→      requestAnimationFrame(() => scrollToBottom("instant"));
  1605→  }, [isLoading, scrollToBottom]);
  1606→
  1607→  useEffect(() => {
  1608→    if (!onMarkAsRead) return;
  1609→    const unread = messages.filter(
  1610→      (m) => m.sender_id !== currentUserId && m.status !== "read",
  1611→    );
  1612→    if (unread.length > 0 && isAutoScrollRef.current)
  1613→      onMarkAsRead(unread.map((m) => m.id));
  1614→  }, [messages, currentUserId, onMarkAsRead]);
  1615→
  1616→  /* ---- Scroll to a specific message ----------------------------------- */
  1617→  useEffect(() => {
  1618→    if (!scrollToMessageId) return;
  1619→    // Find which group contains the target message
  1620→    const groupIdx = groups.findIndex((g) =>
  1621→      g.messages.some((m) => m.id === scrollToMessageId),
  1622→    );
  1623→    if (groupIdx === -1) {
  1624→      onScrolledToMessage?.();
  1625→      return;
  1626→    }
  1627→    virtualizer.scrollToIndex(groupIdx, { align: "center" });
  1628→    onScrolledToMessage?.();
  1629→  }, [scrollToMessageId, groups, virtualizer, onScrolledToMessage]);
  1630→
  1631→  /* ---- Handle message count changes (new or prepended) ------------ */
  1632→  useEffect(() => {
  1633→    const prev = prevCountRef.current;
  1634→    const prevFirstId = prevFirstIdRef.current;
  1635→    prevCountRef.current = messages.length;
  1636→    prevFirstIdRef.current =
  1637→      messages.length > 0 ? messages[0].id : null;
  1638→
  1639→    if (messages.length <= prev) return;
  1640→
  1641→    /* Check if older messages were prepended (first message changed) */
  1642→    const firstId = messages[0]?.id;
  1643→    const wasPrepended = firstId !== prevFirstId && prevFirstId !== null;
  1644→
  1645→    if (wasPrepended) {
  1646→      /* Older messages prepended – preserve scroll position */
  1647→      const el = parentRef.current;
  1648→      if (el) {
  1649→        const oldHeight = prevScrollHeightRef.current;
  1650→        prevScrollHeightRef.current = el.scrollHeight;
  1651→        requestAnimationFrame(() => {
  1652→          const newHeight = el.scrollHeight;
  1653→          el.scrollTop += newHeight - oldHeight;
  1654→        });
  1655→      }
  1656→      return;
  1657→    }
  1658→
  1659→    /* New messages appended at the end */
  1660→    const last = messages[messages.length - 1];
  1661→    const own = last && last.sender_id === currentUserId;
  1662→    if (own) {
  1663→      requestAnimationFrame(() => scrollToBottom("instant"));
  1664→      return;
  1665→    }
  1666→    if (isAutoScrollRef.current)
  1667→      requestAnimationFrame(() => scrollToBottom("smooth"));
  1668→    else {
  1669→      unreadRef.current += messages.length - prev;
  1670→      setShowNewBtn(true);
  1671→    }
  1672→  }, [messages.length, currentUserId, scrollToBottom]);
  1673→
  1674→  /* Track scroll height for scroll preservation */
  1675→  useEffect(() => {
  1676→    const el = parentRef.current;
  1677→    if (!el) return;
  1678→    prevScrollHeightRef.current = el.scrollHeight;
  1679→  }, [messages.length]);
  1680→
  1681→  if (isLoading) return <MessageListSkeleton />;
  1682→  if (messages.length === 0 && !isPartnerTyping)
  1683→    return (
  1684→      <EmptyState partnerName={partnerName} partnerAvatar={partnerAvatar} />
  1685→    );
  1686→
  1687→  const virtualItems = virtualizer.getVirtualItems();
  1688→
  1689→  return (
  1690→    <div
  1691→      className="relative h-full w-full overflow-hidden"
  1692→      style={backgroundStyle}
  1693→    >
  1694→      <div
  1695→        ref={parentRef}
  1696→        onScroll={handleScroll}
  1697→        onClick={() => {
  1698→          onClearSearchHighlight?.();
  1699→          setActiveActionId(null);
  1700→        }}
  1701→        className="h-full w-full select-none overflow-y-auto overscroll-contain"
  1702→        style={{ WebkitOverflowScrolling: "touch" }}
  1703→      >
  1704→        <div
  1705→          style={{
  1706→            height: `${virtualizer.getTotalSize()}px`,
  1707→            width: "100%",
  1708→            position: "relative",
  1709→          }}
  1710→        >
  1711→          {/* Loading more spinner */}
  1712→          {loadingMore && (
  1713→            <div
  1714→              style={{
  1715→                position: "absolute",
  1716→                top: 0,
  1717→                left: 0,
  1718→                width: "100%",
  1719→                height: "48px",
  1720→                zIndex: 5,
  1721→              }}
  1722→            >
  1723→              <div className="flex items-center justify-center py-3">
  1724→                <svg
  1725→                  className="size-5 animate-spin"
  1726→                  viewBox="0 0 24 24"
  1727→                  fill="none"
  1728→                  aria-label="Loading more messages"
  1729→                  style={{ color: "var(--app-accent-light)" }}
  1730→                >
  1731→                  <circle
  1732→                    className="opacity-25"
  1733→                    cx="12"
  1734→                    cy="12"
  1735→                    r="10"
  1736→                    stroke="currentColor"
  1737→                    strokeWidth="4"
  1738→                  />
  1739→                  <path
