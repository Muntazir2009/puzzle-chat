    91→
    92→function EphemeralTimer({ seconds, onExpire }: { seconds: number; onExpire: () => void }) {
    93→  const [remaining, setRemaining] = useState(seconds);
    94→  const intervalRef = useRef<ReturnType<typeof setInterval> | undefined>(undefined);
    95→
    96→  useEffect(() => {
    97→    intervalRef.current = setInterval(() => {
    98→      setRemaining((r) => {
    99→        if (r <= 1) { clearInterval(intervalRef.current); onExpire(); return 0; }
   100→        return r - 1;
   101→      });
   102→    }, 1000);
   103→    return () => clearInterval(intervalRef.current);
   104→  }, [seconds, onExpire]);
   105→
   106→  const pct = remaining / seconds;
   107→  const circ = 2 * Math.PI * 7;
   108→  const offset = circ * (1 - pct);
   109→
   110→  return (
   111→    <svg className="size-3.5 shrink-0 ephemeral-timer-pulse" viewBox="0 0 18 18" aria-label={`${remaining}s remaining`}>
   112→      <circle cx="9" cy="9" r="7" fill="none" stroke="currentColor" strokeWidth="1.5" className="text-white/20" />
   113→      <circle cx="9" cy="9" r="7" fill="none" stroke="currentColor" strokeWidth="1.5" strokeDasharray={circ} strokeDashoffset={offset} strokeLinecap="round" style={{ color: "var(--app-accent-lighter)", transform: "rotate(-90deg)", transformOrigin: "50% 50%", transition: "stroke-dashoffset 1s linear" }} />
   114→    </svg>
   115→  );
   116→}

   199→function VoiceBubble({ message, isOwn, waveform }: {
   200→  message: ChatMessage; isOwn: boolean; waveform: number[];
   201→}) {
   202→  const audioRef = useRef<HTMLAudioElement>(null);
   203→  const [playing, setPlaying] = useState(false);
   204→  const [progress, setProgress] = useState(0);
   205→  const [speed, setSpeed] = useState(1);
   206→  const speeds = [1, 1.5, 2];
   207→  const duration = message.voice_duration ?? 0;
   208→
   209→  useEffect(() => {
   210→    const audio = audioRef.current;
   211→    if (!audio) return;
   212→    const onTime = () => { if (audio.duration) setProgress(audio.currentTime / audio.duration); };
   213→    const onEnd = () => setPlaying(false);
   214→    audio.addEventListener("timeupdate", onTime);
   215→    audio.addEventListener("ended", onEnd);
   216→    return () => { audio.removeEventListener("timeupdate", onTime); audio.removeEventListener("ended", onEnd); };
   217→  }, []);
   218→
   219→  const toggle = useCallback(() => {
   220→    const audio = audioRef.current;
   221→    if (!audio) return;
   222→    if (playing) { audio.pause(); setPlaying(false); }
   223→    else { audio.playbackRate = speed; audio.play().catch(() => {}); setPlaying(true); }
   224→  }, [playing, speed]);
   225→
   226→  const cycleSpeed = useCallback(() => {
   227→    setSpeed((s) => {
   228→      const idx = speeds.indexOf(s);
   229→      const next = speeds[(idx + 1) % speeds.length];
   230→      if (audioRef.current) audioRef.current.playbackRate = next;
   231→      return next;
   232→    });
   233→  }, []);
   234→
   235→  /* Build waveform bars */
   236→  const bars = useMemo(() => {
   237→    const count = Math.min(waveform.length, 40);
   238→    const step = Math.max(1, Math.floor(waveform.length / count));
   239→    return Array.from({ length: count }, (_, i) => waveform[i * step] ?? 0.1);
   240→  }, [waveform]);
   241→
   242→  const progressIdx = Math.floor(progress * bars.length);
   243→
   244→  return (
   245→    <div className={cn("flex items-center gap-3", isOwn ? "flex-row-reverse" : "")}>
   246→      <button type="button" onClick={toggle} className={cn("flex size-9 shrink-0 items-center justify-center rounded-full transition-all duration-150", isOwn ? "bg-white/20 hover:bg-white/30 hover:scale-105" : "bg-zinc-600 hover:bg-zinc-500 hover:scale-105")} aria-label={playing ? "Pause" : "Play"}>
   247→        {playing ? <Pause className="size-3.5 text-white" /> : <Play className="size-3.5 text-white ml-0.5" />}
   248→      </button>
   249→      <div className="flex items-center gap-[2px]" style={{ height: 32 }}>
   250→        {bars.map((amp, i) => (
   251→          <div key={i} className="w-[2.5px] rounded-full transition-all duration-100" style={{ height: `${Math.max(4, amp * 32)}px`, backgroundColor: i < progressIdx ? (isOwn ? "rgba(255,255,255,1)" : "var(--app-accent-light)") : (isOwn ? "rgba(255,255,255,0.5)" : "rgba(161,161,170,0.7)") }} />
   252→        ))}
   253→      </div>
   254→      <button type="button" onClick={cycleSpeed} className={cn("shrink-0 text-[11px] font-semibold tabular-nums transition-colors hover:opacity-80", isOwn ? "text-white/70" : "text-zinc-400")}>{speed}x</button>
   255→      <span className={cn("shrink-0 text-[11px] tabular-nums", isOwn ? "text-white/50" : "text-zinc-500")}>{Math.floor(duration / 60)}:{String(duration % 60).padStart(2, "0")}</span>
   256→      <audio ref={audioRef} src={message.content} preload="metadata" />
   257→    </div>
   258→  );
   259→}

   261→/* ------------------------------------------------------------------ */
   262→/*  LinkifiedText – detects & renders clickable URLs                   */
   263→/* ------------------------------------------------------------------ */
   264→
   265→const URL_REGEX = /(https?:\/\/[^\s<]+)/g;
   266→
   267→type TextSegment = { type: "text"; value: string } | { type: "link"; value: string };
   268→
   269→function parseUrls(text: string): TextSegment[] {
   270→  const segments: TextSegment[] = [];
   271→  let lastIndex = 0;
   272→  let match: RegExpExecArray | null;
   273→  while ((match = URL_REGEX.exec(text)) !== null) {
   274→    if (match.index > lastIndex) {
   275→      segments.push({ type: "text", value: text.slice(lastIndex, match.index) });
   276→    }
   277→    segments.push({ type: "link", value: match[1] });
   278→    lastIndex = URL_REGEX.lastIndex;
   279→  }
   280→  if (lastIndex < text.length) {
   281→    segments.push({ type: "text", value: text.slice(lastIndex) });
   282→  }
   283→  return segments;
   284→}
   285→
   286→/** Split a plain-text string into highlighted / non-highlighted fragments */
   287→function highlightText(text: string, query: string): React.ReactNode[] {
   288→  if (!query) return [text];
   289→  const escaped = query.replace(/[.*+?^${}()|[\]\\]/g, "\\$");
   290→  const re = new RegExp(`(${escaped})`, "gi");
   291→  const parts: React.ReactNode[] = [];
   292→  let lastIndex = 0;
   293→  let match: RegExpExecArray | null;
   294→  while ((match = re.exec(text)) !== null) {
   295→    if (match.index > lastIndex) {
   296→      parts.push(text.slice(lastIndex, match.index));
   297→    }
   298→    parts.push(
   299→      <mark key={match.index} className="bg-yellow-300/60 dark:bg-yellow-400/40 rounded-sm px-0.5">
   300→        {match[1]}
   301→      </mark>,
   302→    );
   303→    lastIndex = re.lastIndex;
   304→  }
   305→  if (lastIndex < text.length) {
   306→    parts.push(text.slice(lastIndex));
   307→  }
   308→  return parts;
   309→}
   310→
   311→export const LinkifiedText = React.memo(function LinkifiedText({ text, className, highlight }: { text: string; className?: string; highlight?: string }) {
   312→  const segments = useMemo(() => parseUrls(text), [text]);
   313→  return (
   314→    <span className={className}>
   315→      {segments.map((seg, i) =>
   316→        seg.type === "link" ? (
   317→          <a
   318→            key={i}
   319→            href={seg.value}
   320→            target="_blank"
   321→            rel="noopener noreferrer"
   322→            className="underline underline-offset-2 transition-colors"
   323→            style={{ textDecorationColor: "var(--app-accent-light)" }}
   324→            onClick={(e) => e.stopPropagation()}
   325→          >
   326→            {seg.value}
   327→          </a>
   328→        ) : (
   329→          <span key={i}>{highlight ? highlightText(seg.value, highlight) : seg.value}</span>
   330→        ),
   331→      )}
   332→    </span>
   333→  );
   334→});

   458→    const now = Date.now();
   459→    if (now - lastTapRef.current < DOUBLE_TAP_MS) triggerHeart();
   460→    lastTapRef.current = now;
   461→  }, [triggerHeart]);
   462→
   463→  const handlePressStart = useCallback(() => {
   464→    longPressRef.current = setTimeout(() => {
   465→      if (typeof navigator !== "undefined" && "vibrate" in navigator) navigator.vibrate(20);
   466→      setShowTapback(true);
   467→    }, LONG_PRESS_MS);
   468→  }, []);
   469→
   470→  const handlePressEnd = useCallback(() => { clearTimeout(longPressRef.current); }, []);
   471→
   472→  const handleContextMenu = useCallback((e: React.MouseEvent) => {
   473→    e.preventDefault(); setShowTapback(true);
   474→  }, []);
   475→
   476→  const handleCopy = useCallback(() => {
   477→    navigator.clipboard?.writeText(message.content).then(() => {
   478→      toast({ title: "Copied to clipboard", description: message.content.length > 50 ? message.content.slice(0, 50) + "..." : message.content });
   479→    }).catch(() => {
   480→      toast({ title: "Failed to copy", description: "Could not access clipboard", variant: "destructive" });
   481→    });
   482→    setShowTapback(false);
   483→  }, [message.content]);
   484→
   485→  const handleDelete = useCallback(() => {
   486→    onDeleteMessage?.(message.id);
   487→    setShowTapback(false);
   488→  }, [message.id, onDeleteMessage]);
   489→
   490→  const isFailed = message.status === "failed";
   491→  const isVoice = message.type === "voice";
   492→  const isText = !isVoice && message.type !== "image";
   493→  const isNew = message.status === "sending" || Date.now() - new Date(message.created_at).getTime() < 2000;
   494→
   495→  const timestampText = formatDistanceToNow(new Date(message.created_at || Date.now()), { addSuffix: true });
   496→
   497→  return (
   498→    <>
   499→    <m.div
   500→      className={cn("relative flex w-full select-none gap-2.5 px-1.5 sm:px-3", isOwn ? "justify-end" : "justify-start")}
   501→      initial={isNew ? { opacity: 0, y: 16, scale: 0.97 } : false}
   502→      animate={{ opacity: 1, y: 0, scale: 1 }}
   503→      transition={{ duration: 0.3, ease: [0.22, 1, 0.36, 1] }}
   504→    >
   505→      <HeartBurst show={showHeart} />
   506→      <AnimatePresence>{showTapback && onReact && (
   507→        <TapbackDock message={message} isOwn={isOwn} onReact={(emoji, add) => { onReact(message.id, emoji, add); setShowTapback(false); }} onReply={() => { onReplyTo?.(message); setShowTapback(false); }} onCopy={handleCopy} onDelete={isOwn ? handleDelete : undefined} />
   508→      )}</AnimatePresence>
   509→
   510→      {/* Reply arrow indicator for swipe-to-reply */}
   511→      <m.div
   512→        className={cn(
   513→          "absolute top-1/2 z-10 -translate-y-1/2",
   514→          isOwn ? "right-1.5 sm:right-3" : "left-1.5 sm:left-3",
   515→        )}
   516→        animate={showReplyArrow ? { scale: 1.2, opacity: 1 } : { scale: 0.5, opacity: 0 }}
   517→        transition={{ type: "spring", stiffness: 500, damping: 35 }}
   518→        style={{ pointerEvents: "none" }}
   519→      >
   520→        <div className="flex items-center justify-center rounded-full p-1.5" style={{ backgroundColor: "var(--app-accent)" }}>
   521→          <Reply className="size-3.5 text-white" />
   522→        </div>
   523→      </m.div>
   524→
   525→      {!isOwn && (
   526→        <Avatar className="mt-auto size-8 shrink-0 ring-1 ring-muted">
   527→          {partnerAvatar && <AvatarImage src={partnerAvatar} alt={partnerName} />}
   528→          <AvatarFallback className="text-xs font-medium" style={{ background: "linear-gradient(to bottom right, var(--app-accent-lighter), var(--app-accent-light))", color: "var(--app-accent-dark)" }}>{partnerName.slice(0, 2).toUpperCase()}</AvatarFallback>
   529→        </Avatar>
   530→      )}
   531→
   532→      <div className={cn("flex max-w-[75%] flex-col gap-1 select-none", isOwn ? "items-end mr-1.5 sm:mr-3" : "items-start ml-1.5 sm:ml-3")}>
   533→        <m.div
   534→          ref={bubbleRef}
   535→          drag="x"
   536→          dragConstraints={{ left: 0, right: 120 }}
   537→          dragElastic={0.3}
   538→          onDrag={(_, info) => {
   539→            if (info.offset.x < 0) return;
   540→            setShowReplyArrow(info.offset.x > 30);
   541→          }}
   542→          onDragEnd={(_, info) => {
   543→            if (info.offset.x > 60 && onReplyTo) {
   544→              onReplyTo(message);
   545→            }
   546→            setShowReplyArrow(false);
   547→          }}
   548→          transition={{ type: "spring", stiffness: 500, damping: 35 }}
   549→          onClick={handleClick}
   550→          onPointerDown={handlePressStart}
   551→          onPointerUp={handlePressEnd}
   552→          onPointerLeave={handlePressEnd}
   553→          onContextMenu={handleContextMenu}
   554→          className={cn(
   555→            "relative cursor-default select-none rounded-2xl px-4 py-2.5 text-sm leading-relaxed transition-all duration-200 active:scale-[0.98]",
   556→            isOwn && "rounded-br-sm text-white",
   557→            !isOwn && "bg-muted text-foreground rounded-bl-sm shadow-sm dark:bg-zinc-800 dark:text-zinc-100",
   558→            isFailed && "ring-2 ring-red-400/50",
   559→            isVoice && "py-2",
   560→            !isVoice && message.type !== "image" && "md:hover:shadow-md md:hover:-translate-y-0.5",
   561→          )}
   562→          style={{
   563→            touchCallout: "none",
   564→            transform: "translate3d(0,0,0)",
   565→            background: isOwn ? "linear-gradient(to right, var(--app-accent-from), var(--app-accent-to))" : undefined,
   566→            boxShadow: isOwn ? "0 4px 6px -1px var(--app-accent-glow)" : "0 1px 2px 0 var(--app-accent-glow)",

   568→        >
   569→          {/* Reply quote */}
   570→          {message.reply_to_id && message.reply_to_content && (
   571→            <ReplyPreview senderName={message.reply_to_sender_name ?? ""} content={message.reply_to_content} isOwn={isOwn} />

   573→

   575→          {isText ? (

   581→                {message.ephemeral_seconds && message.ephemeral_seconds > 0 && (
   582→                  <EphemeralTimer seconds={message.ephemeral_seconds} onExpire={() => { /* trigger vanish */ }} />
   583→                )}
   584→                <span className={cn(
   585→                  "text-[10px] leading-none mt-0.5 whitespace-nowrap",
   586→                  isNew && "animate-[timestamp-fade_0.4s_ease_0.3s_both]",
   587→                  isOwn ? "text-white/50" : "text-muted-foreground/60",
   588→                )}>{timestampText}</span>
   589→                {isOwn && <ReceiptIcon status={message.status} />}
   590→              </div>
   591→            </div>
   592→          ) : isVoice ? (
   593→            message.waveform_data ? (
   594→              <VoiceBubble message={message} isOwn={isOwn} waveform={message.waveform_data} />

   600→                <ZoomIn className="size-6 text-white opacity-0 transition-opacity duration-150 group-hover/img:opacity-80" />
   601→              </div>
   602→            </div>
   603→          )}
   604→
   605→          {/* Vanish indicator */}
   606→          {message.vanish_mode && (
   607→            <div className="mt-1 flex items-center gap-1 text-[10px] text-white/50"><EyeOff className="size-3" /><span>View once</span></div>
   608→          )}
   609→        </m.div>
   610→
   611→        {/* Timestamp + receipt for voice/image (text messages have inline timestamp) */}
   612→        {!isText && (
   613→          <div className={cn("flex items-center gap-1.5 px-1 text-[10px] text-muted-foreground/60", isOwn && "flex-row-reverse")}>
   614→            {message.ephemeral_seconds && message.ephemeral_seconds > 0 && (
   615→              <EphemeralTimer seconds={message.ephemeral_seconds} onExpire={() => { /* trigger vanish */ }} />
   616→            )}
   617→            <span className={cn(isNew && "animate-[timestamp-fade_0.4s_ease_0.3s_both]")}>{timestampText}</span>
   618→            {isOwn && <ReceiptIcon status={message.status} />}
   619→          </div>
   620→        )}
   621→
   622→        {/* Reactions */}
   623→        {onReact && message.reactions && <ReactionPills reactions={message.reactions} isOwn={isOwn} onReact={(e, a) => onReact(message.id, e, a)} currentUserId={""} />}
   624→
   625→        {isFailed && (
   626→          <button
   627→            type="button"
   628→            onClick={(e) => { e.stopPropagation(); onDeleteMessage?.(message.id); }}
   629→            className="mt-1 flex items-center gap-1 px-1 text-[10px] text-red-400 transition-colors hover:text-red-300"
   630→          >
   631→            <AlertCircle className="size-3" />
   632→            <span>Failed to send &middot; Tap to retry</span>
   633→          </button>
   634→        )}
   635→      </div>
   636→
   637→      {isOwn && <div className="w-8 shrink-0" />}
   638→    </m.div>
   639→
   640→    {/* Image Lightbox */}
   641→    <AnimatePresence>
   642→      {showLightbox && (
   643→        <ImageLightbox src={showLightbox} onClose={() => setShowLightbox(null)} />
   644→      )}
   645→    </AnimatePresence>
   646→    </>
   647→  );
   648→}

   654→function formatDateSeparator(dateStr: string): string {
   655→  const d = new Date(dateStr);
   656→  const now = new Date();
   657→  const today = new Date(now.getFullYear(), now.getMonth(), now.getDate());
   658→  const yesterday = new Date(today);
   659→  yesterday.setDate(yesterday.getDate() - 1);
   660→  const msgDay = new Date(d.getFullYear(), d.getMonth(), d.getDate());
   661→
   662→  if (msgDay.getTime() === today.getTime()) return "Today";
   663→  if (msgDay.getTime() === yesterday.getTime()) return "Yesterday";
   664→  return d.toLocaleDateString(undefined, { month: "short", day: "numeric" });
   665→}

   776→
   777→export function MessageFeed({ messages, isLoading, isPartnerTyping, currentUserId, partnerName, partnerAvatar, backgroundStyle, onMarkAsRead, onVanishMessage, onReplyTo, onDeleteMessage, onReact, scrollToMessageId, onScrolledToMessage, searchHighlight, onClearSearchHighlight, loadMore, hasMore, loadingMore }: MessageFeedProps) {
   778→  const parentRef = useRef<HTMLDivElement>(null);
   779→  const isAutoScrollRef = useRef(true);
   780→  const [showNewBtn, setShowNewBtn] = useState(false);
   781→  const unreadRef = useRef(0);
   782→  const prevCountRef = useRef(messages.length);
   783→  const prevFirstIdRef = useRef(messages.length > 0 ? messages[0].id : null);
   784→  const prevScrollHeightRef = useRef(0);
   785→
   786→  const totalItems = useMemo(() => messages.length + (isPartnerTyping ? 1 : 0), [messages.length, isPartnerTyping]);
   787→  const loadMoreRef = useRef(loadMore);
   788→  loadMoreRef.current = loadMore;
   789→
   790→  const virtualizer = useVirtualizer({ count: totalItems, getScrollElement: () => parentRef.current, estimateSize: () => ESTIMATED_ROW_HEIGHT, overscan: OVERSCAN, paddingStart: 16, paddingEnd: 16, gap: 12 });
   791→
   792→  const handleScroll = useCallback((e: UIEvent<HTMLDivElement>) => {
   793→    const el = e.currentTarget;
   794→    const dist = el.scrollHeight - el.scrollTop - el.clientHeight;
   795→    if (dist <= AUTO_SCROLL_THRESHOLD) { isAutoScrollRef.current = true; setShowNewBtn(false); unreadRef.current = 0; }
   796→    else { isAutoScrollRef.current = false; }
   797→    /* Trigger loadMore when scrolled near top */
   798→    if (el.scrollTop < 300 && hasMore && !loadingMore) {
   799→      loadMoreRef.current?.();
   800→    }
   801→  }, [hasMore, loadingMore]);
   802→
   803→  const scrollToBottom = useCallback((behavior: ScrollBehavior = "smooth") => {
   804→    const el = parentRef.current; if (!el) return;
   805→    el.scrollTo({ top: el.scrollHeight, behavior });
   806→    isAutoScrollRef.current = true; setShowNewBtn(false); unreadRef.current = 0;
   807→  }, []);
   808→
   809→  useEffect(() => { if (messages.length > 0) requestAnimationFrame(() => scrollToBottom("instant")); }, [isLoading, scrollToBottom]);
   810→
   811→  useEffect(() => { if (!onMarkAsRead) return; const unread = messages.filter((m) => m.sender_id !== currentUserId && m.status !== "read"); if (unread.length > 0 && isAutoScrollRef.current) onMarkAsRead(unread.map((m) => m.id)); }, [messages, currentUserId, onMarkAsRead]);
   812→
   813→  /* ---- Scroll to a specific message ----------------------------------- */
   814→  useEffect(() => {
   815→    if (!scrollToMessageId) return;
   816→    const idx = messages.findIndex((m) => m.id === scrollToMessageId);
   817→    if (idx === -1) { onScrolledToMessage?.(); return; }
   818→    virtualizer.scrollToIndex(idx, { align: "center" });
   819→    onScrolledToMessage?.();
   820→  }, [scrollToMessageId, messages, virtualizer, onScrolledToMessage]);
   821→
   822→  /* ---- Handle message count changes (new or prepended) ------------ */
   823→  useEffect(() => {
   824→    const prev = prevCountRef.current;
   825→    const prevFirstId = prevFirstIdRef.current;
   826→    prevCountRef.current = messages.length;
   827→    prevFirstIdRef.current = messages.length > 0 ? messages[0].id : null;
   828→
   829→    if (messages.length <= prev) return;
   830→
   831→    /* Check if older messages were prepended (first message changed) */
   832→    const firstId = messages[0]?.id;
   833→    const wasPrepended = firstId !== prevFirstId && prevFirstId !== null;
   834→
   835→    if (wasPrepended) {
   836→      /* Older messages prepended – preserve scroll position */
   837→      const el = parentRef.current;
   838→      if (el) {
   839→        const oldHeight = prevScrollHeightRef.current;
   840→        prevScrollHeightRef.current = el.scrollHeight;
   841→        requestAnimationFrame(() => {
   842→          const newHeight = el.scrollHeight;
   843→          el.scrollTop += newHeight - oldHeight;
   844→        });
   845→      }
   846→      return;
   847→    }
   848→
   849→    /* New messages appended at the end */
   850→    const last = messages[messages.length - 1];
   851→    const own = last && last.sender_id === currentUserId;
   852→    if (own) { requestAnimationFrame(() => scrollToBottom("instant")); return; }
   853→    if (isAutoScrollRef.current) requestAnimationFrame(() => scrollToBottom("smooth"));
   854→    else { unreadRef.current += messages.length - prev; setShowNewBtn(true); }
   855→  }, [messages.length, currentUserId, scrollToBottom]);
   856→
   857→  /* Track scroll height for scroll preservation */
   858→  useEffect(() => {
   859→    const el = parentRef.current;
   860→    if (!el) return;
   861→    prevScrollHeightRef.current = el.scrollHeight;
   862→  }, [messages.length]);
   863→
   864→  if (isLoading) return <MessageListSkeleton />;
   865→  if (messages.length === 0 && !isPartnerTyping) return <EmptyState partnerName={partnerName} partnerAvatar={partnerAvatar} />;
   866→
   867→  const virtualItems = virtualizer.getVirtualItems();
   868→
   869→  return (
   870→    <div className="relative h-full w-full overflow-hidden" style={backgroundStyle}>
   871→      <div ref={parentRef} onScroll={handleScroll} onClick={() => onClearSearchHighlight?.()} className="h-full w-full select-none overflow-y-auto overscroll-contain" style={{ WebkitOverflowScrolling: "touch" }}>
   872→        <div style={{ height: `${virtualizer.getTotalSize()}px`, width: "100%", position: "relative" }}>
   873→          {/* Loading more spinner */}
   874→          {loadingMore && (
   875→            <div style={{ position: "absolute", top: 0, left: 0, width: "100%", height: "48px", zIndex: 5 }}>
   876→              <div className="flex items-center justify-center py-3">
   877→                <svg className="size-5 animate-spin" viewBox="0 0 24 24" fill="none" aria-label="Loading more messages" style={{ color: "var(--app-accent-light)" }}>

   880→                </svg>
   881→              </div>
   882→            </div>
   883→          )}
   884→          {virtualItems.map((vi) => {
   885→            const isTyping = vi.index === messages.length;
   886→            const msg = isTyping ? null : messages[vi.index];
   887→            // Check if we need a date separator before this message
   888→            const showDateSep = msg && vi.index > 0 && messages[vi.index - 1] && (() => {
   889→              const prev = new Date(messages[vi.index - 1].created_at);
   890→              const curr = new Date(msg.created_at);
   891→              return prev.toDateString() !== curr.toDateString();
   892→            })();
   893→            const showFirstDateSep = msg && vi.index === 0;
   894→            return (
   895→              <div key={vi.key} data-index={vi.index} ref={virtualizer.measureElement} style={{ position: "absolute", top: 0, left: 0, width: "100%", transform: `translateY(${vi.start}px)` }}>
   896→                {isTyping ? <TypingIndicator partnerName={partnerName} partnerAvatar={partnerAvatar} /> : msg ? (
   897→                  <>
   898→                    {(showDateSep || showFirstDateSep) && <DateSeparator date={msg.created_at} />}
   899→                    <MessageBubble message={msg} isOwn={msg.sender_id === currentUserId} partnerName={partnerName} partnerAvatar={partnerAvatar} onReplyTo={onReplyTo} onReact={onReact} onDeleteMessage={onDeleteMessage} searchHighlight={searchHighlight} />
   900→                  </>
   901→                ) : null}
   902→              </div>
   903→            );
   904→          })}
   905→        </div>
   906→      </div>
   907→      {showNewBtn && unreadRef.current > 0 && <NewMessagesButton count={unreadRef.current} onClick={() => scrollToBottom("smooth")} />}
   908→    </div>
   909→  );
