    60→  const lastTapRef = useRef(0);
    61→  return useCallback((): boolean => {
    62→    const now = Date.now();
    63→    if (now - lastTapRef.current < delay) {
    64→      lastTapRef.current = 0;
    65→      onDoubleTap();
    66→      return true; // is double tap
    67→    }
    68→    lastTapRef.current = now;
    69→    return false; // is single tap
    70→  }, [onDoubleTap, delay]);
    71→}

   147→function EphemeralTimer({
   148→  seconds,
   149→  onExpire,
   150→}: {
   151→  seconds: number;
   152→  onExpire: () => void;
   153→}) {
   154→  const [remaining, setRemaining] = useState(seconds);
   155→  const intervalRef = useRef<ReturnType<typeof setInterval> | undefined>(
   156→    undefined,
   157→  );
   158→
   159→  useEffect(() => {
   160→    intervalRef.current = setInterval(() => {
   161→      setRemaining((r) => {
   162→        if (r <= 1) {
   163→          clearInterval(intervalRef.current);
   164→          onExpire();
   165→          return 0;
   166→        }
   167→        return r - 1;
   168→      });
   169→    }, 1000);
   170→    return () => clearInterval(intervalRef.current);
   171→  }, [seconds, onExpire]);
   172→
   173→  const pct = remaining / seconds;
   174→  const circ = 2 * Math.PI * 7;
   175→  const offset = circ * (1 - pct);
   176→
   177→  return (
   178→    <svg
   179→      className="size-3.5 shrink-0 ephemeral-timer-pulse"
   180→      viewBox="0 0 18 18"
   181→      aria-label={`${remaining}s remaining`}
   182→    >
   183→      <circle
   184→        cx="9"
   185→        cy="9"
   186→        r="7"
   187→        fill="none"
   188→        stroke="currentColor"
   189→        strokeWidth="1.5"
   190→        className="text-white/20"
   191→      />
   192→      <circle
   193→        cx="9"
   194→        cy="9"
   195→        r="7"
   196→        fill="none"
   197→        stroke="currentColor"
   198→        strokeWidth="1.5"
   199→        strokeDasharray={circ}
   200→        strokeDashoffset={offset}
   201→        strokeLinecap="round"
   202→        style={{
   203→          color: "var(--app-accent-lighter)",

   421→/*  Voice Bubble                                                       */

   423→
   424→function VoiceBubble({
   425→  message,
   426→  isOwn,
   427→  waveform,
   428→}: {
   429→  message: ChatMessage;
   430→  isOwn: boolean;
   431→  waveform: number[];
   432→}) {
   433→  const audioRef = useRef<HTMLAudioElement>(null);
   434→  const [playing, setPlaying] = useState(false);
   435→  const [progress, setProgress] = useState(0);
   436→  const [speed, setSpeed] = useState(1);
   437→  const speeds = [1, 1.5, 2];
   438→  const duration = message.voice_duration ?? 0;
   439→
   440→  useEffect(() => {
   441→    const audio = audioRef.current;
   442→    if (!audio) return;
   443→    const onTime = () => {
   444→      if (audio.duration) setProgress(audio.currentTime / audio.duration);
   445→    };
   446→    const onEnd = () => setPlaying(false);
   447→    audio.addEventListener("timeupdate", onTime);
   448→    audio.addEventListener("ended", onEnd);
   449→    return () => {
   450→      audio.removeEventListener("timeupdate", onTime);
   451→      audio.removeEventListener("ended", onEnd);
   452→    };
   453→  }, []);
   454→
   455→  const toggle = useCallback(() => {
   456→    const audio = audioRef.current;
   457→    if (!audio) return;
   458→    if (playing) {
   459→      audio.pause();
   460→      setPlaying(false);
   461→    } else {
   462→      audio.playbackRate = speed;
   463→      audio.play().catch(() => {});
   464→      setPlaying(true);
   465→    }
   466→  }, [playing, speed]);
   467→
   468→  const cycleSpeed = useCallback(() => {
   469→    setSpeed((s) => {
   470→      const idx = speeds.indexOf(s);
   471→      const next = speeds[(idx + 1) % speeds.length];
   472→      if (audioRef.current) audioRef.current.playbackRate = next;
   473→      return next;
   474→    });
   475→  }, []);
   476→
   477→  /* Build waveform bars */
   478→  const bars = useMemo(() => {
   479→    const count = Math.min(waveform.length, 40);
   480→    const step = Math.max(1, Math.floor(waveform.length / count));
   481→    return Array.from({ length: count }, (_, i) => waveform[i * step] ?? 0.1);
   482→  }, [waveform]);
   483→
   484→  const progressIdx = Math.floor(progress * bars.length);
   485→
   486→  return (
   487→    <div className={cn("flex items-center gap-3", isOwn ? "flex-row-reverse" : "")}>
   488→      <button
   489→        type="button"
   490→        onClick={toggle}
   491→        className={cn(
   492→          "flex size-9 shrink-0 items-center justify-center rounded-full transition-all duration-150 active:scale-95",
   493→          isOwn
   494→            ? "bg-white/20 hover:bg-white/30"
   495→            : "bg-zinc-600 hover:bg-[var(--app-accent-subtle)]",
   496→        )}
   497→        aria-label={playing ? "Pause" : "Play"}
   498→      >
   499→        {playing ? (
   500→          <Pause className="size-3.5 text-white" />
   501→        ) : (
   502→          <Play className="size-3.5 text-white ml-0.5" />
   503→        )}
   504→      </button>
   505→      <div className="flex items-center gap-[2px]" style={{ height: 32 }}>
   506→        {bars.map((amp, i) => (
   507→          <div
   508→            key={i}
   509→            className="w-[2.5px] rounded-full transition-all duration-100"
   510→            style={{
   511→              height: `${Math.max(4, amp * 32)}px`,
   512→              backgroundColor:
   513→                i < progressIdx
   514→                  ? isOwn
   515→                    ? "rgba(255,255,255,1)"
   516→                    : "var(--app-accent-light)"
   517→                  : isOwn
   518→                    ? "rgba(255,255,255,0.5)"
   519→                    : "rgba(161,161,170,0.7)",
   520→            }}
   521→          />
   522→        ))}
   523→      </div>
   524→      <button
   525→        type="button"
   526→        onClick={cycleSpeed}
   527→        className={cn(
   528→          "shrink-0 text-[11px] font-semibold tabular-nums transition-colors hover:opacity-80 active:scale-95",
   529→          isOwn ? "text-white/70" : "text-zinc-400",
   530→        )}
   531→      >
   532→        {speed}x
   533→      </button>
   534→      <span
   535→        className={cn(
   536→          "shrink-0 text-[11px] tabular-nums",
   537→          isOwn ? "text-white/50" : "text-zinc-500",
   538→        )}
   539→      >
   540→        {Math.floor(duration / 60)}:{String(duration % 60).padStart(2, "0")}
   541→      </span>
   542→      <audio ref={audioRef} src={message.content} preload="metadata" />
   543→    </div>
   544→  );
   545→}

   547→/* ------------------------------------------------------------------ */
   548→/*  LinkifiedText – detects & renders clickable URLs                   */
   549→/* ------------------------------------------------------------------ */
   550→
   551→const URL_REGEX = /(https?:\/\/[^\s<]+)/g;
   552→
   553→type TextSegment =
   554→  | { type: "text"; value: string }
   555→  | { type: "link"; value: string };
   556→
   557→function parseUrls(text: string): TextSegment[] {
   558→  const segments: TextSegment[] = [];
   559→  let lastIndex = 0;
   560→  let match: RegExpExecArray | null;
   561→  while ((match = URL_REGEX.exec(text)) !== null) {
   562→    if (match.index > lastIndex) {
   563→      segments.push({ type: "text", value: text.slice(lastIndex, match.index) });
   564→    }
   565→    segments.push({ type: "link", value: match[1] });
   566→    lastIndex = URL_REGEX.lastIndex;
   567→  }
   568→  if (lastIndex < text.length) {
   569→    segments.push({ type: "text", value: text.slice(lastIndex) });
   570→  }
   571→  return segments;
   572→}
   573→
   574→/** Split a plain-text string into highlighted / non-highlighted fragments */
   575→function highlightText(text: string, query: string): React.ReactNode[] {
   576→  if (!query) return [text];
   577→  const escaped = query.replace(/[.*+?^${}()|[\]\\]/g, "\\");
   578→  const re = new RegExp(`(${escaped})`, "gi");
   579→  const parts: React.ReactNode[] = [];
   580→  let lastIndex = 0;
   581→  let match: RegExpExecArray | null;
   582→  while ((match = re.exec(text)) !== null) {
   583→    if (match.index > lastIndex) {
   584→      parts.push(text.slice(lastIndex, match.index));
   585→    }
   586→    parts.push(
   587→      <mark
   588→        key={match.index}
   589→        className="bg-yellow-300/60 dark:bg-yellow-400/40 rounded-sm px-0.5"
   590→      >
   591→        {match[1]}
   592→      </mark>,
   593→    );
   594→    lastIndex = re.lastIndex;
   595→  }
   596→  if (lastIndex < text.length) {
   597→    parts.push(text.slice(lastIndex));
   598→  }
   599→  return parts;
   600→}
   601→
   602→export const LinkifiedText = React.memo(function LinkifiedText({
   603→  text,
   604→  className,
   605→  highlight,
   606→}: {
   607→  text: string;
   608→  className?: string;
   609→  highlight?: string;
   610→}) {
   611→  const segments = useMemo(() => parseUrls(text), [text]);
   612→  return (
   613→    <span className={className}>
   614→      {segments.map((seg, i) =>
   615→        seg.type === "link" ? (
   616→          <a
   617→            key={i}
   618→            href={seg.value}
   619→            target="_blank"
   620→            rel="noopener noreferrer"
   621→            className="underline underline-offset-2 transition-colors"
   622→            style={{ textDecorationColor: "var(--app-accent-light)" }}
   623→            onClick={(e) => e.stopPropagation()}
   624→          >
   625→            {seg.value}
   626→          </a>
   627→        ) : (
   628→          <span key={i}>
   629→            {highlight ? highlightText(seg.value, highlight) : seg.value}
   630→          </span>
   631→        ),
   632→      )}
   633→    </span>
   634→  );
   635→});

   807→/* ------------------------------------------------------------------ */
   808→/*  Bubble rounding helper for grouped messages                        */
   809→/* ------------------------------------------------------------------ */
   810→
   811→function getBubbleRounding({
   812→  isFirst,
   813→  isLast,
   814→  isOwn,
   815→}: {
   816→  isFirst: boolean;
   817→  isLast: boolean;
   818→  isOwn: boolean;
   819→}): string {
   820→  if (isFirst && isLast) {
   821→    // Single message in group (or only message)
   822→    return isOwn
   823→      ? "rounded-2xl rounded-br-sm"
   824→      : "rounded-2xl rounded-bl-sm";
   825→  }
   826→  if (isFirst) {
   827→    // First message of a multi-message group
   828→    return isOwn
   829→      ? "rounded-tl-2xl rounded-bl-2xl rounded-tr-sm rounded-br-sm"
   830→      : "rounded-tl-2xl rounded-bl-sm rounded-tr-2xl rounded-br-sm";
   831→  }
   832→  if (isLast) {
   833→    // Last message of a multi-message group
   834→    return isOwn
   835→      ? "rounded-tl-2xl rounded-bl-2xl rounded-tr-sm rounded-br-2xl"
   836→      : "rounded-tl-2xl rounded-bl-2xl rounded-tr-2xl rounded-br-sm";
   837→  }
   838→  // Middle message
   839→  return isOwn
   840→    ? "rounded-tl-2xl rounded-bl-2xl rounded-tr-sm rounded-br-sm"
   841→    : "rounded-tl-2xl rounded-bl-sm rounded-tr-2xl rounded-br-sm";
   842→}

   856→  currentUserName,
   857→  currentUserAvatar,
   858→  onReplyTo,
   859→  onReact,
   860→  onDeleteMessage,
   861→  searchHighlight,

   863→  onOpenAction,
   864→  activeTapbackId,
   865→  onOpenTapback,
   866→}: {

   874→  currentUserName: string;

   876→  onReplyTo?: (msg: ChatMessage) => void;
   877→  onReact?: (id: string, emoji: string, add: boolean) => void;

   886→  const showActionSheet = activeActionId === message.id;

   888→  const [showReplyArrow, setShowReplyArrow] = useState(false);
   889→  const longPressRef = useRef<ReturnType<typeof setTimeout> | undefined>(
   890→    undefined,
   891→  );

   893→
   894→  /* Double-tap to reaction */
   895→  const handleDoubleTapReaction = useCallback(() => {
   896→    if (!onReact) return;
   897→    onReact(message.id, "\u{1F44D}", true);

   900→  const doubleTap = useDoubleTap(handleDoubleTapReaction);
   901→

   903→    clearTimeout(longPressRef.current);
   904→    if (wasDraggedRef.current) {
   905→      wasDraggedRef.current = false;
   906→      return;

   908→    // Check double-tap first
   909→    const isDouble = doubleTap();
   910→    if (isDouble) {
   911→      // Double tap — close any open menus (reaction was already sent by hook)
   912→      onOpenAction(null);
   913→      onOpenTapback(null);
   914→      return;
   915→    }

   918→      onOpenAction(null);
   919→    } else {
   920→      onOpenAction(message.id);
   921→      onOpenTapback(null);
   922→    }
   923→  }, [activeActionId, message.id, onOpenAction, onOpenTapback, doubleTap]);
   924→
   925→  const handlePressStart = useCallback(() => {
   926→    // Close any previously open menu from ANY bubble (prevent stacking)
   927→    onOpenAction(null);
   928→    onOpenTapback(null);
   929→    longPressRef.current = setTimeout(() => {
   930→      if (typeof navigator !== "undefined" && "vibrate" in navigator)

   937→    clearTimeout(longPressRef.current);

   940→  const handleContextMenu = useCallback((e: React.MouseEvent) => {

   946→  const handleCopy = useCallback(() => {

   948→      ?.writeText(message.content)

   956→        });
   957→      })
   958→      .catch(() => {
   959→        toast({
   960→          title: "Failed to copy",
   961→          description: "Could not access clipboard",
   962→          variant: "destructive",
   963→        });
   964→      });
   965→    onOpenTapback(null);
   966→  }, [message.content, onOpenTapback]);
   967→
   968→  const handleDelete = useCallback(() => {
   969→    onDeleteMessage?.(message.id);
   970→    onOpenTapback(null);
   971→    onOpenAction(null);

  1005→      title: "Forward",
  1006→      description: "Forward coming soon!",
  1007→    });
  1008→    onOpenAction(null);
  1009→    onOpenTapback(null);
  1010→  }, [onOpenAction, onOpenTapback]);

  1013→  const isVoice = message.type === "voice";

  1034→        )}

  1036→        {/* Avatar column for other user's messages — always rendered for stacking */}

  1401→function formatDateSeparator(dateStr: string): string {
  1402→  const d = new Date(dateStr);
  1403→  const now = new Date();
  1404→  const today = new Date(now.getFullYear(), now.getMonth(), now.getDate());
  1405→  const yesterday = new Date(today);
  1406→  yesterday.setDate(yesterday.getDate() - 1);
  1407→  const msgDay = new Date(d.getFullYear(), d.getMonth(), d.getDate());
  1408→
  1409→  if (msgDay.getTime() === today.getTime()) return "Today";
  1410→  if (msgDay.getTime() === yesterday.getTime()) return "Yesterday";
  1411→  return d.toLocaleDateString(undefined, { month: "short", day: "numeric" });
  1412→}

  1595→export function MessageFeed({
  1596→  messages,
  1597→  isLoading,
  1598→  isPartnerTyping,
  1599→  currentUserId,
  1600→  currentUserName,
  1601→  currentUserAvatar,
  1602→  partnerName,
  1603→  partnerAvatar,
  1604→  backgroundStyle,
  1605→  onMarkAsRead,
  1606→  onVanishMessage,
  1607→  onReplyTo,
  1608→  onDeleteMessage,
  1609→  onReact,
  1610→  scrollToMessageId,
  1611→  onScrolledToMessage,
  1612→  searchHighlight,
  1613→  onClearSearchHighlight,
  1614→  loadMore,
  1615→  hasMore,
  1616→  loadingMore,
  1617→}: MessageFeedProps) {
  1618→  const parentRef = useRef<HTMLDivElement>(null);
  1619→  const isAutoScrollRef = useRef(true);
  1620→  const [activeActionId, setActiveActionId] = useState<string | null>(null);
  1621→  const [activeTapbackId, setActiveTapbackId] = useState<string | null>(null);
  1622→  const [showNewBtn, setShowNewBtn] = useState(false);
  1623→  const unreadRef = useRef(0);
  1624→  const prevCountRef = useRef(messages.length);
  1625→  const prevFirstIdRef = useRef(
  1626→    messages.length > 0 ? messages[0].id : null,
  1627→  );
  1628→  const prevScrollHeightRef = useRef(0);
  1629→
  1630→  /* ---- Build message groups ---------------------------------------- */
  1631→  const groups = useMemo(() => {
  1632→    const result: MessageGroupData[] = [];
  1633→    for (const msg of messages) {
  1634→      const last = result[result.length - 1];
  1635→      if (last && last.senderId === msg.sender_id) {
  1636→        last.messages.push(msg);
  1637→      } else {
  1638→        result.push({ senderId: msg.sender_id, messages: [msg] });
  1639→      }
  1640→    }
  1641→    return result;
  1642→  }, [messages]);
  1643→
  1644→  const totalItems = useMemo(
  1645→    () => groups.length + (isPartnerTyping ? 1 : 0),
  1646→    [groups.length, isPartnerTyping],
  1647→  );
  1648→
  1649→  const loadMoreRef = useRef(loadMore);
  1650→  loadMoreRef.current = loadMore;
  1651→
  1652→  const virtualizer = useVirtualizer({
  1653→    count: totalItems,
  1654→    getScrollElement: () => parentRef.current,
  1655→    estimateSize: () => ESTIMATED_ROW_HEIGHT,
  1656→    overscan: OVERSCAN,
  1657→    paddingStart: 16,
  1658→    paddingEnd: 16,
  1659→    gap: 12,
  1660→  });
  1661→
  1662→  const handleScroll = useCallback(
  1663→    (e: UIEvent<HTMLDivElement>) => {
  1664→      const el = e.currentTarget;
  1665→      const dist = el.scrollHeight - el.scrollTop - el.clientHeight;
  1666→      if (dist <= AUTO_SCROLL_THRESHOLD) {
  1667→        isAutoScrollRef.current = true;
  1668→        setShowNewBtn(false);
  1669→        unreadRef.current = 0;
  1670→      } else {
  1671→        isAutoScrollRef.current = false;
  1672→      }
  1673→      /* Trigger loadMore when scrolled near top */
  1674→      if (el.scrollTop < 300 && hasMore && !loadingMore) {
  1675→        loadMoreRef.current?.();
  1676→      }
  1677→    },
  1678→    [hasMore, loadingMore],
  1679→  );
  1680→
  1681→  const scrollToBottom = useCallback(
  1682→    (behavior: ScrollBehavior = "smooth") => {
  1683→      const el = parentRef.current;
  1684→      if (!el) return;
  1685→      el.scrollTo({ top: el.scrollHeight, behavior });
  1686→      isAutoScrollRef.current = true;
  1687→      setShowNewBtn(false);
  1688→      unreadRef.current = 0;
  1689→    },
  1690→    [],
  1691→  );
  1692→
  1693→  useEffect(() => {
  1694→    if (messages.length > 0)
  1695→      requestAnimationFrame(() => scrollToBottom("instant"));
  1696→  }, [isLoading, scrollToBottom]);
  1697→
  1698→  useEffect(() => {
  1699→    if (!onMarkAsRead) return;
  1700→    const unread = messages.filter(
  1701→      (m) => m.sender_id !== currentUserId && m.status !== "read",
  1702→    );
  1703→    if (unread.length > 0 && isAutoScrollRef.current)
  1704→      onMarkAsRead(unread.map((m) => m.id));
  1705→  }, [messages, currentUserId, onMarkAsRead]);
  1706→
  1707→  /* ---- Scroll to a specific message ----------------------------------- */
  1708→  useEffect(() => {
  1709→    if (!scrollToMessageId) return;
  1710→    // Find which group contains the target message
  1711→    const groupIdx = groups.findIndex((g) =>
  1712→      g.messages.some((m) => m.id === scrollToMessageId),
  1713→    );
  1714→    if (groupIdx === -1) {
  1715→      onScrolledToMessage?.();
  1716→      return;
  1717→    }
  1718→    virtualizer.scrollToIndex(groupIdx, { align: "center" });
  1719→    onScrolledToMessage?.();
  1720→  }, [scrollToMessageId, groups, virtualizer, onScrolledToMessage]);
  1721→
  1722→  /* ---- Handle message count changes (new or prepended) ------------ */
  1723→  useEffect(() => {
  1724→    const prev = prevCountRef.current;
  1725→    const prevFirstId = prevFirstIdRef.current;
  1726→    prevCountRef.current = messages.length;
  1727→    prevFirstIdRef.current =
  1728→      messages.length > 0 ? messages[0].id : null;
  1729→
  1730→    if (messages.length <= prev) return;
  1731→
  1732→    /* Check if older messages were prepended (first message changed) */
  1733→    const firstId = messages[0]?.id;
  1734→    const wasPrepended = firstId !== prevFirstId && prevFirstId !== null;
  1735→
  1736→    if (wasPrepended) {
  1737→      /* Older messages prepended – preserve scroll position */
  1738→      const el = parentRef.current;
  1739→      if (el) {
  1740→        const oldHeight = prevScrollHeightRef.current;
  1741→        prevScrollHeightRef.current = el.scrollHeight;
  1742→        requestAnimationFrame(() => {
  1743→          const newHeight = el.scrollHeight;
  1744→          el.scrollTop += newHeight - oldHeight;
  1745→        });
  1746→      }
  1747→      return;
  1748→    }
  1749→
  1750→    /* New messages appended at the end */
  1751→    const last = messages[messages.length - 1];
  1752→    const own = last && last.sender_id === currentUserId;
  1753→    if (own) {
  1754→      requestAnimationFrame(() => scrollToBottom("instant"));
  1755→      return;
  1756→    }
  1757→    if (isAutoScrollRef.current)
  1758→      requestAnimationFrame(() => scrollToBottom("smooth"));
  1759→    else {
  1760→      unreadRef.current += messages.length - prev;
  1761→      setShowNewBtn(true);
  1762→    }
  1763→  }, [messages.length, currentUserId, scrollToBottom]);
  1764→
  1765→  /* Track scroll height for scroll preservation */
  1766→  useEffect(() => {
  1767→    const el = parentRef.current;
  1768→    if (!el) return;
  1769→    prevScrollHeightRef.current = el.scrollHeight;
  1770→  }, [messages.length]);
  1771→
  1772→  if (isLoading) return <MessageListSkeleton />;
  1773→  if (messages.length === 0 && !isPartnerTyping)
  1774→    return (
  1775→      <EmptyState partnerName={partnerName} partnerAvatar={partnerAvatar} />
  1776→    );
  1777→
  1778→  const virtualItems = virtualizer.getVirtualItems();
  1779→
  1780→  return (
  1781→    <div
  1782→      className="relative h-full w-full overflow-hidden"
  1783→      style={backgroundStyle}
  1784→    >
  1785→      <div
  1786→        ref={parentRef}
  1787→        onScroll={handleScroll}
  1788→        onClick={(e) => {
  1789→          onClearSearchHighlight?.();
  1790→          // Close menus if click is directly on the scroll container or background
  1791→          const target = e.target as HTMLElement;
  1792→          if (target === e.currentTarget || target.closest('.message-bubble-row') === null) {
  1793→            setActiveActionId(null);
  1794→            setActiveTapbackId(null);
  1795→          }
  1796→        }}
  1797→        className="h-full w-full select-none overflow-y-auto overscroll-contain"
  1798→        style={{ WebkitOverflowScrolling: "touch" }}
  1799→      >
  1800→        <div
  1801→          style={{
  1802→            height: `${virtualizer.getTotalSize()}px`,
  1803→            width: "100%",
  1804→            position: "relative",
  1805→          }}
  1806→        >
  1807→          {/* Loading more spinner */}
  1808→          {loadingMore && (
  1809→            <div
  1810→              style={{
  1811→                position: "absolute",
  1812→                top: 0,
  1813→                left: 0,
  1814→                width: "100%",
  1815→                height: "48px",
  1816→                zIndex: 5,
  1817→              }}
  1818→            >
  1819→              <div className="flex items-center justify-center py-3">
  1820→                <svg
  1821→                  className="size-5 animate-spin"
  1822→                  viewBox="0 0 24 24"
