    67→function SearchPanel({
    68→  conversationId,
    69→  onResultClick,
    70→  onClose,
    71→}: {
    72→  conversationId: string;
    73→  onResultClick: (messageId: string) => void;
    74→  onClose: () => void;
    75→}) {
    76→  const [query, setQuery] = useState("");
    77→  const [results, setResults] = useState<ChatMessage[]>([]);
    78→  const [isSearching, setIsSearching] = useState(false);
    79→  const inputRef = useRef<HTMLInputElement>(null);
    80→  const debounceRef = useRef<ReturnType<typeof setTimeout>>();
    81→
    82→  /* Auto-focus on open */
    83→  useEffect(() => {
    84→    requestAnimationFrame(() => inputRef.current?.focus());
    85→  }, []);
    86→
    87→  /* Escape key to close */
    88→  useEffect(() => {
    89→    function handleKeyDown(e: KeyboardEvent) {
    90→      if (e.key === "Escape") onClose();
    91→    }
    92→    window.addEventListener("keydown", handleKeyDown);
    93→    return () => window.removeEventListener("keydown", handleKeyDown);
    94→  }, [onClose]);
    95→
    96→  /* Debounced search */
    97→  useEffect(() => {
    98→    if (debounceRef.current) clearTimeout(debounceRef.current);
    99→    if (!query.trim()) { setResults([]); setIsSearching(false); return; }
   100→    setIsSearching(true);
   101→    debounceRef.current = setTimeout(async () => {
   102→      try {
   103→        const res = await fetch(`/api/messages/search?conversation_id=${conversationId}&q=${encodeURIComponent(query)}`);
   104→        if (res.ok) {
   105→          const data: ChatMessage[] = await res.json();
   106→          setResults(data);
   107→        }
   108→      } catch {
   109→        /* silent */
   110→      } finally {
   111→        setIsSearching(false);
   112→      }
   113→    }, 300);
   114→    return () => { if (debounceRef.current) clearTimeout(debounceRef.current); };
   115→  }, [query, conversationId]);
   116→
   117→  return (
   118→    <m.div
   119→      initial={{ height: 0, opacity: 0 }}
   120→      animate={{ height: "auto", opacity: 1 }}
   121→      exit={{ height: 0, opacity: 0 }}
   122→      transition={{ duration: 0.2, ease: "easeOut" }}
   123→      className="overflow-hidden border-b bg-background/80 backdrop-blur-xl"
   124→    >
   125→      <div className="flex items-center gap-2 px-4 py-2.5">
   126→        <Search className="size-4 shrink-0 text-muted-foreground" />
   127→        <input
   128→          ref={inputRef}
   129→          type="text"
   130→          value={query}
   131→          onChange={(e) => setQuery(e.target.value)}
   132→          placeholder="Search messages..."
   133→          className="flex-1 bg-transparent text-sm outline-none placeholder:text-muted-foreground"
   134→          aria-label="Search messages"
   135→        />
   136→        {isSearching ? (
   137→          <Loader2 className="size-4 shrink-0 animate-spin text-muted-foreground" />
   138→        ) : (
   139→          <button
   140→            type="button"
   141→            onClick={() => { setQuery(""); onClose(); }}
   142→            className="flex size-6 items-center justify-center rounded-full text-muted-foreground transition-colors hover:bg-muted"
   143→            aria-label="Close search"
   144→          >
   145→            <X className="size-3.5" />
   146→          </button>
   147→        )}
   148→      </div>
   149→
   150→      {/* Results list */
   151→      {results.length > 0 && (
   152→        <div className="max-h-[200px] overflow-y-auto border-t px-2 pb-2">
   153→          {results.map((msg) => (
   154→            <button
   155→              key={msg.id}
   156→              type="button"
   157→              onClick={() => onResultClick(msg.id)}
   158→              className="flex w-full flex-col gap-0.5 rounded-lg px-3 py-2.5 text-left transition-colors hover:bg-muted/50"
   159→            >
   160→              <p className="truncate text-sm text-foreground">
   161→                {msg.content.length > 60 ? msg.content.slice(0, 60) + "..." : msg.content}
   162→              </p>
   163→              <span className="text-[11px] text-muted-foreground">
   164→                {formatDistanceToNow(new Date(msg.created_at), { addSuffix: true })}
   165→              </span>
   166→            </button>
   167→          ))}
   168→        </div>
   169→      )}
   170→
   171→      {query.trim() && !isSearching && results.length === 0 && (
   172→        <div className="border-t px-4 py-3 text-center text-xs text-muted-foreground">
   173→          No results found
   174→        </div>
   175→      )}
   176→    </m.div>
   177→  );
   178→}

   632→      {/* Partner Info Panel */}
   633→      <PartnerInfoPanel
