   764→                      ? "border-violet-500/30 focus-visible:ring-violet-500/30 focus-visible:shadow-[0_0_0_3px_rgba(139,92,246,0.1)] vanish-input-glow"
   765→                      : "focus-visible:ring-violet-500/30 focus-visible:shadow-[0_0_0_3px_rgba(139,92,246,0.1)]",
   766→                  )}
   767→                />
   768→              </div>
   769→            )}
   770→
   771→            {/* Hidden file input for attachments */}
   772→            <input
   773→              ref={fileInputRef}
   774→              type="file"
   775→              accept={ACCEPTED_TYPES}
   776→              onChange={handleFileSelect}
   777→              className="hidden"
   778→              aria-hidden="true"
   779→            />
   780→
   781→            {/* Attachment button */}
   782→            {!showVoiceWaveform && (
   783→              <m.button type="button" onClick={handlePaperclipClick} whileTap={{ scale: 0.9 }} className={cn("flex size-9 shrink-0 items-center justify-center rounded-xl transition-colors duration-200", hasAttachment ? "text-violet-500 bg-violet-500/15" : "text-muted-foreground hover:bg-violet-500/10")} aria-label="Attach file">
   784→                <Paperclip className="size-4" />
   785→              </m.button>
   786→            )}
   787→
   788→            {/* Emoji button with picker */}
   789→            {!showVoiceWaveform && (
   790→              <EmojiPicker
   791→                onSelect={(emoji) => {
   792→                  const ta = inputRef.current;
   793→                  if (!ta) return;

   795→                  const end = ta.selectionEnd ?? draft.length;

   811→
   812→            {/* Send button (shown when there's text or attachment) */}
   813→            {!showVoiceWaveform && canSend && (
   814→              <AnimatePresence>

   822→
   823→            {/* Mic button (shown when input is empty and no attachment) */}
   824→            {!showVoiceWaveform && !canSend && (
   825→              <AnimatePresence>
   826→                <m.div initial={{ scale: 0, opacity: 0 }} animate={{ scale: 1, opacity: 1 }} exit={{ scale: 0, opacity: 0 }} transition={{ duration: 0.15, type: "spring", stiffness: 500, damping: 30 }}>
   827→                  <m.button type="button" onClick={() => voice.startRecording()} whileTap={{ scale: 0.9 }} className="flex size-9 shrink-0 items-center justify-center rounded-xl text-muted-foreground transition-colors duration-200 hover:bg-violet-500/10" aria-label="Record voice">
   828→                    <Mic className="size-4" />

   833→
   834→            {/* Cancel recording button */}
   835→            {showVoiceWaveform && (
   836→              <m.button type="button" onClick={async () => { await voice.stopRecording(); setShowVoiceWaveform(false); }} whileTap={{ scale: 0.9 }} className="flex size-10 shrink-0 items-center justify-center rounded-xl text-red-500 hover:bg-red-500/10" aria-label="Cancel recording">
   837→                <MicOff className="size-4" />
   838→              </m.button>
   839→            )}
   840→          </form>
   841→        </div>
   842→      </div>
   843→
   844→      {/* Partner Info Panel */}
   845→      <PartnerInfoPanel
   846→        partner={partner}
   847→        partnerStatus={partnerStatus}
   848→        open={infoPanelOpen}
   849→        onClose={() => setInfoPanelOpen(false)}
   850→      />
   851→    </div>
   852→  );
