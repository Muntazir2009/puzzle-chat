    55→    55→
    56→    56→function useDoubleTap(
    57→    57→  onDoubleTap: () => void,
    58→    58→  delay = 300,
    59→    59→) {
    60→    60→  const lastTapRef = useRef(0);
    61→    61→  return useCallback((): boolean => {
    62→    62→    const now = Date.now();
    63→    63→    if (now - lastTapRef.current < delay) {
    64→    64→      lastTapRef.current = 0;
    65→    65→      onDoubleTap();
    66→    66→      return true; // is double tap
    67→    67→    }
    68→    68→    lastTapRef.current = now;
    69→    69→    return false; // is single tap
    70→    70→  }, [onDoubleTap, delay]);
    71→    71→}

   145→   145→/* ------------------------------------------------------------------ */
   146→   146→
   147→   147→function EphemeralTimer({
   148→   148→  seconds,
   149→   149→  onExpire,
   150→   150→}: {
   151→   151→  seconds: number;
   152→   152→  onExpire: () => void;
   153→   153→}) {
   154→   154→  const [remaining, setRemaining] = useState(seconds);
   155→   155→  const intervalRef = useRef<ReturnType<typeof setInterval> | undefined>(
   156→   156→    undefined,
   157→   157→  );
   158→   158→
   159→   159→  useEffect(() => {
   160→   160→    intervalRef.current = setInterval(() => {
   161→   161→      setRemaining((r) => {
   162→   162→        if (r <= 1) {
   163→   163→          clearInterval(intervalRef.current);
   164→   164→          onExpire();
   165→   165→          return 0;
   166→   166→        }
   167→   167→        return r - 1;
   168→   168→      });
   169→   169→    }, 1000);
   170→   170→    return () => clearInterval(intervalRef.current);
   171→   171→  }, [seconds, onExpire]);
   172→   172→
   173→   173→  const pct = remaining / seconds;
   174→   174→  const circ = 2 * Math.PI * 7;
   175→   175→  const offset = circ * (1 - pct);
   176→   176→
   177→   177→  return (
   178→   178→    <svg
   179→   179→      className="size-3.5 shrink-0 ephemeral-timer-pulse"
   180→   180→      viewBox="0 0 18 18"
   181→   181→      aria-label={`${remaining}s remaining`}
   182→   182→    >
   183→   183→      <circle
   184→   184→        cx="9"
   185→   185→        cy="9"
   186→   186→        r="7"
   187→   187→        fill="none"
   188→   188→        stroke="currentColor"
   189→   189→        strokeWidth="1.5"
   190→   190→        className="text-white/20"
   191→   191→      />
   192→   192→      <circle
   193→   193→        cx="9"
   194→   194→        cy="9"
   195→   195→        r="7"
   196→   196→        fill="none"
   197→   197→        stroke="currentColor"
   198→   198→        strokeWidth="1.5"
   199→   199→        strokeDasharray={circ}
   200→   200→        strokeDashoffset={offset}
   201→   201→        strokeLinecap="round"
   202→   202→        style={{
   203→   203→          color: "var(--app-accent-lighter)",
   204→   204→          transform: "rotate(-90deg)",
   205→   205→          transformOrigin: "50% 50%",
   206→   206→          transition: "stroke-dashoffset 1s linear",
   207→   207→        }}
   208→   208→      />
   209→   209→    </svg>
   210→   210→  );
   211→   211→}

   422→   422→/* ------------------------------------------------------------------ */
   423→   423→
   424→   424→function VoiceBubble({
   425→   425→  message,
   426→   426→  isOwn,
   427→   427→  waveform,
   428→   428→}: {
   429→   429→  message: ChatMessage;
   430→   430→  isOwn: boolean;
   431→   431→  waveform: number[];
   432→   432→}) {
   433→   433→  const audioRef = useRef<HTMLAudioElement>(null);
   434→   434→  const [playing, setPlaying] = useState(false);
   435→   435→  const [progress, setProgress] = useState(0);
   436→   436→  const [speed, setSpeed] = useState(1);
   437→   437→  const speeds = [1, 1.5, 2];
   438→   438→  const duration = message.voice_duration ?? 0;
   439→   439→
   440→   440→  useEffect(() => {
   441→   441→    const audio = audioRef.current;
   442→   442→    if (!audio) return;
   443→   443→    const onTime = () => {
   444→   444→      if (audio.duration) setProgress(audio.currentTime / audio.duration);
   445→   445→    };
   446→   446→    const onEnd = () => setPlaying(false);
   447→   447→    audio.addEventListener("timeupdate", onTime);
   448→   448→    audio.addEventListener("ended", onEnd);
   449→   449→    return () => {
   450→   450→      audio.removeEventListener("timeupdate", onTime);
   451→   451→      audio.removeEventListener("ended", onEnd);
   452→   452→    };
   453→   453→  }, []);
   454→   454→
   455→   455→  const toggle = useCallback(() => {
   456→   456→    const audio = audioRef.current;
   457→   457→    if (!audio) return;
   458→   458→    if (playing) {
   459→   459→      audio.pause();
   460→   460→      setPlaying(false);
   461→   461→    } else {
   462→   462→      audio.playbackRate = speed;
   463→   463→      audio.play().catch(() => {});
   464→   464→      setPlaying(true);
   465→   465→    }
   466→   466→  }, [playing, speed]);
   467→   467→
   468→   468→  const cycleSpeed = useCallback(() => {
   469→   469→    setSpeed((s) => {
   470→   470→      const idx = speeds.indexOf(s);
   471→   471→      const next = speeds[(idx + 1) % speeds.length];
   472→   472→      if (audioRef.current) audioRef.current.playbackRate = next;
   473→   473→      return next;
   474→   474→    });
   475→   475→  }, []);
   476→   476→
   477→   477→  /* Build waveform bars */
   478→   478→  const bars = useMemo(() => {
   479→   479→    const count = Math.min(waveform.length, 40);
   480→   480→    const step = Math.max(1, Math.floor(waveform.length / count));
   481→   481→    return Array.from({ length: count }, (_, i) => waveform[i * step] ?? 0.1);
   482→   482→  }, [waveform]);
   483→   483→
   484→   484→  const progressIdx = Math.floor(progress * bars.length);
   485→   485→
   486→   486→  return (
   487→   487→    <div className={cn("flex items-center gap-3", isOwn ? "flex-row-reverse" : "")}>
   488→   488→      <button
   489→   489→        type="button"
   490→   490→        onClick={toggle}
   491→   491→        className={cn(
   492→   492→          "flex size-9 shrink-0 items-center justify-center rounded-full transition-all duration-150 active:scale-95",
   493→   493→          isOwn
   494→   494→            ? "bg-white/20 hover:bg-white/30"
   495→   495→            : "bg-zinc-600 hover:bg-[var(--app-accent-subtle)]",

   548→   548→/*  LinkifiedText – detects & renders clickable URLs                   */
   549→   549→/* ------------------------------------------------------------------ */
   550→   550→
   551→   551→const URL_REGEX = /(https?:\/\/[^\s<]+)/g;
   552→   552→
   553→   553→type TextSegment =
   554→   554→  | { type: "text"; value: string }
   555→   555→  | { type: "link"; value: string };
   556→   556→
   557→   557→function parseUrls(text: string): TextSegment[] {
   558→   558→  const segments: TextSegment[] = [];
   559→   559→  let lastIndex = 0;
   560→   560→  let match: RegExpExecArray | null;
   561→   561→  while ((match = URL_REGEX.exec(text)) !== null) {
   562→   562→    if (match.index > lastIndex) {
   563→   563→      segments.push({ type: "text", value: text.slice(lastIndex, match.index) });
   564→   564→    }
   565→   565→    segments.push({ type: "link", value: match[1] });
   566→   566→    lastIndex = URL_REGEX.lastIndex;
   567→   567→  }
   568→   568→  if (lastIndex < text.length) {
   569→   569→    segments.push({ type: "text", value: text.slice(lastIndex) });
   570→   570→  }
   571→   571→  return segments;
   572→   572→}

   574→   574→/** Split a plain-text string into highlighted / non-highlighted fragments */
   575→   575→function highlightText(text: string, query: string): React.ReactNode[] {
   576→   576→  if (!query) return [text];
   577→   577→  const escaped = query.replace(/[.*+?^${}()|[\]\\]/g, "\\");
   578→   578→  const re = new RegExp(`(${escaped})`, "gi");
   579→   579→  const parts: React.ReactNode[] = [];
   580→   580→  let lastIndex = 0;
   581→   581→  let match: RegExpExecArray | null;
   582→   582→  while ((match = re.exec(text)) !== null) {
   583→   583→    if (match.index > lastIndex) {
   584→   584→      parts.push(text.slice(lastIndex, match.index));
   585→   585→    }
   586→   586→    parts.push(
   587→   587→      <mark
   588→   588→        key={match.index}
   589→   589→        className="bg-yellow-300/60 dark:bg-yellow-400/40 rounded-sm px-0.5"
   590→   590→      >
   591→   591→        {match[1]}
   592→   592→      </mark>,
   593→   593→    );
   594→   594→    lastIndex = re.lastIndex;
   595→   595→  }
   596→   596→  if (lastIndex < text.length) {
   597→   597→    parts.push(text.slice(lastIndex));
   598→   598→  }
   599→   599→  return parts;
   600→   600→}
