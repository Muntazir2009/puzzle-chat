    83→/*  Ephemeral Timer (circular SVG countdown)                           */
    84→/* ------------------------------------------------------------------ */
    85→
    86→function EphemeralTimer({ seconds, onExpire }: { seconds: number; onExpire: () => void }) {
    87→  const [remaining, setRemaining] = useState(seconds);
    88→  const intervalRef = useRef<ReturnType<typeof setInterval> | undefined>(undefined);
    89→
    90→  useEffect(() => {
    91→    intervalRef.current = setInterval(() => {
    92→      setRemaining((r) => {
    93→        if (r <= 1) { clearInterval(intervalRef.current); onExpire(); return 0; }
    94→        return r - 1;
    95→      });
    96→    }, 1000);
    97→    return () => clearInterval(intervalRef.current);
    98→  }, [seconds, onExpire]);
    99→
   100→  const pct = remaining / seconds;
   101→  const circ = 2 * Math.PI * 7;
   102→  const offset = circ * (1 - pct);
   103→
   104→  return (
   105→    <svg className="size-4 shrink-0 ephemeral-timer-pulse" viewBox="0 0 18 18" aria-label={`${remaining}s remaining`}>
   106→      <circle cx="9" cy="9" r="7" fill="none" stroke="currentColor" strokeWidth="1.5" className="text-white/20" />
   107→      <circle cx="9" cy="9" r="7" fill="none" stroke="currentColor" strokeWidth="1.5" strokeDasharray={circ} strokeDashoffset={offset} strokeLinecap="round" className="text-violet-300" style={{ transform: "rotate(-90deg)", transformOrigin: "50% 50%", transition: "stroke-dashoffset 1s linear" }} />
   108→    </svg>
   109→  );
   110→}

   153→function VoiceBubble({ message, isOwn, waveform }: {
   154→  message: ChatMessage; isOwn: boolean; waveform: number[];
   155→}) {
   156→  const audioRef = useRef<HTMLAudioElement>(null);
   157→  const [playing, setPlaying] = useState(false);
   158→  const [progress, setProgress] = useState(0);
   159→  const [speed, setSpeed] = useState(1);
   160→  const speeds = [1, 1.5, 2];
   161→  const duration = message.voice_duration ?? 0;
   162→
   163→  useEffect(() => {
   164→    const audio = audioRef.current;
   165→    if (!audio) return;
   166→    const onTime = () => { if (audio.duration) setProgress(audio.currentTime / audio.duration); };
   167→    const onEnd = () => setPlaying(false);
   168→    audio.addEventListener("timeupdate", onTime);
   169→    audio.addEventListener("ended", onEnd);
   170→    return () => { audio.removeEventListener("timeupdate", onTime); audio.removeEventListener("ended", onEnd); };
   171→  }, []);
   172→
   173→  const toggle = useCallback(() => {
   174→    const audio = audioRef.current;
   175→    if (!audio) return;
   176→    if (playing) { audio.pause(); setPlaying(false); }
   177→    else { audio.playbackRate = speed; audio.play().catch(() => {}); setPlaying(true); }
   178→  }, [playing, speed]);
   179→
   180→  const cycleSpeed = useCallback(() => {
   181→    setSpeed((s) => {
   182→      const idx = speeds.indexOf(s);
   183→      const next = speeds[(idx + 1) % speeds.length];
   184→      if (audioRef.current) audioRef.current.playbackRate = next;
   185→      return next;
   186→    });
   187→  }, []);
   188→
   189→  /* Build waveform bars */
   190→  const bars = useMemo(() => {
   191→    const count = Math.min(waveform.length, 40);
   192→    const step = Math.max(1, Math.floor(waveform.length / count));
   193→    return Array.from({ length: count }, (_, i) => waveform[i * step] ?? 0.1);
   194→  }, [waveform]);
   195→
   196→  const barColor = isOwn ? "bg-white/50" : "bg-zinc-400/70";
   197→  const progressColor = isOwn ? "bg-white" : "bg-violet-400";
   198→  const progressIdx = Math.floor(progress * bars.length);
   199→
   200→  return (
   201→    <div className={cn("flex items-center gap-3", isOwn ? "flex-row-reverse" : "")}>
   202→      <button type="button" onClick={toggle} className={cn("flex size-9 shrink-0 items-center justify-center rounded-full transition-all duration-150", isOwn ? "bg-white/20 hover:bg-white/30 hover:scale-105" : "bg-zinc-600 hover:bg-zinc-500 hover:scale-105")} aria-label={playing ? "Pause" : "Play"}>
   203→        {playing ? <Pause className="size-3.5 text-white" /> : <Play className="size-3.5 text-white ml-0.5" />}
   204→      </button>
   205→      <div className="flex items-center gap-[2px]" style={{ height: 32 }}>
   206→        {bars.map((amp, i) => (
   207→          <div key={i} className={cn("w-[2.5px] rounded-full transition-all duration-100", i < progressIdx ? progressColor : barColor)} style={{ height: `${Math.max(4, amp * 32)}px` }} />
   208→        ))}
   209→      </div>
   210→      <button type="button" onClick={cycleSpeed} className={cn("shrink-0 text-[11px] font-semibold tabular-nums transition-colors hover:opacity-80", isOwn ? "text-white/70" : "text-zinc-400")}>{speed}x</button>
   211→      <span className={cn("shrink-0 text-[11px] tabular-nums", isOwn ? "text-white/50" : "text-zinc-500")}>{Math.floor(duration / 60)}:{String(duration % 60).padStart(2, "0")}</span>
   212→      <audio ref={audioRef} src={message.content} preload="metadata" />
   213→    </div>
   214→  );
   215→}

   269→function MessageBubble({ message, isOwn, partnerName, partnerAvatar, onReplyTo, onReact }: {
   270→  message: ChatMessage; isOwn: boolean; partnerName: string; partnerAvatar: string | null;
   271→  onReplyTo?: (msg: ChatMessage) => void; onReact?: (id: string, emoji: string, add: boolean) => void;
   272→}) {
   273→  const [showHeart, setShowHeart] = useState(false);
   274→  const [showTapback, setShowTapback] = useState(false);
   275→  const lastTapRef = useRef(0);
   276→  const longPressRef = useRef<ReturnType<typeof setTimeout> | undefined>(undefined);
   277→  const bubbleRef = useRef<HTMLDivElement>(null);
   278→
   279→  const triggerHeart = useCallback(() => {
   280→    setShowHeart(true); setTimeout(() => setShowHeart(false), 800);
   281→    if (typeof navigator !== "undefined" && "vibrate" in navigator) navigator.vibrate(10);
   282→  }, []);
   283→
   284→  const handleClick = useCallback(() => {
   285→    clearTimeout(longPressRef.current);
   286→    const now = Date.now();
   287→    if (now - lastTapRef.current < DOUBLE_TAP_MS) triggerHeart();
   288→    lastTapRef.current = now;
   289→  }, [triggerHeart]);
   290→
   291→  const handlePressStart = useCallback(() => {
   292→    longPressRef.current = setTimeout(() => {
   293→      if (typeof navigator !== "undefined" && "vibrate" in navigator) navigator.vibrate(20);
   294→      setShowTapback(true);
   295→    }, LONG_PRESS_MS);
   296→  }, []);
   297→
   298→  const handlePressEnd = useCallback(() => { clearTimeout(longPressRef.current); }, []);
   299→
   300→  const handleContextMenu = useCallback((e: React.MouseEvent) => {
   301→    e.preventDefault(); setShowTapback(true);
   302→  }, []);
   303→
   304→  const handleSwipeEnd = useCallback((_: MouseEvent | TouchEvent | PointerEvent, info: PanInfo) => {
   305→    if (info.offset.x > SWIPE_REPLY_THRESHOLD && info.velocity.x > 200) {
   306→      if (typeof navigator !== "undefined" && "vibrate" in navigator) navigator.vibrate(10);
   307→      onReplyTo?.(message);
   308→    }
   309→  }, [message, onReplyTo]);
   310→
   311→  const isFailed = message.status === "failed";
   312→  const isVoice = message.type === "voice";
   313→  const isNew = message.status === "sending" || Date.now() - new Date(message.created_at).getTime() < 2000;
   314→
   315→  return (
   316→    <m.div
   317→      className={cn("relative flex w-full gap-2.5 px-4", isOwn ? "justify-end" : "justify-start")}
   318→      initial={isNew ? { opacity: 0, y: 16, scale: 0.97 } : false}
   319→      animate={{ opacity: 1, y: 0, scale: 1 }}
   320→      transition={{ duration: 0.3, ease: [0.22, 1, 0.36, 1] }}
   321→    >
   322→      <HeartBurst show={showHeart} />
   323→      <AnimatePresence>{showTapback && onReact && (
   324→        <TapbackDock message={message} isOwn={isOwn} onReact={(emoji, add) => { onReact(message.id, emoji, add); setShowTapback(false); }} onReply={() => { onReplyTo?.(message); setShowTapback(false); }} />
   325→      )}</AnimatePresence>
   326→
   327→      {!isOwn && (
   328→        <Avatar className="mt-auto size-8 shrink-0 ring-1 ring-muted">
   329→          {partnerAvatar && <AvatarImage src={partnerAvatar} alt={partnerName} />}
   330→          <AvatarFallback className="bg-gradient-to-br from-violet-100 to-purple-200 text-xs font-medium text-violet-700 dark:from-violet-500 dark:to-purple-600 dark:text-white">{partnerName.slice(0, 2).toUpperCase()}</AvatarFallback>
   331→        </Avatar>
   332→      )}
   333→
   334→      <div className={cn("flex max-w-[75%] flex-col gap-1", isOwn ? "items-end" : "items-start")}>
   335→        <m.div
   336→          ref={bubbleRef}
   337→          drag="x"
   338→          dragConstraints={{ left: 0, right: 0 }}
   339→          dragElastic={{ left: 0, right: 0.35 }}
   340→          onDragEnd={handleSwipeEnd}
   341→          onClick={handleClick}
   342→          onPointerDown={handlePressStart}
   343→          onPointerUp={handlePressEnd}
   344→          onPointerLeave={handlePressEnd}
   345→          onContextMenu={handleContextMenu}
   346→          className={cn(
   347→            "relative cursor-default select-none rounded-2xl px-4 py-2.5 text-sm leading-relaxed transition-transform duration-100 active:scale-[0.98]",
   348→            // Own messages: vibrant violet-purple gradient with subtle shadow
   349→            isOwn && "bg-gradient-to-r from-violet-500 to-purple-600 text-white rounded-br-md shadow-md shadow-violet-500/15",
   350→            // Partner messages: softer background with shadow
   351→            !isOwn && "bg-muted text-foreground rounded-bl-md shadow-sm dark:bg-zinc-800 dark:text-zinc-100 dark:shadow-violet-950/20",
   352→            isFailed && "ring-2 ring-red-400/50",
   353→            isVoice && "py-2"
   354→          )}
   355→          style={{ transform: "translate3d(0,0,0)" }}
   356→        >
   357→          {/* Reply quote */}
   358→          {message.reply_to_id && message.reply_to_content && (
   359→            <ReplyPreview senderName={message.reply_to_sender_name ?? ""} content={message.reply_to_content} isOwn={isOwn} />
   360→          )}
   361→
   362→          {/* Content */}
   363→          {isVoice ? (
   364→            message.waveform_data ? (
   365→              <VoiceBubble message={message} isOwn={isOwn} waveform={message.waveform_data} />
   366→            ) : <p className="text-white/60 italic">Voice message</p>
   367→          ) : message.type === "image" ? (
   368→            <img src={message.content} alt="Shared image" className="max-h-64 rounded-lg object-contain" />
   369→          ) : (
   370→            <p className="whitespace-pre-wrap break-words">{message.content}</p>
   371→          )}
   372→
   373→          {/* Vanish indicator */}
   374→          {message.vanish_mode && (
   375→            <div className="mt-1 flex items-center gap-1 text-[10px] text-white/50"><EyeOff className="size-3" /><span>View once</span></div>
   376→          )}
   377→        </m.div>
   378→
   379→        {/* Timestamp + receipt + ephemeral timer */}
   380→        <div className={cn("flex items-center gap-1.5 px-1 text-[11px] text-muted-foreground", isOwn && "flex-row-reverse")}>
   381→          {message.ephemeral_seconds && message.ephemeral_seconds > 0 && (
   382→            <EphemeralTimer seconds={message.ephemeral_seconds} onExpire={() => { /* trigger vanish */ }} />
   383→          )}
   384→          <span>{formatDistanceToNow(new Date(message.created_at), { addSuffix: true })}</span>
   385→          {isOwn && <ReceiptIcon status={message.status} />}
   386→        </div>
   387→
   388→        {/* Reactions */}
   389→        {onReact && <ReactionPills reactions={message.reactions} isOwn={isOwn} onReact={(e, a) => onReact(message.id, e, a)} currentUserId={""} />}
   390→
   391→        {isFailed && <span className="px-1 text-[10px] text-red-500">Failed to send. Retry.</span>}
   392→      </div>
   393→
   394→      {isOwn && <div className="w-8 shrink-0" />}
   395→    </m.div>
   396→  );
   397→}

   403→function formatDateSeparator(dateStr: string): string {
   404→  const d = new Date(dateStr);
   405→  const now = new Date();
   406→  const today = new Date(now.getFullYear(), now.getMonth(), now.getDate());
   407→  const yesterday = new Date(today);
   408→  yesterday.setDate(yesterday.getDate() - 1);
   409→  const msgDay = new Date(d.getFullYear(), d.getMonth(), d.getDate());
   410→
   411→  if (msgDay.getTime() === today.getTime()) return "Today";
   412→  if (msgDay.getTime() === yesterday.getTime()) return "Yesterday";
   413→  return d.toLocaleDateString(undefined, { month: "short", day: "numeric" });
   414→}

   512→export function MessageFeed({ messages, isLoading, isPartnerTyping, currentUserId, partnerName, partnerAvatar, onMarkAsRead, onVanishMessage, onReplyTo, onReact, scrollToMessageId, onScrolledToMessage }: MessageFeedProps) {
   513→  const parentRef = useRef<HTMLDivElement>(null);
   514→  const isAutoScrollRef = useRef(true);
   515→  const [showNewBtn, setShowNewBtn] = useState(false);
   516→  const unreadRef = useRef(0);
   517→  const prevCountRef = useRef(messages.length);
   518→
   519→  const totalItems = useMemo(() => messages.length + (isPartnerTyping ? 1 : 0), [messages.length, isPartnerTyping]);
   520→
   521→  const virtualizer = useVirtualizer({ count: totalItems, getScrollElement: () => parentRef.current, estimateSize: () => ESTIMATED_ROW_HEIGHT, overscan: OVERSCAN, paddingStart: 16, paddingEnd: 16, gap: 12 });
   522→
   523→  const scrollToBottom = useCallback((behavior: ScrollBehavior = "smooth") => {
   524→    const el = parentRef.current; if (!el) return;
   525→    el.scrollTo({ top: el.scrollHeight, behavior });
   526→    isAutoScrollRef.current = true; setShowNewBtn(false); unreadRef.current = 0;
   527→  }, []);
   528→
   529→  const handleScroll = useCallback((e: UIEvent<HTMLDivElement>) => {
   530→    const el = e.currentTarget;
   531→    const dist = el.scrollHeight - el.scrollTop - el.clientHeight;
   532→    if (dist <= AUTO_SCROLL_THRESHOLD) { isAutoScrollRef.current = true; setShowNewBtn(false); unreadRef.current = 0; }
   533→    else { isAutoScrollRef.current = false; }
   534→  }, []);
   535→
   536→  useEffect(() => { if (messages.length > 0) requestAnimationFrame(() => scrollToBottom("instant")); }, [isLoading, scrollToBottom]);
   537→
   538→  useEffect(() => { if (!onMarkAsRead) return; const unread = messages.filter((m) => m.sender_id !== currentUserId && m.status !== "read"); if (unread.length > 0 && isAutoScrollRef.current) onMarkAsRead(unread.map((m) => m.id)); }, [messages, currentUserId, onMarkAsRead]);
   539→
   540→  /* ---- Scroll to a specific message ----------------------------------- */
   541→  useEffect(() => {
   542→    if (!scrollToMessageId) return;
   543→    const idx = messages.findIndex((m) => m.id === scrollToMessageId);
   544→    if (idx === -1) { onScrolledToMessage?.(); return; }
   545→    virtualizer.scrollToIndex(idx, { align: "center" });
   546→    onScrolledToMessage?.();
   547→  }, [scrollToMessageId, messages, virtualizer, onScrolledToMessage]);
   548→
   549→  useEffect(() => {
   550→    const prev = prevCountRef.current; const next = messages.length; prevCountRef.current = next;
   551→    if (next <= prev) return;
   552→    const last = messages[next - 1]; const own = last && last.sender_id === currentUserId;
   553→    if (own) { requestAnimationFrame(() => scrollToBottom("instant")); return; }
   554→    if (isAutoScrollRef.current) requestAnimationFrame(() => scrollToBottom("smooth"));
   555→    else { unreadRef.current += next - prev; setShowNewBtn(true); }
   556→  }, [messages.length, currentUserId, scrollToBottom]);
   557→
   558→  if (isLoading) return <MessageListSkeleton />;
   559→  if (messages.length === 0 && !isPartnerTyping) return <EmptyState partnerName={partnerName} partnerAvatar={partnerAvatar} />;
   560→
   561→  const virtualItems = virtualizer.getVirtualItems();
   562→
   563→  return (
   564→    <div className="relative h-full w-full overflow-hidden">
   565→      <div ref={parentRef} onScroll={handleScroll} className="h-full w-full overflow-y-auto overscroll-contain" style={{ WebkitOverflowScrolling: "touch" }}>
   566→        <div style={{ height: `${virtualizer.getTotalSize()}px`, width: "100%", position: "relative" }}>
   567→          {virtualItems.map((vi) => {
   568→            const isTyping = vi.index === messages.length;
   569→            const msg = isTyping ? null : messages[vi.index];
   570→            // Check if we need a date separator before this message
   571→            const showDateSep = msg && vi.index > 0 && messages[vi.index - 1] && (() => {
   572→              const prev = new Date(messages[vi.index - 1].created_at);
   573→              const curr = new Date(msg.created_at);
   574→              return prev.toDateString() !== curr.toDateString();
   575→            })();
   576→            const showFirstDateSep = msg && vi.index === 0;
   577→            return (
   578→              <div key={vi.key} data-index={vi.index} ref={virtualizer.measureElement} style={{ position: "absolute", top: 0, left: 0, width: "100%", transform: `translateY(${vi.start}px)` }}>
   579→                {isTyping ? <TypingIndicator partnerName={partnerName} /> : msg ? (
   580→                  <>
   581→                    {(showDateSep || showFirstDateSep) && <DateSeparator date={msg.created_at} />}
   582→                    <MessageBubble message={msg} isOwn={msg.sender_id === currentUserId} partnerName={partnerName} partnerAvatar={partnerAvatar} onReplyTo={onReplyTo} onReact={onReact} />
   583→                  </>
   584→                ) : null}
   585→              </div>
   586→            );
   587→          })}
   588→        </div>
   589→      </div>
   590→      {showNewBtn && unreadRef.current > 0 && <NewMessagesButton count={unreadRef.current} onClick={() => scrollToBottom("smooth")} />}
   591→    </div>
   592→  );
   593→}
