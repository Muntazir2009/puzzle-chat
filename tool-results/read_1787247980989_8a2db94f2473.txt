    38→    38→function formatFileSize(bytes: number): string {
    39→    39→  if (bytes < 1024) return `${bytes} B`;
    40→    40→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
    41→    41→  return `${(bytes / (1024 * 1024)).toFixed(1)} MB`;
    42→    42→}
    43→    43→
    44→    44→function formatLastSeen(lastSeen: string | null): string {
    45→    45→  if (!lastSeen) return "Offline";
    46→    46→  const diffMs = Date.now() - new Date(lastSeen).getTime();
    47→    47→  const diffSec = Math.floor(diffMs / 1000);
    48→    48→  if (diffSec < 60) return "Seen just now";
    49→    49→  const diffMin = Math.floor(diffSec / 60);
    50→    50→  if (diffMin < 60) return `Seen ${diffMin}m ago`;
    51→    51→  const diffHr = Math.floor(diffMin / 60);
    52→    52→  if (diffHr < 24) return `Seen ${diffHr}h ago`;
    53→    53→  const diffDay = Math.floor(diffHr / 24);
    54→    54→  if (diffDay < 7) return `Seen ${diffDay}d ago`;
    55→    55→  return `Seen ${new Date(lastSeen).toLocaleDateString()}`;
    56→    56→}
