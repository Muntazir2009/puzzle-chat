    37→function formatLastSeen(lastSeen: string | null): string {
    38→  if (!lastSeen) return "Offline";
    39→  const diffMs = Date.now() - new Date(lastSeen).getTime();
    40→  const diffSec = Math.floor(diffMs / 1000);
    41→  if (diffSec < 60) return "Seen just now";
    42→  const diffMin = Math.floor(diffSec / 60);
    43→  if (diffMin < 60) return `Seen ${diffMin}m ago`;
    44→  const diffHr = Math.floor(diffMin / 60);
    45→  if (diffHr < 24) return `Seen ${diffHr}h ago`;
    46→  const diffDay = Math.floor(diffHr / 24);
    47→  if (diffDay < 7) return `Seen ${diffDay}d ago`;
    48→  return `Seen ${new Date(lastSeen).toLocaleDateString()}`;
    49→}

    67→function SearchPanel({
    68→  conversationId,
    69→  onResultClick,
    70→  onClose,
    71→}: {
    72→  conversationId: string;
    73→  onResultClick: (messageId: string) => void;
    74→  onClose: () => void;
    75→}) {
    76→  const [query, setQuery] = useState("");
    77→  const [results, setResults] = useState<ChatMessage[]>([]);
    78→  const [isSearching, setIsSearching] = useState(false);
    79→  const inputRef = useRef<HTMLInputElement>(null);
    80→  const debounceRef = useRef<ReturnType<typeof setTimeout> | undefined>(undefined);
    81→
    82→  /* Auto-focus on open */
    83→  useEffect(() => {
    84→    requestAnimationFrame(() => inputRef.current?.focus());
    85→  }, []);
    86→
    87→  /* Escape key to close */
    88→  useEffect(() => {
    89→    function handleKeyDown(e: KeyboardEvent) {
    90→      if (e.key === "Escape") onClose();
    91→    }
    92→    window.addEventListener("keydown", handleKeyDown);
    93→    return () => window.removeEventListener("keydown", handleKeyDown);
    94→  }, [onClose]);
    95→
    96→  /* Debounced search */
    97→  useEffect(() => {
    98→    if (debounceRef.current) clearTimeout(debounceRef.current);
    99→    if (!query.trim()) { setResults([]); setIsSearching(false); return; }
   100→    setIsSearching(true);
   101→    debounceRef.current = setTimeout(async () => {
   102→      try {
   103→        const res = await fetch(`/api/messages/search?conversation_id=${conversationId}&q=${encodeURIComponent(query)}`);
   104→        if (res.ok) {
   105→          const data: ChatMessage[] = await res.json();
   106→          setResults(data);
   107→        }
   108→      } catch (_err) {
   109→        /* silent */
   110→      } finally {
   111→        setIsSearching(false);
   112→      }
   113→    }, 300);
   114→    return () => { if (debounceRef.current) clearTimeout(debounceRef.current); };
   115→  }, [query, conversationId]);
   116→
   117→  return (
   118→    <div
   119→      className="overflow-hidden border-b bg-background/80 backdrop-blur-xl"
   120→    >
   121→      <div className="flex items-center gap-2 px-4 py-2.5">
   122→        <Search className="size-4 shrink-0 text-muted-foreground" />
   123→        <input
   124→          ref={inputRef}
   125→          type="text"
   126→          value={query}
   127→          onChange={(e) => setQuery(e.target.value)}
   128→          placeholder="Search messages..."
   129→          className="flex-1 bg-transparent text-sm outline-none placeholder:text-muted-foreground"
   130→          aria-label="Search messages"
   131→        />
   132→        {isSearching ? (
   133→          <Loader2 className="size-4 shrink-0 animate-spin text-muted-foreground" />
   134→        ) : (
   135→          <button
   136→            type="button"
   137→            onClick={() => { setQuery(""); onClose(); }}
   138→            className="flex size-6 items-center justify-center rounded-full text-muted-foreground transition-colors hover:bg-muted"
   139→            aria-label="Close search"
   140→          >
   141→            <X className="size-3.5" />
   142→          </button>
   143→        )}
   144→      </div>
   145→
   146→      {/* Results list */}
   147→      {results.length > 0 && (
   148→        <div className="max-h-[200px] overflow-y-auto border-t px-2 pb-2">
   149→          {results.map((msg) => (
   150→            <button
   151→              key={msg.id}
   152→              type="button"
   153→              onClick={() => onResultClick(msg.id)}
   154→              className="flex w-full flex-col gap-0.5 rounded-lg px-3 py-2.5 text-left transition-colors hover:bg-muted/50"
   155→            >
   156→              <p className="truncate text-sm text-foreground">
   157→                {msg.content.length > 60 ? msg.content.slice(0, 60) + "..." : msg.content}
   158→              </p>
   159→              <span className="text-[11px] text-muted-foreground">
   160→                {formatDistanceToNow(new Date(msg.created_at), { addSuffix: true })}
   161→              </span>
   162→            </button>
   163→          ))}
   164→        </div>
   165→      )}
   166→
   167→      {query.trim() && !isSearching && results.length === 0 && (
   168→        <div className="border-t px-4 py-3 text-center text-xs text-muted-foreground">
   169→          No results found
   170→        </div>
   171→      )}
   172→    </div>
   173→  );
   174→}

   334→export function ChatLayout({ currentUserId, otherUserId, conversationId, partner, initialMessages }: ChatLayoutProps) {
   335→  const { messages, isLoading, isPartnerTyping, partnerStatus, sendMessage, onTyping, markAsRead, vanishMessage, sendReaction } =
   336→    useChat({ currentUserId, otherUserId, conversationId, initialMessages });
   337→
   338→  const [draft, setDraft] = useState("");
   339→  const [isSending, setIsSending] = useState(false);
   340→  const [vanishMode, setVanishMode] = useState(false);
   341→  const [replyTo, setReplyTo] = useState<ChatMessage | null>(null);
   342→  const [ephemeralOpen, setEphemeralOpen] = useState(false);
   343→  const [ephemeralSeconds, setEphemeralSeconds] = useState<number | null>(null);
   344→  const [infoPanelOpen, setInfoPanelOpen] = useState(false);
   345→  const [searchOpen, setSearchOpen] = useState(false);
   346→  const [scrollToMessageId, setScrollToMessageId] = useState<string | null>(null);
   347→  const inputRef = useRef<HTMLTextAreaElement>(null);
   348→  const viewport = useVisualViewport();
   349→
   350→  /* Voice */
   351→  const voice = useVoiceRecorder();
   352→  const [showVoiceWaveform, setShowVoiceWaveform] = useState(false);
   353→
   354→  const containerStyle = viewport.height > 0 ? ({ height: `${viewport.height}px` } as React.CSSProperties) : ({ height: "100dvh" } as React.CSSProperties);
   355→  const kbOffset = viewport.isKeyboardVisible && viewport.offsetTop > 0 ? ({ marginTop: `-${viewport.offsetTop}px` } as React.CSSProperties) : undefined;
   356→
   357→  useEffect(() => { inputRef.current?.focus(); }, []);
   358→
   359→  /* Show voice waveform when recording */
   360→  useEffect(() => { setShowVoiceWaveform(voice.isRecording); }, [voice.isRecording]);
   361→
   362→  const handleSubmit = useCallback(async (e: FormEvent) => {
   363→    e.preventDefault();
   364→    const trimmed = draft.trim();
   365→    if (!trimmed || isSending) return;
   366→    setIsSending(true);
   367→    const opts: SendMessageOptions = { vanish_mode: vanishMode, ephemeral_seconds: ephemeralSeconds, reply_to_id: replyTo?.id ?? null };
   368→    setDraft(""); setReplyTo(null);
   369→    if (inputRef.current) inputRef.current.style.height = "auto";
   370→    try { await sendMessage(trimmed, opts); }
   371→    finally { setIsSending(false); requestAnimationFrame(() => inputRef.current?.focus()); }
   372→  }, [draft, isSending, sendMessage, vanishMode, ephemeralSeconds, replyTo]);
   373→
   374→  const handleVoiceSend = useCallback(async () => {
   375→    const result = await voice.stopRecording();
   376→    if (!result || result.duration < 1) return;
   377→    const blob = result.blob;
   378→    const url = URL.createObjectURL(blob);
   379→    await sendMessage(url, { type: "voice", voice_duration: result.duration, waveform_data: result.amplitudes, ephemeral_seconds: ephemeralSeconds });
   380→    URL.revokeObjectURL(url);
   381→  }, [voice, sendMessage, ephemeralSeconds]);
   382→
   383→  const handleKeyDown = useCallback((e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   384→    if (e.key === "Enter" && !e.shiftKey) { e.preventDefault(); const form = e.currentTarget.closest("form"); if (form) form.requestSubmit(); }
   385→  }, []);
   386→
   387→  const handleInputChange = useCallback((e: React.ChangeEvent<HTMLTextAreaElement>) => { setDraft(e.target.value); onTyping(); }, [onTyping]);
   388→  const handleInput = useCallback((e: React.FormEvent<HTMLTextAreaElement>) => { const el = e.currentTarget; el.style.height = "auto"; el.style.height = `${Math.min(el.scrollHeight, 128)}px`; }, []);
   389→
   390→  const hasText = draft.trim().length > 0;
   391→
   392→  /* Search result click → scroll to message */
   393→  const handleSearchResultClick = useCallback((messageId: string) => {
   394→    setScrollToMessageId(messageId);
   395→    setSearchOpen(false);
   396→  }, []);
   397→
   398→  const handleScrolledToMessage = useCallback(() => {
   399→    setScrollToMessageId(null);
   400→  }, []);
   401→
   402→  /* Status text for header */
   403→  const statusText = isPartnerTyping
   404→    ? "typing\u2026"
   405→    : partnerStatus.online
   406→      ? "Online"
   407→      : formatLastSeen(partnerStatus.last_seen);
   408→
   409→  /* ---- Render --------------------------------------------------- */
   410→
   411→  return (
   412→    <div className="flex w-full flex-col bg-background" style={containerStyle}>
   413→      <div style={kbOffset} className="flex min-h-0 flex-1 flex-col">
   414→        {/* Header - sticky with glassmorphic blur */}
   415→        <header className="sticky top-0 z-30 flex h-14 shrink-0 items-center gap-3 border-b border-border/50 bg-background/80 backdrop-blur-xl supports-[backdrop-filter]:bg-background/60 px-4">
   416→          <button
   417→            type="button"
   418→            onClick={() => setInfoPanelOpen(true)}
   419→            className="flex items-center gap-3 rounded-lg p-1 -ml-1 transition-colors hover:bg-muted/50 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-violet-400/50"
   420→            aria-label="Open user info"
   421→          >
   422→            <Avatar className="size-9 ring-2 ring-violet-100 dark:ring-violet-900/50">
   423→              {partner.avatar_url && <AvatarImage src={partner.avatar_url} alt={partner.name} />}
   424→              <AvatarFallback className="bg-gradient-to-br from-violet-500 to-purple-600 text-xs font-semibold text-white">{partner.name.slice(0, 2).toUpperCase()}</AvatarFallback>
   425→            </Avatar>
   426→            <div className="flex flex-col">
   427→              <span className="text-sm font-semibold leading-tight">{partner.name}</span>
   428→              <div className="flex items-center gap-1.5">
   429→                <StatusDot online={partnerStatus.online} />
   430→                <span className={cn(
   431→                  "text-xs",
   432→                  isPartnerTyping ? "text-violet-500" : "text-muted-foreground",
   433→                )}>{statusText}</span>
   434→              </div>
   435→            </div>
   436→          </button>
   437→
   438→          {/* Search icon button - right side of header */}
   439→          <div className="ml-auto">
   440→            <button
   441→              type="button"
   442→              onClick={() => setSearchOpen((v) => !v)}
   443→              className={cn(
   444→                "flex size-9 items-center justify-center rounded-lg transition-colors",
   445→                searchOpen
   446→                  ? "bg-violet-500/15 text-violet-500"
   447→                  : "text-muted-foreground hover:bg-muted",
   448→              )}
   449→              aria-label={searchOpen ? "Close search" : "Search messages"}
   450→            >
   451→              <Search className="size-4" />
   452→            </button>
   453→          </div>
   454→        </header>
   455→
   456→        {/* Search panel (animated slide-down) */}
   457→        <AnimatePresence>
   458→          {searchOpen && (
   459→            <SearchPanel
   460→              conversationId={conversationId}
   461→              onResultClick={handleSearchResultClick}
   462→              onClose={() => setSearchOpen(false)}
   463→            />
   464→          )}
   465→        </AnimatePresence>
   466→
   467→        {/* Feed with dot pattern background */}
   468→        <div className="relative min-h-0 flex-1">
   469→          {/* Subtle dot pattern behind messages */}
   470→          <div
   471→            className="pointer-events-none absolute inset-0"
   472→            style={{
   473→              backgroundImage: DOT_PATTERN,
   474→              backgroundRepeat: "repeat",
   475→              backgroundSize: "20px 20px",
   476→            }}
   477→            aria-hidden="true"
   478→          />
   479→          <MessageFeed
   480→            messages={messages}
   481→            isLoading={isLoading}
   482→            isPartnerTyping={isPartnerTyping}
   483→            currentUserId={currentUserId}
   484→            partnerName={partner.name}
   485→            partnerAvatar={partner.avatar_url}
   486→            onMarkAsRead={markAsRead}
   487→            onVanishMessage={vanishMessage}
   488→            onReplyTo={setReplyTo}
   489→            onReact={sendReaction}
   490→            scrollToMessageId={scrollToMessageId}
   491→            onScrolledToMessage={handleScrolledToMessage}
   492→          />
   493→        </div>
   494→
   495→        {/* Reply preview bar */}
   496→        <AnimatePresence>
   497→          {replyTo && (
   498→            <m.div initial={{ height: 0, opacity: 0 }} animate={{ height: "auto", opacity: 1 }} exit={{ height: 0, opacity: 0 }} transition={{ duration: 0.2 }} className="overflow-hidden border-t bg-muted/30">
   499→              <div className="flex items-center gap-2 px-4 py-2">
   500→                <div className="w-1 h-8 rounded-full bg-gradient-to-b from-violet-500 to-purple-600" />
   501→                <div className="min-w-0 flex-1">
   502→                  <p className="text-[11px] font-semibold text-violet-500">{replyTo.sender_name || "Unknown"}</p>
   503→                  <p className="truncate text-xs text-muted-foreground">{replyTo.content}</p>
   504→                </div>
   505→                <button type="button" onClick={() => setReplyTo(null)} className="flex size-6 items-center justify-center rounded-full text-muted-foreground hover:bg-muted"><X className="size-3.5" /></button>
   506→              </div>
   507→            </m.div>
   508→          )}
   509→        </AnimatePresence>
   510→
   511→        {/* Voice waveform overlay */}
   512→        <AnimatePresence>
   513→          {showVoiceWaveform && (
   514→            <m.div initial={{ height: 0, opacity: 0 }} animate={{ height: "auto", opacity: 1 }} exit={{ height: 0, opacity: 0 }} transition={{ duration: 0.15 }} className="overflow-hidden">
   515→              <div className="flex items-center gap-3 px-4 py-3">
   516→                <div className="flex size-10 items-center justify-center rounded-full bg-red-500/15 text-red-500">
   517→                  {voice.isRecording ? <span className="relative flex size-3"><span className="absolute inline-flex size-full animate-ping rounded-full bg-red-400 opacity-75" /><span className="relative inline-flex size-3 rounded-full bg-red-500" /></span> : null}

   520→                  {voice.amplitudes.slice(-60).map((amp, i) => (

   522→                  ))}
   523→                </div>
   524→                <span className="shrink-0 text-xs font-medium tabular-nums text-muted-foreground">{Math.floor(voice.duration / 60)}:{String(voice.duration % 60).padStart(2, "0")}</span>
   525→                <m.button type="button" onClick={handleVoiceSend} whileTap={{ scale: 0.9 }} className="flex size-10 shrink-0 items-center justify-center rounded-full bg-gradient-to-r from-violet-500 to-purple-600 text-white" aria-label="Send voice">
   526→                  <ArrowUp className="size-4" />
   527→                </m.button>
   528→              </div>
   529→            </m.div>
   530→          )}
   531→        </AnimatePresence>
   532→
   533→        {/* Input bar */}
   534→        <div className="shrink-0 border-t bg-background px-3 py-2.5 sm:px-4 sm:py-3">
   535→          <form onSubmit={handleSubmit} className="flex items-end gap-2">
   536→            {/* Vanish toggle - compact */}
   537→            <m.button type="button" onClick={() => setVanishMode((v) => !v)} whileTap={{ scale: 0.9 }} className={cn("flex size-8 shrink-0 items-center justify-center rounded-lg transition-colors duration-200", vanishMode ? "bg-violet-500/15 text-violet-500" : "text-muted-foreground hover:bg-muted")} aria-label="Vanish mode">
   538→              {vanishMode ? <EyeOff className="size-3.5" /> : <Eye className="size-3.5" />}
   539→            </m.button>
   540→
   541→            {/* Ephemeral timer dropdown - compact */}
   542→            <div className="relative">
   543→              <m.button type="button" onClick={() => setEphemeralOpen((v) => !v)} whileTap={{ scale: 0.9 }} className={cn("flex size-8 shrink-0 items-center justify-center rounded-lg transition-colors duration-200", ephemeralSeconds ? "bg-amber-500/15 text-amber-500" : "text-muted-foreground hover:bg-muted")} aria-label="Ephemeral timer">

   545→              </m.button>
   546→              <AnimatePresence>

   548→                  <m.div initial={{ opacity: 0, y: 4, scale: 0.95 }} animate={{ opacity: 1, y: 0, scale: 1 }} exit={{ opacity: 0, y: 4, scale: 0.95 }} transition={{ duration: 0.15 }} className="absolute bottom-full left-0 z-50 mb-2 w-28 overflow-hidden rounded-xl border bg-popover p-1 shadow-xl">
   549→                    {EPHEMERAL_OPTIONS.map((opt) => (
   550→                      <button key={opt.label} type="button" onClick={() => { setEphemeralSeconds(opt.value); setEphemeralOpen(false); }}
   551→                        className={cn("flex w-full items-center gap-2 rounded-lg px-3 py-2 text-xs transition-colors", ephemeralSeconds === opt.value ? "bg-violet-500/10 text-violet-500 font-medium" : "text-foreground hover:bg-muted")}>
   552→                        <Clock className="size-3" />{opt.label}
   553→                      </button>
   554→                    ))}
   555→                  </m.div>
   556→                )}
   557→              </AnimatePresence>
   558→            </div>
   559→
   560→            {/* Text input (hidden when recording) - pill shaped with gradient border on focus */}
   561→            {!showVoiceWaveform && (
   562→              <div className="relative flex-1">
   563→                <textarea ref={inputRef} value={draft} onChange={handleInputChange} onKeyDown={handleKeyDown} onInput={handleInput}
   564→                  placeholder={vanishMode ? "Send a disappearing message\u2026" : replyTo ? `Reply to ${replyTo.sender_name || "message"}...` : "Type a message\u2026"}
   565→                  rows={1} className={cn(
   566→                    "w-full resize-none rounded-2xl border bg-muted/50 px-4 py-3 text-sm",
   567→                    "placeholder:text-muted-foreground",
   568→                    "focus-visible:outline-none focus-visible:ring-0",
   569→                    "max-h-32 overflow-y-auto",
   570→                    "transition-all duration-200",
   571→                    // Gradient border effect on focus
   572→                    "focus-visible:border-transparent focus-visible:ring-[2px]",
   573→                    vanishMode
   574→                      ? "border-violet-500/30 focus-visible:ring-violet-400/30"
   575→                      : "focus-visible:ring-violet-400/25",
   576→                  )}
   577→                />
   578→                {/* Subtle gradient border overlay when focused - implemented via ring colors */}
   579→              </div>
   580→            )}
   581→
   582→            {/* Attachment button (placeholder) */}
   583→            {!showVoiceWaveform && (
   584→              <m.button type="button" whileTap={{ scale: 0.9 }} className="flex size-9 shrink-0 items-center justify-center rounded-xl text-muted-foreground transition-colors hover:bg-muted" aria-label="Attach file">
   585→                <Paperclip className="size-4" />
   586→              </m.button>
   587→            )}
   588→
   589→            {/* Emoji button (placeholder) */}
   590→            {!showVoiceWaveform && (
   591→              <m.button type="button" whileTap={{ scale: 0.9 }} className="flex size-9 shrink-0 items-center justify-center rounded-xl text-muted-foreground transition-colors hover:bg-muted" aria-label="Emoji">
   592→                <Smile className="size-4" />
   593→              </m.button>
   594→            )}
   595→
   596→            {/* Send button (shown when there's text) */}
   597→            {!showVoiceWaveform && hasText && (
   598→              <AnimatePresence>
   599→                <m.div initial={{ scale: 0, opacity: 0 }} animate={{ scale: 1, opacity: 1 }} exit={{ scale: 0, opacity: 0 }} transition={{ duration: 0.15, type: "spring", stiffness: 500, damping: 30 }}>
   600→                  <Button type="submit" size="icon" disabled={isSending}
   601→                    className={cn("size-9 shrink-0 rounded-xl bg-gradient-to-r from-violet-500 to-purple-600 shadow-md shadow-violet-500/20 hover:from-violet-600 hover:to-purple-700")}
   602→                    aria-label="Send message"><ArrowUp className="size-4" /></Button>
   603→                </m.div>
   604→              </AnimatePresence>
   605→            )}
   606→
   607→            {/* Mic button (shown when input is empty) */}
   608→            {!showVoiceWaveform && !hasText && (
   609→              <AnimatePresence>
   610→                <m.div initial={{ scale: 0, opacity: 0 }} animate={{ scale: 1, opacity: 1 }} exit={{ scale: 0, opacity: 0 }} transition={{ duration: 0.15, type: "spring", stiffness: 500, damping: 30 }}>
   611→                  <m.button type="button" onClick={() => voice.startRecording()} whileTap={{ scale: 0.9 }} className="flex size-9 shrink-0 items-center justify-center rounded-xl text-muted-foreground transition-colors hover:bg-muted" aria-label="Record voice">
   612→                    <Mic className="size-4" />
   613→                  </m.button>
   614→                </m.div>
   615→              </AnimatePresence>
   616→            )}
   617→
   618→            {/* Cancel recording button */}
   619→            {showVoiceWaveform && (
   620→              <m.button type="button" onClick={async () => { await voice.stopRecording(); setShowVoiceWaveform(false); }} whileTap={{ scale: 0.9 }} className="flex size-10 shrink-0 items-center justify-center rounded-xl text-red-500 hover:bg-red-500/10" aria-label="Cancel recording">
   621→                <MicOff className="size-4" />
   622→              </m.button>
   623→            )}
   624→          </form>
   625→        </div>
   626→      </div>
   627→
   628→      {/* Partner Info Panel */}
   629→      <PartnerInfoPanel
   630→        partner={partner}
   631→        partnerStatus={partnerStatus}
   632→        open={infoPanelOpen}
   633→        onClose={() => setInfoPanelOpen(false)}
   634→      />
   635→    </div>
   636→  );
