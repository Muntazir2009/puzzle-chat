    46→function formatFileSize(bytes: number): string {
    47→  if (bytes < 1024) return `${bytes} B`;
    48→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
    49→  return `${(bytes / (1024 * 1024)).toFixed(1)} MB`;
    50→}
    51→
    52→function formatLastSeen(lastSeen: string | null): string {
    53→  if (!lastSeen) return "Offline";
    54→  const diffMs = Date.now() - new Date(lastSeen).getTime();
    55→  const diffSec = Math.floor(diffMs / 1000);
    56→  if (diffSec < 60) return "Seen just now";
    57→  const diffMin = Math.floor(diffSec / 60);
    58→  if (diffMin < 60) return `Seen ${diffMin}m ago`;
    59→  const diffHr = Math.floor(diffMin / 60);
    60→  if (diffHr < 24) return `Seen ${diffHr}h ago`;
    61→  const diffDay = Math.floor(diffHr / 24);
    62→  if (diffDay < 7) return `Seen ${diffDay}d ago`;
    63→  return `Seen ${new Date(lastSeen).toLocaleDateString()}`;
    64→}

   239→export function ChatLayout({ currentUserId, otherUserId, conversationId, partner, initialMessages }: ChatLayoutProps) {
   240→  const { messages, isLoading, isPartnerTyping, partnerStatus, sendMessage, onTyping, markAsRead, vanishMessage, deleteMessage, sendReaction, loadMore, hasMore, loadingMore } =
   241→    useChat({ currentUserId, otherUserId, conversationId, initialMessages });
   242→
   243→  const [draft, setDraft] = useState("");
   244→  const [sendBtnKey, setSendBtnKey] = useState(0);
   245→  const [isSending, setIsSending] = useState(false);
   246→  const [vanishMode, setVanishMode] = useState(false);
   247→  const [replyTo, setReplyTo] = useState<ChatMessage | null>(null);
   248→  const [ephemeralOpen, setEphemeralOpen] = useState(false);
   249→  const [ephemeralSeconds, setEphemeralSeconds] = useState<number | null>(null);
   250→  const [infoPanelOpen, setInfoPanelOpen] = useState(false);
   251→  const inputRef = useRef<HTMLTextAreaElement>(null);
   252→  const viewport = useVisualViewport();
   253→
   254→  /* Background theme + wallpaper */
   255→  const { themeId, theme, selectTheme, wallpaper, setWallpaper } = useChatBackground();
   256→
   257→  /* Voice */
   258→  const voice = useVoiceRecorder();
   259→  const [showVoiceWaveform, setShowVoiceWaveform] = useState(false);
   260→
   261→  /* Attachment */
   262→  const [attachmentFile, setAttachmentFile] = useState<File | null>(null);
   263→  const [attachmentPreview, setAttachmentPreview] = useState<string | null>(null);
   264→  const [isUploading, setIsUploading] = useState(false);
   265→  const fileInputRef = useRef<HTMLInputElement>(null);
   266→
   267→  const containerStyle = viewport.height > 0 ? ({ height: `${viewport.height}px` } as React.CSSProperties) : ({ height: "100dvh" } as React.CSSProperties);
   268→  const kbOffset = viewport.isKeyboardVisible && viewport.offsetTop > 0 ? ({ marginTop: `-${viewport.offsetTop}px` } as React.CSSProperties) : undefined;
   269→
   270→  /* Compute background style — wallpaper overrides theme */
   271→  const bgStyle = wallpaper
   272→    ? { backgroundImage: `url(${wallpaper})`, backgroundSize: "cover", backgroundPosition: "center" }
   273→    : theme.style;
   274→
   275→  useEffect(() => { inputRef.current?.focus(); }, []);
   276→
   277→  /* Show voice waveform when recording */
   278→  useEffect(() => { setShowVoiceWaveform(voice.isRecording); }, [voice.isRecording]);
   279→
   280→  /* Global keyboard shortcuts */
   281→  useEffect(() => {
   282→    function handleGlobalKey(e: KeyboardEvent) {
   283→      if (e.key === "Escape") {
   284→        if (replyTo) { setReplyTo(null); e.preventDefault(); }
   285→        else if (ephemeralOpen) { setEphemeralOpen(false); e.preventDefault(); }
   286→        else if (infoPanelOpen) { setInfoPanelOpen(false); e.preventDefault(); }
   287→      }
   288→    }
   289→    window.addEventListener("keydown", handleGlobalKey);
   290→    return () => window.removeEventListener("keydown", handleGlobalKey);
   291→  }, [replyTo, ephemeralOpen, infoPanelOpen]);
   292→
   293→  /* Derived values — MUST be declared before any useCallback that
   294→     references them in its dependency array to avoid TDZ errors
   295→     in the production webpack bundle. */
   296→  const hasAttachment = attachmentFile !== null;
   297→  const hasText = draft.trim().length > 0;
   298→
   299→  /* ---- Attachment upload helpers (declared BEFORE handleSubmit
   300→     because handleSubmit's dependency array includes uploadAttachment;
   301→     accessing a const before its declaration causes a TDZ crash in the
   302→     production webpack bundle deployed to Cloudflare Workers). ---- */
   303→  const clearAttachment = useCallback(() => {
   304→    if (attachmentPreview) URL.revokeObjectURL(attachmentPreview);
   305→    setAttachmentFile(null);
   306→    setAttachmentPreview(null);
   307→  }, [attachmentPreview]);
   308→
   309→  const uploadAttachment = useCallback(async () => {
   310→    if (!attachmentFile || isUploading) return;
   311→    setIsUploading(true);
   312→    try {
   313→      const fd = new FormData();
   314→      fd.append("file", attachmentFile);
   315→      fd.append("conversation_id", conversationId);
   316→      if (replyTo?.id) fd.append("reply_to_id", replyTo.id);
   317→      if (vanishMode) fd.append("vanish_mode", "true");
   318→      if (ephemeralSeconds) fd.append("ephemeral_seconds", String(ephemeralSeconds));
   319→
   320→      const res = await fetch("/api/messages/upload", { method: "POST", body: fd });
   321→      if (!res.ok) {
   322→        const data = await res.json().catch(() => ({ error: "Upload failed" }));
   323→        throw new Error(data.error || `Upload failed: ${res.status}`);
   324→      }
   325→      /* Message is inserted server-side and pushed via Pusher;
   326→         the useChat hook will pick it up automatically. */
   327→      clearAttachment();
   328→      setReplyTo(null);
   329→    } catch (err) {
   330→      console.error("[ChatLayout] upload error:", err);
   331→      toast({
   332→        title: "Upload failed",
   333→        description: err instanceof Error ? err.message : "Could not send the file.",
   334→        variant: "destructive",
   335→      });
   336→    } finally {
   337→      setIsUploading(false);
   338→    }
   339→  }, [attachmentFile, isUploading, conversationId, replyTo, vanishMode, ephemeralSeconds, clearAttachment]);
   340→
   341→  const handleSubmit = useCallback(async (e: FormEvent) => {
   342→    e.preventDefault();
   343→    /* If an attachment is selected, upload it instead of sending text */
   344→    if (hasAttachment && !hasText) {
   345→      await uploadAttachment();
   346→      return;
   347→    }
   348→    /* If there's both text and an attachment, send text first, then upload */
   349→    const trimmed = draft.trim();
   350→    if (!trimmed || isSending) return;
   351→    setIsSending(true);
   352→    const opts: SendMessageOptions = { vanish_mode: vanishMode, ephemeral_seconds: ephemeralSeconds, reply_to_id: replyTo?.id ?? null };
   353→    setDraft(""); setReplyTo(null);
   354→    if (inputRef.current) inputRef.current.style.height = "auto";
   355→    try {
   356→      await sendMessage(trimmed, opts);
   357→      setSendBtnKey((k) => k + 1);
   358→      /* Upload attachment after text is sent */
   359→      if (hasAttachment) await uploadAttachment();
   360→    }
   361→    finally { setIsSending(false); requestAnimationFrame(() => inputRef.current?.focus()); }
   362→  }, [draft, isSending, sendMessage, vanishMode, ephemeralSeconds, replyTo, hasAttachment, uploadAttachment]);
   363→
   364→  const handleVoiceSend = useCallback(async () => {
   365→    const result = await voice.stopRecording();
   366→    if (!result || result.duration < 1) return;
   367→    const blob = result.blob;
   368→    const url = URL.createObjectURL(blob);
   369→    await sendMessage(url, { type: "voice", voice_duration: result.duration, waveform_data: result.amplitudes, ephemeral_seconds: ephemeralSeconds });
   370→    URL.revokeObjectURL(url);
   371→  }, [voice, sendMessage, ephemeralSeconds]);
   372→
   373→  const handleKeyDown = useCallback((e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   374→    if (e.key === "Enter" && !e.shiftKey) { e.preventDefault(); const form = e.currentTarget.closest("form"); if (form) form.requestSubmit(); }
   375→  }, []);
   376→
   377→  const handleInputChange = useCallback((e: React.ChangeEvent<HTMLTextAreaElement>) => { setDraft(e.target.value); onTyping(); }, [onTyping]);
   378→  const handleInput = useCallback((e: React.FormEvent<HTMLTextAreaElement>) => { const el = e.currentTarget; el.style.height = "auto"; el.style.height = `${Math.min(el.scrollHeight, 128)}px`; }, []);
   379→
   380→  const canSend = hasText || hasAttachment;
   381→
   382→  /* ---- Attachment handlers ---------------------------------------- */
   383→  const handleFileSelect = useCallback((e: ChangeEvent<HTMLInputElement>) => {
   384→    const file = e.target.files?.[0] ?? null;
   385→    if (!file) return;
   386→    if (file.size > MAX_ATTACHMENT_SIZE) {
   387→      toast({ title: "File too large", description: "Maximum size is 10 MB.", variant: "destructive" });
   388→      e.target.value = "";
   389→      return;
   390→    }
   391→    setAttachmentFile(file);
   392→    /* Generate thumbnail preview for images */
   393→    if (IMAGE_MIME_TYPES.has(file.type)) {
   394→      const url = URL.createObjectURL(file);
   395→      setAttachmentPreview(url);
   396→    } else {
   397→      setAttachmentPreview(null);
   398→    }
   399→    /* Reset input so the same file can be re-selected */
   400→    e.target.value = "";
   401→  }, []);
   402→
   403→  const handlePaperclipClick = useCallback(() => {
   404→    fileInputRef.current?.click();
   405→  }, []);
   406→
   407→  /* Status text for header */
   408→  const statusText = isPartnerTyping
   409→    ? "typing\u2026"
   410→    : partnerStatus.online
   411→      ? "Online"
   412→      : formatLastSeen(partnerStatus.last_seen);
   413→
   414→  /* ---- Render --------------------------------------------------- */
   415→
   416→  return (
   417→    <div className="flex w-full flex-col bg-background" style={containerStyle}>
   418→      <div style={kbOffset} className="flex min-h-0 flex-1 flex-col">
   419→        {/* Header - sticky with glassmorphic blur + shadow */}
   420→        <header className="sticky top-0 z-30 flex h-14 shrink-0 items-center gap-3 border-b border-border/40 bg-background/80 shadow-[0_1px_3px_rgba(0,0,0,0.04)] backdrop-blur-2xl supports-[backdrop-filter]:bg-background/60 px-4">
   421→          <button
   422→            type="button"
   423→            onClick={() => setInfoPanelOpen(true)}
   424→            className="flex items-center gap-3 rounded-lg p-1 -ml-1 transition-colors hover:bg-muted/50 focus-visible:outline-none"
   425→            aria-label="Open user info"
   426→          >
   427→            <Avatar className="size-9 ring-2 ring-[var(--app-accent-lighter)]/30 dark:ring-[var(--app-accent)]/40">
   428→              {partner.avatar_url && <AvatarImage src={partner.avatar_url} alt={partner.name} />}
   429→              <AvatarFallback
   430→                className="text-xs font-semibold text-white"
   431→                style={{ background: "linear-gradient(to bottom right, var(--app-accent-from), var(--app-accent-to))" }}
   432→              >{partner.name.slice(0, 2).toUpperCase()}</AvatarFallback>
   433→            </Avatar>
   434→            <div className="flex flex-col">
   435→              <span className="text-sm font-semibold leading-tight">{partner.name}</span>
   436→              <div className="flex items-center gap-1.5">
   437→                <StatusDot online={partnerStatus.online} />
   438→                <span
   439→                  className={cn("text-xs", isPartnerTyping ? "font-medium" : "text-muted-foreground")}
   440→                  style={isPartnerTyping ? { color: "var(--app-accent)" } : undefined}
   441→                >{statusText}</span>
   442→              </div>
   443→            </div>
   444→          </button>
   445→
   446→          {/* Right-side header actions */}
   447→          <div className="ml-auto flex items-center gap-1">
   448→            <ChatBackgroundPicker themeId={themeId} onSelect={selectTheme} wallpaper={wallpaper} onSetWallpaper={setWallpaper} />
   449→          </div>
   450→        </header>
   451→
   452→        {/* Feed area – background theme applied via MessageFeed */}
   453→        <div className="relative min-h-0 flex-1">
   454→          <MessageFeed
   455→            messages={messages}
   456→            isLoading={isLoading}
   457→            isPartnerTyping={isPartnerTyping}
   458→            currentUserId={currentUserId}
   459→            partnerName={partner.name}
   460→            partnerAvatar={partner.avatar_url}
   461→            backgroundStyle={bgStyle}
   462→            onMarkAsRead={markAsRead}
   463→            onVanishMessage={vanishMessage}
   464→            onDeleteMessage={deleteMessage}
   465→            onReplyTo={setReplyTo}
   466→            onReact={sendReaction}
   467→            loadMore={loadMore}
   468→            hasMore={hasMore}
   469→            loadingMore={loadingMore}
   470→          />
   471→        </div>
   472→
   473→        {/* Reply preview bar */}
   474→        <AnimatePresence>
   475→          {replyTo && (
   476→            <m.div initial={{ height: 0, opacity: 0 }} animate={{ height: "auto", opacity: 1 }} exit={{ height: 0, opacity: 0 }} transition={{ duration: 0.2 }} className="overflow-hidden backdrop-blur-sm" style={{ backgroundColor: "var(--app-accent-subtle)", borderLeft: "2px solid var(--app-accent)" }}>
   477→              <div className="flex items-center gap-2 px-4 py-2">
   478→                <div className="w-1 h-8 rounded-full" style={{ background: "linear-gradient(to bottom, var(--app-accent-from), var(--app-accent-to))" }} />
   479→                <div className="min-w-0 flex-1">
   480→                  <p className="text-[11px] font-semibold" style={{ color: "var(--app-accent)" }}>{replyTo.sender_name || "Unknown"}</p>
   481→                  <p className="truncate text-xs text-muted-foreground">{replyTo.content}</p>
   482→                </div>
   483→                <div className="flex items-center gap-1.5">
   484→                  <button type="button" onClick={() => setReplyTo(null)} className="flex size-6 items-center justify-center rounded-full text-muted-foreground hover:bg-muted"><X className="size-3.5" /></button>
   485→                  <kbd className="hidden sm:inline-flex h-5 items-center rounded border bg-muted px-1.5 text-[10px] font-medium text-muted-foreground">Esc</kbd>

   538→            </m.div>

   541→
   542→        {/* Voice waveform overlay */}

   596→                          ? { backgroundColor: "var(--app-accent-subtle)", color: "var(--app-accent)", fontWeight: 600 }

   599→                        <Clock className="size-3 opacity-60" />{opt.label}

   602→                  </m.div>

   604→              </AnimatePresence>
   605→            </div>
   606→
   607→            {/* Text input (hidden when recording) - pill shaped with gradient border on focus */}
   608→            {!showVoiceWaveform && (
   609→              <div className="relative flex-1">

   620→                    vanishMode && "vanish-input-glow",

   622→                  style={vanishMode
   623→                    ? { borderColor: 'var(--app-accent-subtle)' }
