    52→/*  Double-tap hook                                                      */
    53→/* ------------------------------------------------------------------ */
    54→
    55→function useDoubleTap(
    56→  onDoubleTap: () => void,
    57→  delay = 300,
    58→) {
    59→  const lastTapRef = useRef(0);
    60→  return useCallback((): boolean => {
    61→    const now = Date.now();
    62→    if (now - lastTapRef.current < delay) {
    63→      lastTapRef.current = 0;
    64→      // Haptic feedback on double-tap
    65→      if (typeof navigator !== "undefined" && "vibrate" in navigator) {
    66→        navigator.vibrate(10);
    67→      }
    68→      onDoubleTap();
    69→      return true; // is double tap
    70→    }
    71→    lastTapRef.current = now;
    72→    return false; // is single tap
    73→  }, [onDoubleTap, delay]);
    74→}

   147→/*  Ephemeral Timer (circular SVG countdown)                           */
   148→/* ------------------------------------------------------------------ */
   149→
   150→function EphemeralTimer({
   151→  seconds,
   152→  onExpire,
   153→}: {
   154→  seconds: number;
   155→  onExpire: () => void;
   156→}) {
   157→  const [remaining, setRemaining] = useState(seconds);
   158→  const intervalRef = useRef<ReturnType<typeof setInterval> | undefined>(
   159→    undefined,
   160→  );
   161→
   162→  useEffect(() => {
   163→    intervalRef.current = setInterval(() => {
   164→      setRemaining((r) => {
   165→        if (r <= 1) {
   166→          clearInterval(intervalRef.current);
   167→          onExpire();
   168→          return 0;
   169→        }
   170→        return r - 1;
   171→      });
   172→    }, 1000);
   173→    return () => clearInterval(intervalRef.current);
   174→  }, [seconds, onExpire]);
   175→
   176→  const pct = remaining / seconds;
   177→  const circ = 2 * Math.PI * 7;
   178→  const offset = circ * (1 - pct);
   179→
   180→  return (
   181→    <svg
   182→      className="size-3.5 shrink-0 ephemeral-timer-pulse"
   183→      viewBox="0 0 18 18"
   184→      aria-label={`${remaining}s remaining`}
   185→    >
   186→      <circle
   187→        cx="9"
   188→        cy="9"
   189→        r="7"
   190→        fill="none"
   191→        stroke="currentColor"
   192→        strokeWidth="1.5"
   193→        className="text-white/20"
   194→      />
   195→      <circle
   196→        cx="9"
   197→        cy="9"
   198→        r="7"
   199→        fill="none"
   200→        stroke="currentColor"
   201→        strokeWidth="1.5"
   202→        strokeDasharray={circ}
   203→        strokeDashoffset={offset}
   204→        strokeLinecap="round"
   205→        style={{
   206→          color: "var(--app-accent-lighter)",
   207→          transform: "rotate(-90deg)",
   208→          transformOrigin: "50% 50%",
   209→          transition: "stroke-dashoffset 1s linear",
   210→        }}
   211→      />
   212→    </svg>
   213→  );
   214→}

   437→/* ------------------------------------------------------------------ */
   438→/*  Voice Bubble                                                       */
   439→/* ------------------------------------------------------------------ */
   440→
   441→function VoiceBubble({
   442→  message,
   443→  isOwn,
   444→  waveform,
   445→}: {
   446→  message: ChatMessage;
   447→  isOwn: boolean;
   448→  waveform: number[];
   449→}) {
   450→  const audioRef = useRef<HTMLAudioElement>(null);
   451→  const [playing, setPlaying] = useState(false);
   452→  const [progress, setProgress] = useState(0);
   453→  const [speed, setSpeed] = useState(1);
   454→  const speeds = [1, 1.5, 2];
   455→  const duration = message.voice_duration ?? 0;
   456→
   457→  useEffect(() => {
   458→    const audio = audioRef.current;
   459→    if (!audio) return;
   460→    const onTime = () => {
   461→      if (audio.duration) setProgress(audio.currentTime / audio.duration);
   462→    };
   463→    const onEnd = () => setPlaying(false);
   464→    audio.addEventListener("timeupdate", onTime);
   465→    audio.addEventListener("ended", onEnd);
   466→    return () => {
   467→      audio.removeEventListener("timeupdate", onTime);
   468→      audio.removeEventListener("ended", onEnd);
   469→    };
   470→  }, []);
   471→
   472→  const toggle = useCallback(() => {
   473→    const audio = audioRef.current;
   474→    if (!audio) return;
   475→    if (playing) {
   476→      audio.pause();
   477→      setPlaying(false);
   478→    } else {
   479→      audio.playbackRate = speed;
   480→      audio.play().catch(() => {});
   481→      setPlaying(true);
   482→    }
   483→  }, [playing, speed]);
   484→
   485→  const cycleSpeed = useCallback(() => {
   486→    setSpeed((s) => {
   487→      const idx = speeds.indexOf(s);
   488→      const next = speeds[(idx + 1) % speeds.length];
   489→      if (audioRef.current) audioRef.current.playbackRate = next;
   490→      return next;
   491→    });
   492→  }, []);
   493→
   494→  /* Build waveform bars */
   495→  const bars = useMemo(() => {
   496→    const count = Math.min(waveform.length, 40);
   497→    const step = Math.max(1, Math.floor(waveform.length / count));
   498→    return Array.from({ length: count }, (_, i) => waveform[i * step] ?? 0.1);
   499→  }, [waveform]);
   500→
   501→  const progressIdx = Math.floor(progress * bars.length);
   502→
   503→  return (
   504→    <div className={cn("flex items-center gap-3", isOwn ? "flex-row-reverse" : "")}>
   505→      <button
   506→        type="button"
   507→        onClick={toggle}
   508→        className={cn(
   509→          "flex size-9 shrink-0 items-center justify-center rounded-full transition-all duration-150 active:scale-95",
   510→          isOwn
   511→            ? ""
   512→            : "",
   513→        )}
   514→        aria-label={playing ? "Pause" : "Play"}
   515→      >
   516→        {playing ? (
   517→          <Pause className="size-3.5 text-white" />
   518→        ) : (
   519→          <Play className="size-3.5 text-white ml-0.5" />
   520→        )}
   521→      </button>
   522→      <div className="flex items-center gap-[2px]" style={{ height: 32 }}>
   523→        {bars.map((amp, i) => (
   524→          <div
   525→            key={i}
   526→            className="w-[2.5px] rounded-full transition-all duration-100"
   527→            style={{
   528→              height: `${Math.max(4, amp * 32)}px`,
   529→              backgroundColor:
   530→                i < progressIdx
   531→                  ? isOwn
   532→                    ? "rgba(255,255,255,1)"
   533→                    : "var(--app-accent-light)"
   534→                  : isOwn
   535→                    ? "rgba(255,255,255,0.5)"
   536→                    : "rgba(161,161,170,0.7)",
   537→            }}
   538→          />
   539→        ))}
   540→      </div>
   541→      <button
   542→        type="button"
   543→        onClick={cycleSpeed}
   544→        className={cn(
   545→          "shrink-0 text-[11px] font-semibold tabular-nums transition-colors hover:opacity-80 active:scale-95",
   546→          isOwn ? "text-white/70" : "text-zinc-400",
   547→        )}
   548→      >
   549→        {speed}x
   550→      </button>
   551→      <span
   552→        className={cn(
   553→          "shrink-0 text-[11px] tabular-nums",
   554→          isOwn ? "text-white/50" : "text-zinc-500",
   555→        )}
   556→      >
   557→        {Math.floor(duration / 60)}:{String(duration % 60).padStart(2, "0")}
   558→      </span>
   559→      <audio ref={audioRef} src={message.content} preload="metadata" />
   560→    </div>
   561→  );
   562→}

   565→/*  LinkifiedText – detects & renders clickable URLs                   */
   566→/* ------------------------------------------------------------------ */

   568→const URL_REGEX = /(https?:\/\/[^\s<]+)/g;
   569→
   570→type TextSegment =
   571→  | { type: "text"; value: string }
   572→  | { type: "link"; value: string };
   573→
   574→function parseUrls(text: string): TextSegment[] {
   575→  const segments: TextSegment[] = [];
   576→  let lastIndex = 0;
   577→  let match: RegExpExecArray | null;
   578→  while ((match = URL_REGEX.exec(text)) !== null) {
   579→    if (match.index > lastIndex) {
   580→      segments.push({ type: "text", value: text.slice(lastIndex, match.index) });
   581→    }
   582→    segments.push({ type: "link", value: match[1] });
   583→    lastIndex = URL_REGEX.lastIndex;
   584→  }
   585→  if (lastIndex < text.length) {
   586→    segments.push({ type: "text", value: text.slice(lastIndex) });
   587→  }
   588→  return segments;
   589→}

   591→/** Split a plain-text string into highlighted / non-highlighted fragments */
   592→function highlightText(text: string, query: string): React.ReactNode[] {
   593→  if (!query) return [text];
   594→  const escaped = query.replace(/[.*+?^${}()|[\]\\]/g, "\\");
   595→  const re = new RegExp(`(${escaped})`, "gi");
   596→  const parts: React.ReactNode[] = [];
   597→  let lastIndex = 0;
   598→  let match: RegExpExecArray | null;
   599→  while ((match = re.exec(text)) !== null) {
   600→    if (match.index > lastIndex) {
   601→      parts.push(text.slice(lastIndex, match.index));
   602→    }
   603→    parts.push(
   604→      <mark
   605→        key={match.index}
   606→        className="bg-yellow-300/60 dark:bg-yellow-400/40 rounded-sm px-0.5"
   607→      >
   608→        {match[1]}
   609→      </mark>,
   610→    );
   611→    lastIndex = re.lastIndex;
   612→  }
   613→  if (lastIndex < text.length) {
   614→    parts.push(text.slice(lastIndex));
   615→  }
   616→  return parts;
   617→}

   830→function getBubbleRounding({
   831→  isFirst,
   832→  isLast,
   833→  isOwn,
   834→}: {
   835→  isFirst: boolean;
   836→  isLast: boolean;
   837→  isOwn: boolean;
   838→}): string {
   839→  if (isFirst && isLast) {
   840→    // Single message in group (or only message)
   841→    return isOwn
   842→      ? "rounded-2xl rounded-br-sm"
   843→      : "rounded-2xl rounded-bl-sm";
   844→  }
   845→  if (isFirst) {
   846→    // First message of a multi-message group
   847→    return isOwn
   848→      ? "rounded-tl-2xl rounded-bl-2xl rounded-tr-sm rounded-br-sm"
   849→      : "rounded-tl-2xl rounded-bl-sm rounded-tr-2xl rounded-br-sm";
   850→  }
   851→  if (isLast) {
   852→    // Last message of a multi-message group
   853→    return isOwn
   854→      ? "rounded-tl-2xl rounded-bl-2xl rounded-tr-sm rounded-br-2xl"
   855→      : "rounded-tl-2xl rounded-bl-2xl rounded-tr-2xl rounded-br-sm";
   856→  }
   857→  // Middle message
   858→  return isOwn
   859→    ? "rounded-tl-2xl rounded-bl-2xl rounded-tr-sm rounded-br-sm"
   860→    : "rounded-tl-2xl rounded-bl-sm rounded-tr-2xl rounded-br-sm";
   861→}

  1462→function formatDateSeparator(dateStr: string): string {
  1463→  const d = new Date(dateStr);
  1464→  const now = new Date();
  1465→  const today = new Date(now.getFullYear(), now.getMonth(), now.getDate());
  1466→  const yesterday = new Date(today);
  1467→  yesterday.setDate(yesterday.getDate() - 1);
  1468→  const msgDay = new Date(d.getFullYear(), d.getMonth(), d.getDate());
  1469→
  1470→  if (msgDay.getTime() === today.getTime()) return "Today";
  1471→  if (msgDay.getTime() === yesterday.getTime()) return "Yesterday";
  1472→  return d.toLocaleDateString(undefined, { month: "short", day: "numeric" });
  1473→}

  1715→    getScrollElement: () => parentRef.current,
  1716→    estimateSize: () => ESTIMATED_ROW_HEIGHT,
  1717→    overscan: OVERSCAN,
  1718→    paddingStart: 16,
  1719→    paddingEnd: 16,
  1720→    gap: 16,
  1721→  });
  1722→
  1723→  const handleScroll = useCallback(
  1724→    (e: UIEvent<HTMLDivElement>) => {
  1725→      const el = e.currentTarget;
  1726→      const dist = el.scrollHeight - el.scrollTop - el.clientHeight;
  1727→      if (dist <= AUTO_SCROLL_THRESHOLD) {
  1728→        isAutoScrollRef.current = true;
  1729→        setShowNewBtn(false);
  1730→        unreadRef.current = 0;
  1731→      } else {
  1732→        isAutoScrollRef.current = false;
  1733→      }
  1734→      /* Trigger loadMore when scrolled near top */
  1735→      if (el.scrollTop < 300 && hasMore && !loadingMore) {
  1736→        loadMoreRef.current?.();
  1737→      }
  1738→    },
  1739→    [hasMore, loadingMore],
  1740→  );
  1741→
  1742→  const scrollToBottom = useCallback(
  1743→    (behavior: ScrollBehavior = "smooth") => {
  1744→      const el = parentRef.current;
  1745→      if (!el) return;
  1746→      el.scrollTo({ top: el.scrollHeight, behavior });
  1747→      isAutoScrollRef.current = true;
  1748→      setShowNewBtn(false);
  1749→      unreadRef.current = 0;
  1750→    },
  1751→    [],
  1752→  );
  1753→
  1754→  useEffect(() => {
  1755→    if (messages.length > 0)
  1756→      requestAnimationFrame(() => scrollToBottom("instant"));
  1757→  }, [isLoading, scrollToBottom]);
  1758→
  1759→  useEffect(() => {
  1760→    if (!onMarkAsRead) return;
  1761→    const unread = messages.filter(
  1762→      (m) => m.sender_id !== currentUserId && m.status !== "read",
  1763→    );
  1764→    if (unread.length > 0 && isAutoScrollRef.current)
  1765→      onMarkAsRead(unread.map((m) => m.id));
  1766→  }, [messages, currentUserId, onMarkAsRead]);
  1767→
  1768→  /* ---- Scroll to a specific message ----------------------------------- */
  1769→  useEffect(() => {
  1770→    if (!scrollToMessageId) return;
  1771→    // Find which group contains the target message
  1772→    const groupIdx = groups.findIndex((g) =>
  1773→      g.messages.some((m) => m.id === scrollToMessageId),
  1774→    );
  1775→    if (groupIdx === -1) {
  1776→      onScrolledToMessage?.();
  1777→      return;
  1778→    }
  1779→    virtualizer.scrollToIndex(groupIdx, { align: "center" });
  1780→    onScrolledToMessage?.();
  1781→  }, [scrollToMessageId, groups, virtualizer, onScrolledToMessage]);
  1782→
  1783→  /* ---- Handle message count changes (new or prepended) ------------ */
  1784→  useEffect(() => {
  1785→    const prev = prevCountRef.current;
  1786→    const prevFirstId = prevFirstIdRef.current;
  1787→    prevCountRef.current = messages.length;
  1788→    prevFirstIdRef.current =
  1789→      messages.length > 0 ? messages[0].id : null;
  1790→
  1791→    if (messages.length <= prev) return;
  1792→
  1793→    /* Check if older messages were prepended (first message changed) */
  1794→    const firstId = messages[0]?.id;
  1795→    const wasPrepended = firstId !== prevFirstId && prevFirstId !== null;
  1796→
  1797→    if (wasPrepended) {
  1798→      /* Older messages prepended – preserve scroll position */
  1799→      const el = parentRef.current;
  1800→      if (el) {
  1801→        const oldHeight = prevScrollHeightRef.current;
  1802→        prevScrollHeightRef.current = el.scrollHeight;
  1803→        requestAnimationFrame(() => {
  1804→          const newHeight = el.scrollHeight;
  1805→          el.scrollTop += newHeight - oldHeight;
  1806→        });
  1807→      }
  1808→      return;
  1809→    }
  1810→
  1811→    /* New messages appended at the end */
  1812→    const last = messages[messages.length - 1];
  1813→    const own = last && last.sender_id === currentUserId;
  1814→    if (own) {
  1815→      requestAnimationFrame(() => scrollToBottom("instant"));
  1816→      return;
  1817→    }
  1818→    if (isAutoScrollRef.current)
  1819→      requestAnimationFrame(() => scrollToBottom("smooth"));
  1820→    else {
  1821→      unreadRef.current += messages.length - prev;
  1822→      setShowNewBtn(true);
  1823→    }
  1824→  }, [messages.length, currentUserId, scrollToBottom]);
  1825→
  1826→  /* Track scroll height for scroll preservation */
  1827→  useEffect(() => {
  1828→    const el = parentRef.current;
  1829→    if (!el) return;
  1830→    prevScrollHeightRef.current = el.scrollHeight;
  1831→  }, [messages.length]);
  1832→
  1833→  if (isLoading) return <MessageListSkeleton />;
  1834→  if (messages.length === 0 && !isPartnerTyping)
  1835→    return (
  1836→      <EmptyState partnerName={partnerName} partnerAvatar={partnerAvatar} />
  1837→    );
  1838→
  1839→  const virtualItems = virtualizer.getVirtualItems();
  1840→
  1841→  return (
  1842→    <div
  1843→      className="relative h-full w-full overflow-hidden"
  1844→      style={backgroundStyle}
  1845→    >
  1846→      {/* Click-away overlay for action sheets / tapback docks */}
  1847→      {(activeActionId || activeTapbackId) && (
  1848→        <div
  1849→          className="fixed inset-0 z-40"
  1850→          onClick={() => { setActiveActionId(null); setActiveTapbackId(null); }}
  1851→        />
  1852→      )}
  1853→      <div
  1854→        ref={parentRef}
  1855→        onScroll={handleScroll}
  1856→        onClick={(e) => {
  1857→          onClearSearchHighlight?.();
  1858→          // Close menus if click is directly on the scroll container or background
  1859→          const target = e.target as HTMLElement;
  1860→          if (target === e.currentTarget || target.closest('.message-bubble-row') === null) {
  1861→            setActiveActionId(null);
  1862→            setActiveTapbackId(null);
  1863→          }
  1864→        }}
  1865→        className="h-full w-full select-none overflow-y-auto overscroll-contain"
  1866→        style={{ WebkitOverflowScrolling: "touch" }}
  1867→      >
  1868→        <div
  1869→          style={{
  1870→            height: `${virtualizer.getTotalSize()}px`,
  1871→            width: "100%",
  1872→            position: "relative",
  1873→          }}
  1874→        >
  1875→          {/* Loading more spinner */}
  1876→          {loadingMore && (
  1877→            <div
  1878→              style={{
  1879→                position: "absolute",
  1880→                top: 0,
  1881→                left: 0,
  1882→                width: "100%",
  1883→                height: "48px",
  1884→                zIndex: 5,
  1885→              }}
  1886→            >
  1887→              <div className="flex items-center justify-center py-3">
  1888→                <svg
  1889→                  className="size-5 animate-spin"
  1890→                  viewBox="0 0 24 24"
  1891→                  fill="none"
  1892→                  aria-label="Loading more messages"
  1893→                  style={{ color: "var(--app-accent-light)" }}
  1894→                >
  1895→                  <circle
  1896→                    className="opacity-25"
  1897→                    cx="12"
  1898→                    cy="12"
  1899→                    r="10"
  1900→                    stroke="currentColor"
  1901→                    strokeWidth="4"
  1902→                  />
  1903→                  <path
  1904→                    className="opacity-75"
  1905→                    fill="currentColor"
  1906→                    d="M4 12a8 8 0 018-8V0C5.373 0 0 5.373 0 12h4z"
  1907→                  />
  1908→                </svg>
  1909→              </div>
  1910→            </div>
  1911→          )}
  1912→          {virtualItems.map((vi) => {
  1913→            const isTyping = vi.index === groups.length;
  1914→            const group = isTyping ? null : groups[vi.index];
  1915→
  1916→            // Determine if we need a date separator before this group
  1917→            let showDateSep = false;
  1918→            if (group) {
  1919→              const firstMsg = group.messages[0];
  1920→              if (vi.index === 0) {
  1921→                showDateSep = true;
  1922→              } else {
  1923→                const prevGroup = groups[vi.index - 1];
  1924→                const prevLastMsg = prevGroup.messages[prevGroup.messages.length - 1];
  1925→                const prevDate = new Date(prevLastMsg.created_at);
  1926→                const currDate = new Date(firstMsg.created_at);
  1927→                showDateSep = prevDate.toDateString() !== currDate.toDateString();
  1928→              }
  1929→            }
  1930→
  1931→            return (
  1932→              <div
  1933→                key={vi.key}
  1934→                data-index={vi.index}
  1935→                ref={virtualizer.measureElement}
  1936→                style={{
  1937→                  position: "absolute",
  1938→                  top: 0,
  1939→                  left: 0,
  1940→                  width: "100%",
  1941→                  transform: `translateY(${vi.start}px)`,
  1942→                }}
  1943→              >
  1944→                {isTyping ? (
  1945→                  <TypingIndicator
  1946→                    partnerName={partnerName}
  1947→                    partnerAvatar={partnerAvatar}
  1948→                  />
  1949→                ) : group ? (
  1950→                  <>
  1951→                    {showDateSep && (
  1952→                      <DateSeparator date={group.messages[0].created_at} />
  1953→                    )}
  1954→                    <MessageGroup
  1955→                      messages={group.messages}
  1956→                      currentUserId={currentUserId}
  1957→                      currentUserName={currentUserName}
  1958→                      currentUserAvatar={currentUserAvatar}
  1959→                      partnerName={partnerName}
  1960→                      partnerAvatar={partnerAvatar}
  1961→                      onReplyTo={onReplyTo}
  1962→                      onReact={onReact}
  1963→                      onDeleteMessage={onDeleteMessage}
  1964→                      searchHighlight={searchHighlight}
  1965→                      activeActionId={activeActionId}
  1966→                      onOpenAction={setActiveActionId}
  1967→                      activeTapbackId={activeTapbackId}
  1968→                      onOpenTapback={setActiveTapbackId}
  1969→                    />
  1970→                  </>
  1971→                ) : null}
  1972→              </div>
  1973→            );
  1974→          })}
  1975→        </div>
  1976→      </div>
  1977→      {showNewBtn && unreadRef.current > 0 && (
  1978→        <NewMessagesButton
  1979→          count={unreadRef.current}
  1980→          onClick={() => scrollToBottom("smooth")}
  1981→        />
  1982→      )}
  1983→    </div>
  1984→  );
  1985→}
