    94→function EphemeralTimer({ seconds, onExpire }: { seconds: number; onExpire: () => void }) {
    95→  const [remaining, setRemaining] = useState(seconds);
    96→  const intervalRef = useRef<ReturnType<typeof setInterval> | undefined>(undefined);
    97→
    98→  useEffect(() => {
    99→    intervalRef.current = setInterval(() => {
   100→      setRemaining((r) => {
   101→        if (r <= 1) { clearInterval(intervalRef.current); onExpire(); return 0; }
   102→        return r - 1;
   103→      });
   104→    }, 1000);
   105→    return () => clearInterval(intervalRef.current);
   106→  }, [seconds, onExpire]);
   107→
   108→  const pct = remaining / seconds;
   109→  const circ = 2 * Math.PI * 7;
   110→  const offset = circ * (1 - pct);
   111→
   112→  return (
   113→    <svg className="size-4 shrink-0 ephemeral-timer-pulse" viewBox="0 0 18 18" aria-label={`${remaining}s remaining`}>
   114→      <circle cx="9" cy="9" r="7" fill="none" stroke="currentColor" strokeWidth="1.5" className="text-white/20" />
   115→      <circle cx="9" cy="9" r="7" fill="none" stroke="currentColor" strokeWidth="1.5" strokeDasharray={circ} strokeDashoffset={offset} strokeLinecap="round" className="text-violet-300" style={{ transform: "rotate(-90deg)", transformOrigin: "50% 50%", transition: "stroke-dashoffset 1s linear" }} />
   116→    </svg>

   266→/*  LinkifiedText – detects & renders clickable URLs                   */
   267→/* ------------------------------------------------------------------ */
   268→
   269→const URL_REGEX = /(https?:\/\/[^\s<]+)/g;
   270→
   271→type TextSegment = { type: "text"; value: string } | { type: "link"; value: string };
   272→
   273→function parseUrls(text: string): TextSegment[] {
   274→  const segments: TextSegment[] = [];
   275→  let lastIndex = 0;
   276→  let match: RegExpExecArray | null;
   277→  while ((match = URL_REGEX.exec(text)) !== null) {
   278→    if (match.index > lastIndex) {
   279→      segments.push({ type: "text", value: text.slice(lastIndex, match.index) });
   280→    }
   281→    segments.push({ type: "link", value: match[1] });
   282→    lastIndex = URL_REGEX.lastIndex;
   283→  }
   284→  if (lastIndex < text.length) {
   285→    segments.push({ type: "text", value: text.slice(lastIndex) });
   286→  }
   287→  return segments;
   288→}

   290→/** Split a plain-text string into highlighted / non-highlighted fragments */
   291→function highlightText(text: string, query: string): React.ReactNode[] {
   292→  if (!query) return [text];
   293→  const escaped = query.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
   294→  const re = new RegExp(`(${escaped})`, "gi");
   295→  const parts: React.ReactNode[] = [];
   296→  let lastIndex = 0;
   297→  let match: RegExpExecArray | null;
   298→  while ((match = re.exec(text)) !== null) {
   299→    if (match.index > lastIndex) {
   300→      parts.push(text.slice(lastIndex, match.index));
   301→    }
   302→    parts.push(
   303→      <mark key={match.index} className="bg-yellow-300/60 dark:bg-yellow-400/40 rounded-sm px-0.5">
   304→        {match[1]}
   305→      </mark>,
   306→    );
   307→    lastIndex = re.lastIndex;
   308→  }
   309→  if (lastIndex < text.length) {
   310→    parts.push(text.slice(lastIndex));
   311→  }
   312→  return parts;
   313→}
   314→
   315→export const LinkifiedText = React.memo(function LinkifiedText({ text, className, highlight }: { text: string; className?: string; highlight?: string }) {
   316→  const segments = useMemo(() => parseUrls(text), [text]);
   317→  return (
   318→    <span className={className}>
   319→      {segments.map((seg, i) =>
   320→        seg.type === "link" ? (
   321→          <a
   322→            key={i}
   323→            href={seg.value}
   324→            target="_blank"
   325→            rel="noopener noreferrer"
   326→            className="underline decoration-violet-400/50 underline-offset-2 hover:decoration-violet-400 transition-colors"
   327→            onClick={(e) => e.stopPropagation()}
   328→          >
   329→            {seg.value}
   330→          </a>
   331→        ) : (
   332→          <span key={i}>{highlight ? highlightText(seg.value, highlight) : seg.value}</span>
   333→        ),
   334→      )}
   335→    </span>
   336→  );
   337→});

   595→
   596→    {/* Image Lightbox */}
   597→    <AnimatePresence>
   598→      {showLightbox && (

   600→      )}
   601→    </AnimatePresence>

   610→function formatDateSeparator(dateStr: string): string {
   611→  const d = new Date(dateStr);
   612→  const now = new Date();
   613→  const today = new Date(now.getFullYear(), now.getMonth(), now.getDate());
   614→  const yesterday = new Date(today);
   615→  yesterday.setDate(yesterday.getDate() - 1);
   616→  const msgDay = new Date(d.getFullYear(), d.getMonth(), d.getDate());
   617→
   618→  if (msgDay.getTime() === today.getTime()) return "Today";
   619→  if (msgDay.getTime() === yesterday.getTime()) return "Yesterday";
   620→  return d.toLocaleDateString(undefined, { month: "short", day: "numeric" });
   621→}

   731→
   732→export function MessageFeed({ messages, isLoading, isPartnerTyping, currentUserId, partnerName, partnerAvatar, backgroundStyle, onMarkAsRead, onVanishMessage, onReplyTo, onDeleteMessage, onReact, scrollToMessageId, onScrolledToMessage, searchHighlight, onClearSearchHighlight, loadMore, hasMore, loadingMore }: MessageFeedProps) {
   733→  const parentRef = useRef<HTMLDivElement>(null);
   734→  const isAutoScrollRef = useRef(true);
   735→  const [showNewBtn, setShowNewBtn] = useState(false);
   736→  const unreadRef = useRef(0);
   737→  const prevCountRef = useRef(messages.length);
   738→  const prevFirstIdRef = useRef(messages.length > 0 ? messages[0].id : null);
   739→  const prevScrollHeightRef = useRef(0);
   740→
   741→  const totalItems = useMemo(() => messages.length + (isPartnerTyping ? 1 : 0), [messages.length, isPartnerTyping]);
   742→  const loadMoreRef = useRef(loadMore);
   743→  loadMoreRef.current = loadMore;
   744→
   745→  const virtualizer = useVirtualizer({ count: totalItems, getScrollElement: () => parentRef.current, estimateSize: () => ESTIMATED_ROW_HEIGHT, overscan: OVERSCAN, paddingStart: 16, paddingEnd: 16, gap: 12 });
   746→
   747→  const handleScroll = useCallback((e: UIEvent<HTMLDivElement>) => {
   748→    const el = e.currentTarget;
   749→    const dist = el.scrollHeight - el.scrollTop - el.clientHeight;
   750→    if (dist <= AUTO_SCROLL_THRESHOLD) { isAutoScrollRef.current = true; setShowNewBtn(false); unreadRef.current = 0; }
   751→    else { isAutoScrollRef.current = false; }
   752→    /* Trigger loadMore when scrolled near top */
   753→    if (el.scrollTop < 300 && hasMore && !loadingMore) {
   754→      loadMoreRef.current?.();
   755→    }
   756→  }, [hasMore, loadingMore]);
   757→
   758→  const scrollToBottom = useCallback((behavior: ScrollBehavior = "smooth") => {
   759→    const el = parentRef.current; if (!el) return;
   760→    el.scrollTo({ top: el.scrollHeight, behavior });
   761→    isAutoScrollRef.current = true; setShowNewBtn(false); unreadRef.current = 0;
   762→  }, []);
   763→
   764→  useEffect(() => { if (messages.length > 0) requestAnimationFrame(() => scrollToBottom("instant")); }, [isLoading, scrollToBottom]);
   765→
   766→  useEffect(() => { if (!onMarkAsRead) return; const unread = messages.filter((m) => m.sender_id !== currentUserId && m.status !== "read"); if (unread.length > 0 && isAutoScrollRef.current) onMarkAsRead(unread.map((m) => m.id)); }, [messages, currentUserId, onMarkAsRead]);
   767→
   768→  /* ---- Scroll to a specific message ----------------------------------- */
   769→  useEffect(() => {
   770→    if (!scrollToMessageId) return;
   771→    const idx = messages.findIndex((m) => m.id === scrollToMessageId);
   772→    if (idx === -1) { onScrolledToMessage?.(); return; }
   773→    virtualizer.scrollToIndex(idx, { align: "center" });
   774→    onScrolledToMessage?.();
   775→  }, [scrollToMessageId, messages, virtualizer, onScrolledToMessage]);
   776→
   777→  /* ---- Handle message count changes (new or prepended) ------------ */
   778→  useEffect(() => {
   779→    const prev = prevCountRef.current;
   780→    const prevFirstId = prevFirstIdRef.current;
   781→    prevCountRef.current = messages.length;
   782→    prevFirstIdRef.current = messages.length > 0 ? messages[0].id : null;
   783→
   784→    if (messages.length <= prev) return;
   785→
   786→    /* Check if older messages were prepended (first message changed) */
   787→    const firstId = messages[0]?.id;
   788→    const wasPrepended = firstId !== prevFirstId && prevFirstId !== null;
   789→
   790→    if (wasPrepended) {
   791→      /* Older messages prepended – preserve scroll position */
   792→      const el = parentRef.current;
   793→      if (el) {
   794→        const oldHeight = prevScrollHeightRef.current;
   795→        prevScrollHeightRef.current = el.scrollHeight;
   796→        requestAnimationFrame(() => {
   797→          const newHeight = el.scrollHeight;
   798→          el.scrollTop += newHeight - oldHeight;
   799→        });
   800→      }
   801→      return;
   802→    }
   803→
   804→    /* New messages appended at the end */
   805→    const last = messages[messages.length - 1];
   806→    const own = last && last.sender_id === currentUserId;
   807→    if (own) { requestAnimationFrame(() => scrollToBottom("instant")); return; }
   808→    if (isAutoScrollRef.current) requestAnimationFrame(() => scrollToBottom("smooth"));
   809→    else { unreadRef.current += messages.length - prev; setShowNewBtn(true); }
   810→  }, [messages.length, currentUserId, scrollToBottom]);
   811→
   812→  /* Track scroll height for scroll preservation */
   813→  useEffect(() => {
   814→    const el = parentRef.current;
   815→    if (!el) return;
   816→    prevScrollHeightRef.current = el.scrollHeight;
   817→  }, [messages.length]);
   818→
   819→  if (isLoading) return <MessageListSkeleton />;
   820→  if (messages.length === 0 && !isPartnerTyping) return <EmptyState partnerName={partnerName} partnerAvatar={partnerAvatar} />;
   821→
   822→  const virtualItems = virtualizer.getVirtualItems();
   823→
   824→  return (
   825→    <div className="relative h-full w-full overflow-hidden" style={backgroundStyle}>
   826→      <div ref={parentRef} onScroll={handleScroll} onClick={() => onClearSearchHighlight?.()} className="h-full w-full overflow-y-auto overscroll-contain" style={{ WebkitOverflowScrolling: "touch" }}>
   827→        <div style={{ height: `${virtualizer.getTotalSize()}px`, width: "100%", position: "relative" }}>
   828→          {/* Loading more spinner */}
   829→          {loadingMore && (
   830→            <div style={{ position: "absolute", top: 0, left: 0, width: "100%", height: "48px", zIndex: 5 }}>
   831→              <div className="flex items-center justify-center py-3">
   832→                <svg className="size-5 animate-spin text-violet-400" viewBox="0 0 24 24" fill="none" aria-label="Loading more messages">
   833→                  <circle className="opacity-25" cx="12" cy="12" r="10" stroke="currentColor" strokeWidth="4" />
   834→                  <path className="opacity-75" fill="currentColor" d="M4 12a8 8 0 018-8V0C5.373 0 0 5.373 0 12h4z" />
   835→                </svg>
   836→              </div>
   837→            </div>
   838→          )}
   839→          {virtualItems.map((vi) => {
   840→            const isTyping = vi.index === messages.length;
   841→            const msg = isTyping ? null : messages[vi.index];
   842→            // Check if we need a date separator before this message
   843→            const showDateSep = msg && vi.index > 0 && messages[vi.index - 1] && (() => {
   844→              const prev = new Date(messages[vi.index - 1].created_at);
   845→              const curr = new Date(msg.created_at);
   846→              return prev.toDateString() !== curr.toDateString();
   847→            })();
   848→            const showFirstDateSep = msg && vi.index === 0;
   849→            return (
   850→              <div key={vi.key} data-index={vi.index} ref={virtualizer.measureElement} style={{ position: "absolute", top: 0, left: 0, width: "100%", transform: `translateY(${vi.start}px)` }}>
   851→                {isTyping ? <TypingIndicator partnerName={partnerName} partnerAvatar={partnerAvatar} /> : msg ? (
   852→                  <>
   853→                    {(showDateSep || showFirstDateSep) && <DateSeparator date={msg.created_at} />}
   854→                    <MessageBubble message={msg} isOwn={msg.sender_id === currentUserId} partnerName={partnerName} partnerAvatar={partnerAvatar} onReplyTo={onReplyTo} onReact={onReact} onDeleteMessage={onDeleteMessage} searchHighlight={searchHighlight} />
   855→                  </>
   856→                ) : null}
   857→              </div>
   858→            );
   859→          })}
   860→        </div>
   861→      </div>
   862→      {showNewBtn && unreadRef.current > 0 && <NewMessagesButton count={unreadRef.current} onClick={() => scrollToBottom("smooth")} />}
   863→    </div>
   864→  );
   865→}
   866→
