    55→    55→
    56→    56→function useDoubleTap(
    57→    57→  onDoubleTap: () => void,
    58→    58→  delay = 300,
    59→    59→) {
    60→    60→  const lastTapRef = useRef(0);
    61→    61→  return useCallback(() => {
    62→    62→    const now = Date.now();
    63→    63→    if (now - lastTapRef.current < delay) {
    64→    64→      onDoubleTap();
    65→    65→      lastTapRef.current = 0; // reset
    66→    66→    } else {
    67→    67→      lastTapRef.current = now;
    68→    68→    }
    69→    69→  }, [onDoubleTap, delay]);
    70→    70→}

   145→   145→
   146→   146→function EphemeralTimer({
   147→   147→  seconds,
   148→   148→  onExpire,
   149→   149→}: {
   150→   150→  seconds: number;
   151→   151→  onExpire: () => void;
   152→   152→}) {
   153→   153→  const [remaining, setRemaining] = useState(seconds);
   154→   154→  const intervalRef = useRef<ReturnType<typeof setInterval> | undefined>(
   155→   155→    undefined,
   156→   156→  );
   157→   157→
   158→   158→  useEffect(() => {
   159→   159→    intervalRef.current = setInterval(() => {
   160→   160→      setRemaining((r) => {
   161→   161→        if (r <= 1) {
   162→   162→          clearInterval(intervalRef.current);
   163→   163→          onExpire();
   164→   164→          return 0;
   165→   165→        }
   166→   166→        return r - 1;
   167→   167→      });
   168→   168→    }, 1000);
   169→   169→    return () => clearInterval(intervalRef.current);
   170→   170→  }, [seconds, onExpire]);
   171→   171→
   172→   172→  const pct = remaining / seconds;
   173→   173→  const circ = 2 * Math.PI * 7;
   174→   174→  const offset = circ * (1 - pct);
   175→   175→
   176→   176→  return (
   177→   177→    <svg
   178→   178→      className="size-3.5 shrink-0 ephemeral-timer-pulse"
   179→   179→      viewBox="0 0 18 18"
   180→   180→      aria-label={`${remaining}s remaining`}
   181→   181→    >
   182→   182→      <circle
   183→   183→        cx="9"
   184→   184→        cy="9"
   185→   185→        r="7"
   186→   186→        fill="none"
   187→   187→        stroke="currentColor"
   188→   188→        strokeWidth="1.5"
   189→   189→        className="text-white/20"
   190→   190→      />
   191→   191→      <circle
   192→   192→        cx="9"
   193→   193→        cy="9"
   194→   194→        r="7"
   195→   195→        fill="none"
   196→   196→        stroke="currentColor"
   197→   197→        strokeWidth="1.5"
   198→   198→        strokeDasharray={circ}
   199→   199→        strokeDashoffset={offset}
   200→   200→        strokeLinecap="round"
   201→   201→        style={{
   202→   202→          color: "var(--app-accent-lighter)",
   203→   203→          transform: "rotate(-90deg)",
   204→   204→          transformOrigin: "50% 50%",
   205→   205→          transition: "stroke-dashoffset 1s linear",
   206→   206→        }}
   207→   207→      />
   208→   208→    </svg>
   209→   209→  );
   210→   210→}

   421→   421→/* ------------------------------------------------------------------ */
   422→   422→
   423→   423→function VoiceBubble({
   424→   424→  message,
   425→   425→  isOwn,
   426→   426→  waveform,
   427→   427→}: {
   428→   428→  message: ChatMessage;
   429→   429→  isOwn: boolean;
   430→   430→  waveform: number[];
   431→   431→}) {
   432→   432→  const audioRef = useRef<HTMLAudioElement>(null);
   433→   433→  const [playing, setPlaying] = useState(false);
   434→   434→  const [progress, setProgress] = useState(0);
   435→   435→  const [speed, setSpeed] = useState(1);
   436→   436→  const speeds = [1, 1.5, 2];
   437→   437→  const duration = message.voice_duration ?? 0;
   438→   438→
   439→   439→  useEffect(() => {
   440→   440→    const audio = audioRef.current;
   441→   441→    if (!audio) return;
   442→   442→    const onTime = () => {
   443→   443→      if (audio.duration) setProgress(audio.currentTime / audio.duration);
   444→   444→    };
   445→   445→    const onEnd = () => setPlaying(false);
   446→   446→    audio.addEventListener("timeupdate", onTime);
   447→   447→    audio.addEventListener("ended", onEnd);
   448→   448→    return () => {
   449→   449→      audio.removeEventListener("timeupdate", onTime);
   450→   450→      audio.removeEventListener("ended", onEnd);
   451→   451→    };
   452→   452→  }, []);
   453→   453→
   454→   454→  const toggle = useCallback(() => {
   455→   455→    const audio = audioRef.current;
   456→   456→    if (!audio) return;
   457→   457→    if (playing) {
   458→   458→      audio.pause();
   459→   459→      setPlaying(false);
   460→   460→    } else {
   461→   461→      audio.playbackRate = speed;
   462→   462→      audio.play().catch(() => {});
   463→   463→      setPlaying(true);
   464→   464→    }
   465→   465→  }, [playing, speed]);
   466→   466→
   467→   467→  const cycleSpeed = useCallback(() => {
   468→   468→    setSpeed((s) => {
   469→   469→      const idx = speeds.indexOf(s);
   470→   470→      const next = speeds[(idx + 1) % speeds.length];
   471→   471→      if (audioRef.current) audioRef.current.playbackRate = next;
   472→   472→      return next;
   473→   473→    });
   474→   474→  }, []);
   475→   475→
   476→   476→  /* Build waveform bars */
   477→   477→  const bars = useMemo(() => {
   478→   478→    const count = Math.min(waveform.length, 40);
   479→   479→    const step = Math.max(1, Math.floor(waveform.length / count));
   480→   480→    return Array.from({ length: count }, (_, i) => waveform[i * step] ?? 0.1);
   481→   481→  }, [waveform]);
   482→   482→
   483→   483→  const progressIdx = Math.floor(progress * bars.length);
   484→   484→
   485→   485→  return (
   486→   486→    <div className={cn("flex items-center gap-3", isOwn ? "flex-row-reverse" : "")}>
   487→   487→      <button
   488→   488→        type="button"
   489→   489→        onClick={toggle}
   490→   490→        className={cn(
   491→   491→          "flex size-9 shrink-0 items-center justify-center rounded-full transition-all duration-150 active:scale-95",
   492→   492→          isOwn
   493→   493→            ? "bg-white/20 hover:bg-white/30"
   494→   494→            : "bg-zinc-600 hover:bg-[var(--app-accent-subtle)]",
   495→   495→        )}
   496→   496→        aria-label={playing ? "Pause" : "Play"}

   547→   547→/*  LinkifiedText – detects & renders clickable URLs                   */

   549→   549→
   550→   550→const URL_REGEX = /(https?:\/\/[^\s<]+)/g;
   551→   551→
   552→   552→type TextSegment =
   553→   553→  | { type: "text"; value: string }
   554→   554→  | { type: "link"; value: string };
   555→   555→
   556→   556→function parseUrls(text: string): TextSegment[] {
   557→   557→  const segments: TextSegment[] = [];
   558→   558→  let lastIndex = 0;
   559→   559→  let match: RegExpExecArray | null;
   560→   560→  while ((match = URL_REGEX.exec(text)) !== null) {
   561→   561→    if (match.index > lastIndex) {
   562→   562→      segments.push({ type: "text", value: text.slice(lastIndex, match.index) });
   563→   563→    }
   564→   564→    segments.push({ type: "link", value: match[1] });
   565→   565→    lastIndex = URL_REGEX.lastIndex;
   566→   566→  }
   567→   567→  if (lastIndex < text.length) {
   568→   568→    segments.push({ type: "text", value: text.slice(lastIndex) });
   569→   569→  }
   570→   570→  return segments;
   571→   571→}

   573→   573→/** Split a plain-text string into highlighted / non-highlighted fragments */
   574→   574→function highlightText(text: string, query: string): React.ReactNode[] {
   575→   575→  if (!query) return [text];
   576→   576→  const escaped = query.replace(/[.*+?^${}()|[\]\\]/g, "\\");
   577→   577→  const re = new RegExp(`(${escaped})`, "gi");
   578→   578→  const parts: React.ReactNode[] = [];
   579→   579→  let lastIndex = 0;
   580→   580→  let match: RegExpExecArray | null;
   581→   581→  while ((match = re.exec(text)) !== null) {
   582→   582→    if (match.index > lastIndex) {
   583→   583→      parts.push(text.slice(lastIndex, match.index));
   584→   584→    }
   585→   585→    parts.push(
   586→   586→      <mark
   587→   587→        key={match.index}
   588→   588→        className="bg-yellow-300/60 dark:bg-yellow-400/40 rounded-sm px-0.5"
   589→   589→      >
   590→   590→        {match[1]}
   591→   591→      </mark>,
   592→   592→    );
   593→   593→    lastIndex = re.lastIndex;
   594→   594→  }
   595→   595→  if (lastIndex < text.length) {
   596→   596→    parts.push(text.slice(lastIndex));
   597→   597→  }
   598→   598→  return parts;
   599→   599→}
