    38→    38→function formatLastSeen(lastSeen: string | null): string {
    39→    39→  if (!lastSeen) return "Offline";
    40→    40→  const diffMs = Date.now() - new Date(lastSeen).getTime();
    41→    41→  const diffSec = Math.floor(diffMs / 1000);
    42→    42→  if (diffSec < 60) return "Seen just now";
    43→    43→  const diffMin = Math.floor(diffSec / 60);
    44→    44→  if (diffMin < 60) return `Seen ${diffMin}m ago`;
    45→    45→  const diffHr = Math.floor(diffMin / 60);
    46→    46→  if (diffHr < 24) return `Seen ${diffHr}h ago`;
    47→    47→  const diffDay = Math.floor(diffHr / 24);
    48→    48→  if (diffDay < 7) return `Seen ${diffDay}d ago`;
    49→    49→  return `Seen ${new Date(lastSeen).toLocaleDateString()}`;
    50→    50→}
