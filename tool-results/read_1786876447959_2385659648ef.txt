    47→function formatFileSize(bytes: number): string {
    48→  if (bytes < 1024) return `${bytes} B`;
    49→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
    50→  return `${(bytes / (1024 * 1024)).toFixed(1)} MB`;
    51→}
    52→
    53→function formatLastSeen(lastSeen: string | null): string {
    54→  if (!lastSeen) return "Offline";
    55→  const diffMs = Date.now() - new Date(lastSeen).getTime();
    56→  const diffSec = Math.floor(diffMs / 1000);
    57→  if (diffSec < 60) return "Seen just now";
    58→  const diffMin = Math.floor(diffSec / 60);
    59→  if (diffMin < 60) return `Seen ${diffMin}m ago`;
    60→  const diffHr = Math.floor(diffMin / 60);
    61→  if (diffHr < 24) return `Seen ${diffHr}h ago`;
    62→  const diffDay = Math.floor(diffHr / 24);
    63→  if (diffDay < 7) return `Seen ${diffDay}d ago`;
    64→  return `Seen ${new Date(lastSeen).toLocaleDateString()}`;
    65→}

    83→function SearchPanel({
    84→  conversationId,
    85→  onResultClick,
    86→  onClose,
    87→}: {
    88→  conversationId: string;
    89→  onResultClick: (messageId: string) => void;
    90→  onClose: () => void;
    91→}) {
    92→  const [query, setQuery] = useState("");
    93→  const [results, setResults] = useState<ChatMessage[]>([]);
    94→  const [isSearching, setIsSearching] = useState(false);
    95→  const inputRef = useRef<HTMLInputElement>(null);
    96→  const debounceRef = useRef<ReturnType<typeof setTimeout> | undefined>(undefined);
    97→
    98→  /* Auto-focus on open */
    99→  useEffect(() => {
   100→    requestAnimationFrame(() => inputRef.current?.focus());
   101→  }, []);
   102→
   103→  /* Escape key to close */
   104→  useEffect(() => {
   105→    function handleKeyDown(e: KeyboardEvent) {
   106→      if (e.key === "Escape") onClose();
   107→    }
   108→    window.addEventListener("keydown", handleKeyDown);
   109→    return () => window.removeEventListener("keydown", handleKeyDown);
   110→  }, [onClose]);
   111→
   112→  /* Debounced search */
   113→  useEffect(() => {
   114→    if (debounceRef.current) clearTimeout(debounceRef.current);
   115→    if (!query.trim()) { setResults([]); setIsSearching(false); return; }
   116→    setIsSearching(true);
   117→    debounceRef.current = setTimeout(async () => {
   118→      try {
   119→        const res = await fetch(`/api/messages/search?conversation_id=${conversationId}&q=${encodeURIComponent(query)}`);
   120→        if (res.ok) {
   121→          const data: ChatMessage[] = await res.json();
   122→          setResults(data);
   123→        }
   124→      } catch (_err) {
   125→        /* silent */
   126→      } finally {
   127→        setIsSearching(false);
   128→      }
   129→    }, 300);
   130→    return () => { if (debounceRef.current) clearTimeout(debounceRef.current); };
   131→  }, [query, conversationId]);
   132→
   133→  return (
   134→    <div
   135→      className="overflow-hidden border-b bg-background/80 backdrop-blur-xl"
   136→    >
   137→      <div className="flex items-center gap-2 px-4 py-2.5">
   138→        <Search className="size-4 shrink-0 text-muted-foreground" />
   139→        <input
   140→          ref={inputRef}
   141→          type="text"
   142→          value={query}
   143→          onChange={(e) => setQuery(e.target.value)}
   144→          placeholder="Search messages..."
   145→          className="flex-1 bg-transparent text-sm outline-none placeholder:text-muted-foreground"
   146→          aria-label="Search messages"
   147→        />
   148→        {isSearching ? (
   149→          <Loader2 className="size-4 shrink-0 animate-spin text-muted-foreground" />
   150→        ) : (
   151→          <button
   152→            type="button"
   153→            onClick={() => { setQuery(""); onClose(); }}
   154→            className="flex size-6 items-center justify-center rounded-full text-muted-foreground transition-colors hover:bg-muted"
   155→            aria-label="Close search"
   156→          >
   157→            <X className="size-3.5" />
   158→          </button>
   159→        )}
   160→      </div>
   161→
   162→      {/* Results list */}
   163→      {results.length > 0 && (
   164→        <div className="max-h-[200px] overflow-y-auto border-t px-2 pb-2">
   165→          {results.map((msg) => (
   166→            <button
   167→              key={msg.id}
   168→              type="button"
   169→              onClick={() => onResultClick(msg.id)}
   170→              className="flex w-full flex-col gap-0.5 rounded-lg px-3 py-2.5 text-left transition-colors hover:bg-muted/50"
   171→            >
   172→              <p className="truncate text-sm text-foreground">
   173→                {msg.content.length > 60 ? msg.content.slice(0, 60) + "..." : msg.content}
   174→              </p>
   175→              <span className="text-[11px] text-muted-foreground">
   176→                {formatDistanceToNow(new Date(msg.created_at), { addSuffix: true })}
   177→              </span>
   178→            </button>
   179→          ))}
   180→        </div>
   181→      )}
   182→
   183→      {query.trim() && !isSearching && results.length === 0 && (
   184→        <div className="border-t px-4 py-3 text-center text-xs text-muted-foreground">
   185→          No results found
   186→        </div>
   187→      )}
   188→    </div>
   189→  );
   190→}

   350→export function ChatLayout({ currentUserId, otherUserId, conversationId, partner, initialMessages }: ChatLayoutProps) {
   351→  const { messages, isLoading, isPartnerTyping, partnerStatus, sendMessage, onTyping, markAsRead, vanishMessage, deleteMessage, sendReaction } =
   352→    useChat({ currentUserId, otherUserId, conversationId, initialMessages });
   353→
   354→  const [draft, setDraft] = useState("");
   355→  const [sendBtnKey, setSendBtnKey] = useState(0);
   356→  const [isSending, setIsSending] = useState(false);
   357→  const [vanishMode, setVanishMode] = useState(false);
   358→  const [replyTo, setReplyTo] = useState<ChatMessage | null>(null);
   359→  const [ephemeralOpen, setEphemeralOpen] = useState(false);
   360→  const [ephemeralSeconds, setEphemeralSeconds] = useState<number | null>(null);
   361→  const [infoPanelOpen, setInfoPanelOpen] = useState(false);
   362→  const [searchOpen, setSearchOpen] = useState(false);
   363→  const [scrollToMessageId, setScrollToMessageId] = useState<string | null>(null);
   364→  const inputRef = useRef<HTMLTextAreaElement>(null);
   365→  const viewport = useVisualViewport();
   366→
   367→  /* Voice */
   368→  const voice = useVoiceRecorder();
   369→  const [showVoiceWaveform, setShowVoiceWaveform] = useState(false);
   370→
   371→  /* Attachment */
   372→  const [attachmentFile, setAttachmentFile] = useState<File | null>(null);
   373→  const [attachmentPreview, setAttachmentPreview] = useState<string | null>(null);
   374→  const [isUploading, setIsUploading] = useState(false);
   375→  const fileInputRef = useRef<HTMLInputElement>(null);
   376→
   377→  const containerStyle = viewport.height > 0 ? ({ height: `${viewport.height}px` } as React.CSSProperties) : ({ height: "100dvh" } as React.CSSProperties);
   378→  const kbOffset = viewport.isKeyboardVisible && viewport.offsetTop > 0 ? ({ marginTop: `-${viewport.offsetTop}px` } as React.CSSProperties) : undefined;
   379→
   380→  useEffect(() => { inputRef.current?.focus(); }, []);
   381→
   382→  /* Show voice waveform when recording */
   383→  useEffect(() => { setShowVoiceWaveform(voice.isRecording); }, [voice.isRecording]);
   384→
   385→  /* Global keyboard shortcuts */
   386→  useEffect(() => {
   387→    function handleGlobalKey(e: KeyboardEvent) {
   388→      if (e.key === "Escape") {
   389→        if (replyTo) { setReplyTo(null); e.preventDefault(); }
   390→        else if (ephemeralOpen) { setEphemeralOpen(false); e.preventDefault(); }
   391→        else if (infoPanelOpen) { setInfoPanelOpen(false); e.preventDefault(); }
   392→        else if (searchOpen) { setSearchOpen(false); e.preventDefault(); }
   393→      }
   394→    }
   395→    window.addEventListener("keydown", handleGlobalKey);
   396→    return () => window.removeEventListener("keydown", handleGlobalKey);
   397→  }, [replyTo, ephemeralOpen, infoPanelOpen, searchOpen]);
   398→
   399→  const handleSubmit = useCallback(async (e: FormEvent) => {
   400→    e.preventDefault();
   401→    /* If an attachment is selected, upload it instead of sending text */
   402→    if (hasAttachment && !hasText) {
   403→      await uploadAttachment();
   404→      return;
   405→    }
   406→    /* If there's both text and an attachment, send text first, then upload */
   407→    const trimmed = draft.trim();
   408→    if (!trimmed || isSending) return;
   409→    setIsSending(true);
   410→    const opts: SendMessageOptions = { vanish_mode: vanishMode, ephemeral_seconds: ephemeralSeconds, reply_to_id: replyTo?.id ?? null };
   411→    setDraft(""); setReplyTo(null);
   412→    if (inputRef.current) inputRef.current.style.height = "auto";
   413→    try {
   414→      await sendMessage(trimmed, opts);
   415→      setSendBtnKey((k) => k + 1);
   416→      /* Upload attachment after text is sent */
   417→      if (hasAttachment) await uploadAttachment();
   418→    }
   419→    finally { setIsSending(false); requestAnimationFrame(() => inputRef.current?.focus()); }
   420→  }, [draft, isSending, sendMessage, vanishMode, ephemeralSeconds, replyTo, hasAttachment, uploadAttachment]);
   421→
   422→  const handleVoiceSend = useCallback(async () => {
   423→    const result = await voice.stopRecording();
   424→    if (!result || result.duration < 1) return;
   425→    const blob = result.blob;
   426→    const url = URL.createObjectURL(blob);
   427→    await sendMessage(url, { type: "voice", voice_duration: result.duration, waveform_data: result.amplitudes, ephemeral_seconds: ephemeralSeconds });
   428→    URL.revokeObjectURL(url);
   429→  }, [voice, sendMessage, ephemeralSeconds]);
   430→
   431→  const handleKeyDown = useCallback((e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   432→    if (e.key === "Enter" && !e.shiftKey) { e.preventDefault(); const form = e.currentTarget.closest("form"); if (form) form.requestSubmit(); }
   433→  }, []);
   434→
   435→  const handleInputChange = useCallback((e: React.ChangeEvent<HTMLTextAreaElement>) => { setDraft(e.target.value); onTyping(); }, [onTyping]);
   436→  const handleInput = useCallback((e: React.FormEvent<HTMLTextAreaElement>) => { const el = e.currentTarget; el.style.height = "auto"; el.style.height = `${Math.min(el.scrollHeight, 128)}px`; }, []);
   437→
   438→  const hasText = draft.trim().length > 0;
   439→  const canSend = hasText || hasAttachment;
   440→
   441→  /* Search result click → scroll to message */
   442→  const handleSearchResultClick = useCallback((messageId: string) => {
   443→    setScrollToMessageId(messageId);
   444→    setSearchOpen(false);
   445→  }, []);
   446→
   447→  const handleScrolledToMessage = useCallback(() => {
   448→    setScrollToMessageId(null);
   449→  }, []);
   450→
   451→  /* ---- Attachment handlers ---------------------------------------- */
   452→  const handleFileSelect = useCallback((e: ChangeEvent<HTMLInputElement>) => {
   453→    const file = e.target.files?.[0] ?? null;
   454→    if (!file) return;
   455→    if (file.size > MAX_ATTACHMENT_SIZE) {
   456→      toast({ title: "File too large", description: "Maximum size is 10 MB.", variant: "destructive" });
   457→      e.target.value = "";
   458→      return;
   459→    }
   460→    setAttachmentFile(file);
   461→    /* Generate thumbnail preview for images */
   462→    if (IMAGE_MIME_TYPES.has(file.type)) {
   463→      const url = URL.createObjectURL(file);
   464→      setAttachmentPreview(url);
   465→    } else {
   466→      setAttachmentPreview(null);
   467→    }
   468→    /* Reset input so the same file can be re-selected */
   469→    e.target.value = "";
   470→  }, []);
   471→
   472→  const clearAttachment = useCallback(() => {
   473→    if (attachmentPreview) URL.revokeObjectURL(attachmentPreview);
   474→    setAttachmentFile(null);
   475→    setAttachmentPreview(null);
   476→  }, [attachmentPreview]);
   477→
   478→  const uploadAttachment = useCallback(async () => {
   479→    if (!attachmentFile || isUploading) return;
   480→    setIsUploading(true);
   481→    try {
   482→      const fd = new FormData();
   483→      fd.append("file", attachmentFile);
   484→      fd.append("conversation_id", conversationId);
   485→      if (replyTo?.id) fd.append("reply_to_id", replyTo.id);
   486→      if (vanishMode) fd.append("vanish_mode", "true");
   487→      if (ephemeralSeconds) fd.append("ephemeral_seconds", String(ephemeralSeconds));
   488→
   489→      const res = await fetch("/api/messages/upload", { method: "POST", body: fd });
   490→      if (!res.ok) {
   491→        const data = await res.json().catch(() => ({ error: "Upload failed" }));
   492→        throw new Error(data.error || `Upload failed: ${res.status}`);
   493→      }
   494→      /* Message is inserted server-side and pushed via Pusher;
   495→         the useChat hook will pick it up automatically. */
   496→      clearAttachment();
   497→      setReplyTo(null);
   498→    } catch (err) {
   499→      console.error("[ChatLayout] upload error:", err);
   500→      toast({
   501→        title: "Upload failed",
   502→        description: err instanceof Error ? err.message : "Could not send the file.",
   503→        variant: "destructive",
   504→      });
   505→    } finally {
   506→      setIsUploading(false);
   507→    }
   508→  }, [attachmentFile, isUploading, conversationId, replyTo, vanishMode, ephemeralSeconds, clearAttachment]);
   509→
   510→  const handlePaperclipClick = useCallback(() => {
   511→    fileInputRef.current?.click();
   512→  }, []);
   513→
   514→  const hasAttachment = attachmentFile !== null;
   515→
   516→  /* Status text for header */
   517→  const statusText = isPartnerTyping
   518→    ? "typing\u2026"
   519→    : partnerStatus.online
   520→      ? "Online"
   521→      : formatLastSeen(partnerStatus.last_seen);
   522→
   523→  /* ---- Render --------------------------------------------------- */
   524→
