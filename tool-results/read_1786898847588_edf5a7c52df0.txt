    46→    46→function formatFileSize(bytes: number): string {
    47→    47→  if (bytes < 1024) return `${bytes} B`;
    48→    48→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
    49→    49→  return `${(bytes / (1024 * 1024)).toFixed(1)} MB`;
    50→    50→}
    51→    51→
    52→    52→function formatLastSeen(lastSeen: string | null): string {
    53→    53→  if (!lastSeen) return "Offline";
    54→    54→  const diffMs = Date.now() - new Date(lastSeen).getTime();
    55→    55→  const diffSec = Math.floor(diffMs / 1000);
    56→    56→  if (diffSec < 60) return "Seen just now";
    57→    57→  const diffMin = Math.floor(diffSec / 60);
    58→    58→  if (diffMin < 60) return `Seen ${diffMin}m ago`;
    59→    59→  const diffHr = Math.floor(diffMin / 60);
    60→    60→  if (diffHr < 24) return `Seen ${diffHr}h ago`;
    61→    61→  const diffDay = Math.floor(diffHr / 24);
    62→    62→  if (diffDay < 7) return `Seen ${diffDay}d ago`;
    63→    63→  return `Seen ${new Date(lastSeen).toLocaleDateString()}`;
    64→    64→}

   400→   400→    e.target.value = "";

   403→   403→  const handlePaperclipClick = useCallback(() => {

   415→   415→
