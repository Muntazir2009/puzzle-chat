    38→function formatLastSeen(lastSeen: string | null): string {
    39→  if (!lastSeen) return "Offline";
    40→  const diffMs = Date.now() - new Date(lastSeen).getTime();
    41→  const diffSec = Math.floor(diffMs / 1000);
    42→  if (diffSec < 60) return "Seen just now";
    43→  const diffMin = Math.floor(diffSec / 60);
    44→  if (diffMin < 60) return `Seen ${diffMin}m ago`;
    45→  const diffHr = Math.floor(diffMin / 60);
    46→  if (diffHr < 24) return `Seen ${diffHr}h ago`;
    47→  const diffDay = Math.floor(diffHr / 24);
    48→  if (diffDay < 7) return `Seen ${diffDay}d ago`;
    49→  return `Seen ${new Date(lastSeen).toLocaleDateString()}`;
    50→}

    68→function SearchPanel({
    69→  conversationId,
    70→  onResultClick,
    71→  onClose,
    72→}: {
    73→  conversationId: string;
    74→  onResultClick: (messageId: string) => void;
    75→  onClose: () => void;
    76→}) {
    77→  const [query, setQuery] = useState("");
    78→  const [results, setResults] = useState<ChatMessage[]>([]);
    79→  const [isSearching, setIsSearching] = useState(false);
    80→  const inputRef = useRef<HTMLInputElement>(null);
    81→  const debounceRef = useRef<ReturnType<typeof setTimeout> | undefined>(undefined);
    82→
    83→  /* Auto-focus on open */
    84→  useEffect(() => {
    85→    requestAnimationFrame(() => inputRef.current?.focus());
    86→  }, []);
    87→
    88→  /* Escape key to close */
    89→  useEffect(() => {
    90→    function handleKeyDown(e: KeyboardEvent) {
    91→      if (e.key === "Escape") onClose();
    92→    }
    93→    window.addEventListener("keydown", handleKeyDown);
    94→    return () => window.removeEventListener("keydown", handleKeyDown);
    95→  }, [onClose]);
    96→
    97→  /* Debounced search */
    98→  useEffect(() => {
    99→    if (debounceRef.current) clearTimeout(debounceRef.current);
   100→    if (!query.trim()) { setResults([]); setIsSearching(false); return; }
   101→    setIsSearching(true);
   102→    debounceRef.current = setTimeout(async () => {
   103→      try {
   104→        const res = await fetch(`/api/messages/search?conversation_id=${conversationId}&q=${encodeURIComponent(query)}`);
   105→        if (res.ok) {
   106→          const data: ChatMessage[] = await res.json();
   107→          setResults(data);
   108→        }
   109→      } catch (_err) {
   110→        /* silent */
   111→      } finally {
   112→        setIsSearching(false);
   113→      }
   114→    }, 300);
   115→    return () => { if (debounceRef.current) clearTimeout(debounceRef.current); };
   116→  }, [query, conversationId]);
   117→
   118→  return (
   119→    <div
   120→      className="overflow-hidden border-b bg-background/80 backdrop-blur-xl"
   121→    >
   122→      <div className="flex items-center gap-2 px-4 py-2.5">
   123→        <Search className="size-4 shrink-0 text-muted-foreground" />
   124→        <input
   125→          ref={inputRef}
   126→          type="text"
   127→          value={query}
   128→          onChange={(e) => setQuery(e.target.value)}
   129→          placeholder="Search messages..."
   130→          className="flex-1 bg-transparent text-sm outline-none placeholder:text-muted-foreground"
   131→          aria-label="Search messages"
   132→        />
   133→        {isSearching ? (
   134→          <Loader2 className="size-4 shrink-0 animate-spin text-muted-foreground" />
   135→        ) : (
   136→          <button
   137→            type="button"
   138→            onClick={() => { setQuery(""); onClose(); }}
   139→            className="flex size-6 items-center justify-center rounded-full text-muted-foreground transition-colors hover:bg-muted"
   140→            aria-label="Close search"
   141→          >
   142→            <X className="size-3.5" />
   143→          </button>
   144→        )}
   145→      </div>
   146→
   147→      {/* Results list */}
   148→      {results.length > 0 && (
   149→        <div className="max-h-[200px] overflow-y-auto border-t px-2 pb-2">
   150→          {results.map((msg) => (
   151→            <button
   152→              key={msg.id}
   153→              type="button"
   154→              onClick={() => onResultClick(msg.id)}
   155→              className="flex w-full flex-col gap-0.5 rounded-lg px-3 py-2.5 text-left transition-colors hover:bg-muted/50"
   156→            >
   157→              <p className="truncate text-sm text-foreground">
   158→                {msg.content.length > 60 ? msg.content.slice(0, 60) + "..." : msg.content}
   159→              </p>
   160→              <span className="text-[11px] text-muted-foreground">
   161→                {formatDistanceToNow(new Date(msg.created_at), { addSuffix: true })}
   162→              </span>
   163→            </button>
   164→          ))}
   165→        </div>
   166→      )}
   167→
   168→      {query.trim() && !isSearching && results.length === 0 && (
   169→        <div className="border-t px-4 py-3 text-center text-xs text-muted-foreground">
   170→          No results found
   171→        </div>
   172→      )}
   173→    </div>
   174→  );
   175→}

   334→
   335→export function ChatLayout({ currentUserId, otherUserId, conversationId, partner, initialMessages }: ChatLayoutProps) {
   336→  const { messages, isLoading, isPartnerTyping, partnerStatus, sendMessage, onTyping, markAsRead, vanishMessage, deleteMessage, sendReaction } =
   337→    useChat({ currentUserId, otherUserId, conversationId, initialMessages });
   338→
   339→  const [draft, setDraft] = useState("");
   340→  const [sendBtnKey, setSendBtnKey] = useState(0);
   341→  const [isSending, setIsSending] = useState(false);
   342→  const [vanishMode, setVanishMode] = useState(false);
   343→  const [replyTo, setReplyTo] = useState<ChatMessage | null>(null);
   344→  const [ephemeralOpen, setEphemeralOpen] = useState(false);
   345→  const [ephemeralSeconds, setEphemeralSeconds] = useState<number | null>(null);
   346→  const [infoPanelOpen, setInfoPanelOpen] = useState(false);
   347→  const [searchOpen, setSearchOpen] = useState(false);
   348→  const [scrollToMessageId, setScrollToMessageId] = useState<string | null>(null);
   349→  const inputRef = useRef<HTMLTextAreaElement>(null);
   350→  const viewport = useVisualViewport();
   351→
   352→  /* Voice */
   353→  const voice = useVoiceRecorder();
   354→  const [showVoiceWaveform, setShowVoiceWaveform] = useState(false);
   355→
   356→  const containerStyle = viewport.height > 0 ? ({ height: `${viewport.height}px` } as React.CSSProperties) : ({ height: "100dvh" } as React.CSSProperties);
   357→  const kbOffset = viewport.isKeyboardVisible && viewport.offsetTop > 0 ? ({ marginTop: `-${viewport.offsetTop}px` } as React.CSSProperties) : undefined;
   358→
   359→  useEffect(() => { inputRef.current?.focus(); }, []);
   360→
   361→  /* Show voice waveform when recording */
   362→  useEffect(() => { setShowVoiceWaveform(voice.isRecording); }, [voice.isRecording]);
   363→
   364→  /* Global keyboard shortcuts */
   365→  useEffect(() => {
   366→    function handleGlobalKey(e: KeyboardEvent) {
   367→      if (e.key === "Escape") {
   368→        if (replyTo) { setReplyTo(null); e.preventDefault(); }
   369→        else if (ephemeralOpen) { setEphemeralOpen(false); e.preventDefault(); }
   370→        else if (infoPanelOpen) { setInfoPanelOpen(false); e.preventDefault(); }
   371→        else if (searchOpen) { setSearchOpen(false); e.preventDefault(); }
   372→      }
   373→    }
   374→    window.addEventListener("keydown", handleGlobalKey);
   375→    return () => window.removeEventListener("keydown", handleGlobalKey);
   376→  }, [replyTo, ephemeralOpen, infoPanelOpen, searchOpen]);
   377→
   378→  const handleSubmit = useCallback(async (e: FormEvent) => {
   379→    e.preventDefault();
   380→    const trimmed = draft.trim();
   381→    if (!trimmed || isSending) return;
   382→    setIsSending(true);
   383→    const opts: SendMessageOptions = { vanish_mode: vanishMode, ephemeral_seconds: ephemeralSeconds, reply_to_id: replyTo?.id ?? null };
   384→    setDraft(""); setReplyTo(null);
   385→    if (inputRef.current) inputRef.current.style.height = "auto";
   386→    try { await sendMessage(trimmed, opts); setSendBtnKey((k) => k + 1); }
   387→    finally { setIsSending(false); requestAnimationFrame(() => inputRef.current?.focus()); }
   388→  }, [draft, isSending, sendMessage, vanishMode, ephemeralSeconds, replyTo]);
   389→
   390→  const handleVoiceSend = useCallback(async () => {
   391→    const result = await voice.stopRecording();
   392→    if (!result || result.duration < 1) return;
   393→    const blob = result.blob;
   394→    const url = URL.createObjectURL(blob);
   395→    await sendMessage(url, { type: "voice", voice_duration: result.duration, waveform_data: result.amplitudes, ephemeral_seconds: ephemeralSeconds });
   396→    URL.revokeObjectURL(url);
   397→  }, [voice, sendMessage, ephemeralSeconds]);
   398→
   399→  const handleKeyDown = useCallback((e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   400→    if (e.key === "Enter" && !e.shiftKey) { e.preventDefault(); const form = e.currentTarget.closest("form"); if (form) form.requestSubmit(); }
   401→  }, []);
   402→
   403→  const handleInputChange = useCallback((e: React.ChangeEvent<HTMLTextAreaElement>) => { setDraft(e.target.value); onTyping(); }, [onTyping]);
   404→  const handleInput = useCallback((e: React.FormEvent<HTMLTextAreaElement>) => { const el = e.currentTarget; el.style.height = "auto"; el.style.height = `${Math.min(el.scrollHeight, 128)}px`; }, []);
   405→
   406→  const hasText = draft.trim().length > 0;
   407→
   408→  /* Search result click → scroll to message */
   409→  const handleSearchResultClick = useCallback((messageId: string) => {
   410→    setScrollToMessageId(messageId);
   411→    setSearchOpen(false);
   412→  }, []);
   413→
   414→  const handleScrolledToMessage = useCallback(() => {
   415→    setScrollToMessageId(null);
   416→  }, []);
   417→
   418→  /* Status text for header */
   419→  const statusText = isPartnerTyping
   420→    ? "typing\u2026"
   421→    : partnerStatus.online
   422→      ? "Online"
   423→      : formatLastSeen(partnerStatus.last_seen);
   424→
   425→  /* ---- Render --------------------------------------------------- */
   426→
   427→  return (
   428→    <div className="flex w-full flex-col bg-background" style={containerStyle}>
   429→      <div style={kbOffset} className="flex min-h-0 flex-1 flex-col">
   430→        {/* Header - sticky with glassmorphic blur + shadow */}
   431→        <header className="sticky top-0 z-30 flex h-14 shrink-0 items-center gap-3 border-b border-border/40 bg-background/80 shadow-[0_1px_3px_rgba(0,0,0,0.04)] backdrop-blur-2xl supports-[backdrop-filter]:bg-background/60 px-4">
   432→          <button
   433→            type="button"
   434→            onClick={() => setInfoPanelOpen(true)}
   435→            className="flex items-center gap-3 rounded-lg p-1 -ml-1 transition-colors hover:bg-muted/50 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-violet-400/50"
   436→            aria-label="Open user info"
   437→          >
   438→            <Avatar className="size-9 ring-2 ring-violet-100 dark:ring-violet-900/50">
   439→              {partner.avatar_url && <AvatarImage src={partner.avatar_url} alt={partner.name} />}
   440→              <AvatarFallback className="bg-gradient-to-br from-violet-500 to-purple-600 text-xs font-semibold text-white">{partner.name.slice(0, 2).toUpperCase()}</AvatarFallback>
   441→            </Avatar>
   442→            <div className="flex flex-col">
   443→              <span className="text-sm font-semibold leading-tight">{partner.name}</span>
   444→              <div className="flex items-center gap-1.5">
   445→                <StatusDot online={partnerStatus.online} />
   446→                <span className={cn(
   447→                  "text-xs",
   448→                  isPartnerTyping ? "text-violet-500" : "text-muted-foreground",
   449→                )}>{statusText}</span>
   450→              </div>
   451→            </div>
   452→          </button>
   453→
   454→          {/* Search icon button - right side of header */}
   455→          <div className="ml-auto">
   456→            <button
   457→              type="button"
   458→              onClick={() => setSearchOpen((v) => !v)}
   459→              className={cn(
   460→                "flex size-9 items-center justify-center rounded-lg transition-colors",
   461→                searchOpen
   462→                  ? "bg-violet-500/15 text-violet-500"
   463→                  : "text-muted-foreground hover:bg-muted",
   464→              )}
   465→              aria-label={searchOpen ? "Close search" : "Search messages"}
   466→            >
   467→              <Search className="size-4" />
   468→            </button>
   469→          </div>
   470→        </header>
   471→
   472→        {/* Search panel (animated slide-down) */}
   473→        <AnimatePresence>
   474→          {searchOpen && (
   475→            <SearchPanel
   476→              conversationId={conversationId}
   477→              onResultClick={handleSearchResultClick}
   478→              onClose={() => setSearchOpen(false)}
   479→            />
   480→          )}
   481→        </AnimatePresence>
   482→
   483→        {/* Feed with dot pattern background */}
   484→        <div className="relative min-h-0 flex-1">
   485→          {/* Subtle dot pattern behind messages */}
   486→          <div
   487→            className="pointer-events-none absolute inset-0"
   488→            style={{
   489→              backgroundImage: DOT_PATTERN,
   490→              backgroundRepeat: "repeat",
   491→              backgroundSize: "20px 20px",
   492→            }}
   493→            aria-hidden="true"
   494→          />
   495→          <MessageFeed
   496→            messages={messages}
   497→            isLoading={isLoading}
   498→            isPartnerTyping={isPartnerTyping}
   499→            currentUserId={currentUserId}
   500→            partnerName={partner.name}
   501→            partnerAvatar={partner.avatar_url}
   502→            onMarkAsRead={markAsRead}
   503→            onVanishMessage={vanishMessage}
   504→            onDeleteMessage={deleteMessage}
   505→            onReplyTo={setReplyTo}
   506→            onReact={sendReaction}
   507→            scrollToMessageId={scrollToMessageId}
   508→            onScrolledToMessage={handleScrolledToMessage}
   509→          />
   510→        </div>
   511→
   512→        {/* Reply preview bar */}
   513→        <AnimatePresence>
   514→          {replyTo && (
   515→            <m.div initial={{ height: 0, opacity: 0 }} animate={{ height: "auto", opacity: 1 }} exit={{ height: 0, opacity: 0 }} transition={{ duration: 0.2 }} className="overflow-hidden border-t bg-muted/30">
   516→              <div className="flex items-center gap-2 px-4 py-2">
   517→                <div className="w-1 h-8 rounded-full bg-gradient-to-b from-violet-500 to-purple-600" />
   518→                <div className="min-w-0 flex-1">
   519→                  <p className="text-[11px] font-semibold text-violet-500">{replyTo.sender_name || "Unknown"}</p>
   520→                  <p className="truncate text-xs text-muted-foreground">{replyTo.content}</p>
   521→                </div>
   522→                <div className="flex items-center gap-1.5">
   523→                  <button type="button" onClick={() => setReplyTo(null)} className="flex size-6 items-center justify-center rounded-full text-muted-foreground hover:bg-muted"><X className="size-3.5" /></button>
   524→                  <kbd className="hidden sm:inline-flex h-5 items-center rounded border bg-muted px-1.5 text-[10px] font-medium text-muted-foreground">Esc</kbd>
   525→                </div>
   526→              </div>
   527→            </m.div>
   528→          )}
   529→        </AnimatePresence>
   530→
   531→        {/* Voice waveform overlay */}
   532→        <AnimatePresence>
   533→          {showVoiceWaveform && (
   534→            <m.div initial={{ height: 0, opacity: 0 }} animate={{ height: "auto", opacity: 1 }} exit={{ height: 0, opacity: 0 }} transition={{ duration: 0.15 }} className="overflow-hidden">
   535→              <div className="flex items-center gap-3 px-4 py-3">
   536→                <div className="flex size-10 items-center justify-center rounded-full bg-red-500/15 text-red-500">
   537→                  {voice.isRecording ? <span className="relative flex size-3"><span className="absolute inline-flex size-full animate-ping rounded-full bg-red-400 opacity-75" /><span className="relative inline-flex size-3 rounded-full bg-red-500" /></span> : null}
   538→                </div>
   539→                <div className="flex flex-1 items-end justify-center gap-px" style={{ height: 40 }}>
   540→                  {voice.amplitudes.slice(-60).map((amp, i) => (
   541→                    <div key={i} className="w-[2px] rounded-full bg-violet-500/70 transition-all duration-75" style={{ height: `${Math.max(3, amp * 40)}px` }} />
   542→                  ))}
   543→                </div>
   544→                <span className="shrink-0 text-xs font-medium tabular-nums text-muted-foreground">{Math.floor(voice.duration / 60)}:{String(voice.duration % 60).padStart(2, "0")}</span>
   545→                <m.button type="button" onClick={handleVoiceSend} whileTap={{ scale: 0.9 }} className="flex size-10 shrink-0 items-center justify-center rounded-full bg-gradient-to-r from-violet-500 to-purple-600 text-white" aria-label="Send voice">
   546→                  <ArrowUp className="size-4" />
   547→                </m.button>
   548→              </div>
   549→            </m.div>
   550→          )}
   551→        </AnimatePresence>
   552→
   553→        {/* Input bar */}
   554→        <div className="shrink-0 border-t bg-background px-3 py-2.5 sm:px-4 sm:py-3">
   555→          <form onSubmit={handleSubmit} className="flex items-end gap-2">
   556→            {/* Vanish toggle - compact with smooth transition */}
   557→            <m.button type="button" onClick={() => setVanishMode((v) => !v)} whileTap={{ scale: 0.9 }} className={cn("flex size-8 shrink-0 items-center justify-center rounded-lg transition-all duration-200", vanishMode ? "bg-violet-500/15 text-violet-500 shadow-sm shadow-violet-500/10" : "text-muted-foreground hover:bg-muted")} aria-label="Vanish mode">
   558→              {vanishMode ? <EyeOff className="size-3.5" /> : <Eye className="size-3.5" />}
   559→            </m.button>
   560→
   561→            {/* Ephemeral timer dropdown - compact */}
   562→            <div className="relative">
   563→              <m.button type="button" onClick={() => setEphemeralOpen((v) => !v)} whileTap={{ scale: 0.9 }} className={cn("flex size-8 shrink-0 items-center justify-center rounded-lg transition-colors duration-200", ephemeralSeconds ? "bg-amber-500/15 text-amber-500" : "text-muted-foreground hover:bg-muted")} aria-label="Ephemeral timer">
   564→                <Clock className="size-3.5" />
   565→              </m.button>
   566→              <AnimatePresence>
   567→                {ephemeralOpen && (
   568→                  <m.div initial={{ opacity: 0, y: 4, scale: 0.95 }} animate={{ opacity: 1, y: 0, scale: 1 }} exit={{ opacity: 0, y: 4, scale: 0.95 }} transition={{ duration: 0.15 }} className="absolute bottom-full left-0 z-50 mb-2 w-32 overflow-hidden rounded-xl border bg-popover p-1.5 shadow-xl shadow-black/10">
   569→                    {EPHEMERAL_OPTIONS.map((opt) => (
   570→                      <button key={opt.label} type="button" onClick={() => { setEphemeralSeconds(opt.value); setEphemeralOpen(false); }}
   571→                        className={cn("flex w-full items-center gap-2 rounded-lg px-3 py-2 text-xs transition-all duration-150", ephemeralSeconds === opt.value ? "bg-violet-500/10 text-violet-500 font-semibold" : "text-foreground hover:bg-muted")}>
   572→                        <Clock className="size-3 opacity-60" />{opt.label}
   573→                      </button>
   574→                    ))}
   575→                  </m.div>
   576→                )}
   577→              </AnimatePresence>
   578→            </div>
   579→
   580→            {/* Text input (hidden when recording) - pill shaped with gradient border on focus */}
   581→            {!showVoiceWaveform && (
   582→              <div className="relative flex-1">
   583→                <textarea ref={inputRef} value={draft} onChange={handleInputChange} onKeyDown={handleKeyDown} onInput={handleInput}
   584→                  placeholder={vanishMode ? "Send a disappearing message\u2026" : replyTo ? `Reply to ${replyTo.sender_name || "message"}...` : "Type a message\u2026"}
   585→                  rows={1} className={cn(
   586→                    "w-full resize-none rounded-2xl border bg-muted/50 px-4 py-3 text-sm",
   587→                    "placeholder:text-muted-foreground",
   588→                    "focus-visible:outline-none",
   589→                    "max-h-32 overflow-y-auto",
   590→                    "transition-all duration-300",
   591→                    // Glow effect on focus
   592→                    "focus-visible:border-transparent focus-visible:ring-2",
   593→                    vanishMode
   594→                      ? "border-violet-500/30 focus-visible:ring-violet-400/40 focus-visible:shadow-[0_0_12px_rgba(139,92,246,0.15)]"
   595→                      : "focus-visible:ring-violet-400/30 focus-visible:shadow-[0_0_12px_rgba(139,92,246,0.08)]",
   596→                  )}
   597→                />
   598→              </div>
   599→            )}
   600→
   601→            {/* Attachment button (placeholder) */}
   602→            {!showVoiceWaveform && (
   603→              <m.button type="button" whileTap={{ scale: 0.9 }} className="flex size-9 shrink-0 items-center justify-center rounded-xl text-muted-foreground transition-colors hover:bg-muted" aria-label="Attach file">
   604→                <Paperclip className="size-4" />
   605→              </m.button>
   606→            )}
   607→
   608→            {/* Emoji button with picker */}
   609→            {!showVoiceWaveform && (
   610→              <EmojiPicker
   611→                onSelect={(emoji) => {
   612→                  const ta = inputRef.current;
   613→                  if (!ta) return;
   614→                  const start = ta.selectionStart ?? draft.length;
   615→                  const end = ta.selectionEnd ?? draft.length;
   616→                  const before = draft.slice(0, start);
   617→                  const after = draft.slice(end);
   618→                  setDraft(before + emoji + after);
   619→                  requestAnimationFrame(() => {
   620→                    const pos = start + emoji.length;
   621→                    ta.focus();
   622→                    ta.setSelectionRange(pos, pos);
   623→                  });
   624→                }}
   625→              >
   626→                <m.button type="button" whileTap={{ scale: 0.9 }} className="flex size-9 shrink-0 items-center justify-center rounded-xl text-muted-foreground transition-colors hover:bg-muted" aria-label="Emoji">
   627→                  <Smile className="size-4" />
   628→                </m.button>
   629→              </EmojiPicker>
   630→            )}
   631→
   632→            {/* Send button (shown when there's text) */}
   633→            {!showVoiceWaveform && hasText && (
   634→              <AnimatePresence>
   635→                <m.div key={sendBtnKey} initial={{ scale: 0, opacity: 0, rotate: -90 }} animate={{ scale: 1, opacity: 1, rotate: 0 }} exit={{ scale: 0, opacity: 0 }} transition={{ duration: 0.2, type: "spring", stiffness: 500, damping: 25 }}>
   636→                  <Button type="submit" size="icon" disabled={isSending}
   637→                    className={cn("size-9 shrink-0 rounded-xl bg-gradient-to-r from-violet-500 to-purple-600 shadow-md shadow-violet-500/25 transition-all duration-200 hover:shadow-lg hover:shadow-violet-500/30 hover:scale-105")}
   638→                    aria-label="Send message"><ArrowUp className="size-4" /></Button>
   639→                </m.div>
   640→              </AnimatePresence>
   641→            )}
   642→
   643→            {/* Mic button (shown when input is empty) */}
   644→            {!showVoiceWaveform && !hasText && (
   645→              <AnimatePresence>
   646→                <m.div initial={{ scale: 0, opacity: 0 }} animate={{ scale: 1, opacity: 1 }} exit={{ scale: 0, opacity: 0 }} transition={{ duration: 0.15, type: "spring", stiffness: 500, damping: 30 }}>
   647→                  <m.button type="button" onClick={() => voice.startRecording()} whileTap={{ scale: 0.9 }} className="flex size-9 shrink-0 items-center justify-center rounded-xl text-muted-foreground transition-colors hover:bg-muted" aria-label="Record voice">
   648→                    <Mic className="size-4" />
   649→                  </m.button>
   650→                </m.div>
   651→              </AnimatePresence>
   652→            )}
   653→
   654→            {/* Cancel recording button */}
   655→            {showVoiceWaveform && (
   656→              <m.button type="button" onClick={async () => { await voice.stopRecording(); setShowVoiceWaveform(false); }} whileTap={{ scale: 0.9 }} className="flex size-10 shrink-0 items-center justify-center rounded-xl text-red-500 hover:bg-red-500/10" aria-label="Cancel recording">
   657→                <MicOff className="size-4" />
   658→              </m.button>
   659→            )}
   660→          </form>
   661→        </div>
   662→      </div>
   663→
   664→      {/* Partner Info Panel */}
   665→      <PartnerInfoPanel
   666→        partner={partner}
   667→        partnerStatus={partnerStatus}
   668→        open={infoPanelOpen}
   669→        onClose={() => setInfoPanelOpen(false)}
   670→      />
   671→    </div>
   672→  );
   673→}
