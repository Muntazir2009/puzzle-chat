    55→
    56→function useDoubleTap(
    57→  onDoubleTap: () => void,
    58→  delay = 300,
    59→) {
    60→  const lastTapRef = useRef(0);
    61→  return useCallback(() => {
    62→    const now = Date.now();
    63→    if (now - lastTapRef.current < delay) {
    64→      onDoubleTap();
    65→      lastTapRef.current = 0; // reset
    66→    } else {
    67→      lastTapRef.current = now;
    68→    }
    69→  }, [onDoubleTap, delay]);
    70→}

   143→/*  Ephemeral Timer (circular SVG countdown)                           */
   144→/* ------------------------------------------------------------------ */
   145→
   146→function EphemeralTimer({
   147→  seconds,
   148→  onExpire,
   149→}: {
   150→  seconds: number;
   151→  onExpire: () => void;
   152→}) {
   153→  const [remaining, setRemaining] = useState(seconds);
   154→  const intervalRef = useRef<ReturnType<typeof setInterval> | undefined>(
   155→    undefined,
   156→  );
   157→
   158→  useEffect(() => {
   159→    intervalRef.current = setInterval(() => {
   160→      setRemaining((r) => {
   161→        if (r <= 1) {
   162→          clearInterval(intervalRef.current);
   163→          onExpire();
   164→          return 0;
   165→        }
   166→        return r - 1;
   167→      });
   168→    }, 1000);
   169→    return () => clearInterval(intervalRef.current);
   170→  }, [seconds, onExpire]);
   171→
   172→  const pct = remaining / seconds;
   173→  const circ = 2 * Math.PI * 7;
   174→  const offset = circ * (1 - pct);
   175→
   176→  return (
   177→    <svg
   178→      className="size-3.5 shrink-0 ephemeral-timer-pulse"
   179→      viewBox="0 0 18 18"
   180→      aria-label={`${remaining}s remaining`}
   181→    >
   182→      <circle
   183→        cx="9"
   184→        cy="9"
   185→        r="7"
   186→        fill="none"
   187→        stroke="currentColor"
   188→        strokeWidth="1.5"
   189→        className="text-white/20"
   190→      />
   191→      <circle
   192→        cx="9"
   193→        cy="9"
   194→        r="7"
   195→        fill="none"
   196→        stroke="currentColor"
   197→        strokeWidth="1.5"
   198→        strokeDasharray={circ}
   199→        strokeDashoffset={offset}
   200→        strokeLinecap="round"
   201→        style={{
   202→          color: "var(--app-accent-lighter)",
   203→          transform: "rotate(-90deg)",
   204→          transformOrigin: "50% 50%",
   205→          transition: "stroke-dashoffset 1s linear",
   206→        }}
   207→      />
   208→    </svg>
   209→  );
   210→}

   419→/* ------------------------------------------------------------------ */
   420→/*  Voice Bubble                                                       */
   421→/* ------------------------------------------------------------------ */
   422→
   423→function VoiceBubble({
   424→  message,
   425→  isOwn,
   426→  waveform,
   427→}: {
   428→  message: ChatMessage;
   429→  isOwn: boolean;
   430→  waveform: number[];
   431→}) {
   432→  const audioRef = useRef<HTMLAudioElement>(null);
   433→  const [playing, setPlaying] = useState(false);
   434→  const [progress, setProgress] = useState(0);
   435→  const [speed, setSpeed] = useState(1);
   436→  const speeds = [1, 1.5, 2];
   437→  const duration = message.voice_duration ?? 0;
   438→
   439→  useEffect(() => {
   440→    const audio = audioRef.current;
   441→    if (!audio) return;
   442→    const onTime = () => {
   443→      if (audio.duration) setProgress(audio.currentTime / audio.duration);
   444→    };
   445→    const onEnd = () => setPlaying(false);
   446→    audio.addEventListener("timeupdate", onTime);
   447→    audio.addEventListener("ended", onEnd);
   448→    return () => {
   449→      audio.removeEventListener("timeupdate", onTime);
   450→      audio.removeEventListener("ended", onEnd);
   451→    };
   452→  }, []);
   453→
   454→  const toggle = useCallback(() => {
   455→    const audio = audioRef.current;
   456→    if (!audio) return;
   457→    if (playing) {
   458→      audio.pause();
   459→      setPlaying(false);
   460→    } else {
   461→      audio.playbackRate = speed;
   462→      audio.play().catch(() => {});
   463→      setPlaying(true);
   464→    }
   465→  }, [playing, speed]);
   466→
   467→  const cycleSpeed = useCallback(() => {
   468→    setSpeed((s) => {
   469→      const idx = speeds.indexOf(s);
   470→      const next = speeds[(idx + 1) % speeds.length];
   471→      if (audioRef.current) audioRef.current.playbackRate = next;
   472→      return next;
   473→    });
   474→  }, []);
   475→
   476→  /* Build waveform bars */
   477→  const bars = useMemo(() => {
   478→    const count = Math.min(waveform.length, 40);
   479→    const step = Math.max(1, Math.floor(waveform.length / count));
   480→    return Array.from({ length: count }, (_, i) => waveform[i * step] ?? 0.1);
   481→  }, [waveform]);
   482→
   483→  const progressIdx = Math.floor(progress * bars.length);
   484→
   485→  return (
   486→    <div className={cn("flex items-center gap-3", isOwn ? "flex-row-reverse" : "")}>
   487→      <button
   488→        type="button"
   489→        onClick={toggle}
   490→        className={cn(
   491→          "flex size-9 shrink-0 items-center justify-center rounded-full transition-all duration-150 active:scale-95",
   492→          isOwn
   493→            ? "bg-white/20 hover:bg-white/30"
   494→            : "bg-zinc-600 hover:bg-[var(--app-accent-subtle)]",
   495→        )}
   496→        aria-label={playing ? "Pause" : "Play"}
   497→      >
   498→        {playing ? (
   499→          <Pause className="size-3.5 text-white" />
   500→        ) : (
   501→          <Play className="size-3.5 text-white ml-0.5" />
   502→        )}
   503→      </button>
   504→      <div className="flex items-center gap-[2px]" style={{ height: 32 }}>
   505→        {bars.map((amp, i) => (
   506→          <div
   507→            key={i}
   508→            className="w-[2.5px] rounded-full transition-all duration-100"
   509→            style={{
   510→              height: `${Math.max(4, amp * 32)}px`,
   511→              backgroundColor:
   512→                i < progressIdx
   513→                  ? isOwn
   514→                    ? "rgba(255,255,255,1)"
   515→                    : "var(--app-accent-light)"
   516→                  : isOwn
   517→                    ? "rgba(255,255,255,0.5)"
   518→                    : "rgba(161,161,170,0.7)",
   519→            }}
   520→          />
   521→        ))}
   522→      </div>
   523→      <button
   524→        type="button"
   525→        onClick={cycleSpeed}
   526→        className={cn(
   527→          "shrink-0 text-[11px] font-semibold tabular-nums transition-colors hover:opacity-80 active:scale-95",
   528→          isOwn ? "text-white/70" : "text-zinc-400",
   529→        )}
   530→      >
   531→        {speed}x
   532→      </button>
   533→      <span
   534→        className={cn(
   535→          "shrink-0 text-[11px] tabular-nums",
   536→          isOwn ? "text-white/50" : "text-zinc-500",
   537→        )}
   538→      >
   539→        {Math.floor(duration / 60)}:{String(duration % 60).padStart(2, "0")}
   540→      </span>
   541→      <audio ref={audioRef} src={message.content} preload="metadata" />
   542→    </div>
   543→  );
   544→}

   546→/* ------------------------------------------------------------------ */
   547→/*  LinkifiedText – detects & renders clickable URLs                   */
   548→/* ------------------------------------------------------------------ */
   549→
   550→const URL_REGEX = /(https?:\/\/[^\s<]+)/g;
   551→
   552→type TextSegment =
   553→  | { type: "text"; value: string }
   554→  | { type: "link"; value: string };
   555→
   556→function parseUrls(text: string): TextSegment[] {
   557→  const segments: TextSegment[] = [];
   558→  let lastIndex = 0;
   559→  let match: RegExpExecArray | null;
   560→  while ((match = URL_REGEX.exec(text)) !== null) {
   561→    if (match.index > lastIndex) {
   562→      segments.push({ type: "text", value: text.slice(lastIndex, match.index) });
   563→    }
   564→    segments.push({ type: "link", value: match[1] });
   565→    lastIndex = URL_REGEX.lastIndex;
   566→  }
   567→  if (lastIndex < text.length) {
   568→    segments.push({ type: "text", value: text.slice(lastIndex) });
   569→  }
   570→  return segments;
   571→}
   572→
   573→/** Split a plain-text string into highlighted / non-highlighted fragments */
   574→function highlightText(text: string, query: string): React.ReactNode[] {
   575→  if (!query) return [text];
   576→  const escaped = query.replace(/[.*+?^${}()|[\]\\]/g, "\\");
   577→  const re = new RegExp(`(${escaped})`, "gi");
   578→  const parts: React.ReactNode[] = [];
   579→  let lastIndex = 0;
   580→  let match: RegExpExecArray | null;
   581→  while ((match = re.exec(text)) !== null) {
   582→    if (match.index > lastIndex) {
   583→      parts.push(text.slice(lastIndex, match.index));
   584→    }
   585→    parts.push(
   586→      <mark
   587→        key={match.index}
   588→        className="bg-yellow-300/60 dark:bg-yellow-400/40 rounded-sm px-0.5"
   589→      >
   590→        {match[1]}
   591→      </mark>,
   592→    );
   593→    lastIndex = re.lastIndex;
   594→  }
   595→  if (lastIndex < text.length) {
   596→    parts.push(text.slice(lastIndex));
   597→  }
   598→  return parts;
   599→}

   806→/* ------------------------------------------------------------------ */
   807→/*  Bubble rounding helper for grouped messages                        */
   808→/* ------------------------------------------------------------------ */
   809→
   810→function getBubbleRounding({
   811→  isFirst,
   812→  isLast,
   813→  isOwn,
   814→}: {
   815→  isFirst: boolean;
   816→  isLast: boolean;
   817→  isOwn: boolean;
   818→}): string {
   819→  if (isFirst && isLast) {
   820→    // Single message in group (or only message)
   821→    return isOwn
   822→      ? "rounded-2xl rounded-br-sm"
   823→      : "rounded-2xl rounded-bl-sm";
   824→  }
   825→  if (isFirst) {
   826→    // First message of a multi-message group
   827→    return isOwn
   828→      ? "rounded-tl-2xl rounded-bl-2xl rounded-tr-sm rounded-br-sm"
   829→      : "rounded-tl-2xl rounded-bl-sm rounded-tr-2xl rounded-br-sm";
   830→  }
   831→  if (isLast) {
   832→    // Last message of a multi-message group
   833→    return isOwn
   834→      ? "rounded-tl-2xl rounded-bl-2xl rounded-tr-sm rounded-br-2xl"
   835→      : "rounded-tl-2xl rounded-bl-2xl rounded-tr-2xl rounded-br-sm";
   836→  }
   837→  // Middle message
   838→  return isOwn
   839→    ? "rounded-tl-2xl rounded-bl-2xl rounded-tr-sm rounded-br-sm"
   840→    : "rounded-tl-2xl rounded-bl-sm rounded-tr-2xl rounded-br-sm";
   841→}

   847→function MessageBubble({
   848→  message,
   849→  isOwn,
   850→  isFirst,
   851→  isLast,
   852→  partnerName,
   853→  partnerAvatar,
   854→  currentUserId,
   855→  currentUserName,
   856→  currentUserAvatar,
   857→  onReplyTo,
   858→  onReact,
   859→  onDeleteMessage,
   860→  searchHighlight,
   861→  activeActionId,
   862→  onOpenAction,
   863→  activeTapbackId,
   864→  onOpenTapback,
   865→}: {
   866→  message: ChatMessage;
   867→  isOwn: boolean;
   868→  isFirst: boolean;
   869→  isLast: boolean;
   870→  partnerName: string;
   871→  partnerAvatar: string | null;
   872→  currentUserId: string;
   873→  currentUserName: string;
   874→  currentUserAvatar: string | null;
   875→  onReplyTo?: (msg: ChatMessage) => void;
   876→  onReact?: (id: string, emoji: string, add: boolean) => void;
   877→  onDeleteMessage?: (id: string) => void;
   878→  searchHighlight?: string;
   879→  activeActionId: string | null;
   880→  onOpenAction: (id: string | null) => void;
   881→  activeTapbackId: string | null;
   882→  onOpenTapback: (id: string | null) => void;
   883→}) {
   884→  const showTapback = activeTapbackId === message.id;
   885→  const showActionSheet = activeActionId === message.id;
   886→  const [showLightbox, setShowLightbox] = useState<string | null>(null);
   887→  const [showReplyArrow, setShowReplyArrow] = useState(false);
   888→  const longPressRef = useRef<ReturnType<typeof setTimeout> | undefined>(
   889→    undefined,
   890→  );
   891→  const wasDraggedRef = useRef(false);
   892→
   893→  /* Double-tap to reaction */
   894→  const handleDoubleTapReaction = useCallback(() => {
   895→    if (!onReact) return;
   896→    onReact(message.id, "\u{1F44D}", true);
   897→  }, [message.id, onReact]);
   898→
   899→  const doubleTap = useDoubleTap(handleDoubleTapReaction);
   900→
   901→  const handleClick = useCallback(() => {
   902→    clearTimeout(longPressRef.current);
   903→    if (wasDraggedRef.current) {
   904→      wasDraggedRef.current = false;
   905→      return;
   906→    }
   907→    // Double-tap reaction
   908→    doubleTap();
   909→    // Toggle action sheet: close if already open for this bubble
   910→    if (activeActionId === message.id) {
   911→      onOpenAction(null);
   912→    } else {
   913→      onOpenAction(message.id);
   914→    }
   915→    // Always close tapback on single click
   916→    onOpenTapback(null);
   917→  }, [activeActionId, message.id, onOpenAction, onOpenTapback, doubleTap]);
   918→
   919→  const handlePressStart = useCallback(() => {
   920→    onOpenAction(null);
   921→    onOpenTapback(null);
   922→    longPressRef.current = setTimeout(() => {
   923→      if (typeof navigator !== "undefined" && "vibrate" in navigator)
   924→        navigator.vibrate(20);
   925→      onOpenTapback(message.id);
   926→    }, LONG_PRESS_MS);
   927→  }, [onOpenAction, onOpenTapback, message.id]);
   928→
   929→  const handlePressEnd = useCallback(() => {
   930→    clearTimeout(longPressRef.current);
   931→  }, []);
   932→
   933→  const handleContextMenu = useCallback((e: React.MouseEvent) => {
   934→    e.preventDefault();
   935→    onOpenAction(null);
   936→    onOpenTapback(message.id);
   937→  }, [onOpenAction, onOpenTapback, message.id]);
   938→
   939→  const handleCopy = useCallback(() => {
   940→    navigator.clipboard
   941→      ?.writeText(message.content)
   942→      .then(() => {

   944→          title: "Copied to clipboard",

   946→            message.content.length > 50
   947→              ? message.content.slice(0, 50) + "..."
   948→              : message.content,
   949→        });
   950→      })
   951→      .catch(() => {
   952→        toast({
   953→          title: "Failed to copy",
   954→          description: "Could not access clipboard",
   955→          variant: "destructive",
   956→        });
   957→      });
   958→    onOpenTapback(null);
   959→  }, [message.content, onOpenTapback]);
   960→
   961→  const handleDelete = useCallback(() => {
   962→    onDeleteMessage?.(message.id);
   963→    onOpenTapback(null);
   964→    onOpenAction(null);
   965→  }, [message.id, onDeleteMessage, onOpenTapback, onOpenAction]);
   966→
   967→  const handleActionSheetCopy = useCallback(() => {
   968→    navigator.clipboard
   969→      ?.writeText(message.content)
   970→      .then(() => {
   971→        toast({

   981→          title: "Failed to copy",
   982→          description: "Could not access clipboard",
   983→          variant: "destructive",
   984→        });
   985→      });
   986→    onOpenAction(null);
   987→    onOpenTapback(null);
   988→  }, [message.content, onOpenAction, onOpenTapback]);
   989→
   990→  const handleActionSheetReply = useCallback(() => {
   991→    onReplyTo?.(message);
   992→    onOpenAction(null);
   993→    onOpenTapback(null);
   994→  }, [message, onReplyTo, onOpenAction, onOpenTapback]);
   995→
   996→  const handleActionSheetForward = useCallback(() => {
   997→    toast({
   998→      title: "Forward",
   999→      description: "Forward coming soon!",
  1000→    });
  1001→    onOpenAction(null);
  1002→    onOpenTapback(null);
  1003→  }, [onOpenAction, onOpenTapback]);
  1004→
  1005→  const isFailed = message.status === "failed";
  1006→  const isVoice = message.type === "voice";
  1007→  const isText = !isVoice && message.type !== "image";
  1008→  const isNew =
  1009→    message.status === "sending" ||
  1010→    Date.now() - new Date(message.created_at).getTime() < 2000;
  1011→
  1012→  const timestampText = formatDistanceToNow(
  1013→    new Date(message.created_at || Date.now()),
  1014→    { addSuffix: true },
  1015→  );
  1016→
  1017→  const rounding = getBubbleRounding({ isFirst, isLast, isOwn });
  1018→
  1019→  return (
  1020→    <>
  1021→      <div
  1022→        className={cn(
  1023→          "message-bubble-row relative flex w-full select-none gap-1",

  1025→        )}
  1026→      >
  1027→        {/* Avatar column for other user's messages — always rendered for stacking */}
  1028→        {!isOwn && (
  1029→          <Avatar className={cn(
  1030→            "shrink-0 self-end ring-1 ring-white/10",
  1031→            isLast ? "size-7" : "size-7 opacity-0 pointer-events-none",

  1034→              <AvatarImage src={partnerAvatar} alt={partnerName} />
  1035→            )}
  1036→            <AvatarFallback
  1037→              className="text-[10px] font-medium"
  1038→              style={{
  1039→                background:
  1040→                  "linear-gradient(to bottom right, var(--app-accent-lighter), var(--app-accent-light))",
  1041→                color: "var(--app-accent-dark)",

  1049→        <AnimatePresence>

  1380→function formatDateSeparator(dateStr: string): string {
  1381→  const d = new Date(dateStr);
  1382→  const now = new Date();
  1383→  const today = new Date(now.getFullYear(), now.getMonth(), now.getDate());
  1384→  const yesterday = new Date(today);
  1385→  yesterday.setDate(yesterday.getDate() - 1);
  1386→  const msgDay = new Date(d.getFullYear(), d.getMonth(), d.getDate());
  1387→
  1388→  if (msgDay.getTime() === today.getTime()) return "Today";
  1389→  if (msgDay.getTime() === yesterday.getTime()) return "Yesterday";
  1390→  return d.toLocaleDateString(undefined, { month: "short", day: "numeric" });
  1391→}

  1574→export function MessageFeed({
  1575→  messages,
  1576→  isLoading,
  1577→  isPartnerTyping,
  1578→  currentUserId,
  1579→  currentUserName,
  1580→  currentUserAvatar,
  1581→  partnerName,
  1582→  partnerAvatar,
  1583→  backgroundStyle,
  1584→  onMarkAsRead,
  1585→  onVanishMessage,
  1586→  onReplyTo,
  1587→  onDeleteMessage,
  1588→  onReact,
  1589→  scrollToMessageId,
  1590→  onScrolledToMessage,
  1591→  searchHighlight,
  1592→  onClearSearchHighlight,
  1593→  loadMore,
  1594→  hasMore,
  1595→  loadingMore,
  1596→}: MessageFeedProps) {
  1597→  const parentRef = useRef<HTMLDivElement>(null);
  1598→  const isAutoScrollRef = useRef(true);
  1599→  const [activeActionId, setActiveActionId] = useState<string | null>(null);
  1600→  const [activeTapbackId, setActiveTapbackId] = useState<string | null>(null);
  1601→  const [showNewBtn, setShowNewBtn] = useState(false);
  1602→  const unreadRef = useRef(0);
  1603→  const prevCountRef = useRef(messages.length);
  1604→  const prevFirstIdRef = useRef(
  1605→    messages.length > 0 ? messages[0].id : null,
  1606→  );
  1607→  const prevScrollHeightRef = useRef(0);
  1608→
  1609→  /* ---- Build message groups ---------------------------------------- */
  1610→  const groups = useMemo(() => {
  1611→    const result: MessageGroupData[] = [];
  1612→    for (const msg of messages) {
  1613→      const last = result[result.length - 1];
  1614→      if (last && last.senderId === msg.sender_id) {
  1615→        last.messages.push(msg);
  1616→      } else {
  1617→        result.push({ senderId: msg.sender_id, messages: [msg] });
  1618→      }
  1619→    }
  1620→    return result;
  1621→  }, [messages]);
  1622→
  1623→  const totalItems = useMemo(
  1624→    () => groups.length + (isPartnerTyping ? 1 : 0),
  1625→    [groups.length, isPartnerTyping],
  1626→  );
  1627→
  1628→  const loadMoreRef = useRef(loadMore);
  1629→  loadMoreRef.current = loadMore;
  1630→
  1631→  const virtualizer = useVirtualizer({
  1632→    count: totalItems,
  1633→    getScrollElement: () => parentRef.current,
  1634→    estimateSize: () => ESTIMATED_ROW_HEIGHT,
  1635→    overscan: OVERSCAN,
  1636→    paddingStart: 16,
  1637→    paddingEnd: 16,
  1638→    gap: 12,
  1639→  });
  1640→
  1641→  const handleScroll = useCallback(
  1642→    (e: UIEvent<HTMLDivElement>) => {
  1643→      const el = e.currentTarget;
  1644→      const dist = el.scrollHeight - el.scrollTop - el.clientHeight;
  1645→      if (dist <= AUTO_SCROLL_THRESHOLD) {
  1646→        isAutoScrollRef.current = true;
  1647→        setShowNewBtn(false);
  1648→        unreadRef.current = 0;
  1649→      } else {
  1650→        isAutoScrollRef.current = false;
  1651→      }
  1652→      /* Trigger loadMore when scrolled near top */
  1653→      if (el.scrollTop < 300 && hasMore && !loadingMore) {
  1654→        loadMoreRef.current?.();
  1655→      }
  1656→    },
  1657→    [hasMore, loadingMore],
  1658→  );
  1659→
  1660→  const scrollToBottom = useCallback(
  1661→    (behavior: ScrollBehavior = "smooth") => {
  1662→      const el = parentRef.current;
  1663→      if (!el) return;
  1664→      el.scrollTo({ top: el.scrollHeight, behavior });
  1665→      isAutoScrollRef.current = true;
  1666→      setShowNewBtn(false);
  1667→      unreadRef.current = 0;
  1668→    },
  1669→    [],
  1670→  );
  1671→
  1672→  useEffect(() => {
  1673→    if (messages.length > 0)
  1674→      requestAnimationFrame(() => scrollToBottom("instant"));
  1675→  }, [isLoading, scrollToBottom]);
  1676→
  1677→  useEffect(() => {
  1678→    if (!onMarkAsRead) return;
  1679→    const unread = messages.filter(
  1680→      (m) => m.sender_id !== currentUserId && m.status !== "read",
  1681→    );
  1682→    if (unread.length > 0 && isAutoScrollRef.current)
  1683→      onMarkAsRead(unread.map((m) => m.id));
  1684→  }, [messages, currentUserId, onMarkAsRead]);
  1685→
  1686→  /* ---- Scroll to a specific message ----------------------------------- */
  1687→  useEffect(() => {
  1688→    if (!scrollToMessageId) return;
  1689→    // Find which group contains the target message
  1690→    const groupIdx = groups.findIndex((g) =>
  1691→      g.messages.some((m) => m.id === scrollToMessageId),
  1692→    );
  1693→    if (groupIdx === -1) {
  1694→      onScrolledToMessage?.();
  1695→      return;
  1696→    }
  1697→    virtualizer.scrollToIndex(groupIdx, { align: "center" });
  1698→    onScrolledToMessage?.();
  1699→  }, [scrollToMessageId, groups, virtualizer, onScrolledToMessage]);
  1700→
  1701→  /* ---- Handle message count changes (new or prepended) ------------ */
  1702→  useEffect(() => {
  1703→    const prev = prevCountRef.current;
  1704→    const prevFirstId = prevFirstIdRef.current;
  1705→    prevCountRef.current = messages.length;
  1706→    prevFirstIdRef.current =
  1707→      messages.length > 0 ? messages[0].id : null;
  1708→
  1709→    if (messages.length <= prev) return;
  1710→
  1711→    /* Check if older messages were prepended (first message changed) */
  1712→    const firstId = messages[0]?.id;
  1713→    const wasPrepended = firstId !== prevFirstId && prevFirstId !== null;
  1714→
  1715→    if (wasPrepended) {
  1716→      /* Older messages prepended – preserve scroll position */
  1717→      const el = parentRef.current;
  1718→      if (el) {
  1719→        const oldHeight = prevScrollHeightRef.current;
  1720→        prevScrollHeightRef.current = el.scrollHeight;
  1721→        requestAnimationFrame(() => {
  1722→          const newHeight = el.scrollHeight;
  1723→          el.scrollTop += newHeight - oldHeight;
  1724→        });
  1725→      }
  1726→      return;
  1727→    }
  1728→
  1729→    /* New messages appended at the end */
  1730→    const last = messages[messages.length - 1];
  1731→    const own = last && last.sender_id === currentUserId;
  1732→    if (own) {
  1733→      requestAnimationFrame(() => scrollToBottom("instant"));
  1734→      return;
  1735→    }
  1736→    if (isAutoScrollRef.current)
  1737→      requestAnimationFrame(() => scrollToBottom("smooth"));
  1738→    else {
  1739→      unreadRef.current += messages.length - prev;
  1740→      setShowNewBtn(true);
  1741→    }
  1742→  }, [messages.length, currentUserId, scrollToBottom]);
  1743→
  1744→  /* Track scroll height for scroll preservation */
  1745→  useEffect(() => {
  1746→    const el = parentRef.current;
  1747→    if (!el) return;
  1748→    prevScrollHeightRef.current = el.scrollHeight;
  1749→  }, [messages.length]);
  1750→
  1751→  if (isLoading) return <MessageListSkeleton />;
  1752→  if (messages.length === 0 && !isPartnerTyping)
  1753→    return (
  1754→      <EmptyState partnerName={partnerName} partnerAvatar={partnerAvatar} />
  1755→    );
  1756→
  1757→  const virtualItems = virtualizer.getVirtualItems();
  1758→
  1759→  return (
  1760→    <div
  1761→      className="relative h-full w-full overflow-hidden"
  1762→      style={backgroundStyle}
  1763→    >
  1764→      <div
  1765→        ref={parentRef}
  1766→        onScroll={handleScroll}
  1767→        onClick={(e) => {
  1768→          onClearSearchHighlight?.();
